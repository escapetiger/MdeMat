2 % TILEDPLOTTER Multi-panel figure manager for data visualization.
3 %
4 % TiledPlotter provides infrastructure for creating and managing
5 % multi-panel figures using MATLAB's tiledlayout functionality.
6 % This class simplifies the creation of complex visualization layouts
7 % with automatic tile placement and flexible axis specifications.

12 % panel sizes.
13 %
14 % Each plotter instance receives its figure from a parent Visualizer,
15 % providing centralized figure management and better control over

107 function obj = draft(obj, options)
108 % DRAFT Create the tiledlayout and axes.
109 %
110 % obj = draft(obj) creates the MATLAB tiledlayout object and
111 % generates all axes according to the added specifications.
112 % A figure must be assigned using setFigure() before calling
113 % this method.
114 %
115 % obj = draft(obj, 'ParameterName', ParameterValue) passes
116 % additional arguments to the tiledlayout constructor.
117
118 arguments
119 obj physics.visual.TiledPlotter
120 options struct = struct()
121 end
122
123 if ~isempty(obj.Axes), return; end
124
125 core.except.assert(~isempty(obj.FigObj) && isvalid(obj.FigObj), ...
126 'InvalidFigure', 'Figure must be valid and assigned before drafting.');
127
128 %< Create tiledlayout in the assigned figure
129 nRows = obj.Size(1);
130 nCols = obj.Size(2);
131 figure(obj.FigObj);
132 args = namedargs2cell(options);
133 obj.TiledObj = tiledlayout(nRows, nCols, args{:});
134 obj.computeTileIndices();
135
136 %< Create axes according to specifications
137 nAxes = length(obj.Specs);
138 obj.Axes = gobjects(1, nAxes);
139 for iAxis = 1:nAxes
140 spec = obj.Specs(iAxis);
141 tileIdx = obj.TileIndices(iAxis);
142 ax = nexttile(obj.TiledObj, tileIdx, [spec.rows, spec.cols]);
143 obj.Axes(iAxis) = ax;
144 end

164 %

166 % database using the configured strategy with default styling.

177
178 core.except.assert(~isempty(obj.Axes), ...
179 'EmptyAxes', 'Axes must be drafted first.');
180 core.except.assert(~isempty(obj.FigObj) && isvalid(obj.FigObj), ...
181 'InvalidFigure', 'Figure must be valid and drafted first.');
182
183 figure(obj.FigObj);
184
185 if ~isempty(options.renderFn)
186 options.renderFn(obj.Axes, database);
187 else
188 core.except.assert(~isempty(obj.Strategy), ...
189 'EmptyStrategy', 'Strategy must be set first.');
190
191 variables = fieldnames(obj.VarToAxis);
192 for j = 1:length(variables)
193 variable = variables{j};
194 offset = obj.VarToAxis.(variable);
195
196 if offset > length(obj.Axes)
197 core.except.verify(false, 'InvalidAxisIdx', ...
198 'Axis index %d exceeds available axes.', offset);
199 continue;
200 end
201
202 group = database.groupBy(variable);
203
204 switch obj.Strategy.NDims
205 case 1
206 ax = obj.Axes(offset);
207 obj.Strategy.render(ax, group, variable);
208 case {2, 3}
209 axes = obj.Axes(offset+(0:group.nDatasets - 1));
210 obj.Strategy.render(axes, group, variable);
211 end
212 end
213 end
214
215 if ~isempty(titleText)
216 switch obj.Strategy.NDims
217 case 1
218 obj.addSgTitle(titleText);
219 case {2, 3}
220 obj.addAnnotationTitle(titleText);
221 end
222 end
223
224 drawnow;
225 end

242 function obj = computeTileIndices(obj)
243 % COMPUTETILEINDICES Compute tile placement for axis
244 % specifications.
245
246 rows = obj.Size(1);
247 cols = obj.Size(2);
248 occupied = false(rows, cols);
249 nSpecs = length(obj.Specs);
250 obj.TileIndices = zeros(1, nSpecs);
251
252 for iSpec = 1:nSpecs
253 spec = obj.Specs(iSpec);
254 placed = false;
255
256 %< Try to place the current specification
257 for idx = 1:(rows * cols)
258 row = floor((idx - 1)/cols) + 1;
259 col = mod(idx-1, cols) + 1;
260
261 %< Check if specification fits at this position
262 if row + spec.rows - 1 <= rows && col + spec.cols - 1 <= cols
263 block = occupied(row:row+spec.rows-1, col:col+spec.cols-1);
264 if all(block(:) == false)
265 occupied(row:row+spec.rows-1, col:col+spec.cols-1) = true;
266 obj.TileIndices(iSpec) = idx;
267 placed = true;
268 break;
269 end
270 end
271 end
272
273 core.except.assert(placed, 'BadAxisSpec', ...
274 'Cannot place axis spec %d in the grid.', iSpec);
275 end
