1%% EX03_CHEBYSHEV_BASIS Generate Chebyshev orthogonal basis functions.
2%
3% This script generates Chebyshev polynomial basis functions of the first
4% kind up to a specified maximum degree. Configure parameters below and
5% run the script.
6
7%% Configuration Parameters
8
9maxDegree = 10; % Maximum polynomial degree
10variable = sym('x'); % Symbolic variable
11isNormalized = false; % Orthonormal basis flag
12isMonic = false; % Monic polynomials flag
13lowerBound = -1; % Lower domain bound
14upperBound = 1; % Upper domain bound
15fileName = '+core/+symbolic/metadata/chebyshev'; % Save file name
16maxDerivOrder = maxDegree; % Maximum derivative order for compilation
17
18%% Generate Chebyshev Basis Functions
19
20basis = core.symbolic.ChebyshevBasis(variable, maxDegree, ...
21 isNormalized=isNormalized, ...
22 isMonic=isMonic, ...
23 lower=lowerBound, ...
24 upper=upperBound);
25
26basis.compile(maxDerivOrder);
27
28%% Display Results
29fprintf('Generated %d basis functions:\n', basis.NCodims);
30
31for i = 1:basis.NCodims
32 fprintf('P_%d(x) = %s\n', i-1, char(basis.Expressions(i)));
33end
34fprintf('\n');
35
36% Display properties
37fprintf('Basis Properties:\n');
38fprintf(' Max Degree: %d\n', basis.MaxDegree);
39fprintf(' IsNormalized: %s\n', string(basis.IsNormalized));
40fprintf(' Monic: %s\n', string(basis.IsMonic));
41fprintf(' Domain: [%.3f, %.3f]\n', basis.Lower, basis.Upper);
42fprintf(' Input Dims: %d\n', basis.NDims);
43fprintf(' Output Dims: %d\n', basis.NCodims);
44fprintf('\n');
45
46%% Save Results (if specified)
47if ~isempty(fileName)
48 basis.save(fileName);
49 fprintf('Basis saved to: %s\n', [fileName, basis.MetaExt]);
50end
