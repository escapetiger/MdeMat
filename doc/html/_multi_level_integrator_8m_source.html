142 function I = integrate(obj, func, options)
143 % INTEGRATE Compute integral using multi-level extrapolation.
144 %
145 % I = integrate(obj) integrates using stored function values
146 % from all levels and applies Richardson extrapolation.
147 %
148 % I = integrate(obj, func) evaluates the specified function
149 % at all integration levels and computes the extrapolated
150 % integral.
151 %
152 % I = integrate(obj, func, levelIndex=levelIndex) computes
153 % the integral using only the specified level.
154
155 arguments
156 obj approx.integrate.MultiLevelIntegrator
157 func {mustBeFunction}
158 options.levelIndex(1, 1) {mustBePositive, mustBeInteger} = []
159 options.useExtrapolation(1, 1) {mustBeNumericOrLogical} = false
160 end
161
162 if ~isempty(options.levelIndex)
163 I = obj.Integrators(options.levelIndex).integrate(func);
164 return;
165 end
166
167 results = zeros(1, obj.NLevels);
168 for i = 1:obj.NLevels
169 results(i) = obj.Integrators(i).integrate(func);
170 end
171
172 if obj.NLevels == 1 || ~options.useExtrapolation
173 I = results(end);
174 else
175 n = length(results);
176 R = zeros(n, n);
177 R(:, 1) = results(:);
178
179 for j = 2:n
180 for i = j:n
181 q = 4^(j-1);
182 R(i, j) = (q * R(i, j-1) - R(i-1, j-1)) / (q - 1);
183 end
184 end
185
186 I = R(n, n);
187 end
188 end
189 end

202 function [X, w] = setPointsOnOrthotopeImpl(obj, level, orthotope, np)
203 % SETPOINTSONORTHOTOPEIMPL Implementation of refined
204 % integration points for Orthotope geometry.
205
206 nd = orthotope.NDims;
207 if level == 1
208 rule = obj.Integrators(level).Rule;
209 [X, w] = rule.generate(orthotope, np);
210 else
211 a = orthotope.Lower;
212 b = orthotope.Upper;
213 m = 2^nd;
214 c = (a + b) / 2;
215 X0 = cell(1, m);
216 w0 = cell(1, m);
217 for j = 1:m
218 t = dec2bin(j - 1, nd) - '0';
219 p = a .* (1 - t) + c .* t;
220 q = c .* (1 - t) + b .* t;
221 bbox = [p(:).'; q(:).'];
222 g = core.geometry.Orthotope(bbox(:));
223 [X0{j}, w0{j}] = obj.setPointsOnOrthotopeImpl(level - 1, g, np);
224 end
225 X = [X0{:}];
226 w = [w0{:}];
227 end
228 end
