64 function obj = clipVolume(obj, v, ht, hx)
65 % CLIPVOLUME Compute volume piece data for semi-Lagrangian
66 % transport.
67 %
68 % obj = clipVolume(obj, v, ht, hx) computes the geometric
69 % intersection between the current element and the upstream
70 % footprint, creating volume pieces with associated function
71 % data for accurate semi-Lagrangian integration.
72 %
73 % The method performs several key operations:
74 % 1. Computes geometric intersection pieces
75 % 2. Creates function data objects for each piece
76 % 3. Sets up quadrature points and weights
77 % 4. Evaluates basis functions at integration points
78 % 5. Computes index shifts for upstream element mapping
79
80 arguments
81 obj approx.element.L2SLOrthotopeElement
82 v(:, 1) {mustBeReal, mustBeNonempty}
83 ht(1, 1) {mustBeReal, mustBeNonempty}
84 hx(:, 1) {mustBeReal, mustBeNonempty}
85 end
86
87 nd = obj.NDims;
88
89 %< Compute clipping geometry
90 bbox = obj.Geometry.BBox;
91 ratio = v(:) .* ht ./ hx(:);
92 poi = ratio - floor(ratio) - 1 / 2;
93
94 %< Initialize volume pieces structure
95 obj.VolumePieces = struct();
96 obj.VolumePieces.xc = zeros(nd, 2^nd);
97 obj.VolumePieces.hx = zeros(nd, 2^nd);
98 np = 0;
99
100 %< Generate intersection pieces
101 indexer = core.linalg.MultiIndexer(shape = repmat(2, 1, nd));
102 m = indexer.generate();
103 for i = 1:2^nd
104 x1 = poi;
105 x2 = bbox((m(i, :) + (0:nd - 1) * 2)');
106 xc = (x1 + x2) / 2;
107 hx = abs(x2-x1);
108
109 %< Skip degenerate pieces
110 if prod(hx) < 1e-8
111 continue;
112 end
113
114 np = np + 1;
115 obj.VolumePieces.xc(:, np) = xc;
116 obj.VolumePieces.hx(:, np) = hx;
117 end
118
119 %< Handle case with no valid pieces
120 if np == 0
121 obj.VolumePieces.xc = zeros(nd, 1);
122 obj.VolumePieces.hx = ones(nd, 1);
123 np = 1;
124 else
125 obj.VolumePieces.xc = obj.VolumePieces.xc(:, 1:np);
126 obj.VolumePieces.hx = obj.VolumePieces.hx(:, 1:np);
127 end
128
129 %< Compute piece boundaries
130 obj.VolumePieces.xmin = obj.VolumePieces.xc - obj.VolumePieces.hx / 2;
131 obj.VolumePieces.xmax = obj.VolumePieces.xc + obj.VolumePieces.hx / 2;
132 obj.VolumePieces.n = np;
133
134 %< Create function data objects for each piece
135 obj.VolumePieces.eulerian = arrayfun(@(i) obj.Volume.copy(), 1:np);
136 obj.VolumePieces.upstream = arrayfun(@(i) obj.Volume.copy(), 1:np);
137
138 %< Set up integration points for each piece
139 nq = obj.Approximator.Basis.NFactorCodims;
140 for i = 1:np
141 a = obj.VolumePieces.xmin(:, i);
142 b = obj.VolumePieces.xmax(:, i);
143 bbox = [a(:).'; b(:).'];
144 G = core.geometry.Orthotope(bbox(:));
145 obj.VolumePieces.eulerian(i).setPoints(G, nq);
146 bbox = [-b(:).'; -a(:).'];
147 G = core.geometry.Orthotope(bbox(:));
148 obj.VolumePieces.upstream(i).setPoints(G, nq);
149 end
150
151 %< Compute basis function values at integration points
152 f = obj.Approximator.Basis;
153 for i = 1:np
154 Ie = obj.VolumePieces.eulerian(i);
155 Iu = obj.VolumePieces.upstream(i);
156 Xe = Ie.Nodes(1:nd, :);
157 Xu = Iu.Nodes(1:nd, :);
158
159 %< Set function values and derivatives
160 Ie.setValues(f, nodes=Xe);
161 Iu.setValues(f, nodes=Xu);
162 Ie.setPartials(f, 1, nodes=Xe);
163 end
164
165 %< Compute index shifts for upstream element mapping
166 obj.VolumePieces.indexShifts = zeros(nd, np);
167 mm = ceil(ratio) .* (ratio > 0) + (floor(ratio) + 1) .* (ratio < 0);
168 mm0 = -mm;
169 mm1 = -mm + 1;
170 for i = 1:np
171 bb = obj.VolumePieces.xc(:, i) <= 0;
172 obj.VolumePieces.indexShifts(:, i) = mm0 .* bb + mm1 .* (1 - bb);
173 end
174 end

176 function obj = clipFlux(obj, v, ht, hx)
177 % SETFLUXPIECEDATA Set up flux piece data for semi-Lagrangian
178 % transport.
179 %
180 % obj = setFluxPieceData(obj, ht, hx) computes the geometric
181 % intersections between element boundaries and upstream
182 % footprints, creating flux pieces with associated function
183 % data for accurate semi-Lagrangian discontinuous Galerkin
184 % integration. Each boundary face is processed to handle
185 % inflow and outflow contributions.
186 %
187 % The method performs several operations for each boundary:
188 % 1. Computes geometric intersection pieces on boundary faces
189 % 2. Creates function data objects for inflow/outflow directions
190 % 3. Sets up quadrature points and weights on boundary pieces
191 % 4. Evaluates basis functions at boundary integration points
192 % 5. Computes index shifts for upstream element mapping
193
194 arguments
195 obj approx.element.L2SLOrthotopeElement
196 v(:, 1) {mustBeReal, mustBeNonempty}
197 ht(1, 1) {mustBeReal, mustBeNonempty}
198 hx(:, 1) {mustBeReal, mustBeNonempty}
199 end
200
201 bbox = obj.Geometry.BBox;
202 bbox = reshape(bbox, 2, []);
203 ratio = v(:) .* ht ./ hx(:);
204 poi = ratio - floor(ratio) - 1 / 2;
205 nd = obj.NDims;
206 nf = 2 * nd; %< Number of boundary faces
207 np = zeros(1, nf);
208
209 %< Initialize flux pieces structure array
210 obj.FluxPieces = arrayfun(@(i) struct(), 1:nf);
211 for i = 1:nf
212 obj.FluxPieces(i).xc = zeros(nd, 2^(nd - 1));
213 obj.FluxPieces(i).hx = zeros(nd, 2^(nd - 1));
214 end
215
216 %< Generate intersection pieces for each boundary face
217 for i = 1:nf
218 d = floor((i - 1)/2) + 1; %< Spatial dimension index
219 s = mod(i-1, 2) + 1; %< Side index (1=min, 2=max)
220
221 for j = 1:2^(nd - 1)
222 %< Create multi-indexer for face geometry
223 shape = ones(1, nd) * 2;
224 shape(d) = 1; %< Collapsed dimension for face
225 indexer = core.linalg.MultiIndexer(shape = shape);
226 m = indexer.linearToMulti(j);
227
228 %< Compute piece geometry
229 x1 = poi;
230 x2 = bbox((m + (0:nd - 1) * 2)');
231 xc = (x1 + x2) / 2;
232 hx = abs(x2-x1);
233
234 %< Skip degenerate pieces (check reduced dimensions)
235 if prod(hx([1:d - 1, d + 1:end])) < 1e-8
236 continue;
237 end
238
239 %< Set face-specific geometry
240 hx(d) = 0; %< Zero thickness in normal direction
241 xc(d) = bbox(s, d); %< Position on boundary face
242
243 np(i) = np(i) + 1;
244 obj.FluxPieces(i).xc(:, np(i)) = xc;
245 obj.FluxPieces(i).hx(:, np(i)) = hx;
246 end
247
248 %< Handle case with no valid pieces
249 if np(i) == 0
250 obj.FluxPieces(i).xc = zeros(nd, 1);
251 obj.FluxPieces(i).xc(d) = bbox(s, d);
252 obj.FluxPieces(i).hx = ones(nd, 1);
253 obj.FluxPieces(i).hx(d) = 0;
254 np(i) = 1;
255 else
256 obj.FluxPieces(i).xc = obj.FluxPieces(i).xc(:, 1:np(i));
257 obj.FluxPieces(i).hx = obj.FluxPieces(i).hx(:, 1:np(i));
258 end
259
260 %< Compute piece boundaries
261 obj.FluxPieces(i).xmin = obj.FluxPieces(i).xc - obj.FluxPieces(i).hx / 2;
262 obj.FluxPieces(i).xmax = obj.FluxPieces(i).xc + obj.FluxPieces(i).hx / 2;
263
264 %< Store number of flux pieces
265 obj.FluxPieces(i).n = np(i);
266 end
267
268 %< Create function data objects for each boundary and piece
269 for i = 1:nf
270 obj.FluxPieces(i).eulerian = arrayfun( ...
271 @(j, k) obj.Flux(i).copy(), ...
272 repmat((1:2).', 1, np(i)), repmat(1:np(i), 2, 1));
273 obj.FluxPieces(i).upstream = arrayfun( ...
274 @(j, k) obj.Flux(i).copy(), ...
275 repmat((1:2).', 1, np(i)), repmat(1:np(i), 2, 1));
276 end
277
278 %< Set up integration points for each boundary and piece
279 nq = obj.Approximator.Basis.NFactorCodims;
280 for i = 1:nf
281 for j = 1:np(i)
282 a = obj.FluxPieces(i).xmin(:, j);
283 b = obj.FluxPieces(i).xmax(:, j);
284 bbox = [a(:).'; b(:).'];
285 G = core.geometry.Orthotope(bbox(:));
286 obj.FluxPieces(i).eulerian(1, j).setPoints(G, nq);
287 obj.FluxPieces(i).eulerian(2, j).setPoints(G, nq);
288 bbox = [-b(:).'; -a(:).'];
289 G = core.geometry.Orthotope(bbox(:));
290 obj.FluxPieces(i).upstream(1, j).setPoints(G, nq);
291 obj.FluxPieces(i).upstream(2, j).setPoints(G, nq);
292 end
293 end
294
295 %< Compute basis function values at integration points
296 f = obj.Approximator.Basis;
297 for i = 1:nf
298 d = floor((i - 1)/2) + 1; %< Spatial dimension index
299 s = mod(i-1, 2) + 1; %< Side index
300
301 for j = 1:np(i)
302 %< Eulerian inflow data
303 Iei = obj.FluxPieces(i).eulerian(1, j);
304 Xe = Iei.Nodes(1:nd, :);
305 Xe(d, :) = bbox(s, d);
306 Iei.setValues(f, Xe);
307
308 %< Eulerian outflow data
309 Ieo = obj.FluxPieces(i).eulerian(2, j);
310 Xe = Ieo.Nodes(1:nd, :);
311 Xe(d, :) = bbox(3-s, d); %< Opposite side
312 Ieo.setValues(f, Xe);
313
314 %< Upstream inflow data
315 Iui = obj.FluxPieces(i).upstream(1, j);
316 Xu = Iui.Nodes(1:nd, :);
317 Xu(d, :) = -poi(d);
318 Iui.setValues(f, Xu);
319
320 %< Upstream outflow data
321 Iuo = obj.FluxPieces(i).upstream(2, j);
322 Xu = Iuo.Nodes(1:nd, :);
323 if -poi(d) == bbox(1, d)
324 Xu(d, :) = bbox(2, d);
325 elseif -poi(d) == bbox(2, d)
326 Xu(d, :) = bbox(1, d);
327 end
328 Iuo.setValues(f, Xu);
329 end
330 end
331
332 %< Compute index shifts for upstream element mapping
333 for i = 1:nf
334 obj.FluxPieces(i).indexShifts = zeros(nd, np(i));
335 mm = ceil(ratio) .* (ratio > 0) + (floor(ratio) + 1) .* (ratio < 0);
336 mm0 = -mm;
337 mm1 = -mm + 1;
338 for j = 1:np(i)
339 bb = obj.FluxPieces(i).xc(:, j) <= 0;
340 obj.FluxPieces(i).indexShifts(:, j) = mm0 .* bb + mm1 .* (1 - bb);
341 end
342 end
343 end
344 end
