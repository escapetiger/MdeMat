1%% ex02_heat_equation.m
2% Solve the 1D heat equation using method of lines with multiple integrator types
3%
4% Problem: ∂u/∂t = α ∂²u/∂x², x ∈ [0,1], t > 0
5% Boundary conditions: u(0,t) = u(1,t) = 0
6% Initial condition: u(x,0) = sin(πx)
7% Exact solution: u(x,t) = exp(-π²αt) sin(πx)
8%
9% Tests explicit, implicit, and IMEX integrators for parabolic PDEs.
10
11clear; close all; clc;
12
13%% Problem Setup
14alpha = 1.0; % Thermal diffusivity
15n = 100; % Number of interior grid points
16dx = 1 / (n + 1); % Spatial step size
17x = linspace(dx, 1 - dx, n)';
18final_time = 0.2;
19
20% Initial condition
21u0 = sin(pi * x);
22
23% Spatial discretization using finite differences
24e = ones(n, 1);
25L = alpha / dx^2 * spdiags([e, -2*e, e], -1:1, n, n);
26
27% For IMEX methods: split the diffusion operator
28% (Here we use implicit treatment for the entire operator)
29F = []; % No nonlinear term
30S = []; % No source term
31
32% Exact solution for comparison
33exact_solution = @(t) exp(-pi^2 * alpha * t) * sin(pi * x);
34
35%% Setup integrators
36% Different time step sizes based on stability requirements
37dt_implicit = 0.001; % Larger step for implicit methods
38dt_explicit = 0.0001; % Much smaller for explicit (CFL condition)
39dt_imex = 0.001; % Similar to implicit for IMEX
40
41% Stability analysis for explicit methods
42% CFL condition: dt ≤ dx²/(2α) for stability
43dt_cfl = dx^2 / (2 * alpha);
44fprintf('CFL stability limit for explicit methods: dt ≤ %.6f\n', dt_cfl);
45if dt_explicit > dt_cfl
46 fprintf('Warning: Explicit time step (%.6f) exceeds CFL limit!\n', dt_explicit);
47 dt_explicit = 0.8 * dt_cfl; % Use 80% of CFL limit for safety
48 fprintf('Adjusted explicit time step to: %.6f\n', dt_explicit);
49end
50
51% Create integrators
52implicit_integrators = {
53 approx.odeint.BeIntegrator(final_time)
54 approx.odeint.Bdf2Integrator(final_time)
55 approx.odeint.Bdf3Integrator(final_time)
56 approx.odeint.Sdirk2Integrator(final_time)
57};
58
59explicit_integrators = {
60 approx.odeint.FeIntegrator(final_time)
61 approx.odeint.HeunIntegrator(final_time)
62 approx.odeint.Exrk4Integrator(final_time)
63 approx.odeint.Ssprk3Integrator(final_time)
64};
65
66imex_integrators = {
67 approx.odeint.Ars111Integrator(final_time)
68 approx.odeint.Ars222Integrator(final_time)
69 approx.odeint.Ars443Integrator(final_time)
70};
71
72% Combine all integrators
73all_integrators = [implicit_integrators; explicit_integrators; imex_integrators];
74n_methods = length(all_integrators);
75
76% Extract integrator names and types
77integrator_names = cell(n_methods, 1);
78integrator_types = cell(n_methods, 1);
79
80for i = 1:n_methods
81 class_name = class(all_integrators{i});
82 name_parts = strsplit(class_name, '.');
83 base_name = name_parts{end};
84 integrator_names{i} = strrep(base_name, 'Integrator', '');
85
86 % Determine type
87 if i <= length(implicit_integrators)
88 integrator_types{i} = 'Implicit';
89 elseif i <= length(implicit_integrators) + length(explicit_integrators)
90 integrator_types{i} = 'Explicit';
91 else
92 integrator_types{i} = 'IMEX';
93 end
94end
95
96%% Solve with all integrators
97fprintf('\nSolving 1D heat equation with multiple integrator types...\n');
98fprintf('Grid points: %d, dx = %.4f\n', n, dx);
99fprintf('Implicit methods dt: %.5f\n', dt_implicit);
100fprintf('Explicit methods dt: %.5f\n', dt_explicit);
101fprintf('IMEX methods dt: %.5f\n\n', dt_imex);
102
103% Storage for solutions at final time
104U_solutions = zeros(n_methods, n);
105computational_costs = zeros(n_methods, 1);
106
107for i = 1:n_methods
108 fprintf('Solving with %s (%s)...\n', integrator_names{i}, integrator_types{i});
109
110 integrator = all_integrators{i};
111
112 % Choose appropriate time step based on method type
113 if strcmp(integrator_types{i}, 'Explicit')
114 dt = dt_explicit;
115 elseif strcmp(integrator_types{i}, 'IMEX')
116 dt = dt_imex;
117 else
118 dt = dt_implicit;
119 end
120
121 n_steps = ceil(final_time / dt);
122
123 % Initialize solution history
124 integrator.update(u0);
125
126 % Time stepping
127 U_current = u0;
128 tic;
129 for k = 1:n_steps
130 integrator.setTimeStep(dt=dt);
131
132 % Set coefficients for BDF methods
133 if isa(integrator, 'approx.odeint.BdfIntegrator')
134 integrator.setCoefficients();
135 end
136
137 % Call appropriate step method based on integrator type
138 if strcmp(integrator_types{i}, 'Implicit')
139 U_new = integrator.step(L, S, []);
140 elseif strcmp(integrator_types{i}, 'Explicit')
141 % For explicit methods, treat diffusion as nonlinear operator
142 F_eval = L * integrator.U0{1};
143 U_new = integrator.step([], F_eval, [], []);
144 else % IMEX
145 % For IMEX methods, treat diffusion implicitly
146 U_new = integrator.step(L, F, S, []);
147 end
148
149 integrator.update(U_new);
150 integrator.advance();
151 U_current = U_new;
152
153 % Progress indicator for explicit methods (many steps)
154 if strcmp(integrator_types{i}, 'Explicit') && mod(k, round(n_steps/10)) == 0
155 fprintf(' Progress: %d%%\n', round(100*k/n_steps));
156 end
157 end
158 computational_time = toc;
159
160 % Store final solution and computational cost
161 U_solutions(i, :) = U_current;
162 computational_costs(i) = computational_time;
163
164 fprintf(' Completed in %.3f seconds (%d steps)\n', computational_time, n_steps);
165end
166
167%% Compute exact solution at final time
168U_exact = exact_solution(final_time);
169

174colors_implicit = lines(length(implicit_integrators));
