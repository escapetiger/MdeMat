1classdef FluxAssembly < approx.assembly.Assembly
2 % FLUXASSEMBLY Assembly for flux terms in DG methods.
3 %
4 % FluxAssembly handles the assembly of flux terms that arise in
5 % discontinuous Galerkin methods. These flux terms provide
6 % inter-element coupling, where the solution is allowed to be
7 % discontinuous across element boundaries.
8 %
9 % The class supports both grid and non-grid meshes, following the
10 % unified assembly procedure:
11 % 1. iterate over interior elements and sum up interface integrals
12 % 2. iterate over boundary elements and sum up interface integrals
13 %
14 % Flux terms are essential for stability and accuracy in DG methods,
15 % as they control the flow of information across element interfaces
16 % and enforce appropriate boundary conditions. The choice of flux
17 % type affects the stability properties and numerical diffusion
18 % characteristics of the resulting scheme.
19 %
20 % See also:
21 % approx.assembly.Assembly, approx.assembly.VolumeAssembly
22

102 function G = assembleOutwardInteriorPartial(obj, dim)
103 IE = obj.space.mesh.findInteriorElements(); % ne x 1
104 detJ = obj.space.mesh.computeElementJacobianDeterminants(IE); % ne x 1
105 for IFK = 1:obj.space.element.nFluxes
106 n = obj.space.mesh.computeOutwardNormals(IE, IFE); % d x ne
107 detFJ = obj.space.mesh.computeFaceJacobianDeterminants(IE, IFE); % ne x 1
108 h = reshape(detFJ ./ detJ, 1, []);
109 [JE, JFE] = obj.space.mesh.findNeighborElements(IE, IFE, obj.bcType); % ne x 1, ne x 1
110 V = obj.space.element.flux(IFE).values; % nl x nq
111 W = diag(obj.space.element.flux(IFE).weights); % nq x nq
112 U = arrayfun(@(j) obj.space.element.flux(j).values, JFE, 'Un', 0);
113 U = cat(3, U{:}); % nl x nq x ne
114 L = pagemtimes(V * W, U.'); % nl x nl x ne
115 L = L .* reshape(n(dim, :) .* h, 1, 1, []);
116 G = obj.assembleMatrix(IE, [], JE, [], L);
117 end
118 end
