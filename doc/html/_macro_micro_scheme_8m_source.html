116 function state = step(obj, state)
117 % STEP Perform one upwind DG time step.
118
119 m = state.nMacroModes;
120 n = state.nMicroModes;
121
122 %< Update history
123 if m > 0 && n == 0
124 W = state.dofs.U(:);
125 elseif m == 0 && n > 0
126 W = state.dofs.G(:);
127 else
128 W = [state.dofs.U(:); state.dofs.G(:)];
129 end
130 obj.tDisc.update(W);
131
132 %< Integration step
133 if isa(obj.tDisc, 'approx.odeint.BeIntegrator') ...
134 || isa(obj.tDisc, 'approx.odeint.BdfIntegrator') ...
135 || isa(obj.tDisc, 'approx.odeint.DirkIntegrator')
136 W = obj.tDisc.step(obj.L, obj.S, obj.M);
137 else
138 W = obj.tDisc.step(obj.L, [], obj.S, obj.M);
139 end
140 W = reshape(W, [], m+n);
141 state.dofs.U = W(:, 1:m);
142 state.dofs.G = W(:, m+1:m+n);
143 end

216 function obj = addFreeStreaming(obj, state)
217 % Assemble streaming part of linear operator
218
219 m = state.nMacroModes;
220 n = state.nMicroModes;
221 p = state.xDisc.space.nGlobalDofs;
222
223 V = state.microNodes;
224
225 % Macro-macro streaming
226 for i = 1:m
227 for j = 1:m
228 a = squeeze(obj.pattern.streaming(i, j, :));
229 if all(a == 0)
230 obj.L{i, j} = sparse(p, p);
231 continue;
232 end
233 if i < j
234 D = obj.auxiliary.linear(a);
235 if obj.auxiliary.isImplicitDirichletBoundaryJump
236 Dj = obj.auxiliary.linearJumpExtraction(a);
237 obj.L{i, i} = obj.L{i, i} - Dj * obj.pattern.scaling(i, 1+j);
238 end
239 else
240 D = obj.primal.linear(a);
241 end
242 obj.L{i, j} = -D * obj.pattern.scaling(i, 1+j);
243 end
244 end
245
246 % Macro-micro streaming
247 for i = 1:m
248 for j = 1:n
249 a = squeeze(obj.pattern.streaming(i, m+j, :));
250 if all(a == 0)
251 obj.L{i, m+j} = sparse(p, p);
252 continue;
253 end
254 D = obj.auxiliary.linear(a);
255 obj.L{i, m+j} = -D * obj.pattern.scaling(i, 1+m+j);
256 if obj.auxiliary.isImplicitDirichletBoundaryJump
257 DJ = obj.auxiliary.linearJumpExtraction(a);
258 obj.L{i, i} = obj.L{i, i} - DJ * obj.pattern.scaling(i, 1+m+j);

260 end
261 end

274 end

336 function S = addSource(obj, state, t, S)
337 m = state.nMacroModes;
338 n = state.nMicroModes;
339 nl = state.xDisc.space.nLocalDofs;
340 state.coefs.QU = state.macroProject(obj.config.source, [], [], t);
341 state.coefs.QG = state.microProject(obj.config.source, [], [], t);
342 C = reshape(repmat(state.coefs.CA(:).', nl, 1), [], 1);
343 QU = C .* state.coefs.QU .* obj.pattern.scaling(1:m, m+n+3).';

347 end

353 function S = addBc(obj, state, t, S)
354 m = state.nMacroModes;
355 n = state.nMicroModes;
356 ng = state.xDisc.space.nGlobalDofs;
357 nd = state.xDisc.space.nDims;
358 fU = @(i, X, V) state.macroTraceEvaluate(i, obj.config.bc, X, V, t);
359 fG = @(i, X, V) state.microTraceEvaluate(i, obj.config.bc, X, V, t);
360
361 volume = obj.auxiliary.volume;
362 bdrFlux = obj.auxiliary.flux.boundary;
363 BU = cell(1, 2*nd);
364 BG = cell(1, 2*nd);
365 for d = 1:nd
366 C = volume.scaleConstant(d, 1);
367 if m > 0
368 T = bdrFlux.assembleTrace(2*d-1, C, fU, []);
369 BU{2*d-1} = core.linalg.sparseFromTriplets(T, ng, m);
370 end
371 if n > 0
372 T = bdrFlux.assembleTrace(2*d-1, C, fG, []);
373 BG{2*d-1} = core.linalg.sparseFromTriplets(T, ng, n);
374 end
375 if m > 0
376 T = bdrFlux.assembleTrace(2*d, C, fU, []);
377 BU{2*d} = core.linalg.sparseFromTriplets(T, ng, m);
378 end
379 if n > 0
380 T = bdrFlux.assembleTrace(2*d, C, fG, []);
381 BG{2*d} = core.linalg.sparseFromTriplets(T, ng, n);
382 end
383 end
384
385 if obj.auxiliary.isImplicitDirichletBoundaryJump
386 BUJ = cell(1, 2*nd);
387 for d = 1:nd
388 C = obj.auxiliary.volume.scaleConstant(d, 1);
389 T = obj.auxiliary.flux.boundary.assembleTrace(2*d-1, -C, fU, []);
390 BUJ{2*d-1} = core.linalg.sparseFromTriplets(T, ng, m);
391 state.clean();
392 T = obj.auxiliary.flux.boundary.assembleTrace(2*d, -C, fU, []);
393 BUJ{2*d} = core.linalg.sparseFromTriplets(T, ng, m);
394 state.clean();
395 end
396 elseif obj.auxiliary.isExplicitDirichletBoundaryJump
397 gU = @(X) state.xDisc.space.evaluate([], X, state.dofs.U);
398 BUJ = cell(1, 2*nd);
399 for d = 1:nd
400 C = obj.auxiliary.volume.scaleConstant(d, 1);
401 T = obj.auxiliary.flux.boundary.assembleJump(2*d-1, C, fU, gU, []);
402 BUJ{2*d-1} = core.linalg.sparseFromTriplets(T, ng, m);
403 state.clean();
404 T = obj.auxiliary.flux.boundary.assembleJump(2*d, -C, fU, gU, []);
405 BUJ{2*d} = core.linalg.sparseFromTriplets(T, ng, m);
406 state.clean();
407 end
408 end

410 % Macro-macro streaming BCs
411 for i = 1:m
412 for j = 1:m
413 a = squeeze(obj.pattern.streaming(i, j, :));
414 if all(a == 0)
415 continue;
416 end
417 if i < j % add boundary jump for auxiliary variables
418 if obj.auxiliary.hasDirichletBoundaryJump
419 DJ = sparse(ng, 1);
420 for d = 1:nd
421 if a(d) == 0, continue; end
422 switch obj.auxiliary.flux.fluxType
423 case 'left'
424 J = BUJ{2*d-1}(:, i);
425 case 'right'
426 J = BUJ{2*d}(:, i);
427 case 'central'
428 J = (BUJ{2*d-1}(:, i) + BUJ{2*d}(:, i)) / 2;
429 end
430 DJ = DJ + a(d) * J;
431 end
432 S{i} = S{i} - DJ * obj.pattern.scaling(i, 1+j);
433 end
434 else % add boundary condition for primal variables
435 D = 0;
436 for d = 1:nd
437 if a(d) == 0, continue; end
438 D = D + a(d) * (BU{2*d}(:, j) - BU{2*d-1}(:, j));
439 end
440 S{i} = S{i} - D * obj.pattern.scaling(i, 1+j);
441 end
442 end
443 end

446 for i = 1:m
447 for j = 1:n
448 a = squeeze(obj.pattern.streaming(i, m+j, :));
449 if all(a == 0)
450 continue;
451 end
452 % add boundary jump for auxiliary variables
453 if obj.auxiliary.hasDirichletBoundaryJump
454 DJ = 0;
455 for d = 1:nd
456 if a(d) == 0, continue; end
457 switch obj.auxiliary.flux.fluxType
458 case 'left'
459 J = BUJ{2*d-1}(:, i);
460 case 'right'
461 J = BUJ{2*d}(:, i);
462 case 'central'
463 J = (BUJ{2*d-1}(:, i) + BUJ{2*d}(:, i)) / 2;
464 end
465 DJ = DJ + a(d) * J;
466 end
467 S{i} = S{i} - DJ * obj.pattern.scaling(i, 1+m+j);
468 end
469 end
470 end
471
472 % Micro-macro streaming BCs
473 for i = 1:n
474 for j = 1:m
475 a = squeeze(obj.pattern.streaming(m + i, j, :));
476 if all(a == 0)
477 continue;
478 end
479 D = 0;
480 for d = 1:nd
481 D = D + a(d) * (BU{2*d}(:, j) - BU{2*d-1}(:, j));
482 end
483 S{m+i} = S{m+i} - D * obj.pattern.scaling(m+i, 1+j);
484 end
485 end

487 % Micro-micro streaming BCs
488 V = state.microNodes;
489 for i = 1:n

491 for j = 1:n

493 if all(a == 0)
494 continue;
495 end
496 D = 0;
497 for d = 1:nd
498 if V(d, j) > 0
499 D = D - a(d) * BG{2*d-1}(:, j);
500 else
501 D = D + a(d) * BG{2*d}(:, j);
502 end
503 end
504 R = R - D * obj.pattern.scaling(m+i, 1+m+j);
505 end
506 S{m+i} = S{m+i} + R;
507 end
