1function [p,t]=distmeshnd(fdist,fh,h,box,fix,varargin)
2%DISTMESHND N-D Mesh Generator using Distance Functions.
3% [P,T]=DISTMESHND(FDIST,FH,H,BOX,FIX,FDISTPARAMS)
4%
5% P: Node positions (NxNDIM)
6% T: Triangle indices (NTx(NDIM+1))
7% FDIST: Distance function
8% FH: Edge length function
9% H: Smallest edge length
10% BOX: Bounding box [xmin,xmax;ymin,ymax; ...] (NDIMx2)
11% FIX: Fixed node positions (NFIXxNDIM)
12% FDISTPARAMS: Additional parameters passed to FDIST
13%
14% Example: Unit ball
15% dim=3;
16% d=inline('sqrt(sum(p.^2,2))-1','p');
17% [p,t]=distmeshnd(d,@huniform,0.2,[-ones(1,dim);ones(1,dim)],[]);
18%
19% See also: DISTMESH2D, DELAUNAYN, TRIMESH, MESHDEMOND.
20
21% Copyright (C) 2004-2012 Per-Olof Persson. See COPYRIGHT.TXT for details.
22
23dim=size(box,2);
24ptol=.001; ttol=.1; L0mult=1+.4/2^(dim-1); deltat=.1; geps=1e-1*h; deps=sqrt(eps)*h;
25
26% 1. Create initial distribution in bounding box
27if dim==1
28 p=(box(1):h:box(2))';
29else
30 cbox=cell(1,dim);
31 for ii=1:dim
32 cbox{ii}=box(1,ii):h:box(2,ii);
33 end
34 pp=cell(1,dim);
35 [pp{:}]=ndgrid(cbox{:});
36 p=zeros(prod(size(pp{1})),dim);
37 for ii=1:dim
38 p(:,ii)=pp{ii}(:);
39 end
40end
41
42% 2. Remove points outside the region, apply the rejection method
43p=p(feval(fdist,p,varargin{:})<geps,:);
44r0=feval(fh,p);
45p=[fix; p(rand(size(p,1),1)<min(r0)^dim./r0.^dim,:)];
46N=size(p,1);
47
48count=0;
49p0=inf;
50while 1
51 % 3. Retriangulation by Delaunay
52 if max(sqrt(sum((p-p0).^2,2)))>ttol*h
53 p0=p;
54 t=delaunayn(p);
55 pmid=zeros(size(t,1),dim);
56 for ii=1:dim+1
57 pmid=pmid+p(t(:,ii),:)/(dim+1);
58 end
59 t=t(feval(fdist,pmid,varargin{:})<-geps,:);
60 % 4. Describe each edge by a unique pair of nodes
61 pair=zeros(0,2);
62 localpairs=nchoosek(1:dim+1,2);
63 for ii=1:size(localpairs,1)
64 pair=[pair;t(:,localpairs(ii,:))];
65 end
66 pair=unique(sort(pair,2),'rows');
67 % 5. Graphical output of the current mesh
68 if dim==2
69 trimesh(t,p(:,1),p(:,2),zeros(N,1))
70 view(2),axis equal,axis off,drawnow
71 elseif dim==3
72 if mod(count,5)==0
73 simpplot(p,t,'p(:,2)>0');
74 title(['Retriangulation #',int2str(count)])
75 drawnow
76 end
77 else
78 disp(sprintf('Retriangulation #%d',count))
79 end
80 count=count+1;
81 end
82
83 % 6. Move mesh points based on edge lengths L and forces F
84 bars=p(pair(:,1),:)-p(pair(:,2),:);
85 L=sqrt(sum(bars.^2,2));
86 L0=feval(fh,(p(pair(:,1),:)+p(pair(:,2),:))/2);
87 L0=L0*L0mult*(sum(L.^dim)/sum(L0.^dim))^(1/dim);
88 F=max(L0-L,0);
89 Fbar=[bars,-bars].*repmat(F./L,1,2*dim);
90 dp=full(sparse(pair(:,[ones(1,dim),2*ones(1,dim)]), ...
91 ones(size(pair,1),1)*[1:dim,1:dim], ...
92 Fbar,N,dim));
93 dp(1:size(fix,1),:)=0;
94 p=p+deltat*dp;
95
96 % 7. Bring outside points back to the boundary
97 d=feval(fdist,p,varargin{:}); ix=d>0;
98 gradd=zeros(sum(ix),dim);
99 for ii=1:dim
100 a=zeros(1,dim);
101 a(ii)=deps;
102 d1x=feval(fdist,p(ix,:)+ones(sum(ix),1)*a,varargin{:});
103 gradd(:,ii)=(d1x-d(ix))/deps;
104 end
105 p(ix,:)=p(ix,:)-d(ix)*ones(1,dim).*gradd;
106
107 % 8. Termination criterion
108 maxdp=max(deltat*sqrt(sum(dp(d<-geps,:).^2,2)));
109 if maxdp<ptol*h, break; end
110end
