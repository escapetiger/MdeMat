196function Q = sourceImpl(x, v, t, E, omega, lambda, epsilon, alpha, beta, gamma)
197z = omega(:).' * x;
198y = omega(:).' * v;
199sigma_s = scatteringImpl(x);
200sigma_a = absorptionImpl(x);
201alpha = epsilon^alpha;
202beta = epsilon^beta;
203gamma = epsilon^gamma;
204
205%< \partial_t f / E
206T1 = -lambda * exp(-lambda*t) * (sin(z) - epsilon * y .* cos(z) ./ sigma_s);
207%< v\cdot\nabla_x f / E
208T2 = exp(-lambda*t) * y .* (cos(z) + epsilon * y .* sin(z) ./ sigma_s);
209%< sigma_s * g / E
210T3 = -exp(-lambda*t) * y .* cos(z);
211%< sigma_a * f / E
212T4 = exp(-lambda*t) * sigma_a .* (sin(z) - epsilon * y .* cos(z) ./ sigma_s);
213
214%< Q = q*E
215Q = (alpha * T1 + T2 + beta * epsilon * T3 + gamma * T4) ./ (gamma * sigma_a) * E;
216end
