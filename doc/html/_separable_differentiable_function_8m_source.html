164 function dY = derivativeDense(obj, X, r)
165 % DERIVATIVEDENSE Dense derivative evaluation using full tensor
166 % product.
167 %
168 % dY = derivativeDense(obj, X, r) computes derivatives using
169 % the full tensor product structure. Each factor function
170 % is evaluated or differentiated as needed, and results are
171 % combined using Kronecker or Khatri-Rao products.
172 %
173 % Inputs:
174 % obj - The SeparableDifferentiableFunction object
175 % X - Evaluation points (nDims x nPoints or cell array)
176 % r - Derivative order vector (1 x nDims)
177 %
178 % Outputs:
179 % dY - Derivative values (nCodims x nPoints)
180
181 s = obj.splitOrderVector(r);
182 F = obj.factors;
183 k = length(F);
184 dZ = cell(1, k);
185
186 if iscell(X)
187 for i = 1:k
188 if iscell(F)
189 Fi = F{i};
190 else
191 Fi = F(i);
192 end
193 mi = Fi.nCodims;
194 if all(s{i} == 0)
195 dZ{i} = Fi.evaluate(X{i});
196 else
197 dZ{i} = Fi.derivative(X{i}, s{i});
198 end
199 dZ{i} = reshape(dZ{i}, mi, []);
200 end
201 dY = core.linalg.kronecker(dZ);
202 else
203 j = 1;
204 for i = 1:k
205 if iscell(F)
206 Fi = F{i};
207 else
208 Fi = F(i);
209 end
210 ni = Fi.nDims;
211 mi = Fi.nCodims;
212 J = j:(j + ni - 1);
213 if all(s{i} == 0)
214 dZ{i} = Fi.evaluate(X(J, :));
215 else
216 dZ{i} = Fi.derivative(X(J, :), s{i});
217 end
