1% Gauss-Hermite Quadrature (Probabilist's version)
2% ------------------------------------------------
3% This script computes the n-point Gauss-Hermite quadrature rule
4% for the weight exp(-x^2/2), i.e. using probabilist's Hermite polynomials He_n(x).
5% It also verifies exactness up to degree 2n-1.
6
7clear; clc;
8
9% Number of quadrature points
10n = 4; % change this value as needed
11
12% Step 1: Build Jacobi matrix for probabilist's Hermite polynomials
13i = 1:n-1;
14beta = sqrt(i); % sub-diagonal entries
15J = diag(beta,1) + diag(beta,-1); % symmetric tridiagonal
16
17% Step 2: Eigenvalue decomposition
18[V,D] = eig(J);
19x = diag(D); % nodes (roots of He_n)
20[~,idx] = sort(x);
21x = x(idx); % sort ascending
22V = V(:,idx);
23
24% Step 3: Weights
25w = sqrt(2*pi) * V(1,:).^2; % weights for probabilist’s Hermite
26
27% Display nodes and weights
28disp('Nodes (x_i):'); disp(x');
29disp('Weights (w_i):'); disp(w);
30
31% Step 4: Verify exactness
32maxdeg = 2*n - 1;
33fprintf('\nVerification on monomials up to degree %d:\n', maxdeg);
34
35for k = 0:maxdeg
36 % Exact integral: ∫ x^k e^(-x^2/2) dx over (-inf,inf)
37 if mod(k,2)==0
38 exact = sqrt(2*pi) * prod(1:2:(k-1)); % double factorial formula
39 if k==0, exact = sqrt(2*pi); end
40 else
41 exact = 0;
42 end
43
44 % Quadrature approximation
45 approx = sum(w .* (x'.^k));
46
47 fprintf('Degree %d: exact = %.8f, approx = %.8f, error = %.2e\n', ...
48 k, exact, approx, abs(exact-approx));
49end
