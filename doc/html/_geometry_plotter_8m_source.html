3 % visualization.
4 %
5 % GeometryPlotter extends TiledPlotter to provide specialized
6 % functionality for visualizing data on reference geometry elements.
7 % This class automatically configures rendering nodes and coordinates
8 % based on geometry bounds, simplifying visualization workflows that
9 % don't require full finite element space setup.
10 %
11 % The plotter is particularly useful for visualizing data on
12 % canonical domains (unit squares, cubes, etc.) and for prototyping
13 % visualization strategies before applying them to complex finite
14 % element meshes.
15 %
16 % Examples:
17 % % Create plotter for unit square with 2×3 layout
18 % geometry = core.geometry.Orthotope.unit(2);
19 % plotter = GeometryPlotter(geometry, 2, 3);
20 %
21 % % Configure rendering and strategy

62 function obj = setNodes(obj, n)
63 % SETNODES Configure evaluation nodes for rendering.
64 %
65 % obj = setNodes(obj, n) sets up a uniform grid of evaluation
66 % nodes within the reference geometry bounds. These nodes are
67 % used for function evaluation and data visualization.
68 %
69 % Inputs:
70 % obj - The GeometryPlotter object
71 % n - Number of nodes per dimension (scalar or vector)
72 %
73 % Outputs:
74 % obj - The GeometryPlotter object
75
76 d = obj.geometry.nDims;
77 a = obj.geometry.lower;
78 b = obj.geometry.upper;
79
80 %< Handle scalar input for uniform grids
81 if isscalar(n)
82 n = repmat(n, 1, d);
83 end
84
85 %< Create coordinate vectors for each dimension
86 h = (b - a) ./ (n + 1);
87 obj.coords = arrayfun(@(i) (a(i)+h(i)):h(i):(b(i)-h(i)), 1:d, 'Un', false);
88
89 %< Generate tensor product grid
90 [obj.nodes{1:d}] = ndgrid(obj.coords{:});
91 obj.nodes = reshape(cat(d+1, obj.nodes{:}), [], d).';
92 end
93 end
