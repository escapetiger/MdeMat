93 function [state, table] = converge(obj, state)
94 % CONVERGE Execute convergence study with mesh refinement.
95 %
96 % [state, table] = converge(obj, state) executes multiple
97 % simulation runs with progressively refined meshes to study
98 % convergence behavior. Uses the run method for each level
99 % and handles error analysis with proper timing.
100
101 arguments
102 obj physics.scheme.MolScheme
103 state physics.state.SpatialState
104 end
105
106 timer = obj.Config.timer;
107 analyzer = obj.Config.analyzer;
108
109 analyzer.initialize();
110
111 if ~analyzer.IsEnabled
112 return;
113 end
114
115 nLevels = analyzer.NLevels;
116 coarse = [];
117
118 if ~analyzer.HasExact
119 nLevels = nLevels + 1;
120 end
121
122 %< Refinement loop
123 for iLevel = 1:nLevels
124 fprintf('[M] Refinement Level %d/%d \n', iLevel, nLevels);
125
126 %< Run simulation
127 state = obj.run(state);
128
129 %< Start timing
130 fprintf('[M] Error Evaluation.\n');
131 timer.start(record='Analysis');
132
133 %< Add new mesh level
134 analyzer.addLevel(state.XDisc.Mesh);
135
136 %< Error evaluation
137 if analyzer.HasExact
138 analyzer.addAbsolute(state, args={'t', obj.Config.tFinal});
139 else
140 if iLevel > 1
141 analyzer.addRelative(state, coarse);
142 end
143 coarse = state;
144 end
145
146 %< Stop timing
147 timer.stop(record='Analysis');
148
149 %< Report timing information
150 if obj.Config.verbose > 0
151 timer.report();
152 end
153
154 %< Refine state for next iteration
155 if iLevel < nLevels
156 state = state.refine(1);
157 end
158
159 fprintf('=====================================================\n');
160 end
161
162 %< Analyze convergence results
163 table = analyzer.analyze(obj.Config.verbose);
164 end

317 space approx.space.FiniteElementSpace
318 dofs struct
319 end
320
321 t0 = obj.TDisc.Timeline.Now;
322 t1 = obj.TDisc.Timeline.Next;
323
324 if t1 <= t0
325 V0 = [];
326 n0 = 0;
327 return;
328 end
329
330 n0 = min(obj.TDisc.Timeline.NSteps, obj.TDisc.Timeline.Count);
331 V0 = cell(1, n0);
332 for i = 1:n0
333 W0 = reshape(obj.TDisc.U0{i}, space.NGlobalDofs, []);
334 dofNames = fieldnames(dofs);

338 nComponentsPerDof = size(dofs.(dofName), 2);
339 V0{i}.(dofName) = W0(:, offset+(1:nComponentsPerDof));
340 offset = offset + nComponentsPerDof;
341 end
