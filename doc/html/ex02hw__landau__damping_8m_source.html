56function [config, state] = simulate(epsilon, tau0, tFinal, nh)
57configPath = fullfile(fileparts(mfilename('fullpath')), 'config.txt');
58config = core.io.OptionParser().parse(configPath);
59config.nDims = 1; % Dimension
60config.k = 0.5; % Wave length
61config.L = 2*pi / config.k; % Length of domain
62config.xBBox = [0, config.L]; % Spatial bounding box
63config.vBBox = [-6, 6]; % Velocity bounding box
64config.tFinal = tFinal; % Final time
65config.T0 = 1; % Temperature parameter
66config.delta = 0.01; % Perturbation
67config.phiInf = phiInf(config.k); % Potential steady state
68config.EInf = EInf(config.k); % Electrical field steady state
69config.cInf = config.L / integral(@(x) exp(-config.phiInf(x)), 0, config.L); % Normalization parameter
70config.rhoInf = rhoInf(config.k, config.cInf); % Density steady state
71config.nx = 64; % Grid resolution
72config.nh = nh; % Number of Hermite modes - 1
73config.ic.D = DInit(config.k, config.delta, config.cInf, config.nh); % Initial hermite coefficients
74config.ic.omega = omegaInit(config.k, config.delta, config.cInf); % Initial potential
75config.bc = []; % Boundary condition
76config.exact = []; % Exact solution
77config.tau0 = tau0; % Collision parameter
78config.epsilon = epsilon; % Scaling parameter
79config.verbose = 1; % Verbose flag
80
81config.TID = upper(config.tOdeIntName);
82config.XID = sprintf('%s%d', upper(config.xDiscName), config.xBasisOrder);
83config.VID = sprintf('H%d', config.nh);
84config.EID = sprintf('AP%d', config.nDims); % Experiment ID
85config.SID = strjoin({config.XID, config.VID}, '-'); % Scheme ID
86
87config.visualizer = physics.visual.Visualizer();
88config.visualizer.setDensity(1);
89config.visualizer.setTimeline(config.tFinal, 1);
90config.visualizer.setComponents(struct('D', [], 'omega', []));
91config.visualizer.setTitleHead(sprintf('%s with %s', config.EID, config.SID));
92config.visualizer.addDataset(config.XID, 'numeric');
93config.visualizer.addPlotter('profile', '1d');
94
95scheme = physics.vlasov.DghsScheme(config);
96
97vElement = approx.element.C0OrthotopeElement(config.vBBox, ...
98 config.nh + 1, config.vBasisType, lambda=sqrt(config.T0));
99vDisc = approx.space.SpectralSpace(vElement);
100
101xElement = approx.element.L2OrthotopeElement(config.nDims, ...
102 config.xBasisOrder, config.xBasisType, config.xBasisPattern);
103xMesh = approx.mesh.UniformGrid(config.xBBox, config.nx);
104xDisc = approx.space.FiniteElementSpace(xMesh, xElement);
105state = physics.vlasov.HermiteState(xDisc, vDisc, config.T0, config.rhoInf);
106state = scheme.run(state);
107end

115function D = DInitImpl(k, delta, cInf, nh, x)
116rhoInf = rhoInfImpl(k, cInf, x);
117D0 = (rhoInf + delta * cos(k*x)) ./ rhoInf;
118D = [D0, zeros(1, size(x, 2)*nh)];

125function omega = omegaInitImpl(k, delta, cInf, x)
126rhoInf = rhoInfImpl(k, cInf, x);
