1classdef GridSpaceRichardsonMetric < profilers.metrics.Metric
2 % GRIDSPACERICHARDSONMETRIC Richardson extrapolation error metric for
3 % grid spaces.
4 %
5 % GridSpaceRichardsonMetric computes relative errors between
6 % solutions on different grid resolutions using Richardson
7 % extrapolation. This approach is invaluable when exact analytical
8 % solutions are unavailable, providing a method to estimate
9 % discretization errors by comparing fine and coarse grid solutions.
10 %
11 % The metric implements the Richardson extrapolation principle:
12 % assuming that discretization errors behave as E ≈ Ch^p where h is
13 % the grid spacing and p is the order of convergence, the difference
14 % between fine and coarse grid solutions provides an estimate of the
15 % error.
16 %
17 % Examples:
18 % % Create Richardson metric for L2 error estimation
19 % metric = GridSpaceRichardsonMetric({'L2'}, fineSpace, coarseSpace);
20 % metric.setPoints([6, 6]); % 6×6 integration points per element
21 %
22 % % Compute Richardson extrapolation error
23 % fineNodes = metric.nodes{1};
24 % coarseNodes = metric.nodes{2};
25 % fineValues = fineSpace.evaluate(fineNodes, fineCoeffs);
26 % coarseValues = coarseSpace.evaluate(coarseNodes, coarseCoeffs);
27 % error = metric.evaluate(fineValues, coarseValues);
28 %
29 % % Use nested DOF ordering for convenience
30 % reorderedCoeffs = fineCoeffs(metric.iNestDofs, :);
31 % error = metric.evaluate(fineValues, coarseValues);
32 %
33 % Notes:
34 % Richardson extrapolation assumes that the coarse grid is obtained
35 % by uniform coarsening of the fine grid, typically by a factor of 2
36 % in each spatial dimension.
37 %
38 % See also:
39 % profilers.metrics.Metric, profilers.metrics.MeshSpaceAbsoluteMetric,
40 % approx.space.MeshSpace

138 function E = evaluate(obj, U, V)
139 % EVALUATE Compute Richardson extrapolation errors.
140 %
141 % E = evaluate(obj, U, V) computes Richardson extrapolation
142 % errors between fine grid solution U and coarse grid
143 % solution V. The method accounts for proper domain
144 % integration and element scaling to provide meaningful error
145 % estimates.
146 %
147 % Inputs:
148 % obj - The GridSpaceRichardsonMetric object
149 % U - Fine grid solution values at integration points
150 % V - Coarse grid solution values at integration points
151 %
152 % Outputs:
153 % E - Matrix of computed Richardson errors for each reduction type
154
155 I = obj.integrator;
156 w = I.weights{2};
157 U = reshape(U, length(w), []);
158 V = reshape(V, length(w), []);
159
160 %< Compute absolute difference
161 R = abs(U - V);
162
163 %< Get mesh properties for proper scaling
164 Th = obj.coarse.mesh;
165 n = Th.nTotalElements;
166 h = Th.magnitudes;
167
168 %< Compute each requested error norm
169 nTypes = length(obj.reduction);
170 E = cell(nTypes, 1);
171 for i = 1:nTypes
172 red = obj.reduction{i};
173 switch red
174 case 'L1'
175 e = w * R;
176 e = reshape(e, n, []);
177 e = e .* h(:);
178 e = sum(e, 1);
179 case 'L2'
180 e = w * R.^2;
181 e = reshape(e, n, []);
182 e = e .* h(:);
183 e = sqrt(sum(e, 1));
184 case 'Lx'
185 e = max(R, [], 1);
186 e = reshape(e, n, []);
187 e = max(e, [], 1);
188 case 'none'
189 e = R;
190 end
191 E{i} = e(:).';
192 end
193 E = cell2mat(E);
194 end
195 end

198 function obj = setNestElements(obj)
199 % SETNESTELEMENTS Reorder fine elements to align with coarse
200 % elements.
201 %
202 % obj = setNestElements(obj) computes the reordering of fine
203 % grid elements to align with the nested coarse grid
204 % structure. This enables proper Richardson extrapolation
205 % between grids.
206 %
207 % Inputs:
208 % obj - The GridSpaceRichardsonMetric object
209 %
210 % Outputs:
211 % obj - The GridSpaceRichardsonMetric object
212
213 Th0 = obj.coarse.mesh;
214 Th1 = obj.fine.mesh;
215 n0 = Th0.resolution;
216 n1 = Th1.resolution;
217 m0 = n1 ./ n0;
218 T1 = core.linalg.MultiIndexer([prod(m0), prod(n0)]);
219 I = (1:prod(n1)).';
220 LK = T1.linearToMulti(I);
221 T2 = core.linalg.MultiIndexer(m0);
222 L = T2.linearToMulti(LK(:, 1));
223 T3 = core.linalg.MultiIndexer(n0);
224 K = T3.linearToMulti(LK(:, 2));
225 I = (K-1) .* m0 + (L-1) + 1;
226 T4 = core.linalg.MultiIndexer(n1);
227 obj.iNestElements = T4.multiToLinear(I);
228 end
