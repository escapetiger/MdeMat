28function test_polynomial_exactness(rule, np)
29 % Test exactness for polynomials up to degree 2*np-1
30

32 max_exact_degree = 2*np - 1;
33
34 % Generate quadrature points and weights
35 [x, w] = rule.generate(G, np);
36 x = x(:)'; % Row vector
37 w = w(:)'; % Row vector
38
39 fprintf(' Points: '); fprintf('%.4f ', x); fprintf('\n');
40 fprintf(' Weights: '); fprintf('%.4f ', w); fprintf('\n');
41 fprintf(' Weight sum: %.8f\n', sum(w));
42
43 % Test polynomials x^k for k = 0, 1, 2, ..., max_exact_degree + 2
44 all_exact = true;
45
46 for k = 0:(max_exact_degree + 3)
47 % Compute numerical integral of x^k with Gauss-Hermite weight
48 if k == 0

52 end

139function moment = compute_analytical_moment(k)
140 % Compute analytical moments of standard normal distribution
141 % ∫ x^k * exp(-x²/2)/√(2π) dx
142
143 if mod(k, 2) == 1
144 % Odd moments are zero
145 moment = 0;
146 else
147 % Even moments: (k-1)!! = (k-1)(k-3)(k-5)...3·1
148 if k == 0
149 moment = 1;
150 else
151 moment = double_factorial(k - 1);
152 end
153 end
154end

156function result = double_factorial(n)
