196 P = obj.element.projector;
197 B = P.basis;
198
199 cls = 'core.function.SeparableFunction';

202 'InvalidBasis', ...

205 ne = size(L, 1);
206 nl = obj.nLocalDofs;
207 if isempty(varargin)
208 xRef = obj.element.volumeData.nodes;

210 U = f(X);

212 U = f(varargin{:});
213 end
