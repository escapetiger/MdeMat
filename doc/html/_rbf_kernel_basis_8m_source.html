3 %
4 % RbfKernelBasis represents a set of radial basis function (RBF)
5 % interpolation basis functions. Each basis function is a radial

80 % DIFFERENTIATEIMPL Implementation of RBF basis derivative.
81 %
82 % dY = diffImpl(obj, X, order) computes derivatives of the
83 % RBF basis functions using the chain rule and
84 % kernel-specific derivative formulas.
85 %
86 % The derivative is computed by
87 %
88 % \f[
89 % \frac{\partial \phi_i(\|x - c_i\|)}{\partial x_j} =
90 % \phi'_i(r_i) \frac{x_j - c_{i,j}}{r_i},
91 % \f]
92 %
93 % where \f$r_i = \|x - c_i\|\f$ is the distance to center
94 % \f$i\f$.
95
96 m = find(order > 0, 1);
97 nd = obj.NDims;
98 nc = obj.NCodims;
99 np = size(X, 2);
100
101 X = reshape(X, [nd, 1, np]); % [nd, 1, np]
102 X0 = reshape(obj.Nodes, [nd, nc, 1]); % [nd, nc, 1]
103 R = vecnorm(X-X0, 2, 1); % [1, nc, np]
104 R = reshape(R, [nc, np]); % [1, nc, np]
105
106 E = obj.Epsilon;
107 if isscalar(E), E = repmat(E, nc, 1); end
108 E = repmat(E, [1, np]);
109
110 dY = obj.diffKernel(R, E);
111
112 R0 = R;
113 R0(R == 0) = eps;
114 dY = dY .* squeeze(R(m, :, :)) ./ R0;

116 end

139 % DIFFKERNEL Evaluate RBF kernel derivative.
140 %
141 % dphi = diffKernel(obj, r, epsilon) evaluates the derivative
142 % of the RBF kernel with respect to distance \f$r\f$:
143 % \f$\frac{\partial \phi(r)}{\partial r}\f$. Supports
144 % vectorized evaluation for multiple distances and shape
145 % parameters.

147 switch obj.Kernel

149 dphi = -2 * epsilon.^2 .* r .* exp(-(epsilon .* r).^2);
