1classdef ElementBilinearForm < handle
2 % ELEMENTBILINEARFORM Element-level bilinear form representation.
3 %
4 % ElementBilinearForm represents a bilinear form through its matrix
5 % representation, computed from function data and derivatives. Each
6 % bilinear form corresponds to a weighted inner product between
7 % basis functions and their derivatives of the form:
8 %
9 % \f[
10 % a(u, v) = \int_{\Omega} \frac{\partial^i u}{\partial x^i}
11 % w(x) \frac{\partial^j v}{\partial x^j} \, dx
12 % \f]
13 %
14 % where \f$u\f$ and \f$v\f$ are basis functions, \f$i\f$ and \f$j\f$
15 % are derivative orders, and \f$w(x)\f$ are integration weights.
16 % The discrete representation is computed using quadrature rules.
17 %
18 % This class is fundamental in finite element methods for assembling
19 % system matrices from weak formulations of differential equations.
20 % It supports both matrix and vector representations for efficient
21 % integration computations.
22 %
23 % Examples:
24 % % Create empty bilinear form
25 % form = ElementBilinearForm();
26 %
27 % % Create mass matrix (0,0 derivatives)
28 % massForm = ElementBilinearForm(functionData1, functionData2, 0, 0);
29 % M = massForm.matrix;
30 %
31 % % Create stiffness matrix (1,1 derivatives)
32 % stiffForm = ElementBilinearForm(functionData1, functionData2, 1, 1);
33 % K = stiffForm.matrix;
34 %
35 % % Mixed derivative form
36 % mixedForm = ElementBilinearForm(functionData1, functionData2, 0, 1);
37 % A = mixedForm.matrix;
38 % b = mixedForm.vector;
39 %
40 % See also:
41 % approx.element.ElementFunction, approx.integrate.Integrator
42
43 properties
44 matrix % Matrix representation of the bilinear form
45 vector % Vector representation of the weighted test function values
46 end

77 % SETDATA Set the matrix representation of the bilinear form.
78 %
79 % obj = setData(obj, D1, D2, i, j) computes the matrix
80 % representation as the weighted inner product between
81 % function values or derivatives from two function data
82 % objects. The computation follows the discrete bilinear form:
83 %
84 % \f[
85 % A_{mn} = \sum_{k} U_k^{(i)}(x_m) w_k V_k^{(j)}(x_n)
86 % \f]
87 %
88 % where \f$U^{(i)}\f$ and \f$V^{(j)}\f$ are the i-th and j-th
89 % derivatives, and \f$w_k\f$ are integration weights.
90 %
91 % Inputs:
92 % obj - The ElementBilinearForm object
93 % D1 - First function data object (ElementFunction)
94 % D2 - Second function data object (ElementFunction)
95 % i - Derivative order for first function (non-negative integer, 0 for values)
96 % j - Derivative order for second function (non-negative integer, 0 for values)
97 %
98 % Outputs:
99 % obj - The ElementBilinearForm object
100
101 %< Create diagonal weight matrix
102 W = diag(D1.weights);
103
104 %< Extract function values or derivatives for first function
105 if i == 0
106 U = D1.values;
107 else
108 U = D1.derivatives{i};
109 end
110
111 %< Extract function values or derivatives for second function
112 if j == 0
113 V = D2.values;
114 else
115 V = D2.derivatives{j};
116 end
117
118 %< Compute bilinear form matrix: A = U^T * W * V
119 A = U * W * V.';
120 obj.matrix = A;
121
122 %< Compute weighted test function vector: b = U^T * W
123 obj.vector = U * W;
124 end
125 end
