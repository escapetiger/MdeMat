1classdef FunctionMetric < profilers.metrics.Metric
2 % FUNCTIONMETRIC Error computation for function-based solutions.
3 %
4 % FunctionMetric implements error computation between discrete
5 % function values evaluated at specified integration nodes with
6 % associated quadrature weights. This class is designed for computing
7 % various error norms (L1, L2, Lx) between numerical and reference
8 % solutions represented as function values.
9 %
10 % The metric uses numerical integration to compute weighted error
11 % norms, making it suitable for finite element error analysis where
12 % solutions are evaluated at quadrature points with appropriate
13 % integration weights.
14 %
15 % Examples:
16 % % Create function metric with L2 norm
17 % metric = FunctionMetric({'L2'}, gaussIntegrator);
18 %
19 % % Compute error between numerical and exact solutions
20 % numericalValues = [1.1, 2.05, 2.98]; % at integration points
21 % exactValues = [1.0, 2.0, 3.0];
22 % error = metric.evaluate(numericalValues, exactValues);
23 %
24 % % Multiple error norms simultaneously
25 % metric = FunctionMetric({'L1', 'L2', 'Lx'}, integrator);
26 % errors = metric.evaluate(numerical, exact); % returns cell array
27 %

54 function E = evaluate(obj, U, V)
55 % EVALUATE Compute error between function-based solutions.
56 %
57 % E = evaluate(obj, U, V) computes errors between numerical
58 % solution U and reference solution V using the specified
59 % error reduction types. For L1 and L2 norms, integration
60 % weights are applied for proper numerical integration.
61 %
62 % Inputs:
63 % obj - The FunctionMetric object
64 % U - Numerical solution values at integration points
65 % V - Reference solution values (same points) or scalar
66 %
67 % Outputs:
68 % E - Cell array of computed errors for each reduction type
69
70 %< Prepare data for weighted integration
71 if ismember('L1', obj.reduction) || ismember('L2', obj.reduction)
72 I = obj.integrator;
73 w = I.weights;
74 n = I.nPoints;
75 U = reshape(U, n, []);
76 if ~isscalar(V)
77 V = reshape(V, n, []);
78 end
79 end
80
81 %< Compute absolute difference
82 R = abs(U - V);
83
84 %< Compute each requested error norm
85 nReds = length(obj.reduction);
86 E = cell(1, nReds);
87 for i = 1:nReds
88 red = obj.reduction{i};
89 switch red
90 case 'L1'
91 e = sum(w * R, 'all');
92 case 'L2'
93 e = sqrt(sum(w * R.^2, 'all'));
94 case 'Lx'
95 e = max(R, [], 'all');
96 case 'none'
97 e = R;
98 end
99 E{i} = e(:).';
100 end
101 end
102 end
103end
