75function [vertices, elements] = generateUniformBoxMesh(nx, ny, xRange, yRange)
76% GENERATEUNIFORMBOXMESH Generate uniform triangular mesh on rectangular box.
77%
78% [vertices, elements] = generateUniformBoxMesh(nx, ny, xRange, yRange)
79% creates a uniform triangular mesh on a rectangular domain.
80
81% Generate grid points
82xPoints = linspace(xRange(1), xRange(2), nx+1);
83yPoints = linspace(yRange(1), yRange(2), ny+1);
84[X, Y] = meshgrid(xPoints, yPoints);
85
86% Convert to vertex list
87vertices = [X(:), Y(:)];
88nVertices = size(vertices, 1);
89
90% Generate triangular elements
91% Each grid cell (i,j) creates two triangles
92elements = [];
93for j = 1:ny
94 for i = 1:nx
95 % Grid indices for current cell
96 v1 = (j - 1) * (nx + 1) + i; % Bottom-left
97 v2 = (j - 1) * (nx + 1) + i + 1; % Bottom-right
98 v3 = j * (nx + 1) + i; % Top-left
99 v4 = j * (nx + 1) + i + 1; % Top-right
100
101 % Two triangles per cell
102 elements = [elements; v1, v2, v3]; % Lower triangle
103 elements = [elements; v2, v4, v3]; % Upper triangle
104 end
105end
106
107fprintf('Generated %d vertices and %d triangular elements.\n', ...
108 nVertices, size(elements, 1));
109
110end

112function validatePeriodicConnections(mesh, boundingBox, tolerance)
113% VALIDATEPERIODICCONNECTIONS Validate periodic vertex and face mappings.
114%
115% validatePeriodicConnections(mesh, boundingBox, tolerance) checks that
116% periodic connections are correctly established.
117
118fprintf('Validating periodic vertex connections...\n');
119
120vertices = mesh.Vertices;
121nConnections = 0;
122
123% Define expected periodic shifts based on bounding box
124xRange = [boundingBox(1), boundingBox(2)];
125yRange = [boundingBox(3), boundingBox(4)];
126periodicShifts = [; ...
127 diff(xRange), 0; ... % Right boundary to left
128 -diff(xRange), 0; ... % Left boundary to right
129 0, diff(yRange); ... % Top boundary to bottom
130 0, -diff(yRange); ... % Bottom boundary to top
131 diff(xRange), diff(yRange); ... % Top-right to bottom-left
132 -diff(xRange), -diff(yRange); ... % Bottom-left to top-right
133 diff(xRange), -diff(yRange); ... % Bottom-right to top-left
134 -diff(xRange), diff(yRange); ... % Top-left to bottom-right
135 ];
136
137% Test vertex-to-vertex mapping
138if ~isempty(mesh.VertexToVertexTables)
139 for iTable = 1:length(mesh.VertexToVertexTables)
140 V2V = mesh.VertexToVertexTables{iTable};
141 [iVertices, jVertices] = find(V2V);
142
143 for k = 1:length(iVertices)
144 i = iVertices(k);
145 j = jVertices(k);
146
147 if i ~= j
148 % Check if connection corresponds to a periodic shift
149 displacement = vertices(j, :) - vertices(i, :);
150
151 % Find matching shift
152 shiftFound = false;
153 for s = 1:size(periodicShifts, 1)
154 if norm(displacement-periodicShifts(s, :)) < tolerance
155 shiftFound = true;
156 nConnections = nConnections + 1;
157 break;
158 end
159 end
160
161 if ~shiftFound && norm(displacement) > tolerance
162 fprintf('Warning: Unexpected connection between vertices %d and %d\n', i, j);
163 fprintf(' Displacement: [%.6f, %.6f]\n', displacement);
164 end
165 end
166 end
167 end
168end
169
170fprintf('Validated %d periodic vertex connections.\n', nConnections);
171
172% Test face-to-face mapping
173if ~isempty(mesh.FaceToFaceTable)
174 F2F = mesh.FaceToFaceTable;
175 faces = mesh.Faces;
176 nFaceConnections = 0;
177
178 fprintf('Validating periodic face connections...\n');
179
180 [iFaces, jFaces] = find(F2F);
181
182 for k = 1:length(iFaces)
183 i = iFaces(k);
184 j = jFaces(k);
185
186 if i ~= j
187 % Check if face connection is consistent with vertex connections
188 face1Vertices = faces(i, :);
189 face2Vertices = faces(j, :);
190
191 % Check if all vertices in face1 have corresponding vertices in face2
192 mappingConsistent = true;
193 if ~isempty(mesh.VertexToVertexTables)
194 for v1 = face1Vertices
195 mapped = false;
196 for iTable = 1:length(mesh.VertexToVertexTables)
197 V2V = mesh.VertexToVertexTables{iTable};
198 for v2 = face2Vertices
199 if V2V(v1, v2)
200 mapped = true;
201 break;
202 end
203 end
204 if mapped
205 break;
206 end
207 end
208 if ~mapped
209 mappingConsistent = false;
210 break;
211 end
212 end
213 end
214
215 if mappingConsistent
216 nFaceConnections = nFaceConnections + 1;
217 else
218 fprintf('Warning: Inconsistent face mapping between faces %d and %d\n', i, j);
219 end
220 end
221 end
222
223 fprintf('Validated %d periodic face connections.\n', nFaceConnections);
224else
225 fprintf('No face-to-face mapping table found.\n');
226end
227
228end

289function validateJacobianComputations(mesh)
290% VALIDATEJACOBIANCOMPUTATIONS Validate Jacobian computation methods.
291%
292% validateJacobianComputations(mesh) tests and validates various
293% Jacobian-related computations for geometric consistency.
294
295fprintf('Computing element Jacobian determinants...\n');
296detJ = mesh.computeElementJacobianDeterminants();
297fprintf('Jacobian determinants: min=%.6f, max=%.6f, mean=%.6f\n', ...
298 min(detJ), max(detJ), mean(detJ));
299
300% Check for positive determinants (proper orientation)
301negativeDetJ = sum(detJ < 0);
302if negativeDetJ == 0
303 fprintf('✓ All elements have positive orientation.\n');
304else
305 fprintf('✗ Warning: %d elements have negative orientation!\n', negativeDetJ);
306end
307
308fprintf('Computing element inverse Jacobians...\n');
309invJ = mesh.computeElementInverseJacobians();
310fprintf('Inverse Jacobians computed for %d elements.\n', size(invJ, 3));
311
312% Validate J * inv(J) = I for a few elements
313fprintf('Validating Jacobian inversion...\n');
314J = mesh.computeElementJacobians();
315testElements = [1, ceil(mesh.NElements/2), mesh.NElements];
316maxError = 0;
317
318for iElement = testElements
319 if iElement <= mesh.NElements
320 if mesh.NDims == 1
321 Ji = J(iElement);
322 invJi = invJ(iElement);
323 product = Ji * invJi;
324 error = abs(product-1);
325 else
326 Ji = J(:, :, iElement);
327 invJi = invJ(:, :, iElement);
328 product = Ji * invJi;
329 identity = eye(size(product));
330 error = norm(product-identity, 'fro');
331 end
332 maxError = max(maxError, error);
333 end
334end
335
336fprintf('Maximum Jacobian inversion error: %.2e\n', maxError);
337if maxError < 1e-12
338 fprintf('✓ Jacobian inversion is accurate.\n');
339else
340 fprintf('✗ Warning: Jacobian inversion has significant errors!\n');
341end
342
343fprintf('Computing face Jacobian determinants...\n');
344nFacesPerElement = mesh.NVerticesPerElement;
345for iFace = 1:nFacesPerElement
346 detJFace = mesh.computeFaceJacobianDeterminants([], iFace);
347 fprintf('Face %d: min=%.6f, max=%.6f, mean=%.6f\n', ...
348 iFace, min(detJFace), max(detJFace), mean(detJFace));
349end
350
351fprintf('Computing outward normal vectors...\n');
352for iFace = 1:nFacesPerElement
353 normals = mesh.computeOutwardNormals([], iFace);
354
355 % Check unit length
356 normLengths = sqrt(sum(normals.^2, 1));
357 maxNormError = max(abs(normLengths-1));
358
359 fprintf('Face %d normals: max deviation from unit length = %.2e\n', ...
360 iFace, maxNormError);
361end
362
363fprintf('Computing mesh measures...\n');
364detJ = mesh.computeElementJacobianDeterminants();
365elementMeasures = sum(detJ) / 2; % Divide by 2 for triangular elements
366fprintf('Total mesh measure: %.6f\n', elementMeasures);
367
368% For unit square, total area should be 1
369expectedArea = 1.0;
370areaError = abs(elementMeasures-expectedArea);
371fprintf('Area error compared to unit square: %.6f\n', areaError);
372
373if areaError < 1e-10
374 fprintf('✓ Mesh area computation is accurate.\n');
375else
376 fprintf('✗ Warning: Mesh area computation has errors!\n');
377end
378
379end
