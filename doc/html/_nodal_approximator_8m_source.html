1classdef NodalApproximator < approx.linear.LinearApproximator
2 % NODALAPPROXIMATOR Linear approximator based on nodal information.
3 %
4 % NodalApproximator implements a linear approximation method that
5 % uses interpolation basis functions, such as Lagrange polynomials.
6 %
7 % Let \f$x_j\f$ denote the nodal points, \f$p_i\f$ basis functions,
8 % and \f$c_i\f$ the coefficients to be determined. The projection
9 % system is defined as:
10 %
11 % \f[
12 % \sum_i\sum_kw_kc_ip_i(x_k)p_j(x_k) = f_j(x_k), \forall j.
13 % \f]
14 %
15 % This implies that the mass matrix is \f$M=diag(w_k)\f$. The
16 % fitting system is defined as:
17 %
18 % \f[
19 % u(x_j) = \sum_i c_i p_i(x_j), \forall j.
20 % \f]
21 %
22 % The design matrix \f$D_{ji} = p_i(x_j)\f$ represents the evaluation
23 % of basis functions at nodal points. For Lagrangian polynomial,
24 % \f$D\f$ reduces to the identity matrix.
25 %
26 %
27 % See also:
28 % approx.linear.LinearApproximator, approx.linear.ModalApproximator
29

31 Type = 'nodal' % Approximator type identifier
32 end
