105 function obj = setPartials(obj, func, order, options)
106 % SETPARTIALS Compute partial derivatives at integration
107 % points.
108 %
109 % obj = setPartials(obj, func, order) computes partial
110 % derivatives up to the specified order at the stored
111 % integration nodes.
112 %
113 % obj = setPartials(obj, func, order, nodes=nodes) computes
114 % partial derivatives at the specified nodes instead of the
115 % stored nodes.
116
117 arguments
118 obj approx.integrate.Integrator
119 func{mustBeFunction}
120 order(1, 1) {mustBePositive, mustBeInteger}
121 options.nodes{mustBeNumeric} = obj.Nodes
122 end
123
124 core.except.assert(~isempty(options.nodes), 'InvalidState', ...
125 'Integration nodes must be set before evaluating function.');
126
127 nd = obj.NDims;
128 indexer = core.linalg.L1MultiIndexer();
129 obj.Partials = cell(1, order);
130 for k = 1:order
131 indexer.setCache(nd, nd+k, minSum = nd+k);
132 M = indexer.Cache - 1;
133 dY = cell(1, size(M, 1));
134 for j = 1:size(M, 1)
135 dY{j} = func.diff(options.nodes, M(j, :));
136 end
137 obj.Partials{k} = cat(3, dY{:});
138 end
