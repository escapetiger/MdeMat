3% Fit data using the nonlinear decay model y = a - b*(x-c)^(-d).
4%
5% Syntax
6% fitResult = fitNonlinearDecayModel()
7%
8% Description
9% This function fits experimental data to a nonlinear decay model of the
10% form y = a - b*(x-c)^(-d), where a, b, c, and d are parameters to be estimated.
11% The parameter 'a' is constrained to be in the range [9, 12].
12% The function uses nonlinear least squares fitting and displays results
13% with visualization.
14%
15% Output Arguments
16% fitResult - structure containing fitted parameters and statistics
17% * parameters - fitted values of a, b, c, and d
18% * rsquared - coefficient of determination
19% * rmse - root mean square error
20% * residuals - fitting residuals
21%
22% Example
23% result = fitNonlinearDecayModel()
24%
25% See also
26% fitnlm, nlinfit, lsqcurvefit
27
28%% Initialize data
29% X-axis values
30xData = [0, 1, 2, 3, 4, 5, 8, 10, 12, 15, 18, 20, 25, 30, 35, 40, 50, 55, 60, 70, 72, 74, 80]';
31
32% Y-axis values
33yData = [0, 6.2097, 6.7417, 7.2635, 7.5920, 7.8747, 8.4854, 8.7461, 8.9925, ...
34 9.2179, 9.3467, 9.4431, 9.6278, 9.7492, 9.8206, 9.8540, 9.9247, ...
35 9.9654, 9.9722, 10.0169, 10.0064, 10.0092, 10.0231]';
36
37% Validate input data
38validateInputData(xData, yData);

40%% Define model function
41% Model: y = a - b*(x-c)^(-d)
42modelFunction = @(parameters, x) calculateModelValues(parameters, x);
43
44%% Set initial parameter estimates
45% Based on the data pattern, provide reasonable initial guesses
46initialA = 10.0; % Initial guess for asymptote parameter a (within [9,12])
47initialB = 10.0; % Initial guess for parameter b

49initialD = 0.5; % Initial guess for power parameter d (positive for negative exponent)
50initialParameters = [initialA, initialB, initialC, initialD];

52%% Perform nonlinear fitting
53try
54 % Use lsqcurvefit for robust nonlinear fitting
55 options = optimoptions('lsqcurvefit', 'Display', 'off', 'MaxIterations', 1000, ...
56 'FunctionTolerance', 1e-8, 'OptimalityTolerance', 1e-8);
57

97function yValues = calculateModelValues(parameters, x)
98% Calculate model values y = a - b*(x-c)^(-d) with proper handling of edge cases
99%
100% Input Arguments
101% parameters - array containing [a, b, c, d]
102% x - input x values
103%
104% Output Arguments
105% yValues - calculated y values
106
107a = parameters(1);
108b = parameters(2);
109c = parameters(3);
110d = parameters(4);
111
112% Calculate (x-c)
113xShifted = x - c;
114
115% For negative exponents, we need to handle division by zero and negative bases
116% Add small epsilon to avoid division by zero
117epsilon = 1e-10;
118
119% Ensure we don't have zero or negative bases for negative exponents
120xShiftedSafe = sign(xShifted) .* max(abs(xShifted), epsilon);
121
122% Calculate y = a - b*(x-c)^(-d) = a - b / (x-c)^d
123yValues = a - b ./ (xShiftedSafe.^d);
124
125% Replace any NaN or Inf values with a large residual value to penalize invalid solutions
126invalidIndices = ~isfinite(yValues);
127if any(invalidIndices)
128 yValues(invalidIndices) = 1e6; % Large value to penalize invalid solutions
129end
130
131end

152function rSquared = calculateRSquared(observedY, predictedY)
153% Calculate coefficient of determination (R-squared)
154%
155% Input Arguments
156% observedY - observed y values
157% predictedY - predicted y values from model
158%
159% Output Arguments
160% rSquared - coefficient of determination
161
162sumSquaresTotal = sum((observedY - mean(observedY)).^2);
163sumSquaresResidual = sum((observedY - predictedY).^2);
164rSquared = 1 - (sumSquaresResidual / sumSquaresTotal);
165
166end
