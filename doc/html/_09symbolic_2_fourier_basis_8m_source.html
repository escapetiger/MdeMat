48 function computeExpressions(obj)
49 % COMPUTEEXPRESSIONS Compute Fourier basis function
50 % expressions.
51
52 obj.Expressions = repmat(sym([]), 2*obj.MaxDegree+1, 1);
53 L = obj.Upper - obj.Lower;
54 x = obj.Variables(1);
55 x = 2 * sym(pi) * (x - obj.Lower) / L;
56
57 if obj.IsNormalized
58 obj.Expressions(1) = sqrt(2*sym(pi)) / sqrt(2*sym(pi)*L);
59 else
60 obj.Expressions(1) = sym(1);
61 end
62
63 for k = 1:obj.MaxDegree
64
65 if obj.IsNormalized
66 normalizer = sqrt(2*sym(pi)) / sqrt(sym(pi)*L);
67 else
68 normalizer = sym(1);
69 end
70
71 obj.Expressions(2*k) = normalizer * cos(k*x);
72 obj.Expressions(2*k+1) = normalizer * sin(k*x);
73 end
