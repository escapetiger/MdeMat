13 % L2ORTHOTOPEELEMENT Create an \f$L^2\f$ element on the unit
14 % orthotope.
15 %
16 % obj = L2OrthotopeElement(nDims, basisOrder, basisType,
17 % basisPattern) creates an \f$L^2\f$ element on the unit
18 % orthotope with specified basis configuration.
19 %
20 % Basis types:
21 % 'monic_unit_legendre': Monic Legendre basis
22 % 'unit_gauss_legendre_lagrange': Lagrange-Gauss-Legendre basis
23 % 'unit_gauss_lobatto_lagrange': Lagrange-Gauss-Lobatto basis
24 %
25 % Basis patterns:
26 % 'Q': \f$L_\infty\f$ norm ordering (tensor product)
27 % 'P': \f$L_1\f$ norm ordering (total degree)
28
29 arguments
30 nDims {mustBePositive, mustBeInteger}
31 basisOrder {mustBePositive, mustBeInteger}
32 basisType {mustBeMember(basisType, { ...
33 'monic_unit_legendre', ...
34 'unit_gauss_legendre_lagrange', ...
35 'unit_gauss_lobatto_lagrange'})}
36 basisPattern {mustBeMember(basisPattern, {'P', 'Q'})}
37 end
38
39 validateCompatibleBasis(basisType, basisPattern);
40
41 bbox = repmat([-1/2, 1/2], 1, nDims);
42
43 %< Create unit orthotope geometry
44 G = core.geometry.Orthotope(bbox);
45
46 %< Create basis functions based on type
47 switch basisType
48 case 'monic_unit_legendre'
49 U = core.function.CompiledOrthogonalBasis(basisOrder, basisType);
50 case {'unit_gauss_legendre_lagrange', 'unit_gauss_lobatto_lagrange'}
51 U = core.function.CompiledInterpolationBasis(basisOrder, basisType);
52 end
53 U = repmat(U, 1, nDims);
54
55 %< Create separable basis with specified pattern
56 B = core.function.SeparableFunction(Factors=U, ...
57 Pattern=basisPattern);
58
59 %< Create volume integrator
60 switch basisType
61 case {'monic_unit_legendre', 'unit_gauss_legendre_lagrange'}
62 R = approx.integrate.GaussLegendreRule(nDims);
63 case 'unit_gauss_lobatto_lagrange'
64 R = approx.integrate.GaussLobattoRule(nDims);
65 end
66 VI = approx.integrate.Integrator(R);
67 VI.setPoints(G, B.NFactorCodims);
68
69 %< Create face integrators for all boundaries
70 FI = arrayfun(@(i) approx.integrate.Integrator(R), 1:2*nDims);
71 for i = 1:(2 * nDims)
72 a = bbox(1:2:end);
73 b = bbox(2:2:end);
74 d = ceil(i/2);
75 if mod(i, 2) == 0
76 a(d) = b(d);
77 else
78 b(d) = a(d);
79 end
80 bboxF = [a(:).'; b(:).'];
81 FG = core.geometry.Orthotope(bboxF(:));
82 FI(i).setPoints(FG, B.NFactorCodims);
83 end
84
85 %< Create approximator based on basis type
86 switch basisType
87 case 'monic_unit_legendre'
88 A = approx.linear.ModalApproximator(B);
89 A.setMass(VI.Nodes, VI.Weights);
90 A.setDesign(VI.Nodes, VI.Weights);
91 case {'unit_gauss_legendre_lagrange', 'unit_gauss_lobatto_lagrange'}
92 A = approx.linear.NodalApproximator(B);
93 A.setMass(VI.Nodes, VI.Weights);
94 A.setDesign(VI.Nodes);
95 end
96
97 %< Call parent constructor
98 obj@approx.element.L2Element(G, A, VI, FI);
99 end
100 end

104% VALIDATECOMPATIBLEBASIS Validate basis configuration compatibility.
105%
106% validateCompatibleBasis(basisType, basisPattern) validates that

109tf1 = isCompatibleModalBasis(basisType, basisPattern);

111tf = tf1 | tf2;

113end
