16%% Basic 1D SeparableGraph Example
17
18% Create 1D separable graph with uniform spacing
19nVertices1D = 20;
20vertices1D = {linspace(0, 10, nVertices1D)};
21edges1D = {[(1:nVertices1D-1)', (2:nVertices1D)']};
22separableGraph1D = approx.mesh.SeparableGraph(vertices1D, edges1D);
23
24% Visualize 1D separable graph
25figure('Name', '1D SeparableGraph Visualization');
26plotter = approx.mesh.SeparableGraphPlotter();
27plotter.plot(separableGraph1D, ...
28 'VertexColor', 'red', ...
29 'VertexSize', 30);
30title('1D SeparableGraph with Uniform Vertex Distribution');
31
32% Display graph properties
33fprintf('=== 1D SeparableGraph Properties ===\n');
34fprintf('Dimensions: %d\n', separableGraph1D.NDims);
35fprintf('Vertices per dimension: %s\n', mat2str(separableGraph1D.NVertices));
36fprintf('Total vertices: %d\n', separableGraph1D.NTotalVertices);
37fprintf('Total edges: %d\n', separableGraph1D.NTotalEdges);

223fprintf('Full Graph object total: %.2f KB\n', fullInfo.bytes/1024);
224fprintf('\nDetailed breakdown:\n');

226fprintf(' SeparableGraph edges (factored): %.2f KB\n', separableEdgesInfo.bytes/1024);
227fprintf(' Full Graph vertices (expanded): %.2f KB\n', fullVerticesInfo.bytes/1024);
228fprintf(' Full Graph edges (expanded): %.2f KB\n', fullEdgesInfo.bytes/1024);

230% Calculate memory efficiency

233memoryReductionFactor = fullDataTotal / separableDataTotal;

240%% Theoretical Memory Scaling Analysis
241
242fprintf('\n=== Theoretical Scaling Analysis ===\n');
243
244% Analyze theoretical memory requirements for larger grids
245gridSizes = [10, 50, 100, 200];
246nGridSizes = length(gridSizes);
247
248fprintf('Grid Size | SeparableGraph (MB) | Full Graph (MB) | Reduction Factor\n');
249fprintf('----------|---------------------|-----------------|------------------\n');
250
251for iSize = 1:nGridSizes
252 gridSize = gridSizes(iSize);
253
254 % Calculate theoretical memory for SeparableGraph
255 % Each dimension stores n vertices and (n-1) edges
256 separableVerticesMemory = 3 * gridSize * 8; % 3 dims, 8 bytes per double
257 separableEdgesMemory = 3 * (gridSize-1) * 2 * 8; % 3 dims, 2 indices per edge
258 separableTotal = separableVerticesMemory + separableEdgesMemory;
259
260 % Calculate theoretical memory for full Graph
261 % Total vertices: n^3, each with 3 coordinates
262 fullVerticesMemory = (gridSize^3) * 3 * 8;
263 % Total edges: 3*n^2*(n-1), each with 2 vertex indices
264 fullEdgesMemory = 3 * (gridSize^2) * (gridSize-1) * 2 * 8;
265 fullTotal = fullVerticesMemory + fullEdgesMemory;
266
267 % Calculate reduction factor
268 reductionFactor = fullTotal / separableTotal;
269
270 fprintf('%8d |%19.2f |%15.2f |%16.1f\n', ...
271 gridSize, separableTotal/(1024*1024), fullTotal/(1024*1024), reductionFactor);
272end
