1classdef L2ElementOperator < approx.element.C0ElementOperator
2 % L2ELEMENTOPERATOR Operator for discontinuous finite elements.
3 %
4 % L2ElementOperator extends C0 operators with flux contributions
5 % for L2 elements. It manages both volume and flux bilinear
6 % form data required for discontinuous Galerkin (DG) methods, where
7 % inter-element coupling is handled through numerical fluxes.
8 %
9 % Flux data is organized in a tensor structure to accommodate
10 % different flow directions (inflow/outflow) and derivative orders
11 % across multiple element boundaries. This organization enables
12 % efficient computation of DG operators that account for the
13 % discontinuous nature of the solution space.
14 %
15 % The operator constructs discrete representations of differential
16 % operators that include both volume integrals (as in continuous
17 % methods) and boundary flux integrals (specific to DG methods).
18 %
19 % Examples:
20 % % Create L2 element and its operator
21 % element = approx.element.L2Element(geometry, projector, volInt, fluxInt);
22 % element.setVolumeData(1); % Include first derivatives
23 % element.setFluxData(0); % Function values on boundaries
24 % operator = L2ElementOperator(element);
25 %
26 % % Set up operator data and access gradient
27 % operator.setVolumeData();
28 % operator.setFluxData();
29 % gradOp = operator.gradient;
30 %
31 % % Access volume and flux contributions
32 % volGrad = gradOp.volumeData; % Volume gradient terms
33 % fluxGrad = gradOp.fluxData; % Boundary flux terms
34 %
35 % % Use in DG assembly
36 % for i = 1:element.nDims
37 % K_vol = volGrad(i).matrix; % Volume stiffness contribution
38 % end
39 % for i = 1:element.nFluxes
40 % K_flux = fluxGrad(:, i); % Flux contributions for face i
41 % end
42 %
43 % See also:
44 % approx.element.C0ElementOperator, approx.element.L2Element,
45 % approx.element.ElementBilinearForm

78 function obj = setFluxData(obj)
79 % SETFLUXDATA Set up bilinear form data for flux integrals.
80 %
81 % obj = setFluxData(obj) creates a tensor of bilinear form
82 % matrices for flux computations, organizing data by
83 % derivative order, flux boundary, and flow direction. This
84 % structure enables efficient assembly of DG operators with
85 % proper treatment of inflow and outflow contributions.
86 %
87 % The flux data tensor accommodates:
88 % - Inflow contributions: coupling within the same element
89 % - Outflow contributions: coupling with neighboring elements
90 % - Multiple derivative orders for higher-order DG methods
91 % - All element boundaries for complete flux integration
92 %
93 % Inputs:
94 % obj - The L2ElementOperator object
95 %
96 % Outputs:
97 % obj - The L2ElementOperator object
98
99 D = obj.element.fluxData;
100 nf = obj.element.nFluxes;
101 nd = 2; %< Number of directions (inflow/outflow)
102 nr = max([D.nDerivatives]) + 1; %< Maximum derivative order + 1
103
104 %< Initialize flux data tensor
105 f = @(i) approx.element.ElementBilinearForm();
106 obj.fluxData = arrayfun(f, 1:nd*nr*nf);
107 obj.fluxData = reshape(obj.fluxData, nd, nr, nf);
108
109 %< Populate flux data for each boundary
110 for i = 1:nf
111 nr = D(i).nDerivatives;
112 k = 2 * ceil(i/2) - 1 + mod(i, 2); %< Neighboring element index
113
114 for j = 0:nr
115 %< Inflow contribution (same element)
116 obj.fluxData(1, j+1, i).setData(D(i), D(i), j, 0);
117 %< Outflow contribution (neighboring element)
118 obj.fluxData(2, j+1, i).setData(D(i), D(k), j, 0);
119 end
120 end
121 end
