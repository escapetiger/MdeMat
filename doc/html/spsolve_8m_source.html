1function x = spsolve(A, b, varargin)
2%SPSOLVE Selects the best solver for a given matrix and right-hand side.
3%
4% x = SPSOLVE(A, b) solves the system Ax = b by selecting the most
5% suitable solver based on the properties of the matrix A.
6%
7% x = SPSOLVE(A, b, options) allows additional options to be passed
8% as a struct with fields:
9% - tol: Tolerance for iterative solvers (default: 1e-6)
10% - maxit: Maximum iterations for iterative solvers (default: 500)
11% - sizeThreshold: Matrix size threshold for iterative solvers
12% (default: 100,000 non-zeros)
13% - spd: Logical flag for symmetric positive definite property
14% (default: [])
15%
16% Inputs:
17% - A: Coefficient matrix (sparse or dense)
18% - b: Right-hand side vector
19% - options: Struct for solver options (optional)
20%
21% Output:
22% - x: Solution to the system Ax = b
23%
24% Author: Yi Cai
25% Date: 2024-11-21
26
27%=========================================================================
28% Parse inputs
29%=========================================================================
30if nargin < 3
31 options = struct();
32else
33 options = varargin{1};
34end
35
36% Set default options
37% Tolerance for iterative solvers (default: 1e-6)
38tol = 1e-6;
39if isfield(options, 'tol'), tol = options.tol; end
40
41% Maximum iterations for iterative solvers (default: 500)
42maxit = 500;
43if isfield(options, 'maxit'), maxit = options.maxit; end
44
45% Size threshold for selecting iterative solvers (default: 100,000 non-zeros)
46sizeThreshold = 1e5;
47if isfield(options, 'sizeThreshold'), sizeThreshold = options.sizeThreshold; end
48
49% User-provided flag for symmetric positive definiteness (default: empty)
50spd = [];
51if isfield(options, 'spd'), spd = options.spd; end
52
53% Check if A is sparse
54isSparse = issparse(A);
55
56%=========================================================================
57% Analyze matrix properties
58%=========================================================================
59% Ensure the matrix is square
60isSquare = size(A, 1) == size(A, 2);
61if ~isSquare
62 error('Matrix A must be square.');
63end
64
65% Check symmetry and positive definiteness only if not provided by the user
66if isempty(spd)
67 if ~issymmetric(A)
68 spd = false;
69 else
70 [~, p] = chol(A);
71 spd = (p == 0);
72 end
73end
74
75% Determine the size of the matrix
76if isSparse
77 N = nnz(A); % Number of non-zero entries
78else
79 N = numel(A); % Total number of entries for dense matrices
80end
81
82%=========================================================================
83% Select solver based on matrix properties
84%=========================================================================
85if N > sizeThreshold || issparse(A)
86 % Large matrix: Use iterative solvers
87 if spd
88 solver = 'pcg'; % Preconditioned Conjugate Gradient for SPD matrices
89 elseif isSparse
90 solver = 'gmres'; % Generalized Minimum Residual Method for sparse matrices
91 else
92 solver = 'bicgstab'; % Bi-Conjugate Gradient Stabilized Method for dense matrices
93 end
94else
95 % Small matrix: Use direct solvers
96 if spd
97 solver = 'chol'; % Cholesky factorization for SPD matrices
98 elseif isSparse
99 solver = 'backslash'; % Sparse LU/QR solver
100 else
101 solver = 'backslash'; % Dense LU solver
102 end
103end
104
105%=========================================================================
106% Solve the system using the selected solver
107%=========================================================================
108switch solver
109 case 'chol'
110 % Use Cholesky factorization for SPD matrices
111 R = chol(A);
112 x = R \ (R' \ b);
113 case 'pcg'
114 % Use Preconditioned Conjugate Gradient for large SPD matrices
115 x = pcg(A, b, tol, maxit);
116 case 'gmres'
117 % Use Generalized Minimum Residual Method for general sparse matrices
118 x = gmres(A, b, [], tol, maxit);
119 case 'bicgstab'
120 % Use Bi-Conjugate Gradient Stabilized Method for general dense matrices
121 x = bicgstab(A, b, tol, maxit);
122 case 'backslash'
123 % Use MATLAB's backslash operator for direct solvers
124 x = A \ b;
125 otherwise
126 error('No suitable solver found.');
127end
128
129end
