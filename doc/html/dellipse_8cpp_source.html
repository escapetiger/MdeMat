28double dellipse(double x0,double y0,double a,double b)
29{
30 double t1,t2,t4,t6,t8,t9,t11,t15,t16;
31 t1=a*a; t2=b*b; t4=y0*y0; t6=x0*x0; t8=t1*t1;
32 t9=t2*t1; t11=t2*t2; t15=t8*t2; t16=t11*t1;
33 double pol[5]={ 1.0, 2.0*t1+2.0*t2, -t2*t4-t1*t6+t8+4.0*t9+t11,
34 -2.0*t9*t6-2.0*t9*t4+2.0*t15+2.0*t16,
35 -t16*t6+t8*t11-t15*t4 };
36
37 double rr[4],ri[4];
38 roots(pol,rr,ri,4);
39
40 double t=-HUGE_VAL;
41 for (int i=0; i<4; i++)
42 t=max(t,rr[i]);
43
44 double x=sqr(a)*x0/(t+sqr(a));
45 double y=sqr(b)*y0/(t+sqr(b));
46 double d=t*sqrt(sqr(x)/sqr(sqr(a))+sqr(y)/sqr(sqr(b)));
47
48 return d;
49}
