2 % MIXEDGRIDASSEMBLER Grid-based assembler for mixed finite element
3 % systems.
4 %
5 % MixedGridAssembler extends the Eulerian grid assembler with support
6 % for different flux types (left-biased, right-biased, central) used
7 % in mixed finite element methods and discontinuous Galerkin schemes.
8 %
9 % This class provides the foundation for assembling operators in
10 % mixed formulations where different variables may require different
11 % flux treatments. It supports various numerical flux choices
12 % commonly used in discontinuous methods.
13 %
14 % Examples:
15 % % Create mixed assembler with central flux and Dirichlet BC
16 % assembler = MixedGridAssembler(fe, mesh, op, 1, 3);
17 %
18 % % Assemble divergence operator with dimension-specific coefficients
19 % coeffs = [1.0, 0.5]; % coefficients for each dimension
20 % divMatrix = assembler.divergence(coeffs);
21 %
22 % % Assemble partial derivative in specific dimension
23 % partialMatrix = assembler.partial(1, 2.0);

58 function A = divergence(obj, c)
59 % DIVERGENCE Assemble divergence operator with coefficients.
60 %
61 % A = divergence(obj, c) creates the divergence operator
62 % matrix by summing partial derivative operators across all
63 % spatial dimensions, each weighted by the corresponding
64 % coefficient.
65 %
66 % Inputs:
67 % obj - The MixedGridAssembler object
68 % c - Coefficient vector (one per spatial dimension)
69 %
70 % Outputs:
71 % A - Assembled divergence operator sparse matrix
72
73 n = obj.nGlobalDofs;
74 A = sparse(n, n);
75 for d = 1:obj.nDims
76 if c(d) == 0, continue; end
77 T = obj.partial(d, c(d));
78 A = A + T;
79 end
80 end

82 function A = partial(obj, d, c)
83 % PARTIAL Assemble partial derivative operator with
84 % coefficients.
85 %
86 % A = partial(obj, d, c) creates the partial derivative
87 % operator matrix for the specified dimension, including
88 % volume and flux contributions based on the finite element
89 % type and flux choice.
90 %
91 % Inputs:
92 % obj - The MixedGridAssembler object
93 % d - Spatial dimension index (1, 2, or 3)
94 % c - Coefficient value for this dimension
95 %
96 % Outputs:
97 % A - Assembled partial derivative sparse matrix
98
99 ne = obj.nTotalElements;
100
101 %< ------------------------------------------------------------
102 %< COEFFICIENTS
103 %< ------------------------------------------------------------
104 if isa(obj.mesh, 'approx.mesh.UniformGrid')
105 h = obj.mesh.spacings{d};
106 c = ones(1, ne) .* c ./ h;
107 end
108
109 if isa(obj.mesh, 'approx.mesh.NonuniformGrid')
110 h = obj.mesh.spacings;
111 [h{:}] = ndgrid(h{:});
112 c = c ./ h{d}(:);
113 end
114
115 c = c(:);
116
117 %< ------------------------------------------------------------
118 %< VOLUME CONTRIBUTION
119 %< ------------------------------------------------------------
120 T1 = obj.assembleVolumeTerms(d, c);
121
122 %< ------------------------------------------------------------
123 %< FLUX CONTRIBUTION
124 %< ------------------------------------------------------------
125 if isa(obj.fe, 'fem.element.L2FiniteElement')
126 switch obj.fluxType
127 case 1
128 T2 = obj.assembleLeftFluxTerms(d, c);
129 case 2
130 T2 = obj.assembleRightFluxTerms(d, c);
131 case 3
132 T2 = obj.assembleCentralFluxTerms(d, c);
133 end
134 end
135
136 %< ------------------------------------------------------------
137 %< TRIPLETS TO SPARSE MATRIX
138 %< ------------------------------------------------------------
139 n = obj.nGlobalDofs;
140 T = [T1; T2];
141 A = obj.sparseFromTriplets(T, n, n);
142 end
143 end
