1classdef RichardsonIntegrator < approx.integrate.Integrator
2 % RICHARDSONINTEGRATOR Richardson extrapolation for numerical integration.
3 %
4 % RichardsonIntegrator implements Richardson extrapolation to improve
5 % integration accuracy by combining quadrature rules at multiple
6 % resolution levels. This method leverages known convergence rates
7 % to achieve higher accuracy than the base integrator alone would
8 % provide at comparable computational cost.
9 %
10 % Examples:
11 % % Create Richardson integrator with 3 levels
12 % integrator = RichardsonIntegrator(3, 'gauss_legendre', 3);
13 %
14 % % Using existing rule object
15 % rule = approx.integrate.SeparableRule.gaussLobatto(2);
16 % integrator = RichardsonIntegrator(rule, 4);
17 %
18 % Notes:
19 % Richardson extrapolation assumes that the integration error can
20 % be expressed as a power series in the mesh size. The method
21 % recursively refines the domain and combines results to eliminate
22 % leading error terms.
23 %
24 % See also:
25 % approx.integrate.Integrator, approx.integrate.SeparableRule,
26 % approx.integrate.OrthotopeIntegrator

90 % SETPOINTS Generate multi-level integration points.

126 function [X, w] = refine(obj, n, a, b, i)
127 % REFINE Generate refined integration nodes and weights.
128
129 d = obj.nDims;
130 if i == 1
131 args = arrayfun(@(k) {n(k), a(k), b(k)}, 1:d, 'Un', 0);
132 [X, w] = obj.rule.generate(args{:});
133 else
134 m = 2^d;
135 c = (a + b) / 2;
136 Y = cell(1, m);
137 v = cell(1, m);
138 for j = 1:m
139 t = dec2bin(j - 1, d) - '0';
140 p = a .* (1 - t) + c .* t;
141 q = c .* (1 - t) + b .* t;
142 [Y{j}, v{j}] = obj.refine(n, p, q, i - 1);
143 end
144 X = [Y{:}];
145 w = [v{:}];
146 end
147 end
148 end
