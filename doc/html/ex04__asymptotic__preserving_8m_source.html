98function [config, state] = simulate(epsilon, tau0, tFinal)
99configPath = fullfile(fileparts(mfilename('fullpath')), 'config.txt');
100config = core.io.OptionParser().parse(configPath);
101config.nDims = 1; % Dimension
102config.k = 0.5; % Wave length
103config.L = 2*pi / config.k; % Length of domain
104config.xBBox = [0, config.L]; % Spatial bounding box
105config.vBBox = [-6, 6]; % Velocity bounding box
106config.tFinal = tFinal; % Final time
107config.T0 = 1; % Temperature parameter
108config.delta = 0.01; % Perturbation
109config.phiInf = phiInf(config.k); % Potential steady state
110config.EInf = EInf(config.k); % Electrical field steady state
111config.cInf = config.L / integral(@(x) exp(-config.phiInf(x)), 0, config.L); % Normalization parameter
112config.rhoInf = rhoInf(config.k, config.cInf); % Density steady state
113config.nx = repmat(64, 1, config.nDims); % Grid resolution
114config.nh = 80; % Number of Hermite modes - 1
115config.ic.D = DInit(config.k, config.delta, config.cInf, config.nh); % Initial hermite coefficients
116config.ic.omega = omegaInit(config.k, config.delta, config.cInf); % Initial potential
117config.bc = []; % Boundary condition
118config.exact = []; % Exact solution
119config.tau0 = tau0; % Collision parameter
120config.epsilon = epsilon; % Scaling parameter
121config.verbose = 1; % Verbose flag
122
123config.TID = upper(config.tOdeIntName);
124config.XID = sprintf('%s%d', upper(config.xDiscName), config.xBasisOrder);
125config.VID = sprintf('H%d', config.nh);
126config.EID = sprintf('AP%d', config.nDims); % Experiment ID
127config.SID = strjoin({config.XID, config.VID}, '-'); % Scheme ID
128
129config.visualizer = physics.visual.Visualizer();
130config.visualizer.setDensity(1);
131config.visualizer.setTimeline(config.tFinal, 1);
132config.visualizer.setComponents(struct('D', [], 'omega', []));
133config.visualizer.setTitleHead(sprintf('%s with %s', config.EID, config.SID));
134config.visualizer.addDataset(config.XID, 'numeric');
135config.visualizer.addPlotter('profile', '1d');
136
137scheme = physics.vlasov.DghsScheme(config);
138
139vElement = approx.element.C0OrthotopeElement(config.vBBox, ...
140 config.nh + 1, config.vBasisType, lambda=sqrt(config.T0));
141vDisc = approx.space.SpectralSpace(vElement);
142
143xElement = approx.element.L2OrthotopeElement(config.nDims, ...
144 config.xBasisOrder, config.xBasisType, config.xBasisPattern);
145xMesh = approx.mesh.UniformGrid(config.xBBox, config.nx);
146xDisc = approx.space.FiniteElementSpace(xMesh, xElement);
147
148state = physics.vlasov.HermiteState(xDisc, vDisc, config.T0, config.rhoInf);
149state = scheme.run(state);
150end

158function D = DInitImpl(k, delta, cInf, nh, x)
159rhoInf = rhoInfImpl(k, cInf, x);
160D0 = (rhoInf + delta * cos(k*x)) ./ rhoInf;
161D = [D0, zeros(1, size(x, 2)*nh)];
162end

168function omega = omegaInitImpl(k, delta, cInf, x)
169rhoInf = rhoInfImpl(k, cInf, x);
170omega = delta / k^2 * cos(k*x) .* sqrt(rhoInf);
171end
