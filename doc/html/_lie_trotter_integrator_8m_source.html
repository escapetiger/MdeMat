1classdef LieTrotterIntegrator < approx.odeint.SeparableOdeIntegrator
2 % LIETROTTERINTEGRATOR First-order Lie-Trotter operator splitting.
3 %
4 % LieTrotterIntegrator implements the first-order Lie-Trotter
5 % operator splitting method for solving ordinary differential
6 % equations of the form:
7 %
8 % \f[
9 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = \sum_{i=1}^{n} (L_i u +
10 % F_i(u) + S_i)
11 % \f]
12 %
13 % by decomposing it into n separate sub-problems:
14 % \f[
15 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = L_i u + F_i(u) + S_i, \quad
16 % i = 1, 2, \ldots, n
17 % \f]
18 %
19 % The Lie-Trotter method uses the splitting formula:
20 % \f[
21 % e^{(A_1+A_2+\cdots+A_n)\Delta t} \approx e^{A_1\Delta t}
22 % e^{A_2\Delta t} \cdots e^{A_n\Delta t}
23 % \f]
24 %
25 % This achieves first-order accuracy with global error \f$O(\Delta
26 % t)\f$. Each factor integrator solves its respective sub-problem
27 % using the most appropriate method (explicit, implicit, or IMEX).
28 %
29 % See also:
30 % approx.odeint.SeparableOdeIntegrator,
31 % approx.odeint.StrangIntegrator,
32 % approx.odeint.BeIntegrator,
33 % approx.odeint.FeIntegrator
34
35 properties (Constant)
36 Order = 1 % Accuracy order of the Lie-Trotter method
37 end

71 % STEPA Advance one time step using augmented Lie-Trotter

73 %
74 % U = stepA(obj, L, F, S, M, P, A, B, C) performs one time
75 % step by sequentially applying each factor integrator to its
76 % corresponding constrained sub-problem. The method
77 % decomposes the augmented differential-algebraic equation:
78 %
79 % \f[
80 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = \sum_{i=1}^{n} (L_i
81 % u + F_i(u) + P_i v + S_i), \quad 0 = \sum_{i=1}^{n} (A_i
82 % u + B_i v + C_i)
83 % \f]
84 %
85 % into n separable constrained sub-problems that are solved
86 % in sequence. Each factor integrator handles its respective
87 % constraint parameters @a P, @a A, @a B, and @a C through
