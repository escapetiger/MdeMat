1classdef ExrkIntegrator < approx.odeint.OdeIntegrator
2 % EXRKINTEGRATOR Explicit Runge-Kutta integrator.
3 %
4 % ExrkIntegrator implements explicit Runge-Kutta methods for solving
5 % ordinary differential equations of the form:
6 %
7 % \f[
8 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = Lu + F(u) + S,
9 % \f]
10 %
11 % where \f$M\f$ is a mass operator, \f$L\f$ is a linear operator,
12 % \f$F(u)\f$ is a nonlinear operator, and \f$S\f$ is a source term.
13 % Any of these terms can be empty (indicating they vanish from the
14 % equation). When M is empty, it defaults to identity. All non-empty
15 % terms are treated explicitly, making this suitable for non-stiff
16 % problems where stability constraints are manageable.
17 %
18 % Explicit RK methods have excellent stability properties for
19 % non-stiff problems but may require small time steps for stability
20 % when applied to stiff systems.
21 %
22 % See also:
23 % approx.odeint.OdeIntegrator, approx.odeint.DirkIntegrator,
24 % approx.odeint.FeIntegrator
25
26 properties
27 A % Butcher tableau coefficient matrix (strictly lower triangular)
28 b % Stage weights vector for final update
29 c % Stage time coefficients vector
30 K % Cell array of derivative approximations for each stage
31 end

81 function obj = setRhs(obj, i, M)
82 % SETRhs Set the right-hand side vector for stage i.
83 %
84 % obj = setRhs(obj, i, M) constructs the right-hand side
85 % vector using mass operator and accumulated previous stages.
86
87 arguments
88 obj approx.odeint.ExrkIntegrator
89 i {mustBePositive, mustBeInteger}
90 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
91 end
92
93 dt = obj.Timeline.StepSize;
94 U0 = obj.U0{1};
95
96 %< Add mass term contribution from previous solution
97 if isempty(M)
98 obj.Rhs{i} = U0;
99 elseif iscell(M)
100 obj.Rhs{i} = M{2} * U0;
101 if ~isempty(M{1})
102 obj.Rhs{i} = obj.Rhs{i} + M{1};
103 end
104 else
105 obj.Rhs{i} = M * U0;
106 end
107
108 %< Add accumulated term from previous stages
109 for j = 1:(i - 1)
110 obj.Rhs{i} = obj.Rhs{i} + dt * obj.A(i, j) * obj.K{j};
111 end
112 end

114 function U = stage(obj, i, L, F, S, M)
115 % STAGE Compute stage i of the explicit RK method.
116 %
117 % U = stage(obj, i, L, F, S, M) computes the solution at stage i
118 % by evaluating all terms explicitly and computing the derivative
119 % approximation for use in subsequent stages.
120
121 arguments
122 obj approx.odeint.ExrkIntegrator
123 i {mustBePositive, mustBeInteger}
124 L {mustBeA(L, {'numeric', 'function_handle'})} % Linear operator
125 F {mustBeA(F, {'numeric', 'function_handle'})} % Nonlinear operator
126 S {mustBeA(S, {'numeric', 'function_handle'})} % Source term
127 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
128 end
129
130 t0 = obj.Timeline.Now;
131 dt = obj.Timeline.StepSize;
132 U0 = obj.U0{1};
133
134 %< Evaluate linear term at initial time
135 Le = obj.evaluateOperator(L, U0, t0);
136
137 %< Evaluate source term at stage time
138 Se = obj.evaluateOperator(S, U0, t0+obj.c(i)*dt);
139
140 %< Update Lhs matrix if step size changed
141 if obj.Timeline.HasStepSizeChanged
142 %< Evaluate Lhs mass term
143 Me = obj.evaluateLhsMass(M, t0+obj.c(i)*dt);
144
145 %< Construct Lhs matrix
146 obj.setLhs(i, Me);
147 end
148
149 %< Evaluate Rhs mass term at previous time level
150 Me = obj.evaluateRhsMass(i, M, [t0+obj.c(i)*dt, t0]);
151
152 %< Construct Rhs vector
153 obj.setRhs(i, Me);
154
155 %< Solve for stage approximation
156 U = obj.Solver{i}.solve(obj.Lhs{i}, obj.Rhs{i});
157
158 %< Evaluate nonlinear term at stage approximation
159 Fe = obj.evaluateOperator(F, U, t0+obj.c(i)*dt);
160
161 %< Compute derivative approximation for this stage
162 obj.K{i} = 0;
163 if ~isempty(Le)
164 obj.K{i} = obj.K{i} + Le * U;
165 end
166 if ~isempty(Fe)
167 obj.K{i} = obj.K{i} + Fe;
168 end
169 if ~isempty(Se)
170 obj.K{i} = obj.K{i} + Se;
171 end
172 end

174 function U = step(obj, L, F, S, M)
175 % STEP Advance one time step using the explicit RK method.
176 %
177 % U = step(obj, L, F, S, M) performs a complete time step by
178 % evaluating all stages explicitly and combining them according
179 % to the Butcher tableau weights. Empty terms contribute zero
180 % to all stages.
181
182 arguments
183 obj approx.odeint.ExrkIntegrator
184 L {mustBeA(L, {'numeric', 'function_handle'})} % Linear operator
185 F {mustBeA(F, {'numeric', 'function_handle'})} % Nonlinear operator
186 S {mustBeA(S, {'numeric', 'function_handle'})} % Source term
187 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
188 end
189
190 t0 = obj.Timeline.Now;
191 dt = obj.Timeline.StepSize;
192 U0 = obj.U0{1};
193 s = obj.NStages;
194
195 %< Compute all stage approximations
196 for i = 1:s
197 [~] = obj.stage(i, L, F, S, M);
198 end
199
200 %< Update Lhs matrix for final solve if step size changed
201 if obj.Timeline.HasStepSizeChanged
202 %< Evaluate Lhs mass term
203 Me = obj.evaluateLhsMass(M, t0+dt);
204
205 %< Construct Lhs matrix
206 obj.setLhs(s+1, Me);
207 end
208
209 %< Evaluate mass term for final solve
210 Me = obj.evaluateRhsMass(s+1, M, [t0+dt, t0]);
211
212 %< Add mass term contribution from previous solution
213 if isempty(Me)
214 obj.Rhs{s + 1} = U0;
215 elseif iscell(Me)
216 obj.Rhs{s + 1} = Me{2} * U0;
217 if ~isempty(Me{1})
218 obj.Rhs{s + 1} = obj.Rhs{s + 1} + Me{1};
219 end
220 else
221 obj.Rhs{s + 1} = Me * U0;
222 end
223
224 %< Add weighted stage derivatives
225 for i = 1:s
226 obj.Rhs{s + 1} = obj.Rhs{s + 1} + dt * obj.b(i) * obj.K{i};
227 end
228
229 %< Solve for final step approximation
230 U = obj.Solver{s + 1}.solve(obj.Lhs{s + 1}, obj.Rhs{s + 1});
231 end
232 end
