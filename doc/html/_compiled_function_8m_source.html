161 % EVALUATEIMPL Implementation of function evaluation.
162 %
163 % Y = evaluateImpl(obj, X) evaluates the compiled function
164 % at the specified points using the loaded metadata handles.
165 %
166 % Inputs:
167 % obj - The CompiledFunction object
168 % X - Coordinates (nDims x nPoints)
169 %
170 % Outputs:
171 % Y - Function values (nCodims x nPoints)
172
173 core.except.assert(obj.hasMetadata, ...
174 'NoMetadata', 'No compiled metadata is loaded.');
175
176 handles = obj.metadata.handles;
177 Y = zeros(obj.nCodims, size(X, 2));

179 Z = mat2cell(X, ones(1, size(X, 1)), size(X, 2));
180
181 for j = 1:size(handles, 1)
182 if ~isempty(handles{j, 1})
183 Y(j, :) = handles{j, 1}(Z{:});

188 function dY = derivativeImpl(obj, X, r)
189 % DERIVATIVEIMPL Implementation of function derivative
190 % evaluation.
191 %
192 % dY = derivativeImpl(obj, X, r) computes derivatives of the
193 % compiled function at the specified points using precompiled
194 % derivative handles from the loaded metadata.
195 %
196 % Inputs:
197 % obj - The DifferentiableCompiledFunction object
198 % X - Coordinates (nDims x nPoints)
199 % r - Derivative order vector (1 x nDims)
200 %
201 % Outputs:
202 % dY - Derivative values (nCodims x nPoints)
203
204 core.except.assert(obj.hasMetadata, ...
205 'NoMetadata', 'No compiled metadata is loaded.');
206
207 handles = obj.metadata.handles;
208
209 maxOrder = size(handles, 2) - 1;
210
211 derivativeOrder = sum(r);
212
213 core.except.assert(derivativeOrder <= maxOrder, ...
214 'InvalidInput', ...
215 'Derivative order %d exceeds maximum available order %d.', ...

217
218 columnIndex = derivativeOrder + 1;
219 dY = zeros(obj.nCodims, size(X, 2));
220
221 for j = 1:size(handles, 1)
222 if ~isempty(handles{j, columnIndex})
223 dY(j, :) = handles{j, columnIndex}(X);
