1classdef Mesh < handle
2 % MESH Base class for computational meshes.
3 %
4 % Mesh provides foundational functionality for computational meshes
5 % in numerical approximation methods. It defines common interfaces
6 % for coordinate transformations, mesh refinement, and geometric
7 % computations while supporting efficient caching of expensive
8 % operations.
9 %
10 % The class manages coordinate transformations between reference and
11 % physical coordinate systems, computes Jacobian matrices and their
12 % determinants for integration quadrature, and provides mesh
13 % refinement capabilities for adaptive algorithms.
14 %
15 % Transformation data is cached using bitwise flags to avoid
16 % redundant computations:
17 % - 0b00001: Outward normals computed
18 % - 0b00010: Face Jacobian determinants computed
19 % - 0b00100: Element inverse Jacobians computed
20 % - 0b01000: Element Jacobian determinants computed
21 % - 0b10000: Element Jacobians computed
22 %
23 % See also:
24 % approx.mesh.Grid, approx.mesh.PolytopalMesh
25

127 function EJac = computeElementJacobians(obj, EI)
128 % COMPUTEELEMENTJACOBIANS Compute element Jacobian matrices.
129 %
130 % EJac = computeElementJacobians(obj) computes Jacobian
131 % matrices for coordinate transformation from reference to
132 % physical coordinates for all elements.
133 %
134 % EJac = computeElementJacobians(obj, EI) computes Jacobian
135 % matrices only for the specified elements.
136 %
137 % The Jacobian matrix \f$J_{ij} = \frac{\partial
138 % x_i}{\partial \xi_j}\f$ where \f$x\f$ are physical
139 % coordinates and \f$\xi\f$ are reference coordinates. For
140 % coordinate transformations: \f$dx = J d\xi\f$.
141
142 arguments
143 obj approx.mesh.Mesh
144 EI {mustBeInteger} = []
145 end
146
147 if ~bitand(obj.Status, obj.EJacMask)
148 obj.Transform.EJac = obj.computeAllElementJacobians();
149 obj.Status = bitor(obj.Status, obj.EJacMask);
150 end
151
152 if isempty(EI)
153 EJac = obj.Transform.EJac;
154 else
155 EJac = obj.Transform.EJac(:, :, EI);
156 end
157 end

159 function detEJac = computeElementJacobianDeterminants(obj, EI)
160 % COMPUTEELEMENTJACOBIANDETERMINANTS Compute element Jacobian
161 % determinants.
162 %
163 % detEJac = computeElementJacobianDeterminants(obj) computes
164 % Jacobian determinants for coordinate transformation from
165 % reference to physical coordinates for all elements.
166 %
167 % detEJac = computeElementJacobianDeterminants(obj, EI)
168 % computes Jacobian determinants only for the specified
169 % elements.
170 %
171 % The Jacobian determinant appears in integration formulas:
172 %
173 % \f[
174 % \int_{\Omega} f(x) dx = \int_{\hat{\Omega}} f(F(\xi))
175 % |det(J)| d\xi
176 % \f]
177 %
178 % where \f$F\f$ is the coordinate transformation and
179 % \f$\hat{\Omega}\f$ is the reference element.
180
181 arguments
182 obj approx.mesh.Mesh
183 EI {mustBeInteger} = []
184 end
185
186 if ~bitand(obj.Status, obj.DetEJacMask)
187 obj.Transform.detEJac = obj.computeAllElementJacobianDeterminants();
188 obj.Status = bitor(obj.Status, obj.DetEJacMask);
189 end
190
191 if isempty(EI) || isscalar(obj.Transform.detEJac)
192 detEJac = obj.Transform.detEJac;
193 else
194 detEJac = obj.Transform.detEJac(EI);
195 end
196 end
