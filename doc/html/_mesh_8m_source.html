1classdef Mesh < handle
2 % MESH Base class for computational meshes.
3 %
4 % Mesh provides foundational functionality for computational meshes
5 % in numerical approximation methods. It defines common interfaces
6 % for coordinate transformations, mesh refinement, and geometric
7 % computations while supporting efficient caching of expensive
8 % operations.
9 %
10 % The class manages coordinate transformations between reference and
11 % physical coordinate systems, computes Jacobian matrices and their
12 % determinants for integration quadrature, and provides mesh
13 % refinement capabilities for adaptive algorithms.
14 %
15 % Transformation data is cached using bitwise flags to avoid
16 % redundant computations:
17 % - 0b00001: Outward normals computed
18 % - 0b00010: Face Jacobian determinants computed
19 % - 0b00100: Element inverse Jacobians computed
20 % - 0b01000: Element Jacobian determinants computed
21 % - 0b10000: Element Jacobians computed
22 %
23 % See also:
24 % core.geometry.Geometry

73 function EJ = computeElementJacobians(obj, IE)
74 % COMPUTEELEMENTJACOBIANS Compute element Jacobian matrices.
75 %
76 % EJ = computeElementJacobians(obj) computes Jacobian
77 % matrices for coordinate transformation from reference to
78 % physical coordinates for all elements.
79 %
80 % EJ = computeElementJacobians(obj, IE) computes Jacobian
81 % matrices only for the specified elements.
82 %
83 % Inputs:
84 % obj - The Mesh object
85 % IE - Element linear indices (optional, vector)
86 %
87 % Outputs:
88 % EJ - Jacobian matrices (nDims × nDims × nElements)
89 %
90 % Notes:
91 % The Jacobian matrix \f$J_{ij} = \frac{\partial
92 % x_i}{\partial\xi_j}\f$ where \f$x\f$ are physical
93 % coordinates and \f$\xi\f$ are reference coordinates. For
94 % coordinate transformations: \f$dx = J d\xi\f$.
95
96 %< Check if Jacobians are already computed
97 if ~bitand(obj.transform.flag, 0b10000)
98 obj.transform.EJ = obj.computeAllElementJacobians();
99 obj.transform.flag = bitor(obj.transform.flag, 0b10000);
100 end
101
102 %< Return all or subset of Jacobians
103 if nargin < 2 || isempty(IE)
104 EJ = obj.transform.EJ;
105 else
106 EJ = obj.transform.EJ(:, :, IE);
107 end
108 end

110 function detEJ = computeElementJacobianDeterminants(obj, IE)
111 % COMPUTEELEMENTJACOBIANDETERMINANTS Compute element Jacobian
112 % determinants.
113 %
114 % detEJ = computeElementJacobianDeterminants(obj) computes
115 % Jacobian determinants for coordinate transformation from
116 % reference to physical coordinates for all elements.
117 %
118 % detEJ = computeElementJacobianDeterminants(obj, IE)
119 % computes Jacobian determinants only for the specified
120 % elements.
121 %
122 % Inputs:
123 % obj - The Mesh object
124 % IE - Element linear indices (optional, vector)
125 %
126 % Outputs:
127 % detEJ - Element Jacobian determinants (vector)
128 %
129 % Notes:
130 % The Jacobian determinant appears in integration formulas:
131 %
132 % \f[
133 % \int_{\Omega} f(x) dx = \int_{\hat{\Omega}} f(F(\xi))
134 % |det(J)| d\xi
135 % \f]
136 %
137 % where \f$F\f$ is the coordinate transformation and
138 % \f$\partial\hat{Ω̂}\f$ is the reference element.
139
140 %< Check if determinants are already computed
141 if ~bitand(obj.transform.flag, 0b01000)
142 obj.transform.detEJ = obj.computeAllElementJacobianDeterminants();
143 obj.transform.flag = bitor(obj.transform.flag, 0b01000);
144 end
145
146 %< Return all or subset of determinants
147 if nargin < 2 || isempty(IE)
148 detEJ = obj.transform.detEJ;
149 else
150 detEJ = obj.transform.detEJ(IE);
151 end
152 end

197 function detFJ = computeFaceJacobianDeterminants(obj, IE, IFE)
198 % COMPUTEFACEJACOBIANDETERMINANTS Compute face Jacobian
199 % determinants.
200 %
201 % detFJ = computeFaceJacobianDeterminants(obj, IE, IFE)
202 % computes Jacobian determinants for coordinate transformation
203 % on element faces for integration over boundaries.
204 %
205 % Inputs:
206 % obj - The Mesh object
207 % IE - Element linear indices (vector or [])
208 % IFE - Local face index (scalar, 1 to nFaces)
209 %
210 % Outputs:
211 % detFJ - Face Jacobian determinants (vector)
212 %
213 % Notes:
214 % Face Jacobian determinants are used in boundary
215 % integration:
216 %
217 % \f[
218 % \int_{\partial\Omega} f(x) dS =
219 % \int_{\partial\hat{\Omega}} f(F(\xi)) |det(J_F)| d\xi
220 % \f]
221 %
222 % where \f$\partial\Omega\f$ is the boundary and \f$J_F\f$ is
223 % the face Jacobian.
224
225 %< Check if face Jacobians are already computed
226 if ~bitand(obj.transform.flag, 0b00010)
227 obj.transform.detFJ = obj.computeAllFaceJacobianDeterminants();
228 obj.transform.flag = bitor(obj.transform.flag, 0b00010);
229 end
230
231 %< Return all or subset of face determinants
232 if isempty(IE)
233 detFJ = obj.transform.detFJ{IFE};
234 else
235 detFJ = obj.transform.detFJ{IFE}(IE);
236 end
237 end
