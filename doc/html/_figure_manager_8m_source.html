147 function position = calculateNonOverlappingPosition(obj, figureIndex, layoutDims)
148 % CALCULATENONOVERLAPPINGPOSITION Calculate position without overlap.
149 %
150 % position = calculateNonOverlappingPosition(obj, figureIndex, layoutDims)
151 % calculates the screen position for the figure to ensure no
152 % overlap with other figures in the layout.
153
154 nRows = layoutDims(1);
155 nCols = layoutDims(2);
156
157 % Convert linear index to grid coordinates
158 row = floor((figureIndex - 1) / nCols) + 1;
159 col = mod(figureIndex - 1, nCols) + 1;
160
161 % Use default figure size
162 figWidth = obj.FigureSize(1);
163 figHeight = obj.FigureSize(2);
164
165 % Calculate total grid dimensions
166 totalGridWidth = nCols * figWidth + (nCols - 1) * obj.FigureSpacing;
167 totalGridHeight = nRows * figHeight + (nRows - 1) * obj.FigureSpacing;
168
169 % Check if grid fits on screen, if not, scale down
170 availableWidth = obj.ScreenSize(1) - 2 * obj.ScreenMargin;
171 availableHeight = obj.ScreenSize(2) - 2 * obj.ScreenMargin;
172
173 if totalGridWidth > availableWidth || totalGridHeight > availableHeight
174 % Scale down figure size to fit
175 scaleWidth = availableWidth / totalGridWidth;
176 scaleHeight = availableHeight / totalGridHeight;
177 scale = min(scaleWidth, scaleHeight);
178
179 figWidth = floor(figWidth * scale);
180 figHeight = floor(figHeight * scale);
181
182 % Recalculate grid dimensions with scaled figures
183 totalGridWidth = nCols * figWidth + (nCols - 1) * obj.FigureSpacing;
184 totalGridHeight = nRows * figHeight + (nRows - 1) * obj.FigureSpacing;
185 end
186
187 % Center the entire grid on screen
188 gridStartX = (obj.ScreenSize(1) - totalGridWidth) / 2;
189 gridStartY = (obj.ScreenSize(2) - totalGridHeight) / 2;
190
191 % Calculate figure position within grid
192 left = gridStartX + (col - 1) * (figWidth + obj.FigureSpacing);
193 bottom = gridStartY + (nRows - row) * (figHeight + obj.FigureSpacing);
194
195 position = [left, bottom, figWidth, figHeight];
196 end
197 end

210 function layoutDims = calculateLayout(nFigures)
211 % CALCULATELAYOUT Determine optimal layout dimensions.
212 %
213 % layoutDims = calculateLayout(nFigures) calculates
214 % the optimal number of rows and columns for arranging
215 % figures without overlap.
216
217 if nFigures == 1
218 layoutDims = [1, 1];
219 elseif nFigures == 2
220 layoutDims = [1, 2]; % Side by side
221 elseif nFigures <= 4
222 layoutDims = [2, 2]; % 2x2 grid
223 elseif nFigures <= 6
224 layoutDims = [2, 3]; % 2x3 grid
225 elseif nFigures <= 9
226 layoutDims = [3, 3]; % 3x3 grid
227 else
228 % For larger numbers, approximate square layout
229 nCols = ceil(sqrt(nFigures));
230 nRows = ceil(nFigures / nCols);
231 layoutDims = [nRows, nCols];
232 end
233 end
234 end
