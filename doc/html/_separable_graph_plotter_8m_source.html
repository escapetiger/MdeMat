58 % PLOTTENSORDIRECT Direct visualization using tensor structure.
59 %
60 % plotTensorDirect(obj, graph, options) visualizes the
61 % separable graph directly using its tensor-product structure

95 function plotWithSubsampling(obj, graph, options)
96 % PLOTWITHSUBSAMPLING Visualize large graphs using subsampling.
97 %
98 % plotWithSubsampling(obj, graph, options) creates a
99 % subsampled version of the graph for visualization when
100 % the total number of vertices exceeds the threshold.
101
102 %< Calculate subsampling factor
103 subsampleFactor = ceil(sqrt(graph.NTotalVertices/options.MaxVertices));
104
105 %< Create subsampled factors
106 subsampledVertices = cell(1, graph.NDims);
107 subsampledEdges = cell(1, graph.NDims);
108
109 for d = 1:graph.NDims
110 vertices = graph.Vertices{d};
111 edges = graph.Edges{d};
112
113 %< Subsample vertices uniformly
114 n = length(vertices);
115 indices = 1:subsampleFactor:n;
116 subsampledVertices{d} = vertices(indices);
117
118 %< Adjust edges for subsampled vertices
119 if ~isempty(edges)
120 %< Map original edge indices to subsampled indices
121 [~, newV1] = ismember(edges(:, 1), indices);
122 [~, newV2] = ismember(edges(:, 2), indices);
123
124 %< Keep only edges where both vertices are retained
125 validEdges = (newV1 > 0) & (newV2 > 0) & (newV1 ~= newV2);
126 subsampledEdges{d} = [newV1(validEdges), newV2(validEdges)];
127 else
128 subsampledEdges{d} = [];
129 end
130 end
131
132 %< Create subsampled graph and plot
133 subsampledGraph = approx.mesh.SeparableGraph(subsampledVertices, subsampledEdges);
134 obj.plotTensorDirect(subsampledGraph, options);
135
136 %< Update title to indicate subsampling
137 currentTitle = get(get(gca, 'Title'), 'String');
138 newTitle = sprintf('%s (subsampled by factor %d)', currentTitle, subsampleFactor);
139 title(newTitle);
140 end

142 function plotWithProjection(obj, graph, options)
143 % PLOTWITHPROJECTION Visualize high-dimensional graphs using
144 % projection.
145 %
146 % plotWithProjection(obj, graph, options) creates a
147 % lower-dimensional projection of high-dimensional separable
148 % graphs for visualization.
149
150
151 %< Validate and adjust projection dimensions
152 projDims = options.ProjectionDims;
153 projDims = projDims(projDims <= graph.NDims);
154 projDims = projDims(1:min(3, length(projDims)));
155
156 if isempty(projDims)
157 projDims = 1:min(3, graph.NDims);
158 end
159
160 %< Create projected graph factors
161 projVertices = graph.Vertices(projDims);
162 projEdges = graph.Edges(projDims);
163
164 %< Create and plot projected graph
165 projGraph = approx.mesh.SeparableGraph(projVertices, projEdges);
166 obj.plotTensorDirect(projGraph, options);
167
168 %< Update title to indicate projection
169 currentTitle = get(get(gca, 'Title'), 'String');
170 newTitle = sprintf('%s (projected onto dims %s)', ...
171 currentTitle, mat2str(projDims));
172 title(newTitle);
173 end
174 end

178function plotTensor1D(graph, options)
179% PLOTTENSOR1D Plot 1D separable graph.
180%
181% plotTensor1D(obj, graph, options) creates a 1D
182% visualization by plotting vertices along the x-axis.
183
184vertices = graph.Vertices{1};
185edges = graph.Edges{1};
186
187%< Plot vertices
188if options.ShowVertices && ~isequal(options.VertexColor, 'none')
189 scatter(vertices, zeros(size(vertices)), ...
190 options.VertexSize, options.VertexColor, 'filled');
191end
192
193%< Plot edges
194if options.ShowEdges && ~isequal(options.EdgeColor, 'none') && ~isempty(edges)
195 for iEdge = 1:size(edges, 1)
196 v1 = edges(iEdge, 1);
197 v2 = edges(iEdge, 2);
198 plot([vertices(v1), vertices(v2)], [0, 0], '-', ...
199 'Color', options.EdgeColor, 'LineWidth', options.EdgeWidth);
200 end
201end
202
203%< Set axis properties
204ylim([-0.5, 0.5]);
205xlabel('X');
206end
