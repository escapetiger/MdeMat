145 function e = reduce(obj, R, reduction)
146 % REDUCE Apply error reduction over the computational domain.
147
148 n = obj.Space.NMeshElements;
149 h = obj.Space.Mesh.computeElementJacobianDeterminants();
150 w = obj.Integrator.Weights;
151 switch reduction
152 case 'L1'
153 e = w * R;
154 e = reshape(e, n, []);
155 e = e .* h(:);
156 e = sum(e, 1);
157 case 'L2'
158 e = w * R.^2;
159 e = reshape(e, n, []);
160 e = e .* h(:);
161 e = sqrt(sum(e, 1));
162 case 'max'
163 e = max(R, [], 1);
164 e = reshape(e, n, []);
165 e = max(e, [], 1);
166 otherwise
167 e = R;
168 end
169 e = e(:).';
170 end
171 end
