2 % FINITEELEMENTSEMILAGRANGIANASSEMBLY Assembly for semi-Lagrangian finite element methods.
3 %
4 % FiniteElementSemiLagrangianAssembly implements assembly operations for
5 % semi-Lagrangian finite element discretizations of transport equations.
6 % The semi-Lagrangian method follows characteristic curves backward in time
7 % to solve advection-dominated problems with excellent stability properties.
8 %
9 % The assembly process involves clipping volume elements based on the
10 % advection field and time step to handle characteristics that may exit
11 % the computational domain. Status flags track whether clipping has been
12 % performed for matrix and vector assembly operations.
13 %
14 % This implementation is specialized for L2 semi-Lagrangian orthotope
15 % elements on uniform grids, providing optimal efficiency for structured
16 % transport problems.
17 %
18 % The assembly requires L2SLOrthotopeElement and UniformGrid for proper
19 % functioning. Status flags use bitwise operations to track clipping state.
20 %
21 % See also:
22 % FiniteElementAssembly, L2SLOrthotopeElement, UniformGrid
23

85 function G = assembleMatrix(obj, field, tBegin, tEnd)
86 % ASSEMBLEMATRIX Assemble semi-Lagrangian transport matrix.
87 %
88 % G = assembleMatrix(obj, field, tBegin, tEnd) assembles the mass
89 % matrix for semi-Lagrangian transport with advection @a field
90 % from time @a tBegin to @a tEnd. The method clips volume elements
91 % based on characteristic trajectories and assembles contributions
92 % from Eulerian and upstream integration points.
93 %
94 % The time step \f$ \Delta t = tBegin - tEnd \f$ and spatial
95 % discretization determine the clipping parameters. Status flags
96 % are updated to track clipping operations.
97
98 arguments
99 obj approx.assembly.FiniteElementSemiLagrangianAssembly
100 field {mustBeNumeric}
101 tBegin {mustBeNumeric}
102 tEnd {mustBeNumeric}
103 end
104
105 nd = obj.Space.NDims;
106
107 % Parse advection field
108 field = reshape(field, nd, []);
109
110 % Clipping
111 ht = tBegin - tEnd;
112 hx = cellfun(@(h) h(1), obj.Space.Mesh.Spacings);
113 obj.Space.Element.clipVolume(field, ht, hx);
114
115 % Update status
116 obj.Status = bitor(obj.Status, obj.ClipMatrixMask);
117
118 % Assemble global mass matrix
119 switch obj.BcType
120 case 'periodic'
121 G = obj.assembleMatrixPeriodic();
122 case 'dirichlet'
123 G = obj.assembleMatrixDirichlet();
124 end
125 end

174 function G = assembleMatrixPeriodic(obj)
175 % ASSEMBLEMATRIXPERIODIC Assemble matrix for periodic boundary conditions.
176
177 G = obj.createZeros();
178 for i = 1:obj.Space.Element.NVolumePieces
179 %< Compute local mass matrix
180 W = diag(obj.Space.Element.VolumePieces.eulerian(i).Weights); % (nq, nq)
181 V = obj.Space.Element.VolumePieces.eulerian(i).Values; % (nl, nq)
182 U = obj.Space.Element.VolumePieces.upstream(i).Values; % (nl, nq)
183 L = V * W * U.'; % nl x nl
184
185 %< Add contribution to global mass matrix
186 IE = 1:obj.Space.NMeshElements;
187 ME = obj.Space.Mesh.Indexer.linearToMulti(IE);
188 ME = ME + obj.Space.Element.VolumePieces.indexShifts(:, i).';
189 JE = obj.Space.Mesh.Indexer.multiToLinear(ME, pad='wrap');
190 GP = obj.createMatrix(L, TEI=IE, SEI=JE);
191 G = G + GP;
192 end
193 end

195 function G = assembleMatrixDirichlet(obj)
196 % ASSEMBLEMATRIXDIRICHLET Assemble matrix for Dirichlet boundary conditions.
197
198 d = obj.Space.NDims;
199 G = obj.createZeros();
200 for i = 1:obj.Space.Element.NVolumePieces
201 %< Compute local mass matrix
202 W = diag(obj.Space.Element.VolumePieces.eulerian(i).Weights); % (nq, nq)
203 V = obj.Space.Element.VolumePieces.eulerian(i).Values; % (nl, nq)
204 U = obj.Space.Element.VolumePieces.upstream(i).Values; % (nl, nq)
205 L = V * W * U.'; % (nl, nl)
206
207 %< Find elements that have at least one interior upstream root
208 ME = obj.Space.Mesh.Indexer.generate();
209 SE = obj.Space.Element.VolumePieces.indexShifts(:, i).';
210 MU = ME + SE;
211 mask1 = (MU < 1) .* (2 * (1:d) - 1);
212 mask2 = (MU > obj.Space.Mesh.Resolution) .* (2 * (1:d));
213 mask = mask1 + mask2;
214 mask = sum(mask ~= 0, 2) == 0;
215 ME = ME(mask, :);
216
217 %< Add contribution to global mass matrix
218 IE = obj.Space.Mesh.Indexer.multiToLinear(ME, pad='empty');
219 JE = obj.Space.Mesh.Indexer.multiToLinear(ME+SE, pad='empty');
220 GP = obj.createMatrix(L, TEI=IE, SEI=JE);
221 G = G + GP;
222 end
223 end

225 function G = assembleVectorDirichlet(obj, field, tBegin, tEnd, func)
226 % ASSEMBLEVECTORDIRICHLET Assemble Dirichlet boundary vector.
227
228 np = obj.Space.Element.NVolumePieces;
229 nd = obj.Space.NDims;
230 ht = tBegin - tEnd;
231
232 G = obj.createZeros(nCols=1);
233 for i = 1:np
234 %< Find elements whose pth volume piece exits domain
235 ME = obj.Space.Mesh.Indexer.generate();
236 SE = obj.Space.Element.VolumePieces.indexShifts(:, i).';
237 MU = ME + SE;
238 mask1 = (MU < 1) .* (2 * (1:nd) - 1);
239 mask2 = (MU > obj.Space.Mesh.Resolution) .* (2 * (1:nd));
240 mask = mask1 + mask2;
241 mask = sum(mask ~= 0, 2) > 0;
242 ME = ME(mask, :);
243 IE = obj.Space.Mesh.Indexer.multiToLinear(ME, pad='empty');
244 ne = size(ME, 1);
245
246 if ne == 0, return; end
247
248 %< Set up characteristic end points
249 nq = obj.Space.Element.VolumePieces.eulerian(i).NPoints;
250 X1 = zeros(nd+1, ne*nq);
251 X2 = zeros(nd+1, ne*nq);
252 Xq = obj.Space.Element.VolumePieces.eulerian(i).Nodes;
253 X2(1:nd, :) = obj.Space.Mesh.collocate(Xq, IE);
254 X2(nd+1, :) = tBegin;
255 X1(1:nd, :) = X2(1:nd, :) - ht * field(:);
256 X1(nd+1, :) = tEnd;
257
258 %< Find boundary intersections of characteristics
259 intersector = core.geometry.LineSegmentByPlaneIntersector(nd+1);
260 bbox = reshape(obj.Space.Mesh.BBox, 2, []);
261 bbox = -bbox(2*(1:nd)-(field > 0));
262 Xb = zeros(nd+1, nq*ne);
263 for d = 1:nd
264 a = zeros(nd+1, 1);
265 a(d) = 1;
266 [X, TF] = intersector.intersect(X1, X2, a, bbox(d));
267 Xb(:, TF) = X;
268 end
269
270 %< Evaluate boundary function and project to element space
271 V = obj.Space.Element.VolumePieces.eulerian(i).Values; % (nl, nq)
272 W = diag(obj.Space.Element.VolumePieces.eulerian(i).Weights); % (nq, nq)
273 U = func(Xb(1:nd, :), t=Xb(nd+1, :)); % (1, nq*ne)
274 U = reshape(U, nq, ne);
275 L = V * W * U; % (nl, ne)
276 GP = obj.createVector(L, TEI=IE);
277 G = G + GP;
278 end
279 end
280 end
