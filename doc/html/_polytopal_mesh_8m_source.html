1classdef PolytopalMesh < approx.mesh.Mesh
2 % POLYTOPALMESH Base class for polytopal meshes.
3 %
4 % PolytopalMesh provides a foundational representation for polytopal
5 % meshes consisting of vertices, faces, and elements with explicit
6 % connectivity information. It serves as the base class for
7 % unstructured mesh types and integrates seamlessly with the
8 % Graph-based mesh infrastructure for visualization and analysis.
9 %
10 % The mesh topology follows standard finite element conventions:
11 % - In 1D: elements are line segments, faces are points (vertices)
12 % - In 2D: elements are polygons, faces are edges
13 % - In 3D: elements are polyhedra, faces are polygons
14 % - In nD: elements are n-polytopes, faces are (n-1)-polytopes
15 %
16 % Connectivity is stored using explicit arrays and sparse tables:
17 % - vertices: Coordinate matrix (nVertices × nDims)
18 % - elements: Connectivity matrix (nElements × nVerticesPerElement)
19 % - faces: Connectivity matrix (nFaces × nVerticesPerFace)
20 % - faceToElementTable: Sparse connectivity (nFaces × nElements)
21 %
22 % The class supports periodic boundary conditions through
23 % vertex-to-vertex and face-to-face mapping tables that identify
24 % geometrically equivalent entities across periodic boundaries.
25 %
26 % See also:
27 % approx.mesh.Mesh, approx.mesh.Graph, approx.mesh.PolytopalMeshPlotter
28

106 function obj = setPeriodic(obj, bbox, tol)
107 % SETPERIODIC Establish periodic boundary mappings.
108 %
109 % obj = setPeriodic(obj, bbox) builds vertex-to-vertex and
110 % face-to-face connectivity tables to represent periodic
111 % mappings between geometrically equivalent mesh entities.
112 %
113 % obj = setPeriodic(obj, bbox, tol) uses the specified
114 % tolerance for identifying equivalent vertices.
115
116 arguments
117 obj approx.mesh.PolytopalMesh
118 bbox {mustBeNumeric}
119 tol {mustBeNumeric} = 1e-12
120 end
121
122 bbox = reshape(bbox, 2, []);
123 nDims = size(bbox, 2);
124 shift = cell(1, nDims);
125 [shift{:}] = ndgrid(-1:1);
126 shift = cellfun(@(x) x(:), shift, 'Un', 0);
127 shift = [shift{:}];
128 shift = shift(any(shift ~= 0, 2), :);
129 shift = shift .* diff(bbox, 1, 1);
130
131 obj.setVertexToVertexTables(shift, tol);
132 obj.setFaceToFaceTable();
133 end

221 % FINDNEIGHBORELEMENTS Find neighboring elements across faces.
222 %
223 % [NEI, NLFI] = findNeighborElements(obj, EI, LFI, BC) computes
224 % the linear indices of neighbor elements sharing the LFI-th
225 % face of prime elements EI, with the specified boundary
226 % condition treatment.
227
228 arguments
229 obj approx.mesh.PolytopalMesh
230 EI {mustBeInteger} = []
231 LFI {mustBeInteger, mustBePositive} = 1
232 BC {mustBeMember(BC, {'', 'periodic', 'dirichlet'})} = ''
233 end
234
235 E = obj.Elements.';
236 T = nchoosek(1:obj.NVerticesPerElement, obj.NVerticesPerFace).';
237 F = E(T(:, LFI), EI);
238 F = sort(F, 1).';
239 [~, I] = ismember(F, obj.Faces, 'rows');
240
241 switch lower(BC)
242 case 'periodic'
243 P = any(obj.FaceToFaceTable(I, :), 2);
244 I(P) = find(obj.FaceToFaceTable(I(P), :), 1, 'first');

246 I = setdiff(I, obj.Boundary);
247 end

249 R = obj.FaceToElementTable(I, :);
250 R(:, EI) = 0;
251 [NLFI, NEI] = max(R, [], 2);
252 NEI(NLFI == 0) = 0;

295 function obj = setVertexToVertexTables(obj, shift, tol)
296 % SETVERTEXTOVERTEXTABLES Build vertex-to-vertex mapping table.
297
298 n = obj.NVertices;
299 V = obj.Vertices;
300
301 obj.VertexToVertexTables = cell(1, size(shift, 1));
302 for k = 1:size(shift, 1)
303 D = pdist2(V, V-shift(k, :));
304 [I, J] = find(D < tol);
305 Tk = sparse(I, J, 1, n, n);
306 obj.VertexToVertexTables{k} = spones(Tk);
307 end

310 function obj = setFaceToFaceTable(obj)
311 % SETFACETOFACETABLES Build face-to-face mapping table.
312
313 F = obj.Faces; % nF × m (m = NVerticesPerFace)
314 nF = obj.NFaces;
315 m = obj.NVerticesPerFace;
316
317 %< Canonical representation of existing faces
318 sortedFaces = sort(F, 2);
319
320 faceToFace = sparse(nF, nF);
321
322 %< Loop only over small number of shifts (8 in 2D, 26 in 3D)
323 for s = 1:length(obj.VertexToVertexTables)
324 Ts = obj.VertexToVertexTables{s};
325
326 %< Map every face's k-th vertex via this shift, in bulk
327 mappedFaces = zeros(nF, m);
328 for k = 1:m
329 Mk = Ts(F(:, k), :);
330 [val, idx] = max(Mk, [], 2);
331 idx(~val) = 0;
332 mappedFaces(:, k) = idx;
333 end
334
335 %< Keep only faces whose all vertices map under this shift
336 valid = all(mappedFaces > 0, 2);
337 if ~any(valid), continue; end
338
339 %< Find matches in the original face set
340 sortedMapped = sort(mappedFaces(valid, :), 2);
341 [isMatch, loc] = ismember(sortedMapped, sortedFaces, 'rows');
342
343 if any(isMatch)
344 iFaces = find(valid);
345 iFaces = iFaces(isMatch);
346 jFaces = loc(isMatch);
347
348 %< Add symmetric connections
349 T = sparse(iFaces, jFaces, 1, nF, nF);
350 faceToFace = faceToFace + T + T.';
351 end
352 end
353
354 obj.FaceToFaceTable = spones(faceToFace);
355 end
356 end
