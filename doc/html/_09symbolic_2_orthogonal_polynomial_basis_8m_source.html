2 % ORTHOGONALPOLYNOMIALBASIS Abstract base class for orthogonal
3 % polynomial basis functions.
4 %
5 % OrthogonalPolynomialBasis provides a unified framework for
6 % generating orthogonal polynomial basis functions using three-term
7 % recurrence relations. Concrete subclasses implement specific
8 % polynomial families such as Legendre, Chebyshev, and Hermite
9 % polynomials.
10 %
11 % The three-term recurrence relation has the general form:
12 %
13 % \f[
14 % P_{n+1}(x) = (A_n \cdot x + B_n) \cdot P_n(x) + C_n \cdot
15 % P_{n-1}(x)
16 % \f]
17 %
18 % See also:
19 % core.symbolic.LegendreBasis, core.symbolic.ChebyshevBasis,
20 % core.symbolic.HermiteBasis

58 function computeExpressions(obj)
59 % COMPUTEEXPRESSIONS Generate polynomial basis function
60 % expressions using recursion.
61
62 obj.Expressions = repmat(sym([]), obj.MaxDegree+1, 1);
63 x = obj.Variables(1);
64 x = obj.transform(x);
65
66 for n = 0:obj.MaxDegree
67 p = obj.recursion(x, n);
68
69 if obj.IsMonic
70 c = sym2poly(p);
71 p = p / c(1);
72 end
73
74 if obj.IsNormalized
75 normalizer = obj.computeNormalizer(n);
76 expr = simplify(normalizer*p);
77 else
78 expr = simplify(p);
79 end
80
81 expr = horner(simplify(expr));
82 obj.Expressions(n+1) = expr;
83 end
84 end

86 function p = recursion(obj, x, n)
87 % RECURSION Compute orthogonal polynomial using three-term
88 % recurrence relation.
89 %
90 % p = recursion(x, n) computes the nth orthogonal polynomial
91 % using the recurrence coefficients defined by subclasses.
92
93 if n == 0
94 p = sym(1);
95 elseif n == 1
96 [a0, b0, ~] = obj.computeCoefficients(0);
97 p = a0 * x + b0;
98 else
99 [an1, bn1, cn1] = obj.computeCoefficients(n-1);
100 p1 = obj.recursion(x, n-1);
101 p2 = obj.recursion(x, n-2);
102 p = (an1 * x + bn1) * p1 + cn1 * p2;
103 end
104 end
