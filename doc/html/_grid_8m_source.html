1classdef Grid < approx.mesh.Mesh

4 % Grid provides a foundational framework for representing structured

6 % Unlike unstructured meshes that store explicit connectivity,

28 % GRID Constructor for Grid base class.
29 %
30 % obj = Grid(vertices) creates a non-uniform grid from vertex
31 % coordinate vectors that define element boundaries in each
32 % dimension, automatically computing element properties.
33
34 arguments
35 vertices {mustBeA(vertices, 'cell')}
36 end
37
38 elements = cellfun(@(x) 1:(length(x) - 1), vertices, 'Un', 0);
39 obj@approx.mesh.Mesh(vertices, elements);
40 obj.Resolution = cellfun(@(x) length(x)-1, vertices);
41 obj.Centroids = cellfun(@(x) (x(1:end-1) + x(2:end))/2, vertices, 'Un', 0);

202 function Y = collocate(obj, X, EI)
203 % COLLOCATE Map reference coordinates to physical coordinates.
204 %
205 % Y = collocate(obj, X) maps reference coordinates from the
206 % standard hypercube $[-1/2,1/2]^nd$ to physical coordinates
207 % across all grid elements using element-specific
208 % transformations.
209 %
210 % Y = collocate(obj, X, EI) maps coordinates to specific
211 % elements identified by linear indices EI.
212
213 arguments
214 obj approx.mesh.Grid
215 X {mustBeNumeric}
216 EI {mustBeInteger} = []
217 end
218
219 nd = obj.NDims;
220 if ~iscell(X)
221 C = cell(1, nd);
222 [C{:}] = ndgrid(obj.Centroids{:});
223 C = reshape(cat(nd+1, C{:}), [], nd).';
224 if ~isempty(EI), C = C(:, EI); end
225 C = reshape(C, nd, 1, []);
226
227 h = cell(1, nd);
228 [h{:}] = ndgrid(obj.Spacings{:});
229 h = reshape(cat(nd+1, h{:}), [], nd).';
230 if ~isempty(EI), h = h(:, EI); end
231 h = reshape(h, nd, 1, []);
232
233 X = reshape(X, nd, [], 1);
234 Y = C + h .* X;
235 Y = reshape(Y, nd, []);
236 else
237 Y = cell(nd, 1);
238 if isempty(EI)
239 for i = 1:nd
240 C = obj.Centroids{i};
241 h = obj.Spacings{i};
242 Y{i} = C(:).' + h(:).' .* X{i}(:).';
243 end
244 else
245 M = obj.Indexer.linearToMulti(EI);
246 for i = 1:nd
247 C = obj.Centroids{i}(M(:, i));
248 h = obj.Spacings{i}(M(:, i));
249 Y{i} = C(:).' + h(:).' .* X{i}(:).';

255 function newObj = refine(obj, nLevels)
256 % REFINE Create refined grid preserving spacing patterns.
257 %
258 % newObj = refine(obj, nLevels) creates a refined grid by
259 % uniformly subdividing each element while preserving the
260 % relative spacing characteristics of the original grid.
261
262 arguments
263 obj approx.mesh.Grid
264 nLevels {mustBeInteger, mustBeNonnegative} = 1
265 end
266
267 if nLevels == 0
268 newObj = obj;
269 return;
270 end
271
272 nd = obj.NDims;
273 p = 2^nLevels;
274 x = cell(1, nd);
275 for i = 1:nd
276 y = obj.Centroids{i};
277 n = obj.Resolution(i);
278 s = obj.Spacings{i};
279 a = y - s / 2;
280 b = y + s / 2;
281 h = (b - a) ./ p;
282 x{i} = zeros(1, 1+n*p);
283 x{i}(1) = a(1);
284 for j = 1:n
285 x{i}(1 + (j - 1) * p + (1:p)) = (a(j) + h(j)):h(j):b(j);
286 end
287 end
288 newObj = approx.mesh.Grid(x);
289 end
