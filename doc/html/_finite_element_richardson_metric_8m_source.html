1classdef FiniteElementRichardsonMetric < physics.analysis.Metric
2 % FINITEELEMENTRICHARDSONMETRIC Richardson extrapolation error metric for
3 % finite element spaces.
4 %
5 % FiniteElementRichardsonMetric computes relative errors between
6 % solutions on different grid resolutions using Richardson
7 % extrapolation. This approach is invaluable when exact analytical
8 % solutions are unavailable, providing a method to estimate
9 % discretization errors by comparing fine and coarse grid solutions.
10 %
11 % The metric implements the Richardson extrapolation principle:
12 % assuming that discretization errors behave as \f$E ≈ Ch^p\f$ where
13 % \f$h\f$ is the grid spacing and \f$p\f$ is the order of
14 % convergence, the difference between fine and coarse grid solutions
15 % provides an estimate of the error.
16 %
17 % Richardson extrapolation assumes that the coarse grid is obtained
18 % by uniform coarsening of the fine grid, typically by a factor of 2
19 % in each spatial dimension.
20 %
21 % See also:
22 % physics.analysis.Metric,
23 % physics.analysis.FiniteElementSpaceAbsoluteMetric,
24 % approx.space.FiniteElementSpace

130 function E = eval(obj, U, V)
131 % EVAL Compute Richardson extrapolation errors.
132 %
133 % E = eval(obj, U, V) computes Richardson extrapolation
134 % errors between fine grid solution @a U and coarse grid
135 % solution @a V using nested grid structure for error
136 % estimation.
137
138 arguments
139 obj physics.analysis.FiniteElementRichardsonMetric
140 U {mustBeNumeric}
141 V {mustBeNumeric}
142 end
143
144 coarseIntegrator = obj.Integrator.Integrators(2);
145 nq = length(coarseIntegrator.Weights);
146 U = reshape(U, nq, []);
147 V = reshape(V, nq, []);
148 R = abs(U - V);
149
150 nt = length(obj.Reductions);
151 E = cell(nt, 1);
152 for i = 1:nt
153 E{i} = obj.reduce(R, obj.Reductions{i});
154 end
155 E = cell2mat(E);
156 end
157 end

160 function e = reduce(obj, R, reduction)
161 % REDUCE Apply Richardson extrapolation error reduction.
162
163 n = obj.Coarse.NMeshElements;
164 h = obj.Coarse.Mesh.computeElementJacobianDeterminants();
165 w = obj.Integrator.Integrators(2).Weights;
166 switch reduction
167 case 'L1'
168 e = w * R;
169 e = reshape(e, n, []);
170 e = e .* h(:);
171 e = sum(e, 1);
172 case 'L2'
173 e = w * R.^2;
174 e = reshape(e, n, []);
175 e = e .* h(:);
176 e = sqrt(sum(e, 1));
177 case 'max'
178 e = max(R, [], 1);
179 e = reshape(e, n, []);
180 e = max(e, [], 1);
181 otherwise
182 e = R;
183 end
184 e = e(:).';
185 end

187 function obj = setNestElements(obj)
188 % SETNESTELEMENTS Reorder fine elements to align with coarse
189 % elements.
190
191 Th0 = obj.Coarse.Mesh;
192 Th1 = obj.Fine.Mesh;
193 n0 = Th0.Resolution;
194 n1 = Th1.Resolution;
195 m0 = n1 ./ n0;
196 T1 = core.linalg.MultiIndexer(shape=[prod(m0), prod(n0)]);
197 I = (1:prod(n1)).';
198 LK = T1.linearToMulti(I);
199 T2 = core.linalg.MultiIndexer(shape=m0);
200 L = T2.linearToMulti(LK(:, 1));
201 T3 = core.linalg.MultiIndexer(shape=n0);
202 K = T3.linearToMulti(LK(:, 2));
203 I = (K-1) .* m0 + (L-1) + 1;
204 T4 = core.linalg.MultiIndexer(shape=n1);
205 obj.NestEI = T4.multiToLinear(I);
206 end
