1classdef ChebyshevBasisFunction < core.symbolic.OrthogonalPolynomialBasisFunction
2 % CHEBYSHEVBASISFUNCTION Chebyshev polynomial basis functions.
3 %
4 % ChebyshevBasisFunction represents Chebyshev polynomials of the
5 % first kind on a specified interval. The polynomials can be in
6 % standard or monic form and are scaled to the given interval. These
7 % polynomials are orthogonal on [-1, 1] with respect to the weight
8 % function \f$w(x)=\frac{1}{\sqrt{1-x^2}}\f$. They also satifies the
9 % following three-term recurrence relation:
10 % \f{eqnarray*}{
11 % &&T_0(t) = 1, \\
12 % &&T_1(t) = t, \\
13 % &&T_n(t) = 2tT_{n-1}(t) - T_{n-2}(t), \forall n \ge 2.
14 % \f}
15 %
16 % Examples:
17 % % Create Chebyshev polynomial of degree 2 on [0, 1]
18 % basis = core.symbolic.ChebyshevBasisFunction(2, [0, 1]);
19 %
20 % % Create monic form on custom interval
21 % basis = core.symbolic.ChebyshevBasisFunction(3, [-1, 1], sym('t'), true);
22 %
23 % See also:
24 % core.symbolic.OrthogonalPolynomialBasisFunction

27 function expr = generate(obj)
28 % GENERATE Generates the Chebyshev polynomial.
29 %
30 % expr = generate(obj) creates the symbolic expression for
31 % the Chebyshev polynomial using the recurrence relation and
32 % scales it to the specified interval. Handles both standard
33 % and monic forms.
34 %
35 % Inputs:
36 % obj - The ChebyshevBasisFunction object
37 %
38 % Outputs:
39 % expr - Symbolic expression for the Chebyshev polynomial
40
41 x = obj.variables{1};
42 t = obj.scaleToCanonical(x, obj.bbox);
43 T = obj.generateChebyshev(obj.degree, t);
44 T = expand(T);
45 if obj.degree == 0
46 expr = sym(1);
47 else
48 if obj.monic
49 coeffs = sym2poly(T);
50 leadingCoeff = coeffs(1);
51 expr = horner(simplify(T/leadingCoeff), x);
52 else
53 expr = horner(simplify(T), x);
54 end
55 end
56 end
57 end

60 function T = generateChebyshev(~, n, t)
61 % GENERATECHEBYSHEV Generates Chebyshev polynomial using
62 % recurrence.
63
64 if n == 0
65 T = sym(1);
66 elseif n == 1
67 T = t;
68 else
69 T0 = sym(1);
70 T1 = t;
71
72 for k = 2:n
73 T = 2 * t * T1 - T0;
74 T0 = T1;
75 T1 = T;
76 end
77 end
78 end
79 end
