3 % numerical methods.
4 %
5 % ConvergenceProfiler provides comprehensive functionality to analyze
6 % the convergence behavior of numerical methods by computing
7 % convergence orders and generating formatted error tables and plots.
8 % This class is essential for validating numerical schemes and
9 % understanding their theoretical convergence properties.
10 %
11 % The profiler stores analysis configuration using MATLAB table
12 % structures and provides methods for building formatted error tables
13 % with convergence orders and creating log-log convergence
14 % visualizations that clearly show the relationship between mesh
15 % resolution and numerical errors.
16 %
17 % Examples:
18 % % Create profiler for 4 refinement levels
19 % profiler = ConvergenceProfiler(4);
20 %
21 % % Set mesh resolutions and descriptions
22 % profiler.resolutions = [0.1; 0.05; 0.025; 0.0125];
23 % profiler.descriptions = {'h=0.1'; 'h=0.05'; 'h=0.025'; 'h=0.0125'};
24 %

69 function result = buildTable(obj, errors, names)
70 % BUILDTABLE Build formatted convergence table with error
71 % orders.
72 %
73 % result = buildTable(obj, errors, names) creates a formatted
74 % table containing errors and computed convergence orders for
75 % each refinement level. Convergence orders are calculated
76 % using the formula:
77 %
78 % \f[
79 % order = \log(\frac{E_h}{E_{h/2}}) / \log(2)
80 % \f]
81 %
82 % where \f$E_h\f$ is the error at resolution \f$h\f$.
83 %
84 % Inputs:
85 % obj - The ConvergenceProfiler object
86 % errors - (nLevels×nTypes) numerical error matrix
87 % names - (1×nTypes) cell array of error type names (optional)
88 %
89 % Outputs:
90 % result - MATLAB table
91
92 [~, nTypes] = size(errors);
93
94 %< Default names if not provided
95 if nargin < 3 || isempty(names)
96 names = cell(1, nTypes);
97 for i = 1:nTypes
98 names{i} = sprintf('Error %d', i);
99 end
100 end
101
102 %< Validate inputs
103 core.except.assert(length(names) == nTypes, 'SizeMismatch', ...
104 'Number of names (%d) must match number of error types (%d).', ...
105 length(names), nTypes);
106
107 %< Replace empty descriptions with default names
108 for i = 1:obj.nLevels
109 if isempty(obj.descriptions{i})
110 obj.descriptions{i} = sprintf('Level %d', i);
111 end
112 end
113
114 %< Compute convergence orders
115 orders = zeros(size(errors));
116 for j = 1:nTypes
117 for k = 2:obj.nLevels
118 hRatio = obj.resolutions(k-1) / obj.resolutions(k);
119 if errors(k, j) > 0 && errors(k-1, j) > 0
120 eRatio = errors(k-1, j) / errors(k, j);
121 orders(k, j) = log(eRatio) / log(hRatio);
122 else
123 orders(k, j) = 0;
124 end
125 end
126 end
127
128 %< Prepare table data
129 tableData = cell(obj.nLevels, 2*nTypes+1);
130 tableData(:, 1) = obj.descriptions;
131 for j = 1:nTypes
132 tableData(:, 2*j) = num2cell(errors(:, j));
133 tableData(:, 2*j+1) = num2cell(orders(:, j));
134 end
135
136 %< Generate column names
137 columnNames = cell(1, 2*nTypes+1);
138 columnNames{1} = 'Resolution';
139 for j = 1:nTypes
140 columnNames{2*j} = sprintf('%s_Error', names{j});
141 columnNames{2*j+1} = sprintf('%s_Order', names{j});
142 end
143
144 %< Create table
145 result = cell2table(tableData, 'VariableNames', columnNames);
146 end

148 function obj = draw(obj, errors, names, rates)
149 % DRAW Create log-log convergence visualization.
150 %
151 % obj = draw(obj, errors, names, rates) creates a log-log
152 % plot showing the convergence behavior of numerical errors
153 % as a function of mesh resolution. Optional theoretical
154 % convergence rates can be plotted as reference lines for
155 % comparison.
156 %
157 % Inputs:
158 % obj - The ConvergenceProfiler object
159 % errors - (nLevels×nTypes) numerical error matrix
160 % names - (1×nTypes) cell array of error type names (optional)
161 % rates - (1×nTypes) theoretical convergence rates (optional)
162 %
163 % Outputs:
164 % obj - The ConvergenceProfiler object
165
166 [~, nTypes] = size(errors);
167
168 %< Default names if not provided
169 if nargin < 3 || isempty(names)
170 names = cell(1, nTypes);
171 for i = 1:nTypes
172 names{i} = sprintf('Error %d', i);
173 end
174 end
175
176 %< Default rates if not provided
177 if nargin < 4, rates = []; end
178
179 %< Validate inputs
180 core.except.assert(length(names) == nTypes, 'SizeMismatch', ...
181 'Number of names (%d) must match number of error types (%d).', ...
182 length(names), nTypes);
183 core.except.assert(all(obj.resolutions > 0), 'InvalidResolutions', ...
184 'All resolutions must be positive for log-log plot.');
185
186 figure;
187 hold on;
188 plotColors = lines(nTypes);
189
190 %< Plot actual errors
191 for j = 1:nTypes
192 loglog(obj.resolutions, errors(:, j), '-o', ...
193 'Color', plotColors(j, :), ...
194 'DisplayName', sprintf('%s Error', names{j}), ...
195 'LineWidth', 2, 'MarkerSize', 6);
196
197 %< Plot theoretical convergence rates if provided
198 if ~isempty(rates) && length(rates) >= j && rates(j) > 0
199 refErrors = errors(1, j) * (obj.resolutions / obj.resolutions(1)).^(rates(j));
200 loglog(obj.resolutions, refErrors, '--', ...
201 'Color', plotColors(j, :), ...
202 'DisplayName', sprintf('%s Rate %g', names{j}, rates(j)), ...
203 'LineWidth', 1.5);
204 end
205 end
206
207 %< Format plot
208 set(gca, 'XScale', 'log', 'YScale', 'log');
209 xlabel('Resolution (mesh size)');
210 ylabel('Error');
211 legend('show', 'Location', 'best');
212 grid on;
213 title('Error Convergence Analysis');
214 hold off;
215 end
216 end
