1classdef DiffusionOperator < approx.assembly.Assembly
2 % DIFFUSIONOPERATOR Assembly for diffusion operators in discontinuous Galerkin methods.
3 %
4 % DiffusionOperator provides high-level assembly of diffusion
5 % operators using discontinuous Galerkin discretizations. It
6 % implements the Local Discontinuous Galerkin (LDG) method by
7 % combining auxiliary and primal flux formulations to handle
8 % second-order elliptic operators.
9 %
10 % The operator discretizes terms of the form:
11 %
12 % \f[
13 % -\nabla \cdot (D \nabla u)
14 % \f]
15 %
16 % where \f$D\f$ is a diffusion tensor and \f$u\f$ is the diffused
17 % quantity.
18 %
19 % The LDG method introduces auxiliary variables for gradients and
20 % uses both auxiliary and primal flux treatments. Jump terms handle
21 % the enforcement of boundary conditions, with options for implicit
22 % or explicit treatment of Dirichlet conditions.
23 %
24 % Examples:
25 % % Create diffusion operator with Dirichlet boundaries
26 % space = approx.space.MeshSpace(mesh, element);
27 % operator = approx.element.L2ElementOperator(element);
28 % diffOp = DiffusionOperator(space, operator, 'dirichlet', ...
29 % 'central', 'central', 'implicit');
30 %
31 % % Assemble diffusion matrix for isotropic diffusion
32 % diffusionTensor = eye(2); % Isotropic diffusion in 2D
33 % A = diffOp.linear(diffusionTensor);
34 %
35 % % Assemble boundary condition terms
36 % boundaryFunction = @(x) x(1,:).^2 + x(2,:).^2;
37 % interiorFunction = @(x) 2*x(1,:) + 2*x(2,:); % Gradient extraction
38 % B = diffOp.linearBc(diag(diffusionTensor), boundaryFunction, interiorFunction);
39 %
40 % See also:
41 % approx.assembly.Assembly, approx.assembly.VolumeAssembly,
42 % approx.assembly.FluxAssembly, approx.assembly.ConvectionOperator
43

62 %

67 %
68 % Inputs:

70 % operator - ElementOperator object
71 % bcType - Boundary condition type: {'periodic', 'dirichlet'}
72 % auxFluxType - Auxiliary flux type: {'left', 'right', 'central'}
73 % prmFluxType - Primal flux type: {'left', 'right', 'central'}
74 % jumpType - Jump treatment: {'implicit', 'explicit'}

107 function A = linear(obj, coe)
108 % LINEAR Assemble linear diffusion operator matrix.
109 %
110 % A = linear(obj, coe) assembles the diffusion operator
111 % matrix using the Local Discontinuous Galerkin method. The
112 % coefficient matrix @a coe represents the diffusion tensor.
113 %
114 % Inputs:
115 % obj - The DiffusionOperator object
116 % coe - Diffusion tensor (nDims x nDims matrix)
117 %
118 % Outputs:
119 % A - Assembled diffusion operator matrix (sparse)
120
121 n = obj.space.nGlobalDofs;
122
123 for dim = 1:obj.space.nDims
124 C = obj.volume.scaleConstant(dim, 1);
125 T1 = obj.volume.assembleVolumePartial(dim, C);
126 T2 = obj.auxFlux.assembleFluxPartial(dim, C);
127 T = [T1; T2];
128 obj.auxMat{dim} = core.linalg.sparseFromTriplets(T, n, n);
129
130 T2 = obj.prmFlux.assembleFluxPartial(dim, C);
131 T = [T1; T2];
132 obj.prmMat{dim} = core.linalg.sparseFromTriplets(T, n, n);
133
134 if obj.isImplicitDirichletBoundaryJump
135 T = obj.auxFlux.assembleImplicitBoundaryJump(dim, 1);
136 obj.jmpMat{dim} = core.linalg.sparseFromTriplets(T, n, n);
137 end
138 end
139
140 A = sparse(n, n);
141 for i = 1:obj.space.nDims
142 for j = 1:obj.space.nDims
143 if coe(i, j) ~= 0
144 A = A + coe(i, j) * obj.auxMat{i} * obj.prmMat{j};
145 if obj.isImplicitDirichletBoundaryJump
146 A = A + coe(i, j) * obj.jmpMat{i};
147 end
148 end
149 end
150 end
151 end

153 function A = linearBc(obj, coe, f, g)
154 % LINEARBC Assemble boundary condition terms for linear
155 % diffusion.
156 %
157 % A = linearBc(obj, coe, f, g) assembles the boundary
158 % condition contribution matrix for linear diffusion with
159 % Dirichlet boundary conditions. The treatment depends on
160 % whether implicit or explicit jump handling is used.
161 %
162 % Inputs:
163 % obj - The DiffusionOperator object
164 % coe - Diagonal diffusion coefficients (nDims x 1)
165 % f - Boundary condition function handle
166 % g - Interior extraction function handle (for explicit jumps)
167 %
168 % Outputs:
169 % A - Boundary condition contribution matrix (sparse)
170
171 n = obj.space.nGlobalDofs;
172 A = cell(1, obj.space.nDims);
173 for dim = 1:obj.space.nDims
174 if coe(dim) == 0, continue; end
175 C = obj.volume.scaleConstant(dim, coe(dim));
176 T1 = obj.auxFlux.boundary.assembleLeftTrace(dim, C, f);
177 T2 = obj.auxFlux.boundary.assembleRightTrace(dim, C, f);
178 T = [T1; T2];
179 A{dim} = core.linalg.sparseFromTriplets(T, n, max(T(:, 2)));
180 A{dim} = obj.auxMat{dim} * A{dim};
181
182 if obj.isImplicitDirichletBoundaryJump
183 switch obj.auxFlux.fluxType
184 case 'left'
185 T = T1;
186 B = core.linalg.sparseFromTriplets(T, n, max(T(:, 2)));
187 case 'right'
188 T = T2;
189 B = core.linalg.sparseFromTriplets(T, n, max(T(:, 2)));
190 case 'central'
191 T1(:, 3) = T1(:, 3) / 2;
192 T2(:, 3) = T2(:, 3) / 2;
193 T = [T1; T2];
194 B = core.linalg.sparseFromTriplets(T, n, max(T(:, 2)));
195 end
196 A{dim} = A{dim} + B;
197 elseif obj.isExplicitDirichletBoundaryJump
198 switch obj.auxFlux.fluxType
199 case 'left'
200 T = obj.auxFlux.boundary.assembleLeftJump(dim, C, f, g);
201 B = core.linalg.sparseFromTriplets(T, n, max(T(:, 2)));
202 case 'right'
203 T = obj.auxFlux.boundary.assembleRightJump(dim, C, f, g);
204 B = core.linalg.sparseFromTriplets(T, n, max(T(:, 2)));
205 case 'central'
206 T1 = obj.auxFlux.boundary.assembleLeftJump(dim, C / 2, f, g);
207 T2 = obj.auxFlux.boundary.assembleRightJump(dim, C / 2, f, g);
208 T = [T1; T2];
209 B = core.linalg.sparseFromTriplets(T, n, max(T(:, 2)));
210 end
211 A{dim} = A{dim} + B;
212 end
213 end
214
215 for dim = obj.space.nDims:-1:2
216 A{1} = A{1} + A{dim};
217 A{dim} = [];
218 end
219 A = A{1};
220 end
221 end
222end
