2 % VISUALIZER Base class for visualizers.
3 %
4 % Visualizer provides comprehensive plotting capabilities for physics

244 function obj = addPlotter(obj, name, strategy)
245 % ADDPLOTTER Add tiled plotter.
246 %
247 % obj = addPlotter(obj, name, strategy) registers a plotter
248 % with the specified rendering strategy.
249
250 arguments
251 obj physics.visual.Visualizer
252 name {mustBeTextScalar}
253 strategy {mustBeTextScalar}
254 end
255
256 switch lower(strategy)
257 case '1d'
258 nRows = sum(arrayfun(@(x) min(1, x), obj.NComponents));
259 nCols = max(obj.NComponents, [], 1);
260 nAxes = sum(obj.NComponents);
261 nAxesPerComponent = 1;
262 strategyObj = physics.visual.Strategy1d();
263 case '2d'
264 nRows = sum(obj.NComponents);
265 nCols = length(fieldnames(obj.Database.Datasets));
266 nAxesPerComponent = nCols;
267 nAxes = nRows * nCols;
268 strategyObj = physics.visual.Strategy2d();
269 case '3d'
270 nRows = sum(obj.NComponents);
271 nCols = length(fieldnames(obj.Database.Datasets));
272 nAxesPerComponent = nCols;
273 nAxes = nRows * nCols;
274 strategyObj = physics.visual.Strategy3d();
275 case 'slice1d'
276 nRows = sum(arrayfun(@(x) min(1, x), obj.NComponents));
277 nCols = max(obj.NComponents, [], 1);
278 nAxes = sum(obj.NComponents);
279 nAxesPerComponent = 1;
280 strategyObj = physics.visual.StrategySlice1d();
281 otherwise
282 core.except.assert(false, 'InvalidStrategy', ...
283 'Unknown strategy: %s', strategy);
284 end
285
286 plotter = physics.visual.TiledPlotter(nRows, nCols);
287
288 dofNames = fieldnames(obj.Components);
289 offset = 0;
290 for iField = 1:length(obj.NComponents)
291 nFieldComponents = obj.NComponents(iField);
292 dofName = dofNames{iField};
293 for iComponent = 1:nFieldComponents
294 componentName = sprintf('%s%d', dofName, ...
295 obj.Components.(dofName)(iComponent));
296 axisIdx = (offset + iComponent - 1) * nAxesPerComponent + 1;
297 plotter.bind(componentName, axisIdx);
298 end
299 offset = offset + nFieldComponents;
300 end
301
302 plotter.setStrategy(strategyObj);
303
304 for iAxis = 1:nAxes
305 plotter.addAxisSpec(1, 1);
306 end
307
308 obj.Plotters.(name) = plotter;
309 end

350 function obj = update(obj, space, dofs, options)
351 % UPDATE Update datasets with current solution.
352
353 arguments
354 obj physics.visual.Visualizer
355 space approx.space.FiniteElementSpace
356 dofs struct
357 options.timeline approx.mesh.DynamicTimeline
358 options.V0 = []
359 options.n0 = 0
360 end
361
362 xRef = obj.Cache.RefXNodes;
363 V0 = options.V0;
364 n0 = options.n0;
365
366 if ~isempty(obj.Timeline)
367 t = obj.Timeline.Now;
368 else
369 t = [];
370 end
371
372 dofNames = fieldnames(dofs);
373
374 for i = 1:length(dofNames)
375 dofName = dofNames{i};
376 if ~isfield(obj.Components, dofName)
377 continue;
378 end
379
380 component = obj.Components.(dofName);
381 if isempty(component)
382 continue;
383 end
384
385 datasetNames = obj.Database.DatasetNames;
386 for j = 1:length(datasetNames)
387 datasetName = datasetNames{j};
388 dataset = obj.Database.getDataset(datasetName);
389
390 if strcmp(dataset.Type, 'fixed')
391 continue;
392 end
393
394 if strcmp(dataset.Type, 'exact')
395 if isfield(obj.Exacts, dofName)
396 f = obj.Exacts.(dofName);
397 if ~isempty(t)
398 args = {'t', t};
399 else
400 args = {};
401 end
402 U = space.feval(f, xRef, Args=args);
403 U = U(:, component);
404 else
405 continue;
406 end
407 else
408 U = dofs.(dofName)(:, component);
409 U = space.eval(xRef, U);
410 if ~isempty(V0)
411 ht = options.timeline.StepSizeQueue;
412 t1 = options.timeline.Next;
413 U0 = cell(1, n0);
414 for k = 1:n0
415 U0{k} = V0{k}.(dofName)(:, component);
416 U0{k} = space.eval(xRef, U0{k});
417 U0{k} = reshape(U0{k}, [1, size(U0{k})]);
418 end
419 n = size(U);
420 U = reshape(U, [1, n]);
421 U = cat(1, U0{:}, U);
422 ti = [t1 - cumsum(ht(n0:-1:1)), t1].';
423 U = interp1(ti, U, t, 'linear', 'extrap');
424 U = reshape(squeeze(U), n);
425 end
426 end
427
428 dataset.setData(dofName, U);
429 end
430 end

479 function TF = get.IsEnabled(obj)
480 % GET.ISENABLED Determine if visualization is enabled.
481
482 arguments
483 obj physics.visual.Visualizer
484 end
485
486 TF = false;
487
488 if isempty(obj.Density) || ~isnumeric(obj.Density) || any(obj.Density <= 0)
489 return;
490 end
491
492 if isempty(obj.Components) || ~isstruct(obj.Components)
493 return;
494 else
495 names = fieldnames(obj.Components);
496 if isempty(names)
497 return;
498 end
499 flag = true;
500 for i = 1:length(names)
501 name = names{i};
502 if ~isempty(obj.Components.(name))
503 flag = false;
504 break;
505 end
506 end
507 if flag, return; end
508 end
509
510 if bitand(obj.Status, obj.IsInitialized)
511 if isempty(obj.Plotters) || ~isstruct(obj.Plotters)
512 return;
513 end
514
515 plotterNames = fieldnames(obj.Plotters);
516 if isempty(plotterNames)
517 return;
518 end
519 end
520
521 TF = true;
522 end
