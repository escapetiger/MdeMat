41 % obj = OrderedMap() creates empty OrderedMap.
42 %
43 % obj = OrderedMap(keys, values) creates OrderedMap from cell
44 % arrays.
45 %
46 % obj = OrderedMap('key1', value1, 'key2', value2, ...)

48 %
49 % Inputs:

51 %
52 % Outputs:
53 % obj - Constructed OrderedMap object
54

56 obj.constructEmptyMap();
57 obj.keys = {};
58 obj.map = containers.Map();
59 elseif nargin == 2
60 keys = varargin{1};
61 values = varargin{2};
62 if iscell(keys) && iscell(values)
63 obj = obj.constructFromArrays(keys, values);
64 end

67 keys = cell(1, nPairs);

72 end

74
75 else
76 core.except.assert(false, 'InvalidInput', ...
77 ['Invalid number of arguments. Expected 0, 2, or an even ', ...

80 end

337 function obj = move(obj, key, newPosition)
338 % MOVE Move a key to a new position in the order.
339 %
340 % obj = move(obj, key, newPosition) changes position of existing
341 % @a key to @a newPosition in the ordering.
342 %
343 % Inputs:
344 % obj - The OrderedMap object
345 % key - Key to move
346 % newPosition - New position index (1-based)
347 %
348 % Outputs:
349 % obj - The OrderedMap object
350
351 core.except.assert(obj.isKey(key), 'KeyNotFound', ...
352 'Cannot move key "%s" because it does not exist in the map.', ...
353 core.data.toString(key));
354
355 n = length(obj.keys);
356 core.except.assert(newPosition >= 1 && newPosition <= n, ...
357 'InvalidPosition', 'Position must be between 1 and %d, got %d.', ...
358 n, newPosition);
359
360 currentPosition = 0;
361 for i = 1:n
362 if isequal(obj.keys{i}, key)
363 currentPosition = i;
364 break;
365 end
366 end
367
368 if currentPosition == newPosition
369 return;
370 end
371
372 keyToMove = obj.keys{currentPosition};
373 obj.keys(currentPosition) = [];
374
375 if newPosition == 1
376 obj.keys = [{keyToMove}, obj.keys];
377 elseif newPosition > length(obj.keys)
378 obj.keys{end+1} = keyToMove;
379 else
380 obj.keys = [obj.keys(1:newPosition-1), ...
381 {keyToMove}, ...
382 obj.keys(newPosition:end)];
383 end
384 end
