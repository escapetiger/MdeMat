44 function computeExpressions(obj)
45 % COMPUTEEXPRESSIONS Compute spherical harmonic basis function
46 % expressions.
47
48 obj.Expressions = repmat(sym([]), (obj.MaxDegree + 1)^2, 1);
49
50 mu = obj.Variables(1);
51 theta = obj.Variables(2);
52
53 idx = 1;
54
55 for l = 0:obj.MaxDegree
56 for m = 0:l
57 if m == 0
58 expr = obj.computeSphericalHarmonic(mu, theta, l, 0);
59 if obj.IsNormalized
60 normalizer = sqrt((2 * l + 1)/(4 * sym(pi))* ...
61 factorial(l-abs(m))/factorial(l+abs(m)));
62 expr = simplify(normalizer*expr);
63 end
64 obj.Expressions(idx) = expr;
65 idx = idx + 1;
66 else
67 expr1 = obj.computeSphericalHarmonic(mu, theta, l, m);
68 expr2 = obj.computeSphericalHarmonic(mu, theta, l, -m);
69 if obj.IsNormalized
70 normalizer = sqrt((2 * l + 1)/(4 * sym(pi))* ...
71 factorial(l-abs(m))/factorial(l+abs(m)));
72 expr1 = simplify(normalizer*expr1);
73 expr2 = simplify(normalizer*expr2);
74 end
75 obj.Expressions(idx) = -sym(1)^m * expr1;
76 obj.Expressions(idx+1) = -sym(1)^m * expr2;
77 idx = idx + 2;
78 end
79 end

83 function harmonic = computeSphericalHarmonic(obj, mu, theta, l, m)
84 % COMPUTESPHERICALHARMONIC Compute spherical harmonic
85 % Y_l^m(mu, theta).
86
87 p = obj.computeAssociatedLegendre(mu, l, abs(m));
88
89 if m > 0
90 f = cos(m * theta);
91 elseif m < 0
92 f = sin(abs(m)*theta);
93 else
94 f = sym(1);
95 end
96
97 harmonic = simplify(p*f);
98 end

100 function p = computeAssociatedLegendre(obj, mu, l, m)
101 % COMPUTEASSOCIATEDLEGENDRE Compute associated Legendre
102 % polynomial P_l^m(mu).
103
104 if m == 0
105 p = obj.computeLegendre(mu, l);
106 else
107 p = obj.computeLegendre(mu, l);
108 p = (-1)^m * (1 - mu^2)^(m / 2) * diff(p, mu, m);
109 p = simplify(p);
110 p = horner(p);
111 end
112 end

114 function p = computeLegendre(obj, mu, n)
115 % COMPUTELEGENDRE Compute Legendre polynomial P_n(mu).
116
117 if n == 0
118 p = sym(1);
119 elseif n == 1
120 p = mu;
121 else
122 p1 = obj.computeLegendre(mu, n-1);
123 p2 = obj.computeLegendre(mu, n-2);
124 p = ((2 * n - 1) * mu * p1 - (n - 1) * p2) / n;
125 p = simplify(p);
126 p = horner(p);
127 end
