1classdef L2FiniteElementClipper < fem.element.C0FiniteElementClipper
2 % L2FINITEELEMENTCLIPPER Geometric clipper for discontinuous finite elements.
3 %
4 % L2FiniteElementClipper extends the C0 clipper with flux piece
5 % computations for discontinuous finite elements in semi-Lagrangian
6 % methods. It handles both volume intersections (inherited from C0)
7 % and boundary flux intersections required for discontinuous Galerkin
8 % transport schemes.
9 %
10 % The clipper computes intersections between element boundaries and
11 % upstream footprints, creating flux pieces that capture the flow
12 % of information across element interfaces. Each boundary face is
13 % processed separately to handle inflow and outflow contributions
14 % in the semi-Lagrangian formulation.
15 %
16 % This approach enables semi-Lagrangian discontinuous Galerkin methods
17 % that combine the stability benefits of characteristic tracking with
18 % the flexibility of discontinuous function spaces, particularly
19 % effective for problems with discontinuous solutions or strong
20 % convection.
21 %
22 % Examples:
23 % % Create clipper for L2 element with specified velocity
24 % velocity = [0.5; 0.3];
25 % element = fem.element.L2FiniteElement(geometry, projector, volInt, fluxInt);
26 % clipper = L2FiniteElementClipper(element, velocity);
27 %
28 % % Set up volume and flux piece data
29 % timeStep = 0.1;
30 % meshSpacing = [0.05; 0.05];
31 % clipper.setVolumePieceData(timeStep, meshSpacing);
32 % clipper.setFluxPieceData(timeStep, meshSpacing);
33 %
34 % % Access computed pieces and data
35 % nVolPieces = clipper.nVolumePieces;
36 % nFluxPieces = clipper.nFluxPieces; % Array for each boundary
37 % fluxEulerianData = clipper.fluxEulerianData; % Cell array by boundary
38 %
39 % See also:
40 % fem.element.C0FiniteElementClipper, fem.element.L2FiniteElement,
41 % fem.element.L2SemiLagrangianFiniteElementOperator

59 function setFluxPieceData(obj, ht, hx)
60 % SETFLUXPIECEDATA Set up flux piece data for semi-Lagrangian transport.
61 %
62 % setFluxPieceData(obj, ht, hx) computes the geometric
63 % intersections between element boundaries and upstream
64 % footprints, creating flux pieces with associated function
65 % data for accurate semi-Lagrangian discontinuous Galerkin
66 % integration. Each boundary face is processed to handle
67 % inflow and outflow contributions.
68 %
69 % Inputs:
70 % obj - The L2FiniteElementClipper object
71 % ht - Time step for characteristic tracking (positive scalar)
72 % hx - Mesh spacing vector (column vector)
73 %
74 % Outputs:
75 % NULL
76
77 v = obj.velocity;
78 bbox = obj.fe.geometry.bbox;
79 bbox = reshape(bbox, 2, []);
80 ratio = v(:) .* ht ./ hx(:);
81 poi = ratio - floor(ratio) - 1 / 2;
82 nd = obj.fe.nDims;
83 nf = 2 * nd; %< Number of boundary faces
84 np = zeros(1, nf);
85
86 %< Initialize flux pieces structure array
87 obj.fluxPieces = arrayfun(@(i) struct(), 1:nf);
88 for i = 1:nf
89 obj.fluxPieces(i).xc = zeros(nd, 2^(nd - 1));
90 obj.fluxPieces(i).hx = zeros(nd, 2^(nd - 1));
91 end
92
93 %< Generate intersection pieces for each boundary face
94 for i = 1:nf
95 d = floor((i - 1)/2) + 1; %< Spatial dimension index
96 s = mod(i-1, 2) + 1; %< Side index (1=min, 2=max)
97
98 for j = 1:2^(nd - 1)
99 %< Create multi-indexer for face geometry
100 shape = ones(1, nd) * 2;
101 shape(d) = 1; %< Collapsed dimension for face
102 indexer = core.linalg.MultiIndexer(shape);
103 m = indexer.linearToMulti(j);
104
105 %< Compute piece geometry
106 x1 = poi;
107 x2 = bbox((m + (0:nd - 1) * 2)');
108 xc = (x1 + x2) / 2;
109 hx = abs(x2 - x1);
110
111 %< Skip degenerate pieces (check reduced dimensions)
112 if prod(hx([1:d-1, d+1:end])) < 1e-8
113 continue;
114 end
115
116 %< Set face-specific geometry
117 hx(d) = 0; %< Zero thickness in normal direction
118 xc(d) = bbox(s, d); %< Position on boundary face
119
120 np(i) = np(i) + 1;
121 obj.fluxPieces(i).xc(:, np(i)) = xc;
122 obj.fluxPieces(i).hx(:, np(i)) = hx;
123 end
124
125 %< Handle case with no valid pieces
126 if np(i) == 0
127 obj.fluxPieces(i).xc = zeros(nd, 1);
128 obj.fluxPieces(i).xc(d) = bbox(s, d);
129 obj.fluxPieces(i).hx = ones(nd, 1);
130 obj.fluxPieces(i).hx(d) = 0;
131 np(i) = 1;
132 else
133 obj.fluxPieces(i).xc = obj.fluxPieces(i).xc(:, 1:np(i));
134 obj.fluxPieces(i).hx = obj.fluxPieces(i).hx(:, 1:np(i));
135 end
136
137 %< Compute piece boundaries
138 obj.fluxPieces(i).xmin = obj.fluxPieces(i).xc - obj.fluxPieces(i).hx / 2;
139 obj.fluxPieces(i).xmax = obj.fluxPieces(i).xc + obj.fluxPieces(i).hx / 2;
140 end
141 obj.nFluxPieces = np;
142
143 %< Create function data objects for each boundary and piece
144 I = obj.fe.fluxData.integrator;
145 obj.fluxEulerianData = cell(1, nf);
146 obj.fluxUpstreamData = cell(1, nf);
147 for i = 1:nf
148 obj.fluxEulerianData{i} = arrayfun(@(j, k) fem.data.FunctionData(I.copy()), ...
149 repmat((1:2).', 1, np(i)), repmat(1:np(i), 2, 1));
150 obj.fluxUpstreamData{i} = arrayfun(@(j, k) fem.data.FunctionData(I.copy()), ...
151 repmat((1:2).', 1, np(i)), repmat(1:np(i), 2, 1));
152 end
153
154 %< Set up integration points for each boundary and piece
155 nq = obj.fe.projector.basis.nFactorCodims;
156 for i = 1:nf
157 for j = 1:np(i)
158 a = obj.fluxPieces(i).xmin(:, j);
159 b = obj.fluxPieces(i).xmax(:, j);
160 obj.fluxEulerianData{i}(1, j).setPoints(nq, false, a, b);
161 obj.fluxEulerianData{i}(2, j).setPoints(nq, false, a, b);
162 obj.fluxUpstreamData{i}(1, j).setPoints(nq, false, -b, -a);
163 obj.fluxUpstreamData{i}(2, j).setPoints(nq, false, -b, -a);
164 end
165 end
166
167 %< Compute basis function values at integration points
168 f = obj.fe.projector.basis;
169 for i = 1:nf
170 d = floor((i - 1)/2) + 1; %< Spatial dimension index
171 s = mod(i-1, 2) + 1; %< Side index
172
173 for j = 1:np(i)
174 %< Eulerian inflow data
175 Dei = obj.fluxEulerianData{i}(1, j);
176 Iei = Dei.integrator;
177 Xe = Iei.nodes(1:nd, :);
178 Xe(d, :) = bbox(s, d);
179 Dei.setValues(f, Xe);
180
181 %< Eulerian outflow data
182 Deo = obj.fluxEulerianData{i}(2, j);
183 Ieo = Deo.integrator;
184 Xe = Ieo.nodes(1:nd, :);
185 Xe(d, :) = bbox(3-s, d); %< Opposite side
186 Deo.setValues(f, Xe);
187
188 %< Upstream inflow data
189 Dui = obj.fluxUpstreamData{i}(1, j);
190 Iui = Dui.integrator;
191 Xu = Iui.nodes(1:nd, :);
192 Xu(d, :) = -poi(d);
193 Dui.setValues(f, Xu);
194
195 %< Upstream outflow data
196 Duo = obj.fluxUpstreamData{i}(2, j);
197 Iuo = Duo.integrator;
198 Xu = Iuo.nodes(1:nd, :);
199 if -poi(d) == bbox(1, d)
200 Xu(d, :) = bbox(2, d);
201 elseif -poi(d) == bbox(2, d)
202 Xu(d, :) = bbox(1, d);
203 end
204 Duo.setValues(f, Xu);
205 end
206 end
207
208 %< Compute index shifts for upstream element mapping
209 obj.fluxShifts = arrayfun(@(i) zeros(nd, np(i)), 1:nf, 'UniformOutput', false);
210 for i = 1:nf
211 mm = ceil(ratio) .* (ratio > 0) + (floor(ratio) + 1) .* (ratio < 0);
212 mm0 = -mm;
213 mm1 = -mm + 1;
214 for j = 1:np(i)
215 bb = obj.fluxPieces(i).xc(:, j) <= 0;
216 obj.fluxShifts{i}(:, j) = mm0 .* bb + mm1 .* (1 - bb);
217 end
218 end
219 end
220 end
221end
