1classdef OdeIntegrator < handle
2 % ODEINTEGRATOR Abstract base class for ODE integrators.
3 %
4 % OdeIntegrator provides a common interface and shared functionality
5 % for all ODE integration methods. This base class manages solution
6 % history, timeline operations, and linear system solving
7 % infrastructure. Concrete integrators must implement the abstract
8 % methods for specific integration schemes.
9 %
10 % The class focuses on solving ordinary differential equations of
11 % the form:
12 %
13 % \f[
14 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = Lu + F(u) + S(t),
15 % \f]
16 %
17 % where \f$M\f$ is a mass operator, \f$L\f$ is a linear operator,
18 % \f$F(u)\f$ is a nonlinear operator, and \f$S(t)\f$ is a source
19 % term. Any of these terms can be empty. In particular, M = [] means
20 % the mass operator reduces to an identity operator; {L, F, S} = []
21 % means they vanish from the equation.
22 %
23 % The contribution of mass operator M is determined by the physical
24 % simulation framework:
25 % * In the Eulerian framework, M is space-only so that it can be
26 % represented by a unique static matrix.
27 % * In the Semi-Lagrangian framework, M is time-dependent. The Lhs
28 % mass term is forced to be a unique static matrix as in Eulerian
29 % framework. However, the Rhs mass term should be updated as time
30 % evolves. In particular, M is a matrix under periodic boundary
31 % conditions, while M is a pair formed by a matrix and a vector
32 % under Dirichlet boundary conditions.
33 %
34 % See also:
35 % approx.odeint.BeIntegrator, approx.odeint.BdfIntegrator,
36 % approx.odeint.DirkIntegrator, approx.odeint.ExrkIntegrator
37

71 function obj = reset(obj, options)
72 % RESET Reset the integrator to initial state.
73 %
74 % obj = reset(obj) clears solution history, resets the
75 % timeline, and initializes storage arrays for left-hand side
76 % matrices and right-hand side vectors. Also creates a fresh
77 % linear solver instance. This method is called automatically
78 % during construction and can be called to restart
79 % integration.
80
81 arguments
82 obj approx.odeint.OdeIntegrator
83 options.linOpts = struct()
84 end
85
86 obj.U0 = cell(obj.NStages, obj.NSteps);
87 obj.Timeline.reset();
88
89 %< Initialize storage based on method characteristics
90 if obj.NStages > 1 && obj.NSteps == 1
91 %< Multi-stage, single-step methods (RK-type)
92 obj.Lhs = cell(obj.NStages+1, obj.NSteps);
93 obj.Rhs = cell(obj.NStages+1, obj.NSteps);
94 elseif obj.NStages == 1 && obj.NSteps > 1
95 %< Single-stage, multi-step methods (BDF-type)
96 obj.Lhs = cell(1, obj.NSteps);
97 obj.Rhs = cell(1, obj.NSteps);
98 else
99 %< Single-stage, single-step methods
100 obj.Lhs = [];
101 obj.Rhs = [];
102 end
103
104 %< Initialize mass term storage
105 if obj.NStages > 1 && obj.NSteps == 1
106 obj.MLhs = [];
107 obj.MRhs = cell(obj.NStages+1, 1);
108 elseif obj.NStages == 1 && obj.NSteps > 1
109 obj.MLhs = [];
110 obj.MRhs = cell(1, obj.NSteps);
111 else
112 obj.MLhs = [];
113 obj.MRhs = [];
114 end
115
116 %< Create linear solver instances for each stage
117 args = namedargs2cell(options.linOpts);
118 if obj.NStages == 1
119 obj.Solver = core.linalg.LinearSolver(args{:});
120 else
121 obj.Solver = cell(1, obj.NStages+1);
122 for i = 1 : obj.NStages+1
123 obj.Solver{i} = core.linalg.LinearSolver(args{:});
124 end
125 end
126 end

244 function Me = evaluateRhsMass(obj, i, M, t)
245 % EVALUATERhsMASS Evaluate mass operator for right-hand side.
246 %
247 % Me = evaluateRhsMass(obj, i, M, t) evaluates the mass
248 % operator for use in right-hand side vector construction.
249 % Handles caching for efficiency.
250
251 if isempty(M)
252 Me = [];
253 elseif isa(M, 'function_handle')
254 switch nargin(M)
255 case 1
256 if isscalar(t)
257 te = t;
258 else
259 te = t(2);
260 end
261 if obj.shouldUpdateRhsMass(i)
262 obj.setRhsMass(i, M(te));
263 end
264 Me = obj.getRhsMass(i);
265 case 2
266 core.except.assert(length(t) >= 2, 'InvalidInput', ...
267 'Mass function requires at least 2 time values.');
268 Me = cell(1, 2);
269 if obj.shouldUpdateRhsMass(i)
270 [Me{1:2}] = M(t(1), t(2));
271 obj.setRhsMass(i, Me{2});
272 else
273 Me{2} = obj.getRhsMass(i);
274 Me{1} = M(t(1), t(2));
275 end
276 otherwise
277 core.except.assert(0, 'InvalidInput', ...
278 'Mass function requires 1 or 2 arguments.');
279 end
280 else
281 Me = M;
282 end
