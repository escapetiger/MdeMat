72function plot1D(graph, options)
73% PLOT1D Visualize a 1D graph.
74%
75% plot1D(graph, options) creates a 1D visualization by plotting vertices
76% along the x-axis at y=0 and connecting them with edges.
77
78%< Plot edges if enabled
79if options.ShowEdges && ~isequal(options.EdgeColor, 'none') && ...
80 graph.NEdges > 0
81 for iEdge = 1:graph.NEdges
82 v1 = graph.Edges(iEdge, 1);
83 v2 = graph.Edges(iEdge, 2);
84 x = [graph.Vertices(v1), graph.Vertices(v2)];
85 y = [0, 0];
86 plot(x, y, '-', 'Color', options.EdgeColor, ...
87 'LineWidth', options.EdgeWidth);
88 end
89end
90
91%< Plot vertices if enabled
92if options.ShowVertices && ~isequal(options.VertexColor, 'none') && ...
93 graph.NVertices > 0
94 scatter(graph.Vertices, zeros(graph.NVertices, 1), ...
95 options.VertexSize, options.VertexColor, 'filled');
96end
97
98%< Set y-axis limits for better visualization
99ylim([-0.5, 0.5]);
100xlabel('X');
101ylabel('');
102
103%< Add reference line at y=0 if no edges are shown
104if ~options.ShowEdges || isequal(options.EdgeColor, 'none')
105 xLimits = xlim;
106 plot(xLimits, [0, 0], 'k-', 'LineWidth', 0.5, 'Color', [0.7, 0.7, 0.7]);
107end
108end
