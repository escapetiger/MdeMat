183 function obj = addAbsolute(obj, state, options)
184 % ADDABSOLUTE Compute absolute error against exact solutions.
185 %
186 % obj = addAbsolute(obj, state) computes absolute errors by
187 % comparing numerical solutions with exact analytical
188 % solutions. The method evaluates both numerical and exact
189 % solutions at the same sample points and computes specified
190 % error metrics.
191
192 arguments
193 obj physics.analysis.Analyzer
194 state physics.state.SpatialState
195 options.args cell = {}
196 end
197
198 space = state.XDisc;
199 dofs = state.Dofs;
200
201 metric = physics.analysis.FiniteElementAbsoluteMetric( ...
202 space, Reductions=obj.Reductions);
203 metric.setPoints(obj.Density);
204
205 xRef = metric.Nodes;
206 fieldNames = fieldnames(dofs);
207 for iField = 1:length(fieldNames)
208 fieldName = fieldNames{iField};
209 component = obj.Components.(fieldName);
210 C = dofs.(fieldName);
211 f = obj.Exacts.(fieldName);
212 U = space.eval(xRef, C);
213 U = U(:, component);
214 U0 = space.feval(f, xRef, Args=options.args);
215 U0 = U0(:, component);
216 R = metric.eval(U, V=U0);
217 E = zeros(size(R, 1), 1);
218 for j = 1:length(obj.Reductions)
219 switch lower(obj.Reductions{j})
220 case 'l1'
221 E(j) = norm(R(j, :), 1);
222 case 'l2'
223 E(j) = norm(R(j, :), 2);
224 case 'max'
225 E(j) = norm(R(j, :), inf);
226 end
227 end
228 obj.Errors.(fieldName){obj.CurrentLevel} = E;
229 end
230 obj.CurrentLevel = obj.CurrentLevel + 1;
231 end

284 % ANALYZE Generate comprehensive convergence analysis results.
285 %
286 % tables = analyze(obj) processes all collected error data
287 % and generates structured results including convergence
288 % tables for each field component and error reduction type.
289 % The results can be used for convergence verification and
290 % method validation.
291
292 arguments
293 obj physics.analysis.Analyzer
294 verbose {mustBeNumericOrLogical} = false
295 end
296
297 tables = struct();
298 fieldNames = fieldnames(obj.Errors);
299 for iField = 1:length(fieldNames)
300 fieldName = fieldNames{iField};
301 component = obj.Components.(fieldName);
302 if isempty(component), continue; end
303 E = cell2mat(obj.Errors.(fieldName));
304 for iComponent = 1:size(E, 2)
305 Ej = reshape(E(:, iComponent), obj.NReductions, obj.NLevels).';
306 name = sprintf('%s%d', fieldName, component(iComponent));
307 tables.(name) = obj.analyzeImpl(Ej, obj.Reductions);
308 if verbose > 0
309 disp(tables.(name));
310 end
311 end
312 end

316 % DRAW Create convergence visualization plots.
317 %
318 % obj = draw(obj) creates log-log convergence plots using

439 function T = analyzeImpl(obj, errors, names)
440 % ANALYZEIMPL Build formatted convergence table with error orders.
441
442 [~, nTypes] = size(errors);
443
444 %< Default names if not provided
445 if nargin < 3 || isempty(names)
446 names = cell(1, nTypes);
447 for i = 1:nTypes
448 names{i} = sprintf('Error %d', i);
449 end
450 end
451
452 %< Validate inputs
453 core.except.assert(length(names) == nTypes, 'SizeMismatch', ...
454 'Number of names (%d) must match number of error types (%d).', ...
455 length(names), nTypes);
456
457 %< Replace empty descriptions with default names
458 for i = 1:obj.NLevels
459 if isempty(obj.Descriptions{i})
460 obj.Descriptions{i} = sprintf('Level %d', i);
461 end
462 end
463
464 %< Compute convergence orders
465 orders = zeros(size(errors));
466 for j = 1:nTypes
467 for k = 2:obj.NLevels
468 hRatio = obj.Resolutions(k-1) / obj.Resolutions(k);
469 if errors(k, j) > 0 && errors(k-1, j) > 0
470 eRatio = errors(k-1, j) / errors(k, j);
471 orders(k, j) = log(eRatio) / log(hRatio);
472 else
473 orders(k, j) = 0;
474 end
475 end
476 end
477
478 %< Prepare table data
479 tableData = cell(obj.NLevels, 2*nTypes+1);
480 tableData(:, 1) = obj.Descriptions;
481 for j = 1:nTypes
482 tableData(:, 2*j) = num2cell(errors(:, j));
483 tableData(:, 2*j+1) = num2cell(orders(:, j));
484 end
485
486 %< Generate column names
487 columnNames = cell(1, 2*nTypes+1);
488 columnNames{1} = 'Resolution';
489 for j = 1:nTypes
490 columnNames{2*j} = sprintf('%s_Error', names{j});
491 columnNames{2*j+1} = sprintf('%s_Order', names{j});
492 end
493
494 %< Create table
495 T = cell2table(tableData, 'VariableNames', columnNames);
496 end

498 function obj = drawImpl(obj, errors, names, rates, plotTitle)
499 % DRAWIMPL Create single subplot in tiled layout.
500
501 nexttile;
502
503 [~, nTypes] = size(errors);
504 plotColors = lines(nTypes);
505
506 hold on;
507
508 %< Plot actual errors for this component
509 for j = 1:nTypes
510 loglog(obj.Resolutions, errors(:, j), '-o', ...
511 'Color', plotColors(j, :), ...
512 'DisplayName', sprintf('%s Error', names{j}), ...
513 'LineWidth', 2, 'MarkerSize', 6);
514
515 %< Plot theoretical convergence rates if provided
516 if ~isempty(rates) && length(rates) >= j && rates(j) > 0
517 refErrors = errors(1, j) * (obj.Resolutions / obj.Resolutions(1)).^(rates(j));
518 loglog(obj.Resolutions, refErrors, '--', ...
519 'Color', plotColors(j, :), ...
520 'DisplayName', sprintf('%s Rate %g', names{j}, rates(j)), ...
521 'LineWidth', 1.5);
522 end
523 end
524
525 %< Format subplot
526 set(gca, 'XScale', 'log', 'YScale', 'log');
527 xlabel('Resolution');
528 ylabel('Error');
529 title(plotTitle);
530 legend('show', 'Location', 'best');
531 grid on;
532 hold off;
533 end
534 end
