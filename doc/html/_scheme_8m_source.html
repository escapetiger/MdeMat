2 % SCHEME Base class for all numerical schemes.
3 %
4 % Scheme provides the fundamental framework for numerical simulation
5 % schemes including time integration, visualization, analysis, and
6 % performance monitoring. It defines the common interface and
7 % workflow for all concrete scheme implementations.
8 %
9 % The class manages the complete simulation lifecycle through
10 % abstract methods that must be implemented by subclasses:
11 % initialize, preStep, step, postStep, and finalize. It coordinates
12 % temporal discretization, performance timing, visualization, and
13 % error analysis components.
14 %
15 % Notes:
16 % Subclasses must implement all abstract methods. The run method
17 % provides the main time integration loop with automatic performance
18 % monitoring and step management.
19 %
20 % See also:
21 % approx.odeint.OdeIntegrator, core.chrono.Timer,
22 % profilers.analysis.Analyzer, profilers.visual.Visualizer

76 function obj = setTimeDiscretization(obj, tDiscId, tFinal)
77 % SETTIMEDISCRETIZATION Set temporal discretization method.
78 %
79 % obj = setTimeDiscretization(obj, tDiscId, tFinal) creates
80 % and configures the appropriate time integration method
81 % based on the specified identifier. Supports various
82 % implicit methods including backward Euler, BDF, and SDIRK
83 % schemes.
84 %
85 % Inputs:
86 % obj - The Scheme object
87 % tDiscId - Time discretization identifier string:
88 %< 'be' - Backward Euler (1st order)
89 %< 'bdf2' - 2nd order Backward Differentiation Formula
90 %< 'bdf3' - 3rd order Backward Differentiation Formula
91 %< 'sdirk2' - 2nd order Singly Diagonally Implicit Runge-Kutta
92 %< 'sdirk3' - 3rd order Singly Diagonally Implicit Runge-Kutta
93 % tFinal - Final integration time (positive scalar)
94 %
95 % Outputs:
96 % obj - The Scheme object with configured time discretization
97
98 switch tDiscId
99 case 'be'
100 cls = 'BeIntegrator';
101 case 'bdf2'
102 cls = 'Bdf2Integrator';
103 case 'bdf3'
104 cls = 'Bdf3Integrator';
105 case 'sdirk2'
106 cls = 'Sdirk2Integrator';
107 case 'sdirk3'
108 cls = 'Sdirk3Integrator';
109 end
110 obj.tDisc = approx.odeint.(cls)(tFinal);
111 end

149 % RUN Execute the complete time integration loop.
150 %
151 % state = run(obj, state) performs the full time integration
152 % from initial to final time, managing preprocessing,
153 % integration steps, and postprocessing with performance
154 % monitoring throughout the simulation. The method
155 % coordinates the abstract step methods defined by
156 % subclasses.
157 %
158 % Inputs:
159 % obj - The Scheme object
160 % state - Initial solution state structure
161 %
162 % Outputs:
163 % state - Final solution state after time integration
164
165 %< Time integration loop
166 while ~obj.tDisc.timeline.isFinished
167 %< Preprocess step
168 state = obj.preStep(state);
169
170 %< Integration step with performance monitoring
171 msg = sprintf( ...
172 "[%d] now = %f; step size = %f; runtime = %.2f s;", ...
173 obj.tDisc.timeline.count, ...
174 obj.tDisc.timeline.now, ...
175 obj.tDisc.timeline.dt, ...
176 obj.timer.records(1).duration);
177 obj.timer.start(1, msg);
178 state = obj.step(state);
179 obj.timer.stop(1);
180
181 %< Postprocess step

183 end
