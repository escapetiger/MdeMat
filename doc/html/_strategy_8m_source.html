217 function obj = setColorLimits(obj, ax, u)
218 % SETCOLORLIMITS Set color axis limits based on data values.
219 %
220 % setColorLimits(obj, ax, u) configures the color axis limits
221 % for the specified axes based on the data range. Uses
222 % symmetric limits for bipolar data when enabled and applies
223 % padding to prevent visual clipping.
224
225 arguments
226 obj physics.visual.Strategy
227 ax matlab.graphics.axis.Axes
228 u double
229 end
230
231 uMin = min(u(:));
232 uMax = max(u(:));
233 absMax = max(abs(uMin), abs(uMax));
234
235 %< Use symmetric limits for bipolar data
236 if obj.UseSymmetricColorLimits && uMin < 0 && uMax > 0
237 cLimits = [-absMax, absMax];
238 else
239 %< Apply padding to data range
240 padding = obj.DefaultColorPadding * (uMax - uMin);
241 if padding == 0
242 %< Handle constant data case
243 padding = obj.DefaultColorPadding * abs(uMin);
244 if padding == 0
245 padding = 0.1; %< Minimum padding for zero data
246 end
247 end
248 cLimits = [uMin - padding, uMax + padding];
249 end
250
251 caxis(ax, cLimits);
252 end

254 function obj = syncColorLimits(obj, axes, database)
255 % SYNCCOLORLIMITS Set unified color limits for multiple plots.
256 %
257 % obj = syncColorLimits(obj, axes, database) synchronizes
258 % color limits across multiple @a axes based on the combined
259 % data range from @a database. This ensures consistent color
260 % mapping across related visualizations.
261
262 arguments
263 obj physics.visual.Strategy
264 axes matlab.graphics.axis.Axes
265 database physics.visual.Database
266 end
267
268 if database.IsEmpty || isempty(axes)
269 return;
270 end
271
272 uMin = inf;
273 uMax = -inf;
274 datasetNames = database.DatasetNames;
275 for i = 1:length(datasetNames)
276 dataset = database.getDataset(datasetNames{i});
277 u = dataset.getData('u');
278 uMin = min(uMin, min(u(:)));
279 uMax = max(uMax, max(u(:)));
280 end
281
282 absMax = max(abs(uMin), abs(uMax));
283
284 %< Apply consistent limit calculation logic
285 if obj.UseSymmetricColorLimits && uMin < 0 && uMax > 0
286 cLimits = [-absMax, absMax];
287 else
288 padding = obj.DefaultColorPadding * (uMax - uMin);
289 if padding == 0
290 padding = obj.DefaultColorPadding * abs(uMin);
291 if padding == 0
292 padding = 0.1;
293 end
294 end
295 cLimits = [uMin - padding, uMax + padding];
296 end
297
298 %< Apply limits to all axes
299 for i = 1:length(axes)
300 caxis(axes(i), cLimits);
301 end
