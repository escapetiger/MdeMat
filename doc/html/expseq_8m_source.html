1function phi = expseq(Z, n)
2 % EXPSEQ Generate exponential sequence phi_k(Z).
3 %
4 % phi = expseq(Z, n) computes the first (n+1) terms of the
5 % exponential sequence defined by:
6 %
7 % \f[
8 % \phi_0(Z) = \exp(Z)
9 % \f]
10 % \f[
11 % \phi_{k+1}(Z) = \frac{\phi_k(Z) - \phi_k(0)}{Z}
12 % \f]
13 % \f[
14 % \phi_k(0) = \frac{1}{k!}
15 % \f]
16 %
17 % The sequence is fundamental in exponential integrators and provides
18 % exact integration of polynomial source terms multiplied by
19 % exponential decay. These functions are also known as the matrix
20 % \f$\phi\f$-functions and appear in the exact solution of linear
21 % ODEs with polynomial forcing terms.
22 %
23 % For small ||Z||, series expansions are used to avoid numerical
24 % issues with division by Z. The function automatically handles
25 % both sparse and dense matrices appropriately. The implementation
26 % switches between series expansion and recurrence relation based
27 % on the Frobenius norm of Z.
28 %
29 % See also:
30 % expm, factorial, approx.odeint.ExponentialIntegrator
31
32 %< Validate inputs
33 core.except.assert(n >= 0, ...
34 'InvalidIndex', 'Index n must be non-negative.');
35 core.except.assert(ismatrix(Z) && isnumeric(Z), ...
36 'InvalidMatrix', 'Z must be a numeric matrix.');
37 core.except.assert(size(Z, 1) == size(Z, 2), ...
38 'NonSquareMatrix', 'Z must be a square matrix.');
39
40 %< Initialize output
41 phi = cell(1, n + 1);
42
43 %< Determine matrix type for identity matrix
44 if issparse(Z)
45 I = speye(size(Z));
46 else
47 I = eye(size(Z));
48 end
49
50 %< Compute phi_0(Z) = exp(Z)
51 if isinf(Z) && all(Z < 0)
52 phi{1} = zeros(size(Z));
53 else
54 phi{1} = expm(Z);
55 end
56
57 %< Return if only phi_0 is requested
58 if n == 0
59 return;
60 end
61
62 %< Check if we should use series expansion for numerical stability
63 useSeriesExpansion = norm(Z, 'fro') < 1e-2;
64
65 if useSeriesExpansion
66 %< Use series expansion for all terms when Z is small
67 phiTerms = computeSeriesExpansion(Z, I, n);
68 for k = 1:n
69 phi{k + 1} = phiTerms{k};
70 end
71 else
72 %< Use recurrence relation for larger Z
73 phiTerms = computeRecurrenceRelation(Z, I, phi{1}, n);
74 for k = 1:n
75 phi{k + 1} = phiTerms{k};
76 end
77 end
78end

80function phiTerms = computeSeriesExpansion(Z, I, n)
81 % COMPUTESERIESEXPANSION Compute phi_k using series expansion.
82 %
83 % phiTerms = computeSeriesExpansion(Z, I, n) computes phi_k(Z) using
84 % the Taylor series expansion:
85 %
86 % \f[
87 % \phi_k(Z) = \sum_{j=0}^{\infty} \frac{Z^j}{(k+j)!}
88 % \f]
89 %
90 % This series expansion is used for numerical stability when ||Z|| is
91 % small, avoiding the division by Z in the recurrence relation.
92
93 phiTerms = cell(n, 1);
94
95 %< Pre-compute powers of Z for efficiency
96 maxTerms = 20; %< Sufficient for convergence when ||Z|| < 1e-2
97 ZPowers = cell(maxTerms, 1);
98 ZPowers{1} = I; %< Z^0
99
100 for j = 2:maxTerms
101 ZPowers{j} = ZPowers{j-1} * Z;
102 end
103
104 %< Compute each phi_k using series expansion
105 for k = 1:n
106 phiK = zeros(size(Z));
107
108 %< Sum series: phi_k(Z) = sum_{j=0}^{inf} Z^j / (k+j)!
109 for j = 1:maxTerms
110 coefficient = 1 / factorial(k + j - 1);
111 term = coefficient * ZPowers{j};
112 phiK = phiK + term;
113
114 %< Check convergence
115 if j > 5 && norm(term, 'fro') < 1e-15 * norm(phiK, 'fro')
116 break;
117 end
118 end
119
120 phiTerms{k} = phiK;
121 end
122end

124function phiTerms = computeRecurrenceRelation(Z, I, phi0, n)
125 % COMPUTERECURRENCERELATION Compute phi_k using recurrence relation.
126 %
127 % phiTerms = computeRecurrenceRelation(Z, I, phi0, n) computes
128 % phi_{k+1}(Z) = (phi_k(Z) - phi_k(0)) / Z for k = 0, ..., n-1
129 % using the recurrence relation for numerical stability when ||Z|| is
130 % not small.
131
132 phiTerms = cell(n, 1);
133
134 %< Compute phi_k(0) = 1/k! for all needed values
135 phiAtZero = zeros(n + 1, 1);
136 for k = 0:n
137 phiAtZero(k + 1) = 1 / factorial(k);
138 end
139
140 %< Initialize with phi_0
141 currentPhi = phi0;
142
143 %< Apply recurrence relation
144 for k = 1:n
145 %< phi_{k+1}(Z) = (phi_k(Z) - phi_k(0)) / Z
146 numerator = currentPhi - phiAtZero(k) * I;
147
148 %< Solve Z * phi_{k+1} = numerator for phi_{k+1}
149 %< This avoids explicit matrix inversion
150 phiTerms{k} = Z \ numerator;
151
152 %< Update for next iteration
153 currentPhi = phiTerms{k};
154 end
155end
