1classdef NonuniformGrid < approx.mesh.Grid
2 % NONUNIFORMGRID Structured grid with variable element spacing.
3 %
4 % NonuniformGrid represents a structured computational grid where
5 % elements can have different sizes within each dimension while
6 % maintaining the regular tensor-product topology. This flexibility
7 % enables adaptive mesh refinement in specific regions while
8 % preserving the computational advantages of structured grids.
9 %
10 % The grid is defined by coordinate vectors along each dimension,
11 % allowing arbitrary spacing distributions such as geometric
12 % progressions, hyperbolic tangent clustering, or user-defined
13 % refinement patterns. Despite varying element sizes, the
14 % tensor-product structure enables efficient storage O(∑nᵢ) and
15 % algorithmic operations compared to unstructured meshes.
16 %
17 % Key advantages over uniform grids:
18 % - Adaptive resolution: Fine elements where needed, coarse elsewhere
19 % - Boundary layer capture: Geometric clustering near walls/interfaces
20 % - Smooth transitions: Gradual spacing changes avoid numerical artifacts
21 % - Domain conformity: Better approximation of curved boundaries
22 %
23 % Key advantages over unstructured meshes:
24 % - Tensor-product structure: Efficient matrix operations and solvers
25 % - Separable algorithms: Fast Fourier transforms and direct methods
26 % - Memory efficiency: O(d×n) storage vs O(n^d) for explicit connectivity
27 % - Neighbor queries: O(1) algorithmic connectivity vs stored tables
28 %
29 % Examples:
30 % % 1D grid with boundary layer refinement
31 % x_fine = linspace(0, 0.1, 21); % Fine near x=0
32 % x_coarse = linspace(0.1, 1, 11); % Coarse in interior
33 % x = [x_fine, x_coarse(2:end)]; % Remove duplicate
34 % grid1D = NonuniformGrid({x});
35 %
36 % % 2D grid with geometric stretching
37 % x_uniform = linspace(0, 1, 51); % Uniform in x
38 % y_stretched = [0, cumsum(1.05.^(0:29))]; % Geometric in y
39 % y_stretched = y_stretched / y_stretched(end); % Normalize to [0,1]
40 % grid2D = NonuniformGrid({x_uniform, y_stretched});
41 %
42 % % 3D grid with mixed spacing patterns
43 % x = linspace(-1, 1, 21); % Uniform x-direction
44 % y = tanh(linspace(-2, 2, 31)) / tanh(2); % Hyperbolic clustering in y
45 % z = [0, 0.01, 0.05, 0.1:0.1:0.9, 0.95, 0.99, 1]; % Manual refinement in z
46 % grid3D = NonuniformGrid({x, y, z});
47 %
48 % % Query nonuniform grid properties
49 % h = grid2D.computeMeasure(); % Minimum element spacing
50 % detJ = grid2D.computeAllElementJacobianDeterminants(); % Variable per element
51 % spacing_x = grid2D.spacings{1}; % Variable spacing in x
52 % spacing_y = grid2D.spacings{2}; % Variable spacing in y
53 %
54 % % Coordinate transformations (element-dependent)
55 % xRef = [0.3; -0.2]; % Reference coordinates
56 % xPhys = grid2D.collocate(xRef, [100, 200]); % Map to elements 100, 200
57 %
58 % % Mesh refinement (preserves spacing patterns)
59 % fineGrid = grid2D.refine(1); % Each element subdivides uniformly
60 %
61 % % Access variable grid structure
62 % centers_x = grid2D.centroids{1}; % Variable element centers in x
63 % centers_y = grid2D.centroids{2}; % Variable element centers in y
64 % nElements = grid2D.nElements; % Total elements = nx × ny
65 %
66 % Notes:
67 % Node coordinates must be strictly increasing within each dimension
68 % to ensure proper element ordering and positive Jacobian
69 % determinants. The constructor automatically computes element
70 % centers and spacings from the provided node coordinates.
71 %
72 % Refinement preserves the relative spacing ratios within each
73 % original element while subdividing uniformly. This maintains the
74 % adaptive character of the original grid at finer scales.
75 %
76 % Despite variable spacing, face normals remain axis-aligned and the
77 % coordinate transformation retains its separable structure, enabling
78 % efficient tensor-product algorithms.
79 %
80 % See also:
81 % approx.mesh.Grid, approx.mesh.SeparableGraph

84 function obj = NonuniformGrid(nodes)
85 % NONUNIFORMGRID Constructor for NonuniformGrid.
86 %
87 % obj = NonuniformGrid(nodes) creates a nonuniform grid from
88 % coordinate vectors that define element boundaries in each
89 % dimension, automatically computing element properties.
90 %
91 % Inputs:
92 % nodes - Cell array of coordinate vectors {x, y, z, ...}
93 %
94 % Outputs:
95 % obj - Constructed NonuniformGrid object
96
97 %< Validate input format
98 core.except.assert(iscell(nodes), 'InvalidInput', ...
99 'Nodes must be a cell array of coordinate vectors.');
100
101 %< Validate coordinate ordering within each dimension
102 for i = 1:length(nodes)
103 x = nodes{i};
104 core.except.assert(all(diff(x) > 0), 'InvalidCoordinates', ...
105 'Coordinates must be strictly increasing in dimension %d.', i);
106 end
107
108 %< Compute grid properties from node coordinates
109 resolution = cellfun(@(x) length(x)-1, nodes); % Elements = nodes - 1
110 spacings = cellfun(@(x) diff(x), nodes, 'Un', 0); % Element widths
111 centroids = cellfun(@(x) (x(1:end-1) + x(2:end))/2, nodes, 'Un', 0); % Element centers
112
113 %< Compute domain bounding box from coordinate extrema
114 bbox = cellfun(@(x) [min(x), max(x)], nodes, 'Un', 0);
115 bbox = horzcat(bbox{:}); % Flatten to [x_min, x_max, y_min, y_max, ...]
116
117 %< Create domain and reference geometries
118 domain = core.geometry.Orthotope(bbox);
119 reference = core.geometry.Orthotope(repmat([-1 / 2, 1 / 2], 1, length(nodes)));
120
121 %< Call parent constructor with computed properties
122 obj@approx.mesh.Grid(domain, reference, resolution, centroids, spacings);
123 end

125 function graphObj = graphify(obj)
126 % GRAPHIFY Convert nonuniform grid to separable graph representation.
127 %
128 % graphObj = graphify(obj) creates a SeparableGraph that
129 % represents the nonuniform grid's connectivity structure using
130 % 1D factors with variable vertex spacing.
131 %
132 % Inputs:
133 % obj - The NonuniformGrid object
134 %
135 % Outputs:
136 % graphObj - SeparableGraph with tensor-product structure
137
138 d = obj.nDims;
139 n = obj.resolution;
140 h = obj.spacings;
141 c = obj.centroids;
142
143 %< Generate vertex coordinates: element boundaries in each dimension
144 V = arrayfun(@(i) [c{i} - h{i} / 2, c{i}(end) + h{i}(end) / 2].', 1:d, 'Un', 0);
145
146 %< Generate edge connectivity: standard 1D chains
147 E = arrayfun(@(k) [(1:k).', (2:(k + 1)).'], n, 'Un', 0);
148
149 %< Create separable graph from 1D factors
150 graphObj = approx.mesh.SeparableGraph(V, E);
151 end

153 function Y = collocate(obj, X, L)
154 % COLLOCATE Map reference coordinates to physical coordinates.
155 %
156 % Y = collocate(obj, X) maps reference coordinates from the
157 % standard hypercube [-1/2,1/2]^d to physical coordinates
158 % across all grid elements using element-specific transformations.
159 %
160 % Y = collocate(obj, X, L) maps coordinates to specific
161 % elements identified by linear indices L.
162 %
163 % Inputs:
164 % obj - The NonuniformGrid object
165 % X - Reference coordinates (d × nPoints or cell array)
166 % L - Element linear indices (optional, vector)
167 %
168 % Outputs:
169 % Y - Physical coordinates (d × nPoints or cell array)
170
171 %< Use all elements if none specified
172 if nargin < 3, L = []; end
173
174 d = obj.nDims;
175
176 %< Convert matrix input to cell array format for processing
177 if ismatrix(X) && ~iscell(X)
178 X = num2cell(reshape(X, d, []), 2);
179 end
180
181 Y = cell(d, 1);
182
183 if isempty(L)
184 %< Transform to all elements: broadcast over all centers
185 for i = 1:d
186 a = obj.centroids{i}; % All element centers in dimension i
187 h = obj.spacings{i}; % All element spacings in dimension i
188 b = bsxfun(@times, h, X{i}(:)); % Scale by element spacing
189 Y{i} = reshape(bsxfun(@plus, a, b), 1, []); % Translate to centers
190 end
191 else
192 %< Transform to specific elements: use element-specific properties
193 M = obj.indexer.linearToMulti(L); % Convert to multi-indices
194
195 for i = 1:d
196 a = obj.centroids{i}(M(:, i)); % Centers for specified elements
197 h = obj.spacings{i}(M(:, i)); % Spacings for specified elements
198 b = bsxfun(@times, h, X{i}(:)); % Scale by element-specific spacing
199 Y{i} = reshape(bsxfun(@plus, a, b), 1, []); % Translate to centers
200 end
201 end
202
203 %< Convert back to matrix format if specific elements requested
204 if ~isempty(L), Y = cell2mat(Y); end
205 end

207 function newObj = refine(obj, nLevels)
208 % REFINE Create refined nonuniform grid preserving spacing patterns.
209 %
210 % newObj = refine(obj, nLevels) creates a refined nonuniform grid
211 % by uniformly subdividing each element while preserving the
212 % relative spacing characteristics of the original grid.
213 %
214 % Inputs:
215 % obj - The NonuniformGrid object
216 % nLevels - Number of refinement levels (non-negative integer)
217 %
218 % Outputs:
219 % newObj - Refined NonuniformGrid with preserved spacing patterns
220 %
221 % Notes:
222 % Refinement subdivides each original element into 2^nLevels
223 % sub-elements with uniform spacing within the original element.
224 % This preserves the adaptive character: regions that were fine
225 % remain proportionally fine, and coarse regions remain coarse.
226 %
227 % The total number of elements increases by 2^(nLevels×d), but
228 % the relative spacing ratios between different grid regions
229 % are maintained at the refined scale.
230
231 if nLevels == 0
232 newObj = obj;
233 return;
234 end
235
236 d = obj.nDims;
237 p = 2^nLevels;
238 x = cell(1, d);
239 for i = 1:d
240 y = obj.centroids{i};
241 n = obj.resolution(i);
242 s = obj.spacings{i};
243 a = y - s / 2;
244 b = y + s / 2;
245 h = (b - a) ./ p;
246 x{i} = zeros(1, 1+n*p);
247 x{i}(1) = a(1);
248 for j = 1:n
249 x{i}(1 + (j - 1) * p + (1:p)) = (a(j) + h(j)):h(j):b(j);
250 end
251 end
252 newObj = approx.mesh.NonuniformGrid(x);
253 end

255 function h = computeMeasure(obj)
256 % COMPUTEMEASURE Compute minimum element spacing across the grid.
257 %
258 % h = computeMeasure(obj) returns the minimum element spacing
259 % among all elements in all dimensions, providing a characteristic
260 % length scale for the nonuniform grid.
261 %
262 % Inputs:
263 % obj - The NonuniformGrid object
264 %
265 % Outputs:
266 % h - Minimum element spacing (positive scalar)
267 %
268 % Notes:
269 % For nonuniform grids, the measure is min(min(h₁), min(h₂), ..., min(hₐ))
270 % where hᵢ is the spacing array in dimension i. This provides the
271 % finest resolution anywhere in the grid, important for:
272 % - CFL stability conditions (determined by smallest element)
273 % - Error estimates (dominated by finest resolution)
274 % - Solver convergence (limited by most refined regions)
275
276 %< Find minimum spacing in each dimension, then global minimum
277 h = min(cellfun(@min, obj.spacings));
278 end

280 function detJ = computeAllElementJacobianDeterminants(obj)
281 % COMPUTEALLELEMENTJACOBIANDETERMINANTS Compute element-specific
282 % Jacobian determinants.
283 %
284 % detJ = computeAllElementJacobianDeterminants(obj) computes
285 % the Jacobian determinant for coordinate transformation for
286 % each element. For nonuniform grids, determinants vary by element.
287 %
288 % Inputs:
289 % obj - The NonuniformGrid object
290 %
291 % Outputs:
292 % detJ - Jacobian determinants for all elements (nElements × 1)
293 %
294 % Notes:
295 % For element with multi-index (i₁, i₂, ..., iₐ) and spacings
296 % h = [h₁(i₁), h₂(i₂), ..., hₐ(iₐ)]:
297 % det(J) = h₁(i₁) × h₂(i₂) × ... × hₐ(iₐ) = ∏ⱼ hⱼ(iⱼ)
298 %
299 % This element-specific determinant appears in integration:
300 % ∫_Ωₑ f(x) dx = det(Jₑ) ∫_Ω̂ f(Fₑ(ξ)) dξ where Ωₑ is element e.
301
302 %< Create tensor grid of spacings: h{i}(j) at each element
303 h = cell(1, obj.nDims);
304 [h{:}] = ndgrid(obj.spacings{:});
305
306 %< Flatten spacing arrays and compute element-wise products
307 h = cellfun(@(x) x(:).', h, 'Un', 0);
308 detJ = prod(cat(1, h{:}), 1);
309 detJ = detJ(:); % Convert to column vector
310 end

312 function J = computeAllElementJacobians(obj)
313 % COMPUTEALLELEMENTJACOBIANS Compute element-specific Jacobian matrices.
314 %
315 % J = computeAllElementJacobians(obj) computes the Jacobian
316 % matrix for coordinate transformation for each element. For
317 % nonuniform grids, Jacobians vary by element but remain diagonal.
318 %
319 % Inputs:
320 % obj - The NonuniformGrid object
321 %
322 % Outputs:
323 % J - Jacobian matrices (d × d × nElements)
324 %
325 % Notes:
326 % For element with multi-index (i₁, i₂, ..., iₐ) and spacings
327 % h = [h₁(i₁), h₂(i₂), ..., hₐ(iₐ)]:
328 % J(:,:,element) = diag([h₁(i₁), h₂(i₂), ..., hₐ(iₐ)])
329 %
330 % Despite variable spacing, Jacobians remain diagonal due to
331 % the axis-aligned coordinate transformation structure.
332
333 d = obj.nDims;
334
335 %< Create tensor grid of spacings for all elements
336 h = cell(1, d);
337 [h{:}] = ndgrid(obj.spacings{:});
338
339 %< Flatten and arrange spacings for matrix construction
340 h = cellfun(@(x) x(:).', h, 'Un', 0);
341 h = cat(1, h{:}); % d × nElements array
342
343 %< Create diagonal Jacobian matrices for each element
344 J = reshape(h, d, 1, []) .* eye(d, d);
345 end

347 function invJ = computeAllElementInverseJacobians(obj)
348 % COMPUTEALLELEMENTINVERSEJACOBIANS Compute element-specific inverse
349 % Jacobian matrices.
350 %
351 % invJ = computeAllElementInverseJacobians(obj) computes the
352 % inverse Jacobian matrix for coordinate transformation for each
353 % element. For nonuniform grids, inverse Jacobians vary by element.
354 %
355 % Inputs:
356 % obj - The NonuniformGrid object
357 %
358 % Outputs:
359 % invJ - Inverse Jacobian matrices (d × d × nElements)
360 %
361 % Notes:
362 % For element with multi-index (i₁, i₂, ..., iₐ) and spacings
363 % h = [h₁(i₁), h₂(i₂), ..., hₐ(iₐ)]:
364 % J⁻¹(:,:,element) = diag([1/h₁(i₁), 1/h₂(i₂), ..., 1/hₐ(iₐ)])
365 %
366 % Element-specific inverse Jacobians appear in gradient transformations:
367 % ∇_x f = (Jₑ)⁻ᵀ ∇_ξ f for element e.
368
369 d = obj.nDims;
370
371 %< Create tensor grid of spacings for all elements
372 h = cell(1, d);
373 [h{:}] = ndgrid(obj.spacings{:});
374
375 %< Flatten and compute reciprocal spacings
376 h = cellfun(@(x) x(:).', h, 'Un', 0);
377 h = 1 ./ cat(1, h{:}); % Reciprocal spacings: d × nElements
378
379 %< Create diagonal inverse Jacobian matrices for each element
380 invJ = reshape(h, d, 1, []) .* eye(d, d);
381 end

383 function detFJ = computeAllFaceJacobianDeterminants(obj)
384 % COMPUTEALLFACEJACOBIANDETERMINANTS Compute element-specific face
385 % Jacobian determinants.
386 %
387 % detFJ = computeAllFaceJacobianDeterminants(obj) computes
388 % Jacobian determinants for coordinate transformation on element
389 % faces. For nonuniform grids, face determinants vary by element.
390 %
391 % Inputs:
392 % obj - The NonuniformGrid object
393 %
394 % Outputs:
395 % detFJ - Face Jacobian determinants (cell array by face type)
396 %
397 % Notes:
398 % For face perpendicular to dimension k in element (i₁, ..., iₐ):
399 % det(J_face) = ∏_{j≠k} hⱼ(iⱼ) (product of spacings in other dimensions)
400 %
401 % Face determinants vary by element location due to nonuniform
402 % spacing, unlike uniform grids where they are constant per face type.
403
404 detFJ = cell(1, 2*obj.nDims);
405
406 for faceIndex = 1:2 * obj.nDims
407 %< Determine perpendicular dimension for this face type
408 dim = ceil(faceIndex/2);
409
410 %< Identify dimensions parallel to the face (all except perpendicular)
411 otherDims = 1:obj.nDims;
412 otherDims(dim) = [];
413
414 if isempty(otherDims)
415 %< 1D case: faces are points, determinant is 1 for all elements
416 detFJ{faceIndex} = ones(1, obj.nElements);
417 else
418 %< Higher dimensions: product of spacings in parallel dimensions
419 h = cell(1, obj.nDims);
420 [h{:}] = ndgrid(obj.spacings{:});
421
422 %< Extract spacings for dimensions parallel to the face
423 h_parallel = cellfun(@(x) x(:).', h(otherDims), 'Un', 0);
424
425 %< Compute element-wise products for face determinants
426 detFJ{faceIndex} = prod(cat(1, h_parallel{:}), 1);
427 end
428 end

431 function outN = computeAllOutwardNormals(obj)
432 % COMPUTEALLOUTWARDNORMALS Compute outward normal vectors for all faces.
433 %
434 % outN = computeAllOutwardNormals(obj) computes unit outward
435 % normal vectors for each face type. For nonuniform grids, normals
436 % remain axis-aligned and constant for faces of the same type.
437 %
438 % Inputs:
439 % obj - The NonuniformGrid object
440 %
441 % Outputs:
442 % outN - Outward normal vectors (cell array by face type)
443 %
444 % Notes:
445 % Despite variable element spacing, outward normals remain
446 % axis-aligned and independent of element size:
447 % - Face 2k-1: normal = -eₖ (negative k-direction, "lower" face)
448 % - Face 2k: normal = +eₖ (positive k-direction, "upper" face)
449 % where eₖ is the unit vector in dimension k.
450 %
451 % All boundary elements sharing the same face type have identical
452 % outward normals, regardless of their individual element sizes.
453
454 outN = cell(1, 2*obj.nDims);
455
456 %< Generate normal vector for each face type
457 for faceIndex = 1:2 * obj.nDims
458 %< Determine perpendicular dimension and orientation
459 dim = ceil(faceIndex/2); % Which dimension (1, 2, 3, ...)
460 isPositive = (mod(faceIndex, 2) == 0); % Upper face (even) or lower face (odd)
461
462 %< Create unit normal vector (axis-aligned, independent of spacing)
463 outN{faceIndex} = zeros(obj.nDims, 1);
464 if isPositive
465 outN{faceIndex}(dim) = +1; % Outward = positive direction
466 else
467 outN{faceIndex}(dim) = -1; % Outward = negative direction
468 end
