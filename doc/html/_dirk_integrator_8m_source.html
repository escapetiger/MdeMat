1classdef DirkIntegrator < approx.odeint.OdeIntegrator
2 % DIRKINTEGRATOR Diagonally Implicit Runge-Kutta integrator.
3 %
4 % DirkIntegrator implements Diagonally Implicit Runge-Kutta (DIRK)
5 % methods for solving ordinary differential equations of the form:
6 %
7 % \f[
8 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = Lu + S,
9 % \f]
10 %
11 % where \f$M\f$ is a mass operator, \f$L\f$ is a linear operator, and
12 % \f$S\f$ is a source term. Any of these terms can be empty. For mass
13 % operator, it means it reduces to an identity. For linear and source
14 % term, it means they vanish from the equation.
15 %
16 % DIRK methods are characterized by a lower triangular Butcher
17 % tableau with identical diagonal elements, allowing each stage to be
18 % solved independently. This makes them more efficient than fully
19 % implicit Runge-Kutta methods while maintaining good stability
20 % properties.
21 %
22 % See also:
23 % approx.odeint.OdeIntegrator, approx.odeint.BeIntegrator,
24 % approx.odeint.BdfIntegrator
25
26 properties
27 A % Butcher tableau coefficient matrix (lower triangular)
28 b % Stage weights vector for final update
29 c % Stage time coefficients vector
30 K % Cell array of derivative approximations for each stage
31 end

87 function obj = setRhs(obj, i, S, M)
88 % SETRhs Set the right-hand side vector for stage i.
89 %
90 % obj = setRhs(obj, i, S, M) constructs the right-hand side
91 % vector using mass operator, accumulated previous stages,
92 % and source term.
93
94 arguments
95 obj approx.odeint.DirkIntegrator
96 i {mustBePositive, mustBeInteger}
97 S {mustBeA(S, {'numeric', 'function_handle'})} % Source term
98 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
99 end
100
101 dt = obj.Timeline.StepSize;
102 U0 = obj.U0{1};
103
104 %< Add mass term contribution from previous solution
105 if isempty(M)
106 obj.Rhs{i} = U0;
107 elseif iscell(M)
108 obj.Rhs{i} = M{2} * U0;
109 if ~isempty(M{1})
110 obj.Rhs{i} = obj.Rhs{i} + M{1};
111 end
112 else
113 obj.Rhs{i} = M * U0;
114 end
115
116 %< Add accumulated term from previous stages
117 for j = 1:(i - 1)
118 obj.Rhs{i} = obj.Rhs{i} + dt * obj.A(i, j) * obj.K{j};
119 end
120
121 %< Add source term contribution with diagonal coefficient
122 if ~isempty(S)
123 obj.Rhs{i} = obj.Rhs{i} + dt * obj.A(i, i) * S;
124 end
125 end

127 function U = stage(obj, i, L, S, M)
128 % STAGE Compute stage i of the DIRK method.
129 %
130 % U = stage(obj, i, L, S, M) computes the solution at stage i
131 % by solving the implicit system and computing the derivative
132 % approximation for use in subsequent stages.
133
134 arguments
135 obj approx.odeint.DirkIntegrator
136 i {mustBePositive, mustBeInteger}
137 L {mustBeA(L, {'numeric', 'function_handle'})} % Linear operator
138 S {mustBeA(S, {'numeric', 'function_handle'})} % Source term
139 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
140 end
141
142 t0 = obj.Timeline.Now;
143 dt = obj.Timeline.StepSize;
144 U0 = obj.U0{1};
145
146 %< Evaluate linear term at stage time
147 Le = obj.evaluateOperator(L, U0, [t0+obj.c(i)*dt, t0]);
148
149 %< Evaluate source term at stage time
150 Se = obj.evaluateOperator(S, U0, [t0+obj.c(i)*dt, t0]);
151
152 %< Update Lhs matrix if step size changed
153 if obj.Timeline.HasStepSizeChanged
154 %< Evaluate Lhs mass term
155 Me = obj.evaluateLhsMass(M, t0+obj.c(i)*dt);
156
157 %< Construct Lhs matrix
158 obj.setLhs(i, Le, Me);
159 end
160
161 %< Evaluate Rhs mass term at previous time level
162 Me = obj.evaluateRhsMass(i, M, [t0 + obj.c(i)*dt, t0]);
163
164 %< Construct Rhs vector
165 obj.setRhs(i, Se, Me);
166
167 %< Solve linear system for stage approximation
168 U = obj.Solver{i}.solve(obj.Lhs{i}, obj.Rhs{i});
169
170 %< Compute derivative approximation for this stage
171 obj.K{i} = 0;
172 if ~isempty(Le)
173 obj.K{i} = obj.K{i} + Le * U;
174 end
175 if ~isempty(Se)
176 obj.K{i} = obj.K{i} + Se;
177 end
178 end

180 function U = step(obj, L, S, M)
181 % STEP Advance one time step using the DIRK method.
182 %
183 % U = step(obj, L, S, M) performs a complete time step by
184 % computing all stages and combining them according to the
185 % Butcher tableau weights.
186
187 arguments
188 obj approx.odeint.DirkIntegrator
189 L {mustBeA(L, {'numeric', 'function_handle'})} % Linear operator
190 S {mustBeA(S, {'numeric', 'function_handle'})} % Source term
191 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
192 end
193
194 t0 = obj.Timeline.Now;
195 dt = obj.Timeline.StepSize;
196 U0 = obj.U0{1};
197 s = obj.NStages;
198
199 %< Compute all stage approximations
200 for i = 1:s
201 [~] = obj.stage(i, L, S, M);
202 end
203
204 %< Update Lhs matrix for final solve if step size changed
205 if obj.Timeline.HasStepSizeChanged
206 %< Evaluate Lhs mass term
207 Me = obj.evaluateLhsMass(M, t0+dt);
208 obj.Lhs{s+1} = Me;
209 end
210
211 %< Evaluate mass term for final solve
212 Me = obj.evaluateRhsMass(s+1, M, [t0+dt, t0]);
213
214 %< Add mass term contribution from previous solution
215 if isempty(Me)
216 obj.Rhs{s+1} = U0;
217 elseif iscell(Me)
218 obj.Rhs{s+1} = Me{2} * U0;
219 if ~isempty(Me{1})
220 obj.Rhs{s+1} = obj.Rhs{s+1} + Me{1};
221 end
222 else
223 obj.Rhs{s+1} = Me * U0;
224 end
225
226 %< Add weighted stage derivatives
227 for i = 1:s
228 obj.Rhs{s+1} = obj.Rhs{s+1} + dt * obj.b(i) * obj.K{i};
229 end
230
231 %< Solve for final step approximation
232 U = obj.Solver{s+1}.solve(obj.Lhs{s+1}, obj.Rhs{s+1});
233 end

235 function obj = setLhsA(obj, i, L, M, P, A, B)
236 % SETLHSA Set the augmented left-hand side matrix for stage i.
237 %
238 % obj = setLhsA(obj, i, L, M, P, A, B) constructs the
239 % augmented left-hand side block matrix for stage i of the
240 % DIRK system with constraints:
241 %
242 % \f[
243 % \begin{bmatrix}
244 % M - \Delta t \cdot A_{ii} L & -\Delta t \cdot A_{ii}
245 % P \\ -A & -B
246 % \end{bmatrix}
247 % \f]
248 %
249 % where \f$A_{ii}\f$ is the diagonal coefficient of the
250 % Butcher tableau for stage i, and the constraint system
251 % couples the differential variables through @a A and
252 % algebraic variables through @a B. The coupling operator @a
253 % P connects the algebraic variables to the differential
254 % equation.
255
256 arguments
257 obj approx.odeint.DirkIntegrator
258 i {mustBePositive, mustBeInteger}
259 L {mustBeA(L, {'numeric', 'function_handle'})} % Linear operator
260 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
261 P {mustBeA(P, {'numeric', 'function_handle'})} % Coupling operator
262 A {mustBeA(A, {'numeric', 'function_handle'})} % Constraint matrix
263 B {mustBeA(B, {'numeric', 'function_handle'})} % Constraint coupling matrix
264 end
265
266 dt = obj.Timeline.StepSize;
267
268 obj.Lhs{i} = cell(2, 2);
269
270 %< Add mass term
271 obj.Lhs{i}{1, 1} = M;
272
273 %< Add linear term contribution with diagonal coefficient
274 if ~isempty(L)
275 obj.Lhs{i}{1, 1} = obj.Lhs{i}{1, 1} - dt * obj.A(i, i) * L;
276 end
277
278 obj.Lhs{i}{1, 2} = -dt * obj.A(i, i) * P;
279
280 obj.Lhs{i}{2, 1} = -A;
281
282 obj.Lhs{i}{2, 2} = -B;
283
284 obj.Lhs{i} = core.linalg.block(obj.Lhs{i});
285 end

287 function obj = setRhsA(obj, i, S, M, C)
288 % SETRHSA Set the augmented right-hand side vector for stage i.
289 %
290 % obj = setRhsA(obj, i, S, M, C) constructs the augmented
291 % right-hand side block vector for stage i of the DIRK system
292 % with constraints:
293 %
294 % \f[
295 % \begin{bmatrix}
296 % M u^{(n)} + \sum_{j=1}^{i-1} \Delta t \cdot A_{ij}
297 % K_j + \Delta t \cdot A_{ii} S \\ C
298 % \end{bmatrix}
299 % \f]
300 %
301 % where \f$u^{(n)}\f$ is the solution at the previous time
302 % step, \f$K_j\f$ are the derivative approximations from
303 % previous stages, @a S is the source term for the
304 % differential equation, and @a C is the constraint source
305 % term for the algebraic equation.
306
307 arguments
308 obj approx.odeint.DirkIntegrator
309 i {mustBePositive, mustBeInteger}
310 S {mustBeA(S, {'numeric', 'function_handle'})} % Source term
311 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
312 C {mustBeA(C, {'numeric', 'function_handle'})} % Constraint source term
313 end
314
315 dt = obj.Timeline.StepSize;
316 U0 = obj.U0{1};
317
318 obj.Rhs{i} = cell(2, 1);
319
320 %< Add mass term contribution from previous solution
321 if isempty(M)
322 obj.Rhs{i}{1} = U0(1:size(S, 2));
323 elseif iscell(M)
324 obj.Rhs{i}{1} = M{2} * U0(1:size(M{2}, 2));
325 if ~isempty(M{1})
326 obj.Rhs{i}{1} = obj.Rhs{i} + M{1}(1:size(M{2}, 2));
327 end
328 else
329 obj.Rhs{i}{1} = M * U0(1:size(M, 2));
330 end
331
332 %< Add accumulated term from previous stages
333 for j = 1:(i - 1)
334 obj.Rhs{i}{1} = obj.Rhs{i}{1} + dt * obj.A(i, j) * obj.K{j};
335 end
336
337 %< Add source term contribution with diagonal coefficient
338 if ~isempty(S)
339 obj.Rhs{i}{1} = obj.Rhs{i}{1} + dt * obj.A(i, i) * S;
340 end
341
342 obj.Rhs{i}{2} = C;
343
344 obj.Rhs{i} = core.linalg.block(obj.Rhs{i});
345 end

347 function U = stageA(obj, i, L, S, M, P, A, B, C)
348 % STAGEA Compute stage i of the augmented DIRK method.
349 %
350 % U = stageA(obj, i, L, S, M, P, A, B, C) computes the
351 % solution at stage i by solving the augmented implicit
352 % system with constraints. The method handles both
353 % differential and algebraic variables simultaneously:
354 %
355 % \f[
356 % \begin{bmatrix}
357 % M - \Delta t \cdot A_{ii} L & -\Delta t \cdot A_{ii}
358 % P \\ -A & -B
359 % \end{bmatrix}
360 % \begin{bmatrix} u^{(i)} \\ v^{(i)} \end{bmatrix} =
361 % \begin{bmatrix} \text{rhs}_u \\ C \end{bmatrix}
362 % \f]
363 %
364 % where \f$u^{(i)}\f$ are the differential variables,
365 % \f$v^{(i)}\f$ are the algebraic variables, and the
366 % right-hand side includes contributions from the mass
367 % operator, previous stages, and source terms. The derivative
368 % approximation for this stage is computed for use in
369 % subsequent stages.
370
371 arguments
372 obj approx.odeint.DirkIntegrator
373 i {mustBePositive, mustBeInteger}
374 L {mustBeA(L, {'numeric', 'function_handle'})} % Linear operator
375 S {mustBeA(S, {'numeric', 'function_handle'})} % Source term
376 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
377 P {mustBeA(P, {'numeric', 'function_handle'})} % Coupling operator
378 A {mustBeA(A, {'numeric', 'function_handle'})} % Constraint matrix
379 B {mustBeA(B, {'numeric', 'function_handle'})} % Constraint coupling matrix
380 C {mustBeA(C, {'numeric', 'function_handle'})} % Constraint source term
381 end
382
383 t0 = obj.Timeline.Now;
384 dt = obj.Timeline.StepSize;
385 U0 = obj.U0{1};
386
387 %< Evaluate linear term at stage time
388 Le = obj.evaluateOperator(L, U0, [t0+obj.c(i)*dt, t0]);
389
390 %< Evaluate source term at stage time
391 Se = obj.evaluateOperator(S, U0, [t0+obj.c(i)*dt, t0]);
392
393 Pe = obj.evaluateOperator(P, U0, [t0+obj.c(i)*dt, t0]);
394
395 Ae = obj.evaluateOperator(A, U0, [t0+obj.c(i)*dt, t0]);
396
397 Be = obj.evaluateOperator(B, U0, [t0+obj.c(i)*dt, t0]);
398
399 Ce = obj.evaluateOperator(C, U0, [t0+obj.c(i)*dt, t0]);
400
401 %< Update Lhs matrix if step size changed
402 if obj.Timeline.HasStepSizeChanged
403 %< Evaluate Lhs mass term
404 Me = obj.evaluateLhsMass(M, t0+obj.c(i)*dt);
405
406 %< Construct Lhs matrix
407 obj.setLhsA(i, Le, Me, Pe, Ae, Be);
408 end
409
410 %< Evaluate Rhs mass term at previous time level
411 Me = obj.evaluateRhsMass(i, M, [t0 + obj.c(i)*dt, t0]);
412
413 %< Construct Rhs vector
414 obj.setRhsA(i, Se, Me, Ce);
415
416 %< Solve linear system for stage approximation
417 U = obj.Solver{i}.solve(obj.Lhs{i}, obj.Rhs{i});
418
419 Pe = obj.evaluateOperator(P, U, [t0+obj.c(i)*dt, t0]);
420
421 Le = obj.evaluateOperator(L, U, [t0+obj.c(i)*dt, t0]);
422
423 Se = obj.evaluateOperator(S, U, [t0+obj.c(i)*dt, t0]);
424
425 %< Compute derivative approximation for this stage
426 obj.K{i} = 0;
427 if ~isempty(Le)
428 obj.K{i} = obj.K{i} + Le * U(1:size(Le, 2));
429 end
430 if ~isempty(Pe)
431 obj.K{i} = obj.K{i} + Pe * U(end-size(Pe,2)+1:end);
432 end
433 if ~isempty(Se)
434 obj.K{i} = obj.K{i} + Se;
435 end
436 end

438 function U = stepA(obj, L, S, M, P, A, B, C)
439 % STEPA Advance one time step using the augmented DIRK method.
440 %
441 % U = stepA(obj, L, S, M, P, A, B, C) performs a complete time step
442 % for the constrained system by computing all augmented stages and
443 % combining them according to the Butcher tableau weights. The method
444 % solves the differential-algebraic equation system:
445 %
446 % \f[
447 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = Lu + Pv + S, \quad
448 % 0 = Au + Bv + C
449 % \f]
450 %
451 % where @a u represents the differential variables, @a v represents
452 % the algebraic variables, and the constraint equation couples them
453 % through operators @a A and @a B. Each stage is solved implicitly
454 % using the diagonal structure of the DIRK method.
455
456 arguments
457 obj approx.odeint.DirkIntegrator
458 L {mustBeA(L, {'numeric', 'function_handle'})} % Linear operator
459 S {mustBeA(S, {'numeric', 'function_handle'})} % Source term
460 M {mustBeA(M, {'numeric', 'function_handle', 'cell'})} % Mass operator
461 P {mustBeA(P, {'numeric', 'function_handle'})} % Coupling operator
462 A {mustBeA(A, {'numeric', 'function_handle'})} % Constraint matrix
463 B {mustBeA(B, {'numeric', 'function_handle'})} % Constraint coupling matrix
464 C {mustBeA(C, {'numeric', 'function_handle'})} % Constraint source term
465 end
466
467 t0 = obj.Timeline.Now;
468 dt = obj.Timeline.StepSize;
469 U0 = obj.U0{1};
470 s = obj.NStages;
471
472 %< Compute all stage approximations
473 for i = 1:s
474 [~] = obj.stageA(i, L, S, M, P, A, B, C);
475 end
476
477 %< Update Lhs matrix for final solve if step size changed
478 if obj.Timeline.HasStepSizeChanged
479 %< Evaluate Lhs mass term
480 Me = obj.evaluateLhsMass(M, t0+dt);
481 obj.Lhs{s+1} = cell(2, 2);
482 obj.Lhs{s+1}{1, 1} = Me;
483 obj.Lhs{s+1}{2, 1} = -A;
484 obj.Lhs{s+1}{2, 2} = -B;
485 obj.Lhs{s+1} = core.linalg.block(obj.Lhs{s+1});
486 end
487
488 %< Evaluate mass term for final solve
489 Me = obj.evaluateRhsMass(s+1, M, [t0+dt, t0]);
490
491 obj.Rhs{s+1} = cell(2, 1);
492
493 %< Add mass term contribution from previous solution
494 if isempty(Me)
495 obj.Rhs{s+1}{1} = U0(1:size(S, 2));
496 elseif iscell(Me)
497 obj.Rhs{s+1}{1} = Me{2} * U0(1:size(Me{2}, 2));
498 if ~isempty(Me{1})
499 obj.Rhs{s+1}{1} = obj.Rhs{s+1} + Me{1}(1:size(Me{2}, 2));
500 end
501 else
502 obj.Rhs{s+1}{1} = Me * U0(1:size(Me, 2));
503 end
504
505 %< Add weighted stage derivatives
506 for i = 1:s
507 obj.Rhs{s+1}{1} = obj.Rhs{s+1}{1} + dt * obj.b(i) * obj.K{i};
508 end
509
510 Ce = obj.evaluateOperator(C, U0, [t0+dt, t0]);
511
512 obj.Rhs{s+1}{2} = Ce;
513 obj.Rhs{s+1} = core.linalg.block(obj.Rhs{s+1});
514
515 %< Solve for final step approximation
516 U = obj.Solver{s+1}.solve(obj.Lhs{s+1}, obj.Rhs{s+1});
517 end
518 end
