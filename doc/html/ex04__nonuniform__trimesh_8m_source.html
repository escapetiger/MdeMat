172function quality = analyzeMeshQuality(mesh)
173% ANALYZEMESHQUALITY Compute basic mesh quality metrics.
174%
175% quality = analyzeMeshQuality(mesh) computes various quality metrics
176% for a triangular mesh including element areas and shape quality.
177
178nElements = mesh.NElements;
179areas = zeros(nElements, 1);
180qualities = zeros(nElements, 1);
181
182vertices = mesh.Vertices;
183elements = mesh.Elements;
184
185for iElement = 1:nElements
186 % Get element vertices
187 elementVertices = vertices(elements(iElement, :), :);
188
189 if size(elementVertices, 1) == 3
190 % Triangle area using cross product
191 v1 = elementVertices(2, :) - elementVertices(1, :);
192 v2 = elementVertices(3, :) - elementVertices(1, :);
193 area = 0.5 * abs(v1(1)*v2(2) - v1(2)*v2(1));
194 areas(iElement) = area;
195
196 % Triangle quality (ratio of inscribed to circumscribed circle radii)
197 % Higher values indicate better quality (equilateral triangle = 1)
198 sideLength1 = norm(elementVertices(2, :) - elementVertices(1, :));
199 sideLength2 = norm(elementVertices(3, :) - elementVertices(2, :));
200 sideLength3 = norm(elementVertices(1, :) - elementVertices(3, :));
201
202 semiPerimeter = (sideLength1 + sideLength2 + sideLength3) / 2;
203
204 if semiPerimeter > 0 && area > 0
205 inRadius = area / semiPerimeter;
206 circumRadius = (sideLength1 * sideLength2 * sideLength3) / (4 * area);
207 qualities(iElement) = 2 * inRadius / circumRadius;
208 else
209 qualities(iElement) = 0;
210 end
211 end
212end
213
214% Remove zero entries
215validAreas = areas(areas > 0);
216validQualities = qualities(qualities > 0);
217
218quality.avgArea = mean(validAreas);
219quality.minArea = min(validAreas);
220quality.maxArea = max(validAreas);
221quality.areaRatio = quality.maxArea / quality.minArea;
222quality.avgQuality = mean(validQualities);
223quality.minQuality = min(validQualities);
224end
