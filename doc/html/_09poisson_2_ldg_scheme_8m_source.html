79 function state = step(obj, state)
80 % STEP Solve the Poisson equation.
81 %
82 % state = step(obj, state) solves the linear system using
83 % the assembled matrices. The method applies appropriate
84 % boundary conditions and returns the updated state with
85 % the solution coefficients.
86
87 arguments
88 obj physics.poisson.PoissonScheme
89 state physics.state.SpatialState
90 end
91
92 switch obj.Config.bcType
93 case 'periodic'
94 [obj.A, obj.b] = obj.Elliptic.forceCompletion(obj.A, obj.b);
95 U = obj.Solver.solve(obj.A, obj.b);
96 state.Dofs.U = U(1:end-1);
97 case 'dirichlet'
98 state.Dofs.U = obj.Solver.solve(obj.A, obj.b);
99 end
