94 function L = multiToLinear(obj, M, options)
95 % MULTITOLINEAR Convert multi-indices to linear indices.
96 %
97 % L = multiToLinear(obj, M) converts multi-indices @a M to
98 % linear indices based on tensor dimensions.
99 %
100 % L = multiToLinear(obj, M, Name=Value) uses boundary
101 % condition specified by options.
102 %
103 % Notes:
104 % Padding modes include 'empty' (out-of-range set to 0),
105 % 'wrap' (wraps around), 'edge' (clamps to edges),
106 % 'reflect' (reflects excluding edges), and 'symmetric'
107 % (reflects including edges).
108
109 arguments
110 obj core.linalg.MultiIndexer
111 M {mustBeInteger}
112 options.pad{mustBeTextScalar} = 'empty'
113 end
114
115 core.except.assert(ismember(options.pad, obj.ValidPads), ...
116 'InvalidInput', 'Padding mode is invalid.');
117
118 core.except.assert(~isempty(obj.Shape), 'MissingShape', ...
119 'Shape must be set before using this method.');
120
121 nd = obj.NDims;
122 core.except.assert(ismatrix(M) && size(M, 2) == nd, ...
123 'InvalidInput', 'Multi-indices must have %d columns.', nd);
124
125 ns = obj.Shape;
126 switch lower(options.pad)
127 case 'wrap'
128 M = mod(M - 1, ns) + 1;
129 case 'edge'
130 M = min(M, ns);
131 M = max(M, ones(1, nd));
132 case 'reflect'
133 period = 2 * ns - 2;
134 P = mod(M - 1, period);
135 mask = P >= ns;
136 Q = repmat(period, size(P, 1), 1);
137 P(mask) = Q(mask) - P(mask);
138 M = P + 1;
139 case 'symmetric'
140 period = 2 * ns;
141 P = mod(M - 1, period);
142 mask = P >= ns;
143 Q = repmat(period-1, size(P, 1), 1);
144 P(mask) = Q(mask) - P(mask);
145 M = P + 1;
146 otherwise
147 end
148 L = (M - 1) * obj.Strides(:) + 1;
149 L(any(M < 1, 2) | any(M > obj.Shape, 2)) = 0;
150 end
