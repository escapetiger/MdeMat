110 function x = iterative(obj, A, b)
111 % ITERATIVE Call iterative methods.
112 if obj.checkSpd(A)
113 maxIter = obj.IterOpts.maxIter;
114 tol = obj.IterOpts.tol;
115 [x, f, rel, iter] = pcg(A, b, tol, maxIter);
116 core.except.verify(f == 0, 'ConvergenceFailed', ...
117 ['PCG did not converge. ', ...
118 'Flag: %d, Relative: %g, Iterations: %d'], f, rel, iter);
119 elseif issparse(A)
120 maxIter = min(obj.IterOpts.maxIter, size(A, 1));
121 restart = obj.IterOpts.restart;
122 tol = obj.IterOpts.tol;
123 if obj.fastCondest(A) > obj.CondTh
124 p = amd(A); Ap = A(p,p); bp = b(p);
125 setup = struct('type','ilutp','droptol',1e-12,'udiag',1);
126 [L,U] = ilu(Ap,setup);
127 [xp, f, rel, iter] = gmres(Ap, bp, restart, tol, maxIter, L, U);
128 x(p) = xp;
129 x = x.';
130 else
131 [x, f, rel, iter] = gmres(A, b, restart, tol, maxIter);
132 end

134% if obj.fastCondest(A) > obj.CondTh
135% amgData = obj.amgSetup(A);
136% P = @(x) obj.amgVCycle(amgData, 1, x);
137% x0 = zeros(size(b));
138% [x, f, rel, iter] = gmres(A, b, restart, tol, maxIter, P, [], x0);
139% else
140% [x, f, rel, iter] = gmres(A, b, restart, tol, maxIter);
141% end

144 ['GMRES did not converge. ', ...

170 function amgData = amgSetup(obj, A)
171 amgData = cell(1, obj.AmgOpts.maxLevels);
172
173 A0 = A;
174 maxRetries = 5; %< Limit retries to avoid infinite loops
175
176 fprintf('Building AMG hierarchy:\n');
177
178 for level = 1:obj.AmgOpts.maxLevels
179 n = size(A0, 1);
180
181 if n <= obj.AmgOpts.coarseSize
182 break;
183 end
184
185 fprintf(' Level %d (%d dofs): ', level, n);
186
187 amgData{level} = struct();
188 amgData{level}.A = A0;
189 amgData{level}.size = n;
190
191 retryCount = 0;
192 successful = false;
193
194 while ~successful && retryCount < maxRetries
195 [P, Ac] = obj.amgCoarsen(A0);
196 nc = size(Ac, 1);
197 coarseRatio = nc / n;
198
199 if coarseRatio < obj.AmgOpts.minCoarseRatio
200 obj.AmgOpts.theta = min(0.9, obj.AmgOpts.theta*1.5); %< less aggressive
201 retryCount = retryCount + 1;
202 continue;
203 end
204
205 if coarseRatio > obj.AmgOpts.maxCoarseRatio
206 obj.AmgOpts.theta = max(0.1, obj.AmgOpts.theta*0.8); %< more aggressive
207 retryCount = retryCount + 1;
208 continue;
209 end
210
211 successful = true;
212 fprintf('-> %d (ratio=%.2f)\n', nc, coarseRatio);
213 amgData{level}.P = P;
214 amgData{level}.R = P';
215 A0 = Ac;
216 end
217
218 if ~successful
219 fprintf('failed, stopping\n');
220 break;
221 end
222 end
223
224 amgData{level}.A = A0;
225 amgData{level}.size = size(A0, 1);
226 amgData = amgData(1:level);
227
228 %< Setup smoothers
229 for level = 1:length(amgData)
230 A0 = amgData{level}.A;
231 n0 = size(A0, 1);
232
233 if level == length(amgData) && n0 <= 1000
234 %< Direct solve for small coarsest level
235 amgData{level}.smoother = 'direct';
236 try
237 [L, U, PLU] = lu(double(A0));
238 amgData{level}.L = L;
239 amgData{level}.U = U;
240 amgData{level}.PLU = PLU;
241 amgData{level}.useLU = true;
242 catch
243 amgData{level}.useLU = false;
244 end
245 else
246 %< Jacobi smoother
247 amgData{level}.smoother = 'jacobi';
248 d = diag(A0);
249 dSafe = d;
250 smallDiag = abs(d) < 1e-14;
251 if any(smallDiag)
252 rowNorms = sum(abs(A0), 2);
253 dSafe(smallDiag) = max(rowNorms(smallDiag), 1e-12);
254 end
255 nd = length(d);
256 nds = length(dSafe);
257 amgData{level}.DInv = spdiags(1./dSafe, 0, nds, nds);
258 amgData{level}.N = A0 - spdiags(d, 0, nd, nd);
259 end
260 end

263 function [P, Ac] = amgCoarsen(obj, A)
264 %< Build coarse level using simple aggregation
265 n = size(A, 1);
266
267 %< Build strong connections
268 S = obj.amgStrongConnect(A);
269
270 %< Simple aggregation-based coarsening
271 [J, nc] = obj.amgAggregate(S);
272
273 %< Build prolongation
274 P = sparse(1:n, J, 1, n, nc);
275
276 %< Galerkin coarse operator
277 R = P';
278 Ac = R * A * P;
279
280 %< Ensure coarse matrix is well-conditioned
281 Ac = Ac + 1e-12 * speye(nc);
282 end

284 function S = amgStrongConnect(obj, A)
285 %< Build strong connection matrix
286 [i, j, v] = find(A);
287 n = size(A, 1);
288
289 %< Remove diagonal
290 offDiag = (i ~= j);
291 iOff = i(offDiag);
292 jOff = j(offDiag);
293 vOff = abs(v(offDiag));
294
295 %< Find strong connections
296 maxRow = accumarray(iOff, vOff, [n, 1], @max, 0);
297 maxRow(maxRow == 0) = 1;
298 strongMask = vOff >= obj.AmgOpts.theta * maxRow(iOff);
299
300 S = sparse(iOff(strongMask), jOff(strongMask), 1, n, n);
301 S = S | S'; %< Make symmetric
302 end

304 function [J, nc] = amgAggregate(obj, S)
305 %< Optimized simple maximal independent set aggregation
306 n = size(S, 1);
307 J = zeros(n, 1);
308 visited = false(n, 1);
309 nc = 0;
310
311 %< Pre-extract sparse matrix structure for efficiency
312 [rows, cols] = find(S);
313
314 %< Build adjacency list for faster neighbor access
315 adjList = cell(n, 1);
316 for i = 1:n
317 adjList{i} = [];
318 end
319
320 for k = 1:length(rows)
321 i = rows(k);
322 j = cols(k);
323 if i ~= j % Skip diagonal
324 adjList{i}(end +1) = j;
325 end
326 end
327
328 %< Process nodes in order of decreasing connectivity (greedy strategy)
329 degrees = zeros(n, 1);
330 for i = 1:n
331 degrees(i) = length(adjList{i});
332 end
333 [~, order] = sort(degrees, 'descend');
334
335 for idx = 1:n
336 i = order(idx);
337 if ~visited(i)
338 nc = nc + 1;
339 J(i) = nc;
340 visited(i) = true;
341
342 %< Add unvisited neighbors efficiently
343 neighbors = adjList{i};
344 if ~isempty(neighbors)
345 unvisitedMask = ~visited(neighbors);
346 unvisitedNeighbors = neighbors(unvisitedMask);
347 if ~isempty(unvisitedNeighbors)
348 J(unvisitedNeighbors) = nc;
349 visited(unvisitedNeighbors) = true;
350 end
351 end
352 end
353 end

356 function x = amgVCycle(obj, amgData, level, b)
357 %< V-cycle implementation
358 if level == length(amgData)
359 %< Coarsest level solve
360 x = obj.direct(amgData{level}.A, b);
361 return;
362 end
363
364 %< Initialize
365 x = zeros(size(b));
366
367 %< Pre-smoothing
368 x = obj.amgSmooth(amgData{level}, b, x, obj.AmgOpts.maxPreStep);
369
370 %< Coarse grid correction
371 A = amgData{level}.A;
372 r = b - A * x;
373
374 %< Restrict
375 R = amgData{level}.R;
376 rc = R * r;
377
378 %< Recursive solve
379 ec = obj.amgVCycle(amgData, level+1, rc);
380
381 %< Prolongate and correct
382 P = amgData{level}.P;
383 x = x + P * ec;
384
385 %< Post-smoothing
386 x = obj.amgSmooth(amgData{level}, b, x, obj.AmgOpts.maxPostStep);
387 end

403 function kappa = fastCondest(A)
404 % FASTCONDEST Fast condition number estimate for large sparse
405 % matrices Uses power iteration and inverse power iteration for
406 % largest/smallest eigenvalues
407
408 n = size(A, 1);
409 maxIter = 20; % Keep iterations low for speed
410 tol = 1e-3; % Relaxed tolerance for estimate
411
412 %< Estimate largest eigenvalue using power iteration
413 v = randn(n, 1);
414 v = v / norm(v);
415
416 for iter = 1:maxIter
417 vOld = v;
418 v = A * v;
419 maxLambda = norm(v);
420 v = v / maxLambda;
421
422 if norm(v-vOld) < tol
423 break;
424 end
425 end
426
427 %< Estimate smallest eigenvalue using diagonal dominance
428 %< approximation This is much faster than inverse power
429 %< iteration
430 d = abs(diag(A));
431 offDiagSums = sum(abs(A), 2) - d;
432
433 %< Gershgorin circle estimate for smallest eigenvalue
434 minLambdaEst = min(d-offDiagSums);
435
436 %< If matrix might be nearly singular, use a conservative
437 %< estimate
438 if minLambdaEst <= 0
439 % Use Frobenius norm scaling approach
440 minLambdaEst = norm(A, 'fro') / n^2;
441 end
442
443 %< Condition number estimate
444 cond_est = abs(maxLambda/minLambdaEst);
445
446 %< Cap at reasonable value to avoid infinity
447 kappa = min(cond_est, 1e16);
448 end
449 end
