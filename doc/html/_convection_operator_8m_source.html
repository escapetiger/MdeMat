1classdef ConvectionOperator < approx.assembly.Assembly
2 % CONVECTIONOPERATOR Assembly for convection operators in DG methods.
3 %
4 % ConvectionOperator provides high-level assembly of convection
5 % (advection) operators using discontinuous Galerkin discretizations.
6 % It combines volume and flux assembly components to create complete
7 % convection operators with appropriate upwind stabilization.
8 %
9 % The operator automatically selects upwind flux directions based
10 % on the velocity field, ensuring stability for convection-dominated
11 % problems. It supports both periodic and Dirichlet boundary
12 % conditions with appropriate flux treatments.
13 %
14 % The convection operator discretizes terms of the form:
15 %
16 % \f[
17 % \nabla \cdot (c \mathbf{v} u)
18 % \f]
19 %
20 % where \f$c\f$ is a coefficient, \f$v\f$ is the velocity field,
21 % and \f$u\f$ is the transported quantity.
22 %
23 % Examples:
24 % % Create convection operator with periodic boundaries
25 % space = approx.space.MeshSpace(mesh, element);
26 % operator = approx.element.L2ElementOperator(element);
27 % convOp = ConvectionOperator(space, operator, 'periodic');
28 %
29 % % Assemble convection matrix for constant velocity
30 % coefficients = [1.0; 0.5]; % [c_x; c_y]
31 % velocity = [2.0; 1.0]; % [v_x; v_y]
32 % A = convOp.linear(coefficients, velocity);
33 %
34 % % Assemble boundary condition terms
35 % boundaryFunction = @(x, t) sin(pi*x(1,:)) .* cos(pi*x(2,:));
36 % B = convOp.linearBc(coefficients, velocity, boundaryFunction);
37 %
38 % See also:
39 % approx.assembly.Assembly, approx.assembly.VolumeAssembly,
40 % approx.assembly.FluxAssembly
41
42 properties
43 volume % Volume assembly for interior terms (VolumeAssembly)
44 flux % Flux assembly for interface terms (FluxAssembly)
45 end

49 % CONVECTIONOPERATOR Constructor for ConvectionOperator.
50 %
51 % obj = ConvectionOperator(space, operator, bcType) creates a
52 % convection operator with the specified mesh space, element
53 % operator, and boundary condition type. For Dirichlet
54 % boundaries, upwind flux is automatically selected.
55 %
56 % Inputs:
57 % space - MeshSpace object
58 % operator - ElementOperator object
59 % bcType - Boundary condition type: {'periodic', 'dirichlet'}
60 %
61 % Outputs:
62 % obj - Constructed ConvectionOperator object
63
64 obj@approx.assembly.Assembly(space, operator);
65 obj.volume = approx.assembly.VolumeAssembly(space, operator);
66 obj.flux = approx.assembly.FluxAssembly(space, operator, bcType, []);
67 if strcmpi(bcType, 'dirichlet')
68 obj.flux.setFluxMode('upwind');
69 end
70 end

72 function A = linear(obj, coe, vel)
73 % LINEAR Assemble linear convection operator matrix.
74 %
75 % A = linear(obj, coe) assembles the convection operator
76 % matrix with coefficients coe, using coe as both
77 % coefficients and velocity field.
78 %
79 % A = linear(obj, coe, vel) assembles the convection operator
80 % matrix with coefficients coe and velocity field vel. The
81 % flux direction (left/right bias) is automatically selected
82 % based on the velocity sign in each dimension.
83 %
84 % Inputs:
85 % obj - The ConvectionOperator object
86 % coe - Coefficient vector for each dimension (nDims x 1)
87 % vel - Velocity vector for each dimension (nDims x 1, optional)
88 %
89 % Outputs:
90 % A - Assembled convection operator matrix (sparse)
91
92 if nargin < 3 || isempty(vel), vel = coe; end
93
94 n = obj.space.nGlobalDofs;
95 A = sparse(n, n);
96 for dim = 1:obj.space.nDims
97 if coe(dim) == 0, continue; end
98 if vel(dim) > 0
99 obj.flux.setFluxType('left');
100 else
101 obj.flux.setFluxType('right');
102 end
103 C = obj.volume.scaleConstant(dim, coe(dim));
104 T1 = obj.volume.assembleVolumePartial(dim, C);
105 T2 = obj.flux.assembleFluxPartial(dim, C);
106 T = [T1; T2];
107 B = core.linalg.sparseFromTriplets(T, n, n);
108 A = A + B;
109 end
110 end

112 function A = linearBc(obj, coe, vel, f, varargin)
113 % LINEARBC Assemble boundary condition terms for linear
114 % convection.
115 %
116 % A = linearBc(obj, coe, vel, f) assembles the boundary
117 % condition contribution matrix for linear convection with
118 % Dirichlet boundary conditions. The boundary flux direction
119 % depends on the velocity sign (inflow boundaries only).
120 %
121 % Inputs:
122 % obj - The ConvectionOperator object
123 % coe - Coefficient vector for each dimension (nDims x 1)
124 % vel - Velocity vector for each dimension (nDims x 1)
125 % f - Boundary condition function handle
126 % varargin - Input arguments passed to boundary condition
127 %
128 % Outputs:
129 % A - Boundary condition contribution matrix (sparse)
130
131 n = obj.space.nGlobalDofs;
132 A = cell(1, obj.space.nDims);
133 for dim = 1:obj.space.nDims
134 if coe(dim) == 0, continue; end
135 C = obj.volume.scaleConstant(dim, coe(dim));
136 if vel(dim) > 0
137 T = obj.flux.boundary.assembleLeftTrace(dim, C, f, varargin{:});
138 else
139 T = obj.flux.boundary.assembleRightTrace(dim, C, f, varargin{:});
140 end
141 A{dim} = core.linalg.sparseFromTriplets(T, n, max(T(:, 2)));
142 end
143
144 for dim = obj.space.nDims:-1:2
145 A{1} = A{1} + A{dim};
146 A{dim} = [];
147 end
148 A = A{1};
149 end
150 end
151end
