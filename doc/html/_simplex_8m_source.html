24 function obj = Simplex(V)
25 % SIMPLEX Construct an instance of Simplex.
26 %
27 % obj = Simplex(V) creates a simplex from the given vertices,
28 % with strict validation of vertex configuration. The
29 % vertices must be affinely independent and form a valid
30 % simplex. A nd-dimensional simplex requires exactly nd+1
31 % vertices.
32 %
33 % The @a V parameter must be a matrix of size nd×(nd+1) where
34 % each column represents a vertex. Vertices must be affinely
35 % independent to ensure the simplex is non-degenerate.
36
37 arguments
38 V {mustBeNumeric}
39 end
40
41 core.except.assert(ismatrix(V), 'InvalidInput', ...
42 'Vertices must be a matrix.');
43
44 [nd, nv] = size(V);
45 core.except.assert(nd+1 == nv, 'InvalidInput', ...
46 'Vertices must be of shape (%d, %d).', nd, nd+1);
47
48 E = V(:, 2:end) - V(:, 1);
49 core.except.assert(rank(E) >= nd, 'InvalidInput', ...
50 'Vertices must be affinely independent.');
51
52 F = arrayfun(@(i) setdiff(1:nv, i), 1:nv, 'Un', 0);
53
54 obj@core.geometry.Polytope(V, F);
55 end

63 function N = get.OuterNormals(obj)
64 % GET.OUTNORMALS Returns the outward normals of boundaries.
65
66 V = obj.Vertices;
67 nd = obj.NDims;
68 normal = zeros(nd, nd+1);
69 for i = 1:(nd + 1)
70 idx = setdiff(1:(nd + 1), i);
71 U = V(:, idx);
72 U0 = U(:, 1);
73 E = U(:, 2:end) - U0;
74 ni = null(E.');
75 if dot(U0-V(:, i), ni) < 0, ni = -ni; end
76 normal(:, i) = ni / norm(ni);
77 end
78 N = normal;
79 end

146 % CARTESIANTOBARYCENTRIC Converts Cartesian to barycentric
147 % coordinates.
148 %
149 % B = cartesianToBarycentric(obj, C) transforms points from
150 % Cartesian coordinates to barycentric coordinates with
151 % respect to the simplex vertices. Barycentric coordinates
152 % sum to 1 and are non-negative for interior points. Each
153 % coordinate represents the weight of the corresponding
154 % vertex.
155
156 arguments
157 obj core.geometry.Simplex

159 end
