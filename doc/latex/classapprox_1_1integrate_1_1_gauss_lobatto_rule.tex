\doxysubsection{Detailed Description}
Gauss-\/\+Lobatto numerical integration rule. 

\doxylink{classapprox_1_1integrate_1_1_gauss_lobatto_rule}{Gauss\+Lobatto\+Rule} implements numerical integration using the Gauss-\/\+Lobatto method, which includes both interval endpoints. This method provides high accuracy while constraining the nodes to include the boundary points.


\begin{DoxyCode}{0}
\DoxyCodeLine{\%\ Create\ a\ Gauss-\/Lobatto\ rule}
\DoxyCodeLine{rule\ =\ GaussLobattoRule();}
\DoxyCodeLine{\%\ Generate\ 5\ nodes\ and\ weights\ \textcolor{keywordflow}{for}\ interval\ [0,\ 1]}
\DoxyCodeLine{[x,\ w]\ =\ rule.generate(5);}
\DoxyCodeLine{\%\ Generate\ nodes\ and\ weights\ \textcolor{keywordflow}{for}\ custom\ interval}
\DoxyCodeLine{[x,\ w]\ =\ rule.generate(4,\ \textcolor{keyword}{false},\ -\/1,\ 1);}

\end{DoxyCode}


\begin{DoxyNote}{Note}
The Gauss-\/\+Lobatto quadrature formula includes both endpoints of 

the integration interval. For n nodes, the method has polynomial 

exactness of degree 2n-\/3. Requires n \texorpdfstring{$>$}{>}= 2 for meaningful results.
\end{DoxyNote}
