# Environment configuration

# =======================================================================
# Algorithm
# =======================================================================

OPTION=schemeName='radiation'

# -----------------------------------------------------------------------
# Time discretization
# -----------------------------------------------------------------------
# tDiscId='be': backward Euler scheme
# tDiscId='bdf2': BDF2 scheme
# tDiscId='bdf3': BDF3 scheme
# tDiscId='sdirk2': SDIRK2 scheme
# tDiscId='sdirk3': SDIRK3 scheme
OPTION=tDiscId='bdf2'

# CFL number
OPTION=cfl=0.3

# -----------------------------------------------------------------------
# Space discretization
# -----------------------------------------------------------------------

# xDiscName='uwldg': upwind DG
# xDiscName='slldg': semi-Lagrangian DG
OPTION=xDiscName='uwldg'

# spatial order
OPTION=xBasisOrder=2

# xBasisType='modal_legendre': Monic Legendre basis, 
# xBasisType='nodal_gauss_legendre': Lagrange basis on Gauss-Legendre nodes
# xBasisType='nodal_gauss_lobatto': Lagrange basis on Gauss-Lobatto nodes
OPTION=xBasisType='modal_legendre'

# xBasisPattern='Q': Qk polynomial space
# xBasisPattern='P': Pk polynomial space
OPTION=xBasisPattern='P'

# xAuxFluxType='left': alternating-left flux
# xAuxFluxType='right': alternating-right flux
# xAuxFluxType='central': central flux
OPTION=xAuxFluxType='right'

# xPrmFluxType='left': alternating-left flux
# xPrmFluxType='right': alternating-right flux
# xPrmFluxType='central': central flux
OPTION=xPrmFluxType='left'

# xAuxBoundaryJumpType='none': no jump at boundary
# xAuxBoundaryJumpType='implicit': implicit jump at boundary
# xAuxBoundaryJumpType='explicit': explicit jump at boundary
OPTION=xAuxBoundaryJumpType='implicit'

# -----------------------------------------------------------------------
# Velocity discretization
# -----------------------------------------------------------------------

# vReductionType=1: Dimension reduction by topology invariance
# vReductionType=2: Dimension reduction by symmetry
OPTION=vReductionType=1
