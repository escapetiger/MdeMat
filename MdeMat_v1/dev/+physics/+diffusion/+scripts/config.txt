# Environment configuration

# =======================================================================
# Filesystem
# =======================================================================
MATLABPATH=
SUBDIRS=
OPTION=outDir=assets

# =======================================================================
# Algorithm
# =======================================================================

# -----------------------------------------------------------------------
# Input/Output
# -----------------------------------------------------------------------
OPTION=verbose=true

# -----------------------------------------------------------------------
# Time discretization
# -----------------------------------------------------------------------
# tDiscId='be': backward Euler scheme
# tDiscId='bdf2': BDF2 scheme
# tDiscId='bdf3': BDF3 scheme
# tDiscId='sdirk2': SDIRK2 scheme
# tDiscId='sdirk3': SDIRK3 scheme
OPTION=tDiscId='sdirk3'

# CFL number
OPTION=cfl=0.5

# -----------------------------------------------------------------------
# Space discretization
# -----------------------------------------------------------------------

# xDiscId='dg': discontinuous Galerkin
OPTION=xDiscId='dg'

# spatial order
OPTION=xBasisOrder=3

# xBasisId=1: Monic Legendre basis, 
# xBasisId=2: Lagrange basis on Gauss-Legendre nodes
# xBasisId=3: Lagrange basis on Gauss-Lobatto nodes
OPTION=xBasisId=1

# xBasisPattern=1: Qk polynomial space
# xBasisPattern=2: Pk polynomial space
OPTION=xBasisPattern=2

# xDiffusionFlux=1: negative flux
# xDiffusionFlux=2: positive flux
# xDiffusionFlux=3: central flux
OPTION=xDiffusionFlux=[2,1]

# xDiffusionBoundaryJump=1: implicit jump at boundary
# xDiffusionBoundaryJump=2: explicit jump at boundary (order reduction)
OPTION=xDiffusionBoundaryJump=1
