1classdef L2FiniteElementOperator < fem.element.C0FiniteElementOperator
2 % L2FINITEELEMENTOPERATOR Operator for discontinuous finite elements.
3 %
4 % L2FiniteElementOperator extends C0 operators with flux contributions
5 % for L2 finite elements. It manages both volume and flux bilinear
6 % form data required for discontinuous Galerkin (DG) methods, where
7 % inter-element coupling is handled through numerical fluxes.
8 %
9 % The class handles the assembly of discrete differential operators
10 % that include both volume terms (from element interiors) and flux
11 % terms (from element boundaries). This dual structure is essential
12 % for DG methods where function discontinuities across element
13 % interfaces require special treatment.
14 %
15 % Flux data is organized in a tensor structure to accommodate
16 % different flow directions (inflow/outflow) and derivative orders
17 % across multiple element boundaries.
18 %
19 % Examples:
20 % % Create L2 element and its operator
21 % element = fem.element.L2FiniteElement(geometry, projector, volInt, fluxInt);
22 % element.setVolumeData(1); % Include first derivatives
23 % element.setFluxData(0); % Function values on boundaries
24 % operator = L2FiniteElementOperator(element);
25 %
26 % % Set up operator data and access gradient
27 % operator.setVolumeData();
28 % operator.setFluxData();
29 % gradOp = operator.gradient;
30 %
31 % % Access volume and flux contributions
32 % volGrad = gradOp.volumeData; % Volume gradient terms
33 % fluxGrad = gradOp.fluxData; % Boundary flux terms
34 %
35 % See also:
36 % fem.element.C0FiniteElementOperator, fem.element.L2FiniteElement,
37 % fem.data.BilinearFormData

69 function setFluxData(obj)
70 % SETFLUXDATA Set up bilinear form data for flux integrals.
71 %
72 % setFluxData(obj) creates a tensor of bilinear form matrices
73 % for flux computations, organizing data by derivative order,
74 % flux boundary, and flow direction. This structure enables
75 % efficient assembly of DG operators with proper treatment
76 % of inflow and outflow contributions.
77 %
78 % Inputs:
79 % obj - The L2FiniteElementOperator object
80 %
81 % Outputs:
82 % NULL
83
84 D = obj.fe.fluxData;
85 nf = obj.fe.nFluxes;
86 nd = 2; %< Number of directions (inflow/outflow)
87 nr = max([D.nDerivatives]) + 1; %< Maximum derivative order + 1
88
89 %< Initialize flux data tensor
90 obj.fluxData = arrayfun(@(i) fem.data.BilinearFormData(), 1:nd*nr*nf);
91 obj.fluxData = reshape(obj.fluxData, nd, nr, nf);
92
93 %< Populate flux data for each boundary
94 for i = 1:nf
95 nr = D(i).nDerivatives;
96 k = 2 * ceil(i/2) - 1 + mod(i, 2); %< Neighboring element index
97
98 for j = 0:nr
99 %< Inflow contribution (same element)
100 obj.fluxData(1, j+1, i).setMatrix(D(i), D(i), j, 0);
101 %< Outflow contribution (neighboring element)
102 obj.fluxData(2, j+1, i).setMatrix(D(i), D(k), j, 0);
103 end
104 end
105 end
