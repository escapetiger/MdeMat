1classdef EulerianGridAssembler < fem.assembly.GridAssembler
2 % EULERIANGRIDASSEMBLER Eulerian grid-based assembler.
3 %
4 % EulerianGridAssembler provides assembly methods for finite element
5 % operators on structured grid meshes using an Eulerian approach.
6 % This class handles volume and flux contributions with support for
7 % different boundary condition types including periodic and Dirichlet
8 % conditions.
9 %
10 % The assembler supports various flux formulations (left-biased,
11 % right-biased, and central) and handles both interior and boundary
12 % contributions for each type.
13 %
14 % Examples:
15 % % Create Eulerian grid assembler
16 % assembler = EulerianGridAssembler(fe, mesh, op, 0); % Periodic BC
17 %
18 % % Assemble different flux types
19 % T_left = assembler.assembleLeftFluxTerms(1, coeffs);
20 % T_central = assembler.assembleCentralFluxTerms(1, coeffs);
21 %
22 % See also:
23 % fem.assembly.GridAssembler, fem.assembly.Assembler

26 function T = assembleVolumeTerms(obj, d, c)
27 % ASSEMBLEVOLUMETERMS Assemble volume integral terms.
28 %
29 % T = assembleVolumeTerms(obj, d, c) creates triplets for the
30 % volume contribution of differential operators along the
31 % specified spatial dimension.
32 %
33 % Inputs:
34 % obj - The EulerianGridAssembler object
35 % d - Spatial dimension index (1, 2, or 3)
36 % c - Coefficient vector, one per element
37 %
38 % Outputs:
39 % T - Triplet matrix [row, col, value] for volume terms
40
41 ne = obj.nTotalElements;
42 np = obj.nLocalDofs;
43 P = obj.fe.projector;
44 G = obj.op.gradient;
45 F = G.volumeData(d).matrix;
46
47 T = zeros(np^2*ne, 3);
48 [i, j, v] = find(-P.project(F));
49 nv = numel(v);
50 k = 1:(nv * ne);
51 T(k, 1) = reshape(bsxfun(@plus, (0:ne - 1)*np, i(:)), [], 1);
52 T(k, 2) = reshape(bsxfun(@plus, (0:ne - 1)*np, j(:)), [], 1);
53 T(k, 3) = kron(c(:), v(:));
54 end

101 % ASSEMBLECENTRALFLUXTERMS Assemble central flux terms.
102 %
103 % T = assembleCentralFluxTerms(obj, d, c) combines left and
104 % right flux contributions with equal weighting to create

107 % Inputs:

188 function T = assembleInnerLeftFluxInteriorTerms(obj, d, c)
189 % ASSEMBLEINNERLEFTFLUXINTERIORTERMS Assemble inner left flux
190 % interior terms.
191 %
192 % T = assembleInnerLeftFluxInteriorTerms(obj, d, c) computes
193 % the inner contribution to a left flux along the specified

312 function T = assembleOuterLeftFluxInteriorTerms(obj, d, c)
313 % ASSEMBLEOUTERLEFTFLUXINTERIORTERMS Assemble outer left flux
314 % interior terms.
315 %
316 % T = assembleOuterLeftFluxInteriorTerms(obj, d, c) computes
317 % the outer contribution to a left flux along the specified
318 % dimension for elements in the domain interior. This handles
319 % coupling between adjacent elements.
320 %
321 % Inputs:
322 % obj - The EulerianGridAssembler object
323 % d - Spatial dimension index (1, 2, or 3)
324 % c - Coefficient vector, one per element
325 %
326 % Outputs:
327 % T - Triplet matrix for outer contribution
328
329 np = obj.nLocalDofs;
330 P = obj.fe.projector;
331 G = obj.op.gradient;
332
333 m = obj.mesh.allElementMultiIndices;
334 e = m(:, d) > 1 & m(:, d) < obj.mesh.resolution(d);
335 ne = numel(find(e));
336 m = m(e, :);
337 l1 = obj.mesh.indexer.multiToLinear(m, 0);
338 m(:, d) = m(:, d) - 1;
339 l2 = obj.mesh.indexer.multiToLinear(m, 0);
340
341 T = zeros(np^2*ne, 3);
342
343 F = G.fluxData(2, 2*d-1).matrix;
344 [i, j, v] = find(-P.project(F));
345 nv = numel(v);
346 k = 1:(nv * ne);
347 T(k, 1) = reshape(bsxfun(@plus, (l1(:).' - 1)*np, i(:)), [], 1);
348 T(k, 2) = reshape(bsxfun(@plus, (l2(:).' - 1)*np, j(:)), [], 1);
349 T(k, 3) = kron(c(l1), v(:));
350 end
