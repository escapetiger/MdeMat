56 % JACOBIAN Compute the Jacobian matrix at specified points.
57 %
58 % J = jacobian(obj, X) computes the Jacobian matrix of the
59 % function at the specified points. For a function from
60 % \f$R^n\f$ to \f$R^m\f$, the Jacobian is an \f$m\times n\f$
61 % matrix of first partial derivatives.
62 %
63 % Inputs:
64 % obj - The DifferentiableFunction object
65 % X - Coordinates (nDims x ...)
66 %
67 % Outputs:
68 % J - Jacobian tensor (nCodims x nDims x ...)
69
70 core.except.assert(obj.isWellDefined, ...
71 'NotWellDefined', 'Function is not well-defined.');
72
73 n = obj.nDims;
74 core.except.assert(size(X, 1) == n, ...
75 'InvalidInput', 'Input dimension mismatch.');
76
77 s = size(X);
78 X = reshape(X, n, []);
79
80 p = size(X, 2);
81 m = obj.nCodims;
82 J = zeros(m, n, p);
83
84 for i = 1:m
85 for j = 1:n
86 r = zeros(1, n);
