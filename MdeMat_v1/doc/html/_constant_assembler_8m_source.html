1classdef ConstantAssembler < fem.assembly.Assembler
2 % CONSTANTASSEMBLER Assembler for constant coefficient operators.
3 %
4 % ConstantAssembler assembles diagonal matrices with constant
5 % coefficients for each element. This is particularly useful for
6 % mass matrices and other operators where each element contributes
7 % a uniform coefficient to all its local degrees of freedom.
8 %
9 % The assembler creates block-diagonal structures where each
10 % element's contribution is a constant value multiplied by the
11 % identity matrix of size equal to the number of local degrees of
12 % freedom.
13 %
14 % Examples:
15 % % Create constant assembler
16 % assembler = ConstantAssembler(fe, mesh, op);
17 %
18 % % Assemble diagonal matrix with element-wise coefficients
19 % coeffs = [1.0, 2.0, 1.5]; % One coefficient per element
20 % matrix = assembler.assemble(coeffs);
21 %
22 % % Mass matrix assembly
23 % density = ones(mesh.nTotalElements, 1);
24 % massMatrix = assembler.assemble(density);
25 %
26 % See also:
27 % fem.assembly.Assembler, fem.assembly.GridAssembler

30 function A = assemble(obj, c)
31 % ASSEMBLE Assemble diagonal matrix with constant coefficients.
32 %
33 % A = assemble(obj, c) creates a diagonal sparse matrix where
34 % each element contributes a constant coefficient repeated for
35 % all its local degrees of freedom. The resulting matrix is
36 % block-diagonal with each block being a scaled identity.
37 %
38 % Inputs:
39 % obj - The ConstantAssembler object
40 % c - Vector of coefficients, one per element (length nTotalElements)
41 %
42 % Outputs:
43 % A - Assembled diagonal sparse matrix of size nGlobalDofs×nGlobalDofs
44
45 n = obj.nGlobalDofs;
46 np = obj.nLocalDofs;
47 v = kron(c(:), ones(np, 1));
48 A = spdiags(v, 0, n, n);
49 end
50 end
