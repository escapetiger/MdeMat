1classdef ScaledAffineSpace < fem.space.AffineSpace
2 % SCALEDAFFINESPACE Scaled affine transformation of finite element space.
3 %
4 % ScaledAffineSpace extends the affine space with individual scaling
5 % parameters for both the finite element coefficients and the
6 % directional component. Functions in this space have the form:
7 %
8 % \f[
9 % f(x) = \varepsilon_u \cdot u(x) + \varepsilon_g \cdot g
10 % \f]
11 %
12 % where \f$s_u\f$ are scaling factors applied to the finite element
13 % coefficients, \f$u(x)\f$ is the finite element function, \f$s_g\f$
14 % is the scaling factor for the directional component, and \f$g\f$
15 % is the directional shift.
16 %
17 % This formulation provides additional flexibility in function
18 % approximation and optimization by allowing independent scaling
19 % of different components, which can improve conditioning and
20 % convergence properties.
21 %
22 % Examples:
23 % % Create scaled affine space with coefficient and direction scaling
24 % coeffScales = [1.5; 2.0; 0.8]; % scale each FE coefficient
25 % directionScale = 0.5; % scale directional component
26 % scales = [coeffScales; directionScale];
27 % space = ScaledAffineSpace(finiteElement, scales);
28 %
29 % % Evaluate with automatic scaling applied
30 % points = [0, 0.5, 1.0; 0, 0, 0];
31 % coeffs = [1; 1; 1]; % unit coefficients
32 % direction = [1, 1, 1]; % unit direction
33 % values = space.evaluate(points, coeffs, direction);
34 % % Result: values = 1.5*fe1 + 2.0*fe2 + 0.8*fe3 + 0.5*[1,1,1]
35 %
36 % See also:
37 % fem.space.AffineSpace, fem.space.FiniteElementSpace
38
