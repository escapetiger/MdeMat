2 % VISUALIZER Base class for visualizers.
3 %
4 % The Visualizer class provides comprehensive plotting capabilities
5 % for advection equation solutions, including support for exact
6 % solution comparisons and temporal interpolation. It manages plot
7 % timelines, rendering strategies, and coordinate visualization
8 % workflows.
9 %
10 % Key features include:
11 % - Multi-dimensional visualization support (1D, 2D, 3D)
12 % - Exact solution overlay capabilities
13 % - Temporal interpolation for smooth animations
14 % - Flexible component-based plotting configuration
15 % - Automatic subplot arrangement and styling
16 %
17 % Examples:
18 % % Basic 1D visualization setup
19 % config = struct('nDims', 1, 'density', [50], 'final', 1.0);
20 % visualizer = Visualizer(config);
21 %
22 % % With exact solution comparison
23 % config.exacts = struct('u', @(x,t) sin(x-t));
24 % config.components = struct('u', [1]);
25 % visualizer = Visualizer(config);

109 function reset(obj)
110 % RESET Reset visualizer to initial state.
111 %
112 % reset(obj) reinitializes the visualizer by clearing
113 % datasets, resetting the timeline, and reconstructing the
114 % plotting strategy based on current component configuration.
115 %
116 % Inputs:
117 % obj - The Visualizer object
118 %
119 % Outputs:
120 % None
121
122 strategy = profilers.visual.Strategy(obj.nDims, 'map');
123 fieldNames = fieldnames(obj.components);
124 offset = 0;
125 for iField = 1:length(obj.nComponents)
126 nFieldComponents = obj.nComponents(iField);
127 fieldName = fieldNames{iField};
128 for iComponent = 1:nFieldComponents
129 componentName = sprintf('%s%d', fieldName, ...
130 obj.components.(fieldName)(iComponent));
131 axisIdx = (offset + iField - 1) * obj.nAxesPerComponent + 1;
132 strategy.bind(componentName, axisIdx);
133 end
134 offset = offset + nFieldComponents;
135 end
136 obj.plotter = profilers.visual.GeometryPlotter( ...
137 core.geometry.Orthotope.unit(obj.nDims), ...
138 obj.nRows, obj.nCols);
139 obj.plotter.setNodes(obj.density);
140 obj.plotter.setStrategy(strategy);
141 for iAxis = 1:obj.nAxes
142 obj.plotter.addAxisSpec(1, 1);
143 end
144 obj.plotter.draft();
145 obj.timeline.reset();
146 obj.isPrepared = false;
147 obj.dataset = struct();
148 end

168 function plot(obj, space, dofs, tDisc)
169 % PLOT Generate visualization for current solution state.
170 %
171 % plot(obj, space, dofs, tDisc) creates plots for the current
172 % solution state, including exact solution comparisons if
173 % configured. Handles temporal interpolation and multi-component
174 % visualization automatically.
175 %
176 % Inputs:
177 % obj - The Visualizer object
178 % space - Spatial discretization object
179 % dofs - Degrees of freedom structure containing solution data
180 % tDisc - Temporal discretization object
181 %
182 % Outputs:
183 % None
184 %
185 % Notes:
186 % The method automatically handles coordinate preparation on
187 % first call and manages timeline advancement for animations.
188
189 if ~obj.isPrepared
190 x = space.mesh.collocate(obj.plotter.coords);
191 fieldNames = fieldnames(obj.dataset);
192 for i = 1:length(fieldNames)
193 fieldName = fieldNames{i};
194 obj.dataset.(fieldName).setCoordinates(x);
195 end
196 obj.isPrepared = true;
197 end
198
199 xRef = obj.plotter.nodes;
200 t0 = tDisc.timeline.next;
201 t = obj.timeline.now;
202
203 while t0 >= t - 1e-14
204 fieldNames = fieldnames(dofs);
205 for i = 1:length(fieldNames)
206 fieldName = fieldNames{i};
207 component = obj.components.(fieldName);
208 if ~isempty(component)
209 dsNames = fieldnames(obj.dataset);
210 for j = 1:length(dsNames)
211 dsName = dsNames{j};
212 if strcmp(dsName, 'REF')
213 f = obj.exacts.(fieldName);
214 U = space.evaluate(f, xRef, t);
215 else
216 U = obj.interpolate(space, dofs, tDisc, xRef, fieldName);
217 end
218 obj.dataset.(dsName).data.(fieldName) = U(:, component);
219 end
220 end
221 end
222
223 %< Create title and render
224 title = sprintf('t = %0.2f', t);
225
226 %< Render
227 obj.plotter.render(title, obj.dataset, obj.style, []);
228
229 %< Advance output timeline
230 obj.timeline.advance();
231 t = obj.timeline.now;
232 end
233 end

294 function U = interpolate(obj, space, dofs, tDisc, refNodes, field)
295 % INTERPOLATE Interpolate solution to current output time.
296 %
297 % Notes:
298 % Uses linear interpolation between time steps when temporal
299 % discretization provides multiple historical states.
300
301 k = min(tDisc.nSteps, tDisc.timeline.count);
302
303 t0 = tDisc.timeline.now;
304 t1 = tDisc.timeline.next;
305
306 U = space.evaluate(refNodes, dofs.(field));
307
308 if t1 > t0
309 U0 = cell(1, k);
310 for i = 1:k
311 U0{i} = space.evaluate(refNodes, tDisc.U0{i});
312 end
313
314 U0 = cellfun(@(x) reshape(x, [1, size(x)]), U0, 'Un', 0);
315 U = reshape(U, [1, size(U)]);
316 U = cat(1, U0{:}, U);
317 t = [t1 - cumsum(tDisc.timeline.h(k:-1:1)), t1];
318
319 U = interp1(t, U, t0);
320 U = reshape(U, [], 1);
321 end
322 end
