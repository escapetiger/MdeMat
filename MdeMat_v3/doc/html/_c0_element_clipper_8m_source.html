1classdef C0ElementClipper < approx.element.ElementClipper
2 % C0ELEMENTCLIPPER Geometric clipper for C0 elements.
3 %
4 % C0ElementClipper implements geometric clipping operations for C0
5 % elements in semi-Lagrangian methods. It computes the intersection
6 % between the current Eulerian element and the upstream footprint
7 % traced along characteristic curves, creating volume pieces for
8 % accurate transport calculations.
9 %
10 % The clipper handles the complex geometry resulting from backward
11 % characteristic tracing by decomposing the intersection region
12 % into simpler geometric pieces. Each piece is equipped with
13 % appropriate quadrature rules and function data for both Eulerian
14 % and upstream (Lagrangian) representations.
15 %
16 % This approach enables stable and accurate semi-Lagrangian schemes
17 % for convection-dominated problems without the CFL restriction
18 % that limits explicit Eulerian methods. The geometric decomposition
19 % ensures accurate integration over potentially complex intersection
20 % regions while maintaining computational efficiency.
21 %
22 % Examples:
23 % % Create clipper for C0 element with specified velocity
24 % velocity = [0.5; 0.3];
25 % element = approx.element.C0Element(geometry, projector, integrator);
26 % clipper = C0ElementClipper(element, velocity);
27 %
28 % % Set up volume piece data for time step and mesh spacing
29 % timeStep = 0.1;
30 % meshSpacing = [0.05; 0.05];
31 % clipper.setVolumePieceData(timeStep, meshSpacing);
32 %
33 % % Access computed pieces and data
34 % nPieces = clipper.nVolumePieces;
35 % eulerianData = clipper.volumeEulerianData;
36 % upstreamData = clipper.volumeUpstreamData;
37 %
38 % % Process geometric information
39 % pieceCenters = clipper.volumePieces.xc;
40 % pieceDimensions = clipper.volumePieces.hx;
41 % indexShifts = clipper.volumeShifts;
42 %
43 % See also:
44 % approx.element.ElementClipper, approx.element.L2ElementClipper,
45 % approx.element.C0Element
46

61 function obj = setVolumePieceData(obj, ht, hx)
62 % SETVOLUMEPIECEDATA Set up volume piece data for
63 % semi-Lagrangian transport.
64 %
65 % obj = setVolumePieceData(obj, ht, hx) computes the
66 % geometric intersection between the current element and the
67 % upstream footprint, creating volume pieces with associated
68 % function data for accurate semi-Lagrangian integration.
69 %
70 % The method performs several key operations:
71 % 1. Computes geometric intersection pieces
72 % 2. Creates function data objects for each piece
73 % 3. Sets up quadrature points and weights
74 % 4. Evaluates basis functions at integration points
75 % 5. Computes index shifts for upstream element mapping
76 %
77 % Inputs:
78 % obj - The C0ElementClipper object
79 % ht - Time step for characteristic tracking (positive scalar)
80 % hx - Mesh spacing vector (nDims x 1 or nDims x 1)
81 %
82 % Outputs:
83 % obj - The C0ElementClipper object
84
85 nd = obj.element.nDims;
86 v = obj.velocity;
87
88 %< Compute clipping geometry
89 bbox = obj.element.geometry.bbox;
90 ratio = v(:) .* ht ./ hx(:);
91 poi = ratio - floor(ratio) - 1 / 2;
92
93 %< Initialize volume pieces structure
94 obj.volumePieces = struct();
95 obj.volumePieces.xc = zeros(nd, 2^nd);
96 obj.volumePieces.hx = zeros(nd, 2^nd);
97 np = 0;
98
99 %< Generate intersection pieces
100 indexer = core.linalg.MultiIndexer(ones(1, nd) * 2);
101 m = indexer.generate();
102 for i = 1:2^nd
103 x1 = poi;
104 x2 = bbox((m(i, :) + (0:nd - 1) * 2)');
105 xc = (x1 + x2) / 2;
106 hx = abs(x2 - x1);
107
108 %< Skip degenerate pieces
109 if prod(hx) < 1e-8
110 continue;
111 end
112
113 np = np + 1;
114 obj.volumePieces.xc(:, np) = xc;
115 obj.volumePieces.hx(:, np) = hx;
116 end
117
118 %< Handle case with no valid pieces
119 if np == 0
120 obj.volumePieces.xc = zeros(nd, 1);
121 obj.volumePieces.hx = ones(nd, 1);
122 np = 1;
123 else
124 obj.volumePieces.xc = obj.volumePieces.xc(:, 1:np);
125 obj.volumePieces.hx = obj.volumePieces.hx(:, 1:np);
126 end
127
128 %< Compute piece boundaries
129 obj.volumePieces.xmin = obj.volumePieces.xc - obj.volumePieces.hx / 2;
130 obj.volumePieces.xmax = obj.volumePieces.xc + obj.volumePieces.hx / 2;
131 obj.nVolumePieces = np;
132
133 %< Create function data objects for each piece
134 I = obj.element.volumeData.integrator;
135 obj.volumeEulerianData = arrayfun(@(i) approx.element.ElementFunction(I.copy()), 1:np);
136 obj.volumeUpstreamData = arrayfun(@(i) approx.element.ElementFunction(I.copy()), 1:np);
137
138 %< Set up integration points for each piece
139 nq = obj.element.projector.basis.nFactorCodims;
140 for i = 1:np
141 a = obj.volumePieces.xmin(:, i);
142 b = obj.volumePieces.xmax(:, i);
143 obj.volumeEulerianData(i).setPoints(nq, a, b);
144 obj.volumeUpstreamData(i).setPoints(nq, -b, -a);
145 end
146
147 %< Compute basis function values at integration points
148 f = obj.element.projector.basis;
149 for i = 1:np
150 De = obj.volumeEulerianData(i);
151 Du = obj.volumeUpstreamData(i);
152 Ie = De.integrator;
153 Iu = Du.integrator;
154 Xe = Ie.nodes(1:nd, :);
155 Xu = Iu.nodes(1:nd, :);
156
157 %< Set function values and derivatives
158 De.setValues(f, Xe);
159 Du.setValues(f, Xu);
160 De.setDerivatives(f, Xe, 1);
161 end
162
163 %< Compute index shifts for upstream element mapping
164 obj.volumeShifts = zeros(nd, np);
165 mm = ceil(ratio) .* (ratio > 0) + (floor(ratio) + 1) .* (ratio < 0);
166 mm0 = -mm;
167 mm1 = -mm + 1;
168 for i = 1:np
169 bb = obj.volumePieces.xc(:, i) <= 0;
170 obj.volumeShifts(:, i) = mm0 .* bb + mm1 .* (1 - bb);
171 end
172 end
173 end
174end
