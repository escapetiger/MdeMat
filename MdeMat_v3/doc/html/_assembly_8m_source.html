35 function G = assembleMatrix(obj, IE, IL, JE, JL, L)
36 % ASSEMBLEMATRIX Assemble matrix contribution.
37 %
38 % G = assembleMatrix(obj, IE, IL, JE, JL, L) assembles local
39 % matrices @a L to a global block diagonal matrix @a G with
40 % specified elements (@a IE, @a JE) and local Dofs (@a IL, @a
41 % JL) for test space and trial space, respectively.
42 %
43 % Inputs:
44 % obj - The MeshOperator object
45 % IE - Element linear indices for test space (ne x 1 vector)
46 % IL - Local DoF indices for test space (ni x 1 vector)
47 % JE - Element linear indices for trial space (ne x 1 vector)
48 % JL - Local DoF indices for trial space (nj x 1 vector)
49 % L - Local matrices (nl x nl matrix or nl × nl × ne array)
50 %
51 % Outputs:
52 % G - Global matrix (ng x ng sparse)
53
54 if isempty(IE) && ~isempty(JE)
55 IE = JE;
56 elseif ~isempty(IE) && isempty(JE)
57 JE = IE;
58 elseif isempty(IE) && isempty(JE)
59 IE = 1:obj.space.nMeshElements;
60 JE = IE;
61 else
62 core.except.assert(length(IE) == length(JE), ...
63 'InvalidInput', ...
64 'Incomptiable row and columen element indices.');
65 end
66 ne = length(IE);
67
68 nl = obj.space.nLocalDofs;
69 if isempty(IL), IL = 1:nl; end
70 if isempty(JL), JL = 1:nl; end
71
72 if ismatrix(L)
73 [I, J, V] = find(L(IL, JL));
74 nnz = numel(V);
75 T = zeros(nnz * ne, 3);
76 T(:, 1) = obj.space.localToGlobal(IE, I);
77 T(:, 2) = obj.space.localToGlobal(JE, J);
78 T(:, 3) = V(:);
79 else
80 L = reshape(L, nl, nl, []);
81 core.except.assert(size(L, 3) == ne, ...
82 'InvalidInput', 'Invalid local matrices.');
83
84 ni = length(IL);
85 nj = length(JL);
86 T = zeros(ni*nj*ne, 3);
87
88 [IL, JL] = ndgrid(IL, JL);
89 IL = IL(:);
90 JL = JL(:);
91 T(:, 1) = obj.space.localToGlobal(IE, IL);
92 T(:, 2) = obj.space.localToGlobal(JE, JL);
93

102 function G = assembleVector(obj, IE, IL, L)
103 % ASSEMBLEVECTOR Assemble vector contribution.
104 %
105 % G = assembleVector(obj, IE, IL, L) assembles local vectors
106 % @a L to a global block diagonal matrix @a G with specified
107 % elements @a IE and local Dofs @a IL.
108 %
109 % Inputs:
110 % obj - The MeshOperator object
111 % IE - Element linear indices (ne x 1 vector)
112 % IL - Local DoF indices (ni x 1 vector)
113 % L - Local vectors (nl x nc matrix or nl x ne x nc array)
114 %
115 % Outputs:
116 % G - Global matrix (ng x nc sparse)
117
118 if isempty(IE), IE = 1:obj.space.nMeshElements; end
119 ne = length(IE);
120
121 nl = obj.space.nLocalDofs;
122 if isempty(IL), IL = 1:nl; end
123
124 if ismatrix(L)
125 L = reshape(L, nl, []); % nl × nc
126 L = repmat(reshape(L, nl, 1, []), 1, ne, 1); % nl × ne × nc
127 else
128 L = reshape(L, nl, ne, []); % nl × ne × nc
129 core.except.assert(size(L, 2) == ne, ...
130 'InvalidInput', 'Invalid local vectors.');
131 end
132 L = L(IL, :, :); % ni × ne × nc
133 ni = length(IL);
134 nc = size(L, 3);
135
136 T = zeros(ni*ne*nc, 3);
137 T(:, 1) = repmat(obj.space.localToGlobal(IE, []), nc, 1);
138 T(:, 2) = reshape(repmat(1:nc, ni*ne, 1), [], 1);
139 T(:, 3) = L(:);
140
141 ng = obj.space.nGlobalDofs;
