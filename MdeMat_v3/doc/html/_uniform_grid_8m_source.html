1classdef UniformGrid < approx.mesh.Grid
2 % UNIFORMGRID Structured grid with uniform element spacing.
3 %
4 % UniformGrid represents a structured computational grid where
5 % elements have constant spacing within each dimension. This
6 % regularity enables highly efficient storage and computation since
7 % geometric properties (Jacobians, volumes, outN) are identical for
8 % all elements and can be computed once and reused.
9 %
10 % The grid is defined by domain bounds and resolution, automatically
11 % generating element centers and uniform spacings. All coordinate
12 % transformations use simple affine mappings with constant
13 % coefficients, making uniform grids ideal for:
14 % - Finite difference methods requiring regular stencils
15 % - Spectral methods with tensor-product basis functions
16 % - Fast Fourier Transform-based solvers
17 % - Applications where geometric simplicity aids analysis
18 %
19 % Storage efficiency: O(d) memory for grid definition vs O(n^d) for
20 % explicit storage, where d is spatial dimension and n is resolution
21 % per dimension. All geometric computations have O(1) complexity.
22 %
23 % Examples:
24 % % 2D uniform grid: 50×30 elements on unit square
25 % bbox2D = [0, 1, 0, 1]; % [x_min, x_max, y_min, y_max]
26 % resolution2D = [50, 30];
27 % grid2D = UniformGrid(bbox2D, resolution2D);
28 %
29 % % 3D uniform grid: 20×20×10 elements on custom domain
30 % bbox3D = [-1, 1, -2, 2, 0, 5]; % [x_min, x_max, y_min, y_max, z_min, z_max]
31 % resolution3D = [20, 20, 10];
32 % grid3D = UniformGrid(bbox3D, resolution3D);
33 %
34 % % Query uniform grid properties
35 % h = grid2D.computeMeasure(); % Minimum element spacing
36 % detJ = grid2D.computeElementJacobianDeterminants(); % Constant for all elements
37 % graph = grid2D.graphify(); % SeparableGraph representation
38 %
39 % % Coordinate transformations
40 % xRef = [0.2; -0.1]; % Reference coordinates in [-1/2, 1/2]^2
41 % xPhys = grid2D.collocate(xRef, 100); % Map to element 100
42 %
43 % % Mesh refinement (preserves uniformity)
44 % fineGrid = grid2D.refine(2); % 2 levels: 200×120 elements
45 %
46 % % Access grid structure
47 % spacing = grid2D.spacings; % {h_x, h_y} constant per dimension
48 % centers = grid2D.centroids; % {x_centers, y_centers} vectors
49 % nTotal = grid2D.nElements; % Total elements = 50×30 = 1500
50 %
51 % % Boundary and connectivity queries
52 % interior = grid2D.findInteriorElements(); % Interior elements
53 % leftBdry = grid2D.findBoundaryElements(1); % Left boundary (x_min)
54 % rightBdry = grid2D.findBoundaryElements(2); % Right boundary (x_max)
55 % [neighbors, faces] = grid2D.findNeighborElements(interior, 1, 'periodic');
56 %
57 % Notes:
58 % Domain specification uses the format [a_1, b_1, a_2, b_2, ..., a_d,
59 % b_d] where [a_i, b_i] defines the interval for dimension i. The
60 % constructor validates that b_i > a_i for all dimensions.
61 %
62 % Element spacing in dimension i is h_i = (b_i - a_i)/n_i where n_i
63 % is the number of elements in that dimension. Element centers are at
64 % a_i + (k+0.5)h_i for k = 0, 1, ..., n_i-1.
65 %
66 % All geometric computations exploit uniformity: Jacobian matrices
67 % are diagonal with constant entries, face outN are axis-aligned,
68 % and volume elements are identical throughout the grid.
69 %
70 % See also:
71 % approx.mesh.Grid, approx.mesh.SeparableGraph

74 function obj = UniformGrid(bbox, resolution)
75 % UNIFORMGRID Constructor for UniformGrid.
76 %
77 % obj = UniformGrid(bbox, resolution) creates a uniform grid
78 % with constant element spacing determined by the domain bounds
79 % and specified resolution in each dimension.
80 %
81 % Inputs:
82 % bbox - Domain bounding box [a₁, b₁, a₂, b₂, ..., aₐ, bₐ]
83 % resolution - Number of elements per dimension (1 × d vector)
84 %
85 % Outputs:
86 % obj - Constructed UniformGrid object
87 %
88 % Notes:
89 % The constructor automatically computes:
90 % 1. Element spacing: hᵢ = (bᵢ - aᵢ)/nᵢ for each dimension i
91 % 2. Element centers: aᵢ + (k+0.5)hᵢ for k = 0,1,...,nᵢ-1
92 % 3. Node coordinates: aᵢ + k·hᵢ for k = 0,1,...,nᵢ
93 % 4. Domain and reference element geometries
94 %
95 % Input validation ensures bᵢ > aᵢ for proper domain definition.
96
97 %< Reshape and parse bounding box coordinates
98 bbox = bbox(:).';
99 d = length(bbox) / 2;
100 a = bbox(1:2:end); % Lower bounds [a₁, a₂, ..., aₐ]
101 b = bbox(2:2:end); % Upper bounds [b₁, b₂, ..., bₐ]
102
103 %< Validate domain bounds
104 core.except.assert(all(b > a), 'InvalidInput', ...
105 'Upper bounds must be greater than lower bounds.');
106
107 %< Create domain and reference geometries
108 domain = core.geometry.Orthotope(bbox);
109 reference = core.geometry.Orthotope(repmat([-1 / 2, 1 / 2], 1, d));
110
111 %< Compute uniform grid properties
112 %< Node coordinates: linearly spaced points including boundaries
113 nodes = arrayfun(@(i) linspace(a(i), b(i), resolution(i)+1), 1:d, 'Un', 0);
114
115 %< Element centroids: midpoints between consecutive nodes
116 centroids = cellfun(@(x) (x(1:end-1) + x(2:end))/2, nodes, 'Un', 0);
117
118 %< Uniform element spacings: constant within each dimension
119 spacings = num2cell((b - a)./resolution);
120
121 %< Call parent constructor with computed grid properties
122 obj@approx.mesh.Grid(domain, reference, resolution, centroids, spacings);
123 end

125 function graphObj = graphify(obj)
126 % GRAPHIFY Convert uniform grid to separable graph representation.
127 %
128 % graphObj = graphify(obj) creates a SeparableGraph that
129 % efficiently represents the uniform grid's tensor-product
130 % connectivity structure using 1D factors.
131 %
132 % Inputs:
133 % obj - The UniformGrid object
134 %
135 % Outputs:
136 % graphObj - SeparableGraph with tensor-product structure
137 %
138 % Notes:
139 % The separable representation stores d 1D graphs instead of
140 % one d-dimensional graph, reducing memory from O(n^d) to O(d×n).
141 % Each 1D factor contains vertex coordinates and edge connectivity
142 % for one dimension, with the full grid formed by tensor products.
143
144 d = obj.nDims;
145 n = obj.resolution;
146 h = obj.spacings;
147 c = obj.centroids;
148
149 %< Generate vertex coordinates for each dimension
150 %< Include element boundaries: from first center - h/2 to last center + h/2
151 V = arrayfun(@(i) (c{i}(1) - h{i} / 2):h{i}:(c{i}(end) + h{i} / 2), 1:d, 'Un', 0);
152
153 %< Generate edge connectivity for each dimension
154 %< Standard 1D chain: vertex k connects to vertex k+1
155 E = arrayfun(@(k) [(1:k).', (2:(k + 1)).'], n, 'Un', 0);
156
157 %< Create separable graph from 1D factors
158 graphObj = approx.mesh.SeparableGraph(V, E);
159 end

161 function Y = collocate(obj, X, L)
162 % COLLOCATE Map reference coordinates to physical coordinates.
163 %
164 % Y = collocate(obj, X) maps reference coordinates from the
165 % standard hypercube [-1/2,1/2]^d to physical coordinates
166 % across all grid elements.
167 %
168 % Y = collocate(obj, X, L) maps coordinates to specific
169 % elements identified by linear indices L.
170 %
171 % Inputs:
172 % obj - The UniformGrid object
173 % X - Reference coordinates (d × nPoints or cell array)
174 % L - Element linear indices (optional, vector)
175 %
176 % Outputs:
177 % Y - Physical coordinates (d × nPoints or cell array)
178 %
179 % Notes:
180 % The uniform grid coordinate transformation is the simple
181 % affine mapping: x = c + h⊙ξ where c is element center,
182 % h is element spacing, and ξ are reference coordinates.
183 % This transformation is identical for all elements, differing
184 % only in the translation vector c.
185
186 %< Use all elements if none specified
187 if nargin < 3, L = []; end
188
189 d = obj.nDims;
190
191 %< Convert matrix input to cell array format for processing
192 if ismatrix(X) && ~iscell(X)
193 X = num2cell(reshape(X, d, []), 2).';
194 end
195
196 Y = cell(d, 1);
197
198 if isempty(L)
199 %< Transform to all elements: use broadcast operations
200 for i = 1:d
201 h = obj.spacings{i}; % Element spacing in dimension i
202 a = obj.centroids{i}; % All element centers in dimension i
203 b = bsxfun(@times, h, X{i}(:)); % Scale reference coordinates
204 Y{i} = reshape(bsxfun(@plus, a, b), 1, []); % Translate to centers
205 end
206 else
207 %< Transform to specific elements: extract relevant centers
208 M = obj.indexer.linearToMulti(L); % Convert to multi-indices
209
210 for i = 1:d
211 h = obj.spacings{i}; % Element spacing in dimension i
212 a = obj.centroids{i}(M(:, i)); % Centers for specified elements
213 b = bsxfun(@times, h, X{i}(:)); % Scale reference coordinates
214 Y{i} = reshape(bsxfun(@plus, a, b), 1, []); % Translate to centers
215 end
216 end
217
218 %< Convert back to matrix format if specific elements requested
219 if ~isempty(L), Y = cell2mat(Y); end
220 end

222 function newObj = refine(obj, nLevels)
223 % REFINE Create uniformly refined grid with smaller elements.
224 %
225 % newObj = refine(obj, nLevels) creates a refined uniform grid
226 % by subdividing each element uniformly. The refinement preserves
227 % the uniform spacing property while increasing resolution.
228 %
229 % Inputs:
230 % obj - The UniformGrid object
231 % nLevels - Number of refinement levels (non-negative integer)
232 %
233 % Outputs:
234 % newObj - Refined UniformGrid with 2^(nLevels) times more elements per dimension
235 %
236 % Notes:
237 % Uniform refinement multiplies the resolution by 2^nLevels in
238 % each dimension, resulting in 2^(nLevels×d) total elements.
239 % Element spacing decreases by factor 2^nLevels in each dimension.
240 %
241 % Examples:
242 % - Level 0: Original grid (no refinement)
243 % - Level 1: 2× resolution per dimension (4× elements in 2D)
244 % - Level 2: 4× resolution per dimension (16× elements in 2D)
245 %
246 % The refined grid maintains identical domain bounds and
247 % uniform spacing properties as the original.
248
249 %< Return original grid if no refinement requested
250 if nLevels == 0
251 newObj = obj;
252 return;
253 end
254
255 %< Compute refined resolution: multiply by 2^nLevels
256 originalResolution = obj.resolution;
257 refinementFactor = 2^nLevels;
258 refinedResolution = originalResolution * refinementFactor;
259
260 %< Create new uniform grid with same domain but higher resolution
261 newObj = approx.mesh.UniformGrid(obj.domain.bbox, refinedResolution);
262 end

264 function h = computeMeasure(obj)
265 % COMPUTEMEASURE Compute characteristic element size.
266 %
267 % h = computeMeasure(obj) returns the minimum element spacing
268 % among all dimensions, providing a characteristic length scale
269 % for the uniform grid.
270 %
271 % Inputs:
272 % obj - The UniformGrid object
273 %
274 % Outputs:
275 % h - Minimum element spacing (positive scalar)
276 %
277 % Notes:
278 % For uniform grids, the measure is simply min(h₁, h₂, ..., hₐ)
279 % where hᵢ is the constant spacing in dimension i. This quantity
280 % appears in:
281 % - CFL stability conditions for explicit time integration
282 % - Error estimates for finite difference and finite element methods
283 % - Convergence analysis for iterative solvers
284
285 %< Extract all spacing values and find minimum
286 h = min([obj.spacings{:}]);
287 end

350 function detFJ = computeAllFaceJacobianDeterminants(obj)
351 % COMPUTEALLFACEJACOBIANDETERMINANTS Compute face Jacobian
352 % determinants for boundary integration.
353 %
354 % detFJ = computeAllFaceJacobianDeterminants(obj) computes
355 % Jacobian determinants for coordinate transformation on
356 % element faces. For uniform grids, faces of the same type
357 % share identical determinants.
358 %
359 % Inputs:
360 % obj - The UniformGrid object
361 %
362 % Outputs:
363 % detFJ - Face Jacobian determinants (cell array by face type)
364
365 detFJ = cell(1, 2*obj.nDims);
366 spacings = cell2mat(obj.spacings);
367
368 %< Compute determinant for each face type
369 for i = 1:2 * obj.nDims
370 %< Determine perpendicular dimension for this face
371 dim = ceil(i / 2);
372
373 %< Face Jacobian = product of spacings in other dimensions
374 otherDims = 1:obj.nDims;
375 otherDims(dim) = [];
376
377 if isempty(otherDims)
378 %< 1D case: face is a point, determinant is 1
379 detFJ{i} = 1;
380 else
381 %< Higher dimensions: product of transverse spacings
382 detFJ{i} = prod(spacings(otherDims));
383 end
384 end

387 function outN = computeAllOutwardNormals(obj)
388 % COMPUTEALLOUTWARDNORMALS Compute outward normal vectors for
389 % all faces.
390 %
391 % outN = computeAllOutwardNormals(obj) computes unit outward
392 % normal vectors for each face type. For uniform grids, outN
393 % are axis-aligned and constant for faces of the same type.
394 %
395 % Inputs:
396 % obj - The UniformGrid object
397 %
398 % Outputs:
399 % outN - Outward normal vectors (cell array by face type)
400 %
401 % Notes:
402 % Face normal directions follow the standard convention:
403 % - Face 2k-1: normal = -e_k (negative k-direction, "lower" face)
404 % - Face 2k: normal = +e_k (positive k-direction, "upper" face)
405 % where e_k is the unit vector in dimension k.
406 %
407 % For example, in 2D:
408 % - Face 1 (left): normal = [-1, 0]^T
409 % - Face 2 (right): normal = [+1, 0]^T
410 % - Face 3 (bottom): normal = [0, -1]^T
411 % - Face 4 (top): normal = [0, +1]^T
412
413 outN = cell(1, 2*obj.nDims);
414
415 %< Generate normal vector for each face type
416 for i = 1:2 * obj.nDims
417 %< Determine perpendicular dimension and orientation
418 dim = ceil(i / 2); % Which dimension (1, 2, 3, ...)
419 isPositive = (mod(i, 2) == 0); % Upper face (even index) or lower face (odd)
420
421 %< Create unit normal vector (axis-aligned)
422 outN{i} = zeros(obj.nDims, 1);
423 if isPositive
424 outN{i}(dim) = +1; % Outward = positive direction
425 else
426 outN{i}(dim) = -1; % Outward = negative direction
427 end
428 end
429 end
