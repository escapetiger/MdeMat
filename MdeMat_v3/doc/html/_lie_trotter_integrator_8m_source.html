2 % LIETROTTERINTEGRATOR First-order Lie-Trotter operator splitting.
3 %
4 % LieTrotterIntegrator implements the first-order Lie-Trotter
5 % operator splitting method for solving ordinary differential
6 % equations of the form:
7 %
8 % \f[
9 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = \sum_{i=1}^{n} (L_i u + F_i(u) + S_i)
10 % \f]
11 %
12 % by decomposing it into n separate sub-problems:
13 % \f[
14 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = L_i u + F_i(u) + S_i, \quad i = 1, 2, \ldots, n
15 % \f]
16 %
17 % The Lie-Trotter method uses the splitting formula:
18 % \f[
19 % e^{(A_1+A_2+\cdots+A_n)\Delta t} \approx e^{A_1\Delta t} e^{A_2\Delta t} \cdots e^{A_n\Delta t}
20 % \f]
21 %
22 % This achieves first-order accuracy with global error O(Δt).
23 % Each factor integrator solves its respective sub-problem using the
24 % most appropriate method (explicit, implicit, or IMEX).
25 %
26 % Examples:
27 % % Reaction-diffusion: implicit diffusion + explicit reaction
28 % integrators = {approx.odeint.BeIntegrator(1.0), ...
29 % approx.odeint.FeIntegrator(1.0)};
30 % splitter = approx.odeint.LieTrotterIntegrator(integrators);
31 %
32 % % Step with operators [diffusion, reaction]
33 % L = {diffusionMatrix, []};
34 % F = {[], @(u,t) reactionTerm(u)};
35 % S = {[], []};
36 % U_new = splitter.step(L, F, S, M);
37 %
38 % % Three-way split: diffusion + advection + reaction
39 % integrators = {approx.odeint.BeIntegrator(1.0), ...
40 % approx.odeint.FeIntegrator(1.0), ...
41 % approx.odeint.FeIntegrator(1.0)};
42 % L = {diffusionMatrix, advectionMatrix, []};
43 % F = {[], [], @(u,t) reactionTerm(u)};
44 % S = {sourceVector, [], []};
45 %
46 % See also:
47 % approx.odeint.SeparableOdeIntegrator,
48 % approx.odeint.StrangIntegrator,
49 % approx.odeint.BeIntegrator,
50 % approx.odeint.FeIntegrator
51

58 % STEP Advance one time step using Lie-Trotter splitting.
59 %
60 % U = step(obj, L, F, S, M) performs one time step by
61 % sequentially applying each factor integrator to its
62 % corresponding sub-problem.
63 %
64 % Inputs:
65 % obj - The LieTrotterIntegrator object
66 % L - Cell array of linear operators {L1, L2, ..., Ln}
67 % Each Li can be matrix, function, or empty
68 % F - Cell array of nonlinear operators {F1, F2, ..., Fn}
69 % Each Fi can be function or empty
70 % S - Cell array of source terms {S1, S2, ..., Sn}
71 % Each Si can be vector, function, or empty
72 % M - Mass operator (matrix, function, or empty)
73 % Same mass operator used for all sub-problems
74 %
75 % Outputs:
76 % U - Solution at the next time step
77
78 U0 = obj.getFactor(1).U0{1};
79 dt = obj.timeline.dt;
80
81 for i = 1:obj.nFactors
82 U = obj.applyFactorStep(i, dt, L{i}, F{i}, S{i}, M);
83 if i < obj.nFactors
84 obj.getFactor(i+1).U0{1} = U;
85 end
86 end
87
88 for i = 1:obj.nFactors
89 obj.getFactor(i).U0{1} = U0;
90 end
91 end
92 end
93end
