1classdef Maxwellian < core.function.Function
2 % MAXWELLIAN Shifted Maxwell-Boltzmann distribution function.
3 %
4 % Maxwellian represents a shifted Maxwell-Boltzmann distribution:
5 %
6 % \f[
7 % f(x;y,T) = (1/(2*pi*T)^{d/2}) * exp(-|x-y|^2/(2*T))
8 % \f]
9 %
10 % where \f$T\f$ is the temperature, \f$y\f$ is the center point, and
11 % \f$d\f$ is the dimension. This represents a Gaussian distribution
12 % centered at point y with covariance matrix T*I.
13 %
14 % Examples:
15 % % 1D Maxwellian centered at y=2 with T T=1
16 % f = MaxwellianFunction(1, 1, 2); % R^1 -> R^1
17 %
18 % % 3D velocity distribution centered at drift velocity
19 % y = [1; 0; -0.5]; % Drift velocity
20 % f = MaxwellianFunction(3, 0.5, y); % R^3 -> R^1
21 %
22 % % Evaluate at points
23 % X = randn(3, 1000);
24 % Y = f.evaluate(X);
25 %
