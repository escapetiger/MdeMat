1classdef Grid < approx.mesh.Mesh
2 % GRID Base class for multidimensional structured grids.
3 %
4 % Grid provides a foundational framework for representing structured
5 % computational grids using memory-efficient tensor-based storage.
6 % Unlike unstructured meshes that store explicit connectivity,
7 % structured grids leverage their regular topology to achieve O(d×n)
8 % memory usage instead of O(n^d) for d-dimensional grids with n
9 % elements per dimension.
10 %
11 % The class implements a tensor-product structure where:
12 % - Grid vertices form a Cartesian product of 1D coordinate arrays
13 % - Element connectivity follows regular patterns in each dimension
14 % - Neighbor relationships are computed algorithmically rather than stored
15 % - Boundary faces are identified by coordinate extrema
16 %
17 % Reference coordinate system uses the hypercube [-1/2, 1/2]^d,
18 % providing a canonical domain for finite element basis functions and
19 % quadrature rules. Coordinate transformations between reference and
20 % physical spaces are handled by concrete subclasses (UniformGrid,
21 % NonuniformGrid).
22 %
23 % The MultiIndexer facilitates efficient conversion between:
24 % - Linear indices: Single integers identifying grid elements [1, N]
25 % - Multi-indices: d-tuples specifying position in each dimension
26 % - Factor indices: Separate indexing within each dimensional factor
27 %
28 % See also:
29 % approx.mesh.Mesh, approx.mesh.SeparableGraph

46 function obj = Grid(domain, reference, resolution, centroids, spacings)
47 % GRID Constructor for Grid base class.
48 %
49 % obj = Grid(domain, reference, resolution, centroids,
50 % spacings) creates a structured grid with the specified
51 % geometric domain, reference element, resolution, and
52 % spacing information.
53 %
54 % Inputs:
55 % domain - Physical domain geometry (core.geometry.Geometry)
56 % reference - Reference element geometry (core.geometry.Geometry)
57 % resolution - Elements per dimension (1 × nDims vector)
58 % centroids - Element centers per dimension (1 × nDims cell)
59 % spacings - Element spacings per dimension (1 × nDims cell)
60 %
61 % Outputs:
62 % obj - Constructed Grid object
63
64 %< Call parent constructor with domain and reference geometry
65 obj@approx.mesh.Mesh(domain, reference);
66
67 %< Store grid structure information
68 obj.resolution = resolution;
69 obj.centroids = centroids;
70 obj.spacings = spacings;
71
72 %< Set up multi-index converter for efficient transformations
73 obj.indexer = core.linalg.MultiIndexer(obj.resolution);
74
75 %< Initialize element indices: [1, 2, ..., n_i] for dimension i
76 obj.elements = arrayfun(@(n) 1:n, obj.resolution, 'Un', 0);
77
78 %< Initialize boundary face specifications
79 obj.boundary = cell(1, 2*obj.nDims);
80 for i = 1:obj.nDims
81 %< Create template with all elements in each dimension
82 f = arrayfun(@(k) 1:k, obj.resolution, 'Un', 0);
83
84 %< Lower boundary: first element in dimension i
85 f{i} = 1;
86 obj.boundary{2*i-1} = f;
87
88 %< Upper boundary: last element in dimension i
89 f{i} = obj.resolution(i);
90 obj.boundary{2*i} = f;
91 end
92 end

106 function IE = findInteriorElements(obj)
107 % FINDINTERIORELEMENTS Find linear indices of interior elements.
108 %
109 % IE = findInteriorElements(obj) computes linear indices of
110 % elements that do not touch the domain boundary, useful for
111 % applying different numerical schemes to interior and boundary
112 % regions.
113 %
114 % Inputs:
115 % obj - The Grid object
116 %
117 % Outputs:
118 % IE - Linear indices of interior elements (vector)
119 %
120 % Notes:
121 % Interior elements are those with indices [2, n_i-1] in
122 % every dimension i, where n_i is the number of elements in
123 % dimension i. This excludes the first and last layer of
124 % elements which touch the boundary.
125 %
126 % For a 2D grid with resolution [nx, ny], interior elements
127 % have multi-indices (i,j) where 2 ≤ i ≤ nx-1 and 2 ≤ j ≤
128 % ny-1.
129
130 %< Extract interior element ranges: exclude boundary layers
131 ME = cellfun(@(I) I(2:end-1), obj.elements, 'Un', 0);
132
133 %< Convert factor indices to multi-indices
134 ME = obj.indexer.factorToMulti(ME);
135
136 %< Convert multi-indices to linear indices
137 IE = obj.indexer.multiToLinear(ME);
138 end

140 function IE = findBoundaryElements(obj, IFE)
141 % FINDBOUNDARYELEMENTS Find linear indices of boundary elements.
142 %
143 % IE = findBoundaryElements(obj, IFE) computes linear indices
144 % of elements that share the specified boundary face,
145 % essential for applying boundary conditions in numerical
146 % methods.
147 %
148 % Inputs:
149 % obj - The Grid object
150 % IFE - Boundary face index (1 to 2×nDims)
151 %
152 % Outputs:
153 % IE - Linear indices of boundary elements (vector)
154 %
155 % Notes:
156 % Face numbering follows the convention:
157 % - Face 1: Lower boundary in dimension 1 (x_min)
158 % - Face 2: Upper boundary in dimension 1 (x_max)
159 % - Face 3: Lower boundary in dimension 2 (y_min)
160 % - Face 4: Upper boundary in dimension 2 (y_max)
161 % - And so forth for higher dimensions
162 %
163 % Boundary elements have one coordinate at the domain extremum
164 % while others vary over the full range.
165
166 %< Get boundary face specification for the requested face
167 IF = obj.boundary{IFE};
168
169 %< Convert factor indices to multi-indices for boundary elements
170 ME = obj.indexer.factorToMulti(IF);
171
172 %< Convert multi-indices to linear indices
173 IE = obj.indexer.multiToLinear(ME);
174 end

176 function [JE, JFE] = findNeighborElements(obj, IE, IFE, BC)
177 % FINDNEIGHBORELEMENTS Find neighboring elements across faces.
178 %
179 % [JE, JFE] = findNeighborElements(obj, IE, IFE, BC) computes
180 % neighbor elements sharing the specified local face of given
181 % elements, with appropriate boundary condition treatment.
182 %
183 % Inputs:
184 % obj - The Grid object
185 % IE - Linear indices of primary elements (vector)
186 % IFE - Local face index (1 to 2×nDims)
187 % BC - Boundary condition ('periodic' or 'dirichlet')
188 %
189 % Outputs:
190 % JE - Linear indices of neighbor elements (vector)
191 % JFE - Local face indices of neighbors (vector)
192 %
193 % Notes:
194 % Local face numbering matches global boundary face
195 % numbering. For structured grids, neighbors are found by
196 % shifting element coordinates by ±1 in the appropriate
197 % dimension.
198 %
199 % Boundary condition handling:
200 % - 'periodic': Wrap around at domain boundaries (torus topology)
201 % - 'dirichlet': No neighbors across domain boundaries (JE = 0)
202 %
203 % The neighbor's local face index JFE is always opposite to
204 % IFE: if element E has neighbor N across face IFE, then N
205 % has E across face JFE where JFE = 2×ceil(IFE/2) -
206 % mod(IFE-1,2).
207
208 %< Convert element linear indices to multi-indices
209 ME = obj.indexer.linearToMulti(IE);
210
211 %< Determine which dimension contains the face
212 dim = ceil(IFE / 2);
213
214 %< Shift coordinates to find neighbors and determine opposite face
215 if mod(IFE, 2) == 1
216 %< Lower face: neighbor is at higher coordinate
217 ME(:, dim) = ME(:, dim) - 1;
218 JFE = 2 * dim; % Opposite is upper face
219 else
220 %< Upper face: neighbor is at lower coordinate
221 ME(:, dim) = ME(:, dim) + 1;
222 JFE = 2 * dim - 1; % Opposite is lower face
223 end
224
225 %< Apply boundary condition treatment
226 switch lower(BC)
227 case 'periodic'
228 %< Wrap around at boundaries (modular arithmetic)
229 JE = obj.indexer.multiToLinear(ME, 'wrap');
230
231 case 'dirichlet'
232 %< Set boundary neighbors to empty/invalid
233 JE = obj.indexer.multiToLinear(ME, 'empty');
234
235 otherwise
236 core.except.assert(false, 'InvalidBoundaryCondition', ...
237 'Boundary condition must be ''periodic'' or ''dirichlet''.');
238 end
239 end
240 end
241end
