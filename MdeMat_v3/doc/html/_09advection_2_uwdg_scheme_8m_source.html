1classdef UwdgScheme < physics.Scheme
2 % UWDGSCHEME Upwind Discontinuous Galerkin scheme.
3 %
4 % UwdgScheme implements an Upwind Discontinuous Galerkin (UWDG)
5 % method for solving advection equations. The method uses upwind
6 % numerical fluxes to handle the hyperbolic nature of advection
7 % problems, providing stability and sharp resolution of solution
8 % features.
9 %
10 % The UWDG method discretizes the spatial domain using discontinuous
11 % Galerkin finite elements and handles inter-element communication
12 % through upwind fluxes, making it particularly suitable for
13 % transport-dominated problems with sharp gradients or
14 % discontinuities.
15 %
16 % Examples:
17 % % Create UWDG scheme with configuration
18 % config = struct('ic', @(x) sin(x), 'advection', [1], 'cfl', 0.5, ...
19 % 'bc', [], 'verbose', true);
20 % scheme = UwdgScheme(config, timeIntegrator, visualizer, analyzer);
21 %
22 % % Full simulation workflow
23 % state = scheme.initialize(initialState);
24 % finalState = scheme.run(state);
25 % scheme.finalize(finalState);
26 %
27 % Notes:
28 % The upwind flux selection provides natural stabilization for
29 % hyperbolic problems but may require smaller time steps compared
30 % to semi-Lagrangian methods for stability.
31 %
32 % See Also:
33 % physics.Scheme, physics.advection.SldgScheme,

120 function state = step(obj, state)
121 % STEP Perform one upwind DG time step.
122 %
123 % state = step(obj, state) advances the solution by one time
124 % step using the upwind discontinuous Galerkin method. The
125 % method updates the solution history and applies the linear
126 % advection operator with appropriate boundary conditions.
127 %
128 % Inputs:
129 % obj - The UwdgScheme object
130 % state - Current solution state
131 %
132 % Outputs:
133 % state - Updated solution state after one time step
134
135 if ~isempty(obj.config.bc)
136 c = obj.config.advection;
137 S = @(t) -obj.assembler.divergenceBc(c, c, obj.config.bc, t);
138 else
139 S = [];
140 end
141
142 obj.tDisc.update(state.dofs.U);
143 L = -obj.operator;
144 state.dofs.U = obj.tDisc.step(L, S, []);
145 end
