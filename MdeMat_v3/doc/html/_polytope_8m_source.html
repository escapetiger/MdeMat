64 function TF = isInside(obj, X)
65 % ISINSIDE Tests if points are inside the polytope.
66 %
67 % TF = isInside(obj, X) uses barycentric coordinate
68 % computation to determine if points lie within the convex
69 % hull of the polytope vertices. Points are inside if their
70 % barycentric coordinates are all non-negative and sum to 1.
71 %
72 % Inputs:
73 % obj - The Polytope object
74 % X - A matrix of size d×m where each column represents a point
75 %
76 % Outputs:
77 % TF - True if the point is inside
78
79 d = obj.nDims;
80 n = size(X);
81 core.except.assert(n(1) == d, 'DimensionMismatch', ...
82 'Point dimension must match geometry dimension.');
83
84 X = reshape(X, n(1), []);
85 tol = sqrt(eps);
86 V = obj.vertices;
87 A = [V; ones(1, size(V, 2))];
88 B = [X; ones(1, size(X, 2))];
89 W = pinv(A) * B;
90 TF = all(W >= -tol, 1) & (abs(sum(W, 1) - 1) < tol) & (sum((A*W-B).^2, 1) < tol);
91 if length(n) > 2
92 TF = reshape(TF, n(2:end));
93 else
94 TF = reshape(TF, 1, n(2));
95 end
96 end

99 % ISONBOUNDARY Tests if points are on the polytope boundary.
100 %
101 % TF = isOnBoundary(obj, X) checks if points lie on any of
102 % the defined faces of the polytope. This method requires
103 % that faces be properly defined in the polytope
104 % construction.
105 %
106 % Inputs:
107 % obj - The Polytope object
108 % X - A matrix of size d×m where each column represents a point
109 %
110 % Outputs:
111 % TF - True if the point is on the boundary
112
113 d = obj.nDims;
114 n = size(X);
115 core.except.assert(n(1) == d, 'DimensionMismatch', ...
116 'Point dimension must match geometry dimension.');
117 core.except.assert(~isempty(obj.faces), 'MissingFaces', ...
118 'Faces must be defined to determine boundary points.')
119
120 V = obj.vertices;
121 X = reshape(X, n(1), []);
122 TF = false(1, size(X, 2));
123 for j = 1:length(obj.faces)
124 TF = TF | obj.isOnFace(X, V(:, obj.faces{j}));
125 end
126 if length(n) > 2
127 TF = reshape(TF, n(2:end));

137 %
138 % TF = isOnFace(obj, X, V) determines if points lie on the
139 % face defined by the given vertices. The method dispatches
140 % to dimension-specific implementations for computational
141 % efficiency and accuracy.
142 %
143 % Inputs:
144 % obj - The Polytope object
145 % X - A matrix of size d×m where each column represents a point

162 function TF = isOnFace2D(~, X, V, tol)
163 % ISONFACE2D Tests if points lie on a 2D face (line segment).
164 %
165 % TF = isOnFace2D(obj, X, V, tol) provides a specialized
166 % method for testing point containment on line segments in
167 % 2D space. The method checks both distance to the line and
168 % position within the segment bounds.
169 %
170 % Inputs:
171 % X - A matrix of size 2×m where each column represents a point
172 % V - A matrix where each column is a vertex of the face
173 % tol - Numerical tolerance for distance calculations
174 %
175 % Outputs:
176 % TF - A logical row vector indicating which points are on the face
177
178 %< Get dimensions
179 [~, m] = size(X);
180 TF = false(1, m);
181
182 %< In 2D, a face is a line segment
183 v1 = V(:, 1);
184 v2 = V(:, 2);
185
186 %< Vector from v1 to v2 (edge vector)
187 v = v2 - v1;
188
189 %< Norm of edge vector
190 vLen = norm(v);
191 if vLen < eps
192 return; %< Degenerate edge, no points can be on it
193 end
194
195 %< Unit vector along edge
196 vUnit = v / vLen;
197
198 %< Vectors from v1 to all points
199 vps = X - v1;
200
201 %< Project vps onto vUnit
202 projs = vUnit' * vps; % 1×m vector of projections
203
204 %< Points are within segment bounds if 0 <= proj <= vLen
205 withinBounds = (projs >= 0) & (projs <= vLen);
206
207 %< Distance from points to line
208 projPoints = v1 + vUnit * projs; % Points projected onto line
209 distToLine = sqrt(sum((X - projPoints).^2, 1)); % Distance vector
210
211 %< Points are on line if distance is very small
212 onLine = distToLine < tol;
213
214 %< Points are on segment if they're on the line and within bounds
215 TF = withinBounds & onLine;
216 end

218 function TF = isOnFace3D(~, X, V, tol)
219 % ISONFACE3D Tests if points lie on a 3D face (polygon).
220 %
221 % TF = isOnFace3D(obj, X, V, tol) provides a specialized
222 % method for testing point containment on polygonal faces in
223 % 3D space. The method computes the face normal and checks
224 % if points lie on the corresponding plane.
225 %
226 % Inputs:
227 % X - A matrix of size 3×m where each column represents a point
228 % V - A matrix where each column is a vertex of the face
229 % tol - Numerical tolerance for distance calculations
230 %
231 % Outputs:
232 % TF - A logical row vector indicating which points are on the face
233
234 %< Get dimensions
235 [~, m] = size(X);
236 TF = false(1, m);
237
238 if size(V, 2) < 3
239 return; %< Degenerate face, no points can be on it
240 end
241
242 %< In 3D, compute face normal from three points
243 v1 = V(:, 1);
244 v2 = V(:, 2);
245 v3 = V(:, 3);
246
247 %< Compute face normal
248 normal = cross(v2 - v1, v3 - v1);
249 normalLen = norm(normal);
250
251 if normalLen < eps
252 return; %< Degenerate face, no points can be on it
253 end
254
255 %< Normalize normal vector
256 normal = normal / normalLen;
257
258 %< Compute plane constant for the equation: normal·x + planeConst = 0
259 planeConst = -dot(normal, v1);
260
261 %< Check if points lie on the plane
262 distToPlane = abs(normal' * X + planeConst);
263 onPlane = distToPlane < tol;
264
265 %< For simplicity, we'll assume all points on the plane are also within the face
266 TF = onPlane;
267 end

269 function TF = isOnFaceND(obj, X, V, tol)
270 % ISONFACEND Tests if points lie on an n-dimensional face.
271 %
272 % TF = isOnFaceND(obj, X, V, tol) provides a generalized
273 % method for testing point containment on hyperplanar faces
274 % in n-dimensional space. The method uses the null space to
275 % compute the hyperplane normal.
276 %
277 % Inputs:
278 % obj - The Polytope object
279 % X - A matrix of size nDims×m where each column represents a point
280 % V - A matrix where each column is a vertex of the face
281 % tol - Numerical tolerance for distance calculations
282 %
283 % Outputs:
284 % TF - A logical row vector indicating which points are on the face
285
286 %< Get dimensions
287 [~, m] = size(X);
288 TF = false(1, m);
289
290 %< Start by computing (nDims-1) edge vectors
291 if size(V, 2) < obj.nDims
292 return; %< Degenerate face, need at least nDims vertices
293 end
294
295 %< Use the first vertex as reference
296 v1 = V(:, 1);
297
298 %< Compute edge vectors
299 edges = V(:, 2:obj.nDims) - v1;
300
301 %< Compute the normal to the hyperplane (using null space)
302 normal = null(edges');
303
304 %< If multiple vectors in null space, use the first one
305 if size(normal, 2) > 1
306 normal = normal(:, 1);
307 end
308
309 %< Normalize the normal vector
310 normal = normal / norm(normal);
311
312 %< Compute hyperplane constant
313 planeConst = -dot(normal, v1);
314
315 %< Check if points lie on the hyperplane
316 distToPlane = abs(normal' * X + planeConst);
317 onPlane = distToPlane < tol;
318
319 %< For simplicity, we'll assume all points on the hyperplane are within the face
320 TF = onPlane;
321 end
322 end
