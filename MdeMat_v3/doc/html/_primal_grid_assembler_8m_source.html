4 % PrimalGridAssembler implements Dirichlet boundary treatment for
5 % primal variables in mixed finite element systems. This class
6 % provides specific boundary flux implementations optimized for
7 % the primal variable formulation in mixed methods.
8 %

27 function T = assembleInnerLeftFluxDirichletBoundaryTerms(obj, d, c)
28 % ASSEMBLEINNERLEFTFLUXDIRICHLETBOUNDARYTERMS Assemble inner
29 % left flux Dirichlet boundary terms.
30 %
31 % T = assembleInnerLeftFluxDirichletBoundaryTerms(obj, d, c)
32 % computes the inner contribution to a left flux along the
33 % specified dimension on boundaries with Dirichlet boundary
34 % conditions for primal variables.
35 %
36 % Inputs:
37 % obj - The PrimalGridAssembler object
38 % d - Spatial dimension index (1, 2, or 3)
39 % c - Coefficient vector, one per element
40 %
41 % Outputs:
42 % T - Triplet matrix [row, col, value] for boundary contribution
43
44 np = obj.nLocalDofs;
45 P = obj.fe.projector;
46 G = obj.op.gradient;
47
48 m = obj.mesh.allElementMultiIndices;
49 e = m(:, d) == 1;
50 ne = numel(find(e));
51 m = m(e, :);
52 l = obj.mesh.indexer.multiToLinear(m, 1);
53
54 T = zeros(np^2*ne, 3);
55
56 F = G.fluxData(1, 2*d).matrix;
57 [i, j, v] = find(P.project(F));
58 nv = numel(v);
59 k = 1:(nv * ne);
60 T(k, 1) = reshape(bsxfun(@plus, (l(:).' - 1)*np, i(:)), [], 1);
61 T(k, 2) = reshape(bsxfun(@plus, (l(:).' - 1)*np, j(:)), [], 1);
62 T(k, 3) = kron(c(l), v(:));
63 end
