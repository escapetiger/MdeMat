1classdef GalerkinProjector < handle
2 % GALERKINPROJECTOR Base class for all Galerkin-type projectors.
3 %
4 % GalerkinProjector provides the foundation for Galerkin
5 % approximation methods where functions are projected onto
6 % finite-dimensional spaces using mass operators to solve linear
7 % systems. The class implements the core Galerkin methodology for
8 % function approximation.
9 %
10 % The Galerkin method seeks an approximation \f$u_h \in V_h\f$ of a
11 % function \f$u\f$ such that
12 % \f[
13 % \langle u_h, v_h\rangle = \langle f, v_h\rangle, \forall v_h ∈ V_h,
14 % \f]
15 % where \f$V_h\f$ is the finite-dimensional function space spanned
16 % by the basis.
17 %
18 % This class defines the interface for concrete implementations such
19 % as modal and nodal projectors, which differ in how they construct
20 % the mass operator and embed data into the latent space.
21 %
22 % See Also:
23 % approx.project.ModalProjector, approx.project.NodalProjector
24
25 properties
26 basis % Basis functions that span the approximation space
27 mass % Mass operator matrix for the Galerkin system
28 end

65 function Y = evaluate(obj, varargin)
66 % EVALUATE Evaluate functions in the approximation space at
67 % given points.
68 %
69 % Y = evaluate(obj, Z) evaluates functions represented by
70 % coefficients embedded in the augmented vector Z = [X; C]
71 % at evaluation points X.
72 %
73 % Y = evaluate(obj, X, C) evaluates functions with
74 % coefficient matrix C at evaluation points X.
75 %
76 % Inputs:
77 % obj - The GalerkinProjector object
78 % varargin - Input arguments
79 %< Z - (n+m, p) matrix representing [X; C]
80 %< X - (n, p) matrix representing evaluation points
81 %< C - (m, q) matrix representing coefficients
82 %
83 % Outputs:
84 % Y - Function values ((1 x p) or (q x p))
85
86 switch length(varargin)
87 case 1
88 n = obj.basis.nDims;
89 Z = varargin{1};
90 Z = reshape(Z, size(Z, 1), []);
91
92 core.except.assert(size(Z, 1) > n, 'InvalidInput', ...
93 'Input Z must have more rows than basis dimensions.');
94
95 C = Z((n+1):end, :);
96 X = Z(1:n, :);
97 B = obj.basis.evaluate(X);
98 Y = dot(B, C, 1);
99
100 case 2
101 [X, C] = varargin{:};
102 X = reshape(X, size(X, 1), []);
103 C = reshape(C, size(C, 1), []);
104 B = obj.basis.evaluate(X);
105 Y = C.' * B;
106
107 otherwise
108 core.except.assert(0, 'InvalidInput', ...
109 'Invalid number of input arguments.');
110 end
111 end

113 function U = project(obj, F)
114 % PROJECT Project data onto the function space.
115 %
116 % U = project(obj, F) projects data from the latent space
117 % onto the function space by solving the Galerkin system M*U
118 % = F, where M is the mass operator. Handles well-posed and
119 % overdetermined systems automatically.
120 %
121 % Inputs:
122 % obj - The GalerkinProjector object
123 % F - Data in the latent space (m x ...)
124 %
125 % Outputs:
126 % U - Coefficients of the basis functions (m x ...)
127
128 M = obj.mass;
129 core.except.assert(~isempty(M), 'EmptyMass', ...
130 'The mass operator has not been set. Call setMass() first.');
131
132 m = size(M, 1);
133 core.except.assert(size(F, 1) == m, 'InvalidInput', ...
134 'Data dimension (%d) must match mass operator rows (%d).', ...
135 size(F, 1), m);
136
137 s = size(F);
138 F = reshape(F, m, []);
139
140 if obj.isWellPosed
141 U = M \ F;
142 elseif obj.isOverdetermined
143 U = (M.' * M) \ (M.' * F);
144 else
145 core.except.assert(0, 'UnderdeterminedSystem', ...
146 'Mass operator cannot be underdetermined.');
147 end
148
149 U = reshape(U, [size(U, 1), s(2:end)]);
150 end
