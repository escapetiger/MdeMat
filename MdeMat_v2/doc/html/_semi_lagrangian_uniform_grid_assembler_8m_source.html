2 % SEMILAGRANGIANUNIFORMGRIDASSEMBLER Grid-based assembler for
3 % semi-Lagrangian methods.
4 %
5 % SemiLagrangianUniformGridAssembler provides assembly methods for
6 % semi-Lagrangian finite element operators on uniform structured
7 % grids. This specialized assembler handles transport operators and
8 % boundary conditions for semi-Lagrangian discretizations.
9 %
10 % The assembler supports both free transport operators and boundary
11 % condition assembly, with particular emphasis on handling
12 % characteristic tracing and interpolation required by
13 % semi-Lagrangian methods.
14 %
15 % Examples:
16 % % Create semi-Lagrangian assembler with periodic boundaries
17 % assembler = SemiLagrangianUniformGridAssembler(fe, mesh, op, 0);
18 %
19 % % Assemble transport operator only
20 % A = assembler.transport();
21 %
22 % % Assemble both transport and boundary conditions
23 % [b, A] = assembler.transport(t0, t1, boundaryFunction);

54 function [b, A] = transport(obj, varargin)
55 % TRANSPORT Assemble semi-Lagrangian transport operator.
56 %
57 % b = transport(obj) assembles boundary condition vector for
58 % the transport operator with current boundary conditions.
59 %
60 % [b, A] = transport(obj) assembles both the boundary condition
61 % vector and the free transport operator matrix.
62 %
63 % [b, A] = transport(obj, tBegin, tEnd, f) assembles transport
64 % components with specified time interval and boundary function.
65 %
66 % Inputs:
67 % obj - The SemiLagrangianUniformGridAssembler object
68 % varargin - Input arguments
69 %< tBegin - Beginning time for boundary conditions (optional)
70 %< tEnd - End time for boundary conditions (optional)
71 %< f - Boundary condition function handle (optional)
72 %
73 % Outputs:
74 % b - Boundary condition vector (sparse column vector)
75 % A - Free transport operator matrix (sparse matrix)
76
77 %< ------------------------------------------------------------
78 %< COMMON PARAMETERS
79 %< ------------------------------------------------------------
80 n = obj.nGlobalDofs;
81
82 %< ------------------------------------------------------------
83 %< FREE TRANSPORT
84 %< ------------------------------------------------------------
85 if nargout >= 2
86 T = obj.assembleFreeTransport();
87 A = obj.sparseFromTriplets(T, n, n);
88 end
89
90 %< ------------------------------------------------------------
91 %< BOUNDARY CONDITION
92 %< ------------------------------------------------------------
93 if nargout >= 1
94 b = [];
95 if obj.bcType == 1
96 [tBegin, tEnd, f] = varargin{:};
97 if abs(tBegin - tEnd) < 1e-8, return; end
98 T = obj.assembleBoundaryCondition(tBegin, tEnd, f);
99 b = obj.sparseFromTriplets(T, n, 1);
100 end
101 end
102 end

106 function T = assembleFreeTransport(obj)
107 % ASSEMBLEFREETRANSPORT Assemble free transport operator.
108 %
109 % T = assembleFreeTransport(obj) assembles the triplet
110 % representation of the free transport operator matrix,
111 % handling characteristic tracing and interpolation for
112 % semi-Lagrangian methods.
113 %
114 % Inputs:
115 % obj - The SemiLagrangianUniformGridAssembler object
116 %
117 % Outputs:
118 % T - Triplet matrix [row, col, value] for transport operator
119
120 nl = obj.nLocalDofs;
121 np = obj.op.clipper.nVolumePieces;
122 ne = obj.mesh.nTotalElements;
123
124 M = arrayfun(@(i)obj.fe.projector.project(obj.op.mass(i).matrix), 1:np, 'Un', 0);
125 s = obj.op.clipper.volumeShifts;
126 d = obj.nDims;
127
128 T = cell(np, 1);
129 for p = 1:np
130 T{p} = zeros(nl^2*ne, 3);
131 [i, j, v] = find(M{p});
132 nv = numel(v);
133 m = obj.mesh.allElementMultiIndices;
134 if obj.bcType == 0
135 m = m + s(:, p).';
136 l = obj.mesh.indexer.multiToLinear(m, 0);
137 k = 1:(nv * ne);
138 T{p}(k, 1) = reshape(bsxfun(@plus, (0:ne - 1)*nl, i(:)), [], 1);
139 T{p}(k, 2) = reshape(bsxfun(@plus, (l(:).' - 1)*nl, j(:)), [], 1);
140 T{p}(k, 3) = kron(ones(ne, 1), v(:));
141 else
142 b = m + s(:, p).';
143 b = (b < 1) .* (2 * (1:d) - 1) + ...
144 (b > obj.mesh.resolution) .* (2 * (1:d));
145 m = m(sum(b ~= 0, 2) == 0, :);
146 nb = size(m, 1);
147 k = 1:(nv * nb);
148 l1 = obj.mesh.indexer.multiToLinear(m, 1);
149 l2 = obj.mesh.indexer.multiToLinear(m + s(:, p).', 1);
150 T{p}(k, 1) = reshape(bsxfun(@plus, (l1(:).' - 1)*nl, i(:)), [], 1);
151 T{p}(k, 2) = reshape(bsxfun(@plus, (l2(:).' - 1)*nl, j(:)), [], 1);
152 T{p}(k, 3) = kron(ones(nb, 1), v(:));
153 end
154 end
155 T = vertcat(T{:});
156 end

158 function T = assembleBoundaryCondition(obj, tBegin, tEnd, f)
159 % ASSEMBLEBOUNDARYCONDITION Assemble boundary condition terms.
160 %
161 % T = assembleBoundaryCondition(obj, tBegin, tEnd, f)
162 % assembles the boundary condition contribution for
163 % semi-Lagrangian transport, handling characteristic tracing
164 % to domain boundaries and evaluation of boundary data.
165 %
166 % Inputs:
167 % obj - The SemiLagrangianUniformGridAssembler object
168 % tBegin - Beginning time for integration
169 % tEnd - End time for integration
170 % f - Boundary condition function handle f(x, t)
171 %
172 % Outputs:
173 % T - Triplet matrix [row, col, value] for boundary conditions
174
175 nl = obj.nLocalDofs;
176 np = obj.op.clipper.nVolumePieces;
177 nd = obj.nDims;
178 ht = tBegin - tEnd;
179 v = obj.op.clipper.velocity;
180
181 T = cell(1, np);
182 for p = 1:np
183 %< Find all elements whose pth volume piece goes outside
184 %< the computational region
185 m = obj.mesh.allElementMultiIndices;
186 s = obj.op.clipper.volumeShifts(:, p).';
187 b = (m + s < 1) .* (2 * (1:nd) - 1) + (m + s > obj.mesh.resolution) .* (2 * (1:nd));
188 b = m(sum(b ~= 0, 2) > 0, :);
189 nb = size(b, 1);
190 k = obj.mesh.indexer.multiToLinear(b, 1);
191 T{p} = zeros(nb*nl, 3);
192 T{p}(:, 1) = reshape(bsxfun(@plus, (k(:)-1).'*nl, (1:nl).'), [], 1);
193 T{p}(:, 2) = 1;
194
195 if nb == 0, continue; end
196
197 %< Find all quadrature points to evaluate
198 nq = obj.op.clipper.volumeEulerianData(1, p).nPoints;
199 X1 = zeros(nd + 1, nb*nq);
200 X2 = zeros(nd + 1, nb*nq);
201 Xq = obj.op.clipper.volumeEulerianData(1, p).nodes;
202 X2(1:nd, :) = obj.mesh.collocate(Xq, b);
203 X2(nd+1, :) = tBegin;
204 X1(1:nd, :) = X2(1:nd, :) - ht * v(:);
205 X1(nd+1, :) = tEnd;
206
207 %< Find intersection of characteristics and boundaries
208 I = core.geometry.LineSegmentByPlaneIntersector(nd+1);
209 b = reshape(obj.mesh.bbox, 2, []);
210 b = -b(2*(1:nd) - (v > 0));
211 Xb = zeros(nd + 1, nb*nq);
212 for d = 1:nd
213 a = zeros(nd+1, 1);
214 a(d) = 1;
215 [X, TF] = I.intersect(X1, X2, a, b(d));
216 Xb(:, TF) = X;
217 end
218
219 %< Evaluate boundary function and project
220 U = f(Xb(1:nd, :), Xb(nd+1, :));
221 U = reshape(U, nq, nb);
222 P = obj.op.volumePieceData(1, p).vector;
223 T{p}(:, 3) = reshape(obj.fe.projector.project(P * U), [], 1);
224 end
225 T = vertcat(T{:});
226 end
227 end
