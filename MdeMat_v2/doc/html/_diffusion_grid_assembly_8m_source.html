1classdef DiffusionGridAssembly < fem.assembly.GridAssembly
2 % DIFFUSIONGRIDASSEMBLY Grid-based assembly for diffusion operators.
3 %
4 % DiffusionGridAssembly provides assembly methods for diffusion-type
5 % operators on structured grid meshes. This class combines auxiliary
6 % and primal variable assemblys to construct complete diffusion
7 % operators with various jump treatments and boundary conditions.
8 %
9 % The assembly supports both implicit and explicit jump treatments
10 % for handling discontinuities at element interfaces, which is
11 % essential for discontinuous Galerkin and mixed finite element
12 % methods.
13 %
14 % Examples:
15 % % Create diffusion assembly with implicit jumps
16 % assembly = DiffusionGridAssembly(fe, mesh, op, 1, 2, 1, 1);
17 %
18 % % Assemble isotropic diffusion operator
19 % diffusionMatrix = assembly.hessian(eye(2));
20 %
21 % % Assemble anisotropic diffusion with boundary conditions
22 % anisotropyTensor = [2.0, 0.1; 0.1, 0.5];
23 % bcVector = assembly.hessianBc(anisotropyTensor, @bcFunc, @jumpFunc);
24 %

37 % DIFFUSIONGRIDASSEMBLY Constructor for DiffusionGridAssembly.
38 %
39 % obj = DiffusionGridAssembly(fe, mesh, op, bcType, auxFlux,
40 % prmFlux, jumpType) creates an assembly for diffusion
41 % operators using mixed formulations with auxiliary and
42 % primal variables.
43 %
44 % Inputs:
45 % fe - Finite element object
46 % mesh - Grid mesh object (must be approx.mesh.Grid)
47 % op - Finite element operator object
48 % bcType - Boundary condition type (0=periodic, 1=Dirichlet)
49 % auxFlux - Flux type for auxiliary variables (1=left, 2=right, 3=central)
50 % prmFlux - Flux type for primal variables (1=left, 2=right, 3=central)
51 % jumpType - Jump treatment (1=implicit, 2=explicit)
52 %
53 % Outputs:
54 % obj - Constructed DiffusionGridAssembly object
55
56 core.except.assert(isa(mesh, 'approx.mesh.Grid'), ...
57 'InvalidInput', ...
58 'Finite element space must be based on grid.');
59
60 obj@fem.assembly.GridAssembly(fe, mesh, op, bcType);
61 obj.aux.asm = fem.assembly.AuxiliaryGridAssembly(fe, mesh, op, bcType, auxFlux);
62 obj.aux.mat = cell(1, obj.nDims);
63 obj.prm.asm = fem.assembly.PrimalGridAssembly(fe, mesh, op, bcType, prmFlux);
64 obj.prm.mat = cell(1, obj.nDims);
65 obj.jumpType = jumpType;
66 end

68 function A = hessian(obj, c)
69 % HESSIAN Assemble diffusion Hessian operator.
70 %
71 % A = hessian(obj, c) assembles the complete diffusion
72 % operator matrix using the specified diffusion tensor. The
73 % method combines auxiliary and primal variable contributions
74 % with appropriate jump terms based on the jump type.
75 %
76 % Inputs:
77 % obj - The DiffusionGridAssembly object
78 % c - Diffusion tensor (nDims×nDims matrix)
79 %
80 % Outputs:
81 % A - Assembled diffusion operator sparse matrix
82 %
83 % Examples:
84 % % Isotropic diffusion
85 % isotropicMatrix = assembly.hessian(eye(2));
86 %
87 % % Anisotropic diffusion
88 % tensor = [2.0, 0.5; 0.5, 1.0];
89 % anisotropicMatrix = assembly.hessian(tensor);
90
91 for i = 1:obj.nDims
92 obj.aux.mat{i} = obj.aux.asm.partial(i, 1);
93 end
94 for i = 1:obj.nDims
95 obj.prm.mat{i} = obj.prm.asm.partial(i, 1);
96 end
97 n = obj.nGlobalDofs;
98 A = sparse(n, n);
99 for i = 1:obj.nDims
100 for j = 1:obj.nDims
101 if c(i, j) ~= 0
102 A = A + c(i, j) * obj.aux.mat{i} * obj.prm.mat{j};
103 if obj.bcType && obj.jumpType == 1
104 B = obj.aux.asm.outerJump(j, c(i, j));
105 A = A + c(i, j) * B;
106 end
107 end
108 end
109 end
110 end

112 function b = hessianBc(obj, c, f, g, varargin)
113 % HESSIANBC Assemble boundary condition vector for diffusion
114 % operator.
115 %
116 % b = hessianBc(obj, c, f, g) assembles the boundary
117 % condition contribution vector for the diffusion operator
118 % with specified boundary functions and diffusion tensor.
119 %
120 % Inputs:
121 % obj - The DiffusionGridAssembly object
122 % c - Diffusion tensor (nDims×nDims matrix)
123 % f - Boundary condition function handle
124 % g - Extraction function handle (optional for explicit jumps)
125 % varargin - Additional arguments for boundary functions
126 %
127 % Outputs:
128 % b - Assembled boundary condition sparse vector
129 %
130 % Examples:
131 % % Simple Dirichlet boundary conditions
132 % bcVector = assembly.hessianBc(eye(2), @(x) sin(x), []);
133 %
134 % % With explicit jump treatment
135 % bcVector = assembly.hessianBc(tensor, @bcFunc, @jumpFunc, param1);
136
137 n = obj.nGlobalDofs;
138 b = sparse(n, 1);
139 for i = 1:obj.nDims
140 for j = 1:obj.nDims
141 b = b + obj.partialBc(i, j, c(i, j), f, g, varargin{:});
142 end
143 end
144 end

146 function b = partialBc(obj, i, j, c, f, g, varargin)
147 % PARTIALBC Assemble partial boundary condition contribution.
148 %
149 % b = partialBc(obj, i, j, c, f, g) assembles the boundary
150 % condition contribution for a specific component of the
151 % diffusion tensor, handling both Dirichlet conditions and
152 % jump terms as appropriate.
153 %
154 % Inputs:
155 % obj - The DiffusionGridAssembly object
156 % i - First tensor index (row)
157 % j - Second tensor index (column)
158 % c - Tensor component value
159 % f - Boundary condition function handle
160 % g - Jump function handle (for explicit jumps)
161 % varargin - Additional arguments for boundary functions
162 %
163 % Outputs:
164 % b - Boundary condition contribution sparse vector
165
166 ne = obj.nTotalElements;
167
168 %< ------------------------------------------------------------
169 %< COEFFICIENTS
170 %< ------------------------------------------------------------
171 if isa(obj.mesh, 'approx.mesh.UniformGrid')
172 h = obj.mesh.spacings{j};
173 c = ones(1, ne) .* c ./ h;
174 end
175
176 if isa(obj.mesh, 'approx.mesh.NonuniformGrid')
177 h = obj.mesh.spacings;
178 [h{:}] = ndgrid(h{:});
179 c = c ./ h{j}(:);
180 end
181
182 c = c(:);
183
184 %< ------------------------------------------------------------
185 %< FLUX CONTRIBUTION
186 %< ------------------------------------------------------------
187 if isa(obj.fe, 'fem.element.L2FiniteElement')
188 T1 = obj.assembleLeftDirichletBoundaryConditionTerms(j, c, f, varargin{:});
189 T2 = obj.assembleRightDirichletBoundaryConditionTerms(j, c, f, varargin{:});
190 end
191
192 %< ------------------------------------------------------------
193 %< TRIPLETS TO SPARSE MATRIX
194 %< ------------------------------------------------------------
195 n = obj.nGlobalDofs;
196 T = [T1; T2];
197
198 b = obj.sparseFromTriplets(T, n, 1);
199 b = obj.aux.mat{i} * b;
200 if obj.jumpType == 1
201 if obj.aux.asm.fluxType == 1
202 b = b + obj.sparseFromTriplets(T1, n, 1);
203 elseif obj.aux.asm.fluxType == 2
204 b = b + obj.sparseFromTriplets(T2, n, 1);
205 elseif obj.aux.asm.fluxType == 3
206 b = b + b / 2;
207 end
208 end
209 if ~isempty(g) && obj.jumpType == 2
210 if obj.prm.asm.fluxType == 1
211 TT = obj.assembleLeftDirichletBoundaryJumpTerms(j, c, f, g, varargin{:});
212 elseif obj.prm.asm.fluxType == 2
213 TT = obj.assembleRightDirichletBoundaryJumpTerms(j, c, f, g, varargin{:});
214 elseif obj.prm.asm.fluxType == 3
215 TT1 = obj.assembleLeftDirichletBoundaryJumpTerms(j, c/2, f, g, varargin{:});
216 TT2 = obj.assembleRightDirichletBoundaryJumpTerms(j, c/2, f, g, varargin{:});
217 TT = [TT1; TT2];
218 end
219 b = b + obj.sparseFromTriplets(TT, n, 1);
220 end
221 end
222 end
