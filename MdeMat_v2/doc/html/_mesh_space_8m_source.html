1classdef MeshSpace < approx.space.FunctionSpace
2 % MESHSPACE Function space defined on a geometric mesh.
3 %
4 % MeshSpace handles the mapping between reference elements and
5 % physical elements in the mesh, allowing for efficient computation
6 % of global operations such as L2 projections and weighted averages.
7 % It serves as the foundation for finite element computations on
8 % complex geometries.
9 %
10 % Examples:
11 % % Create mesh space from element and mesh
12 % space = MeshSpace(element, mesh);
13 %
14 % % Project analytical function onto the mesh space
15 % analyticFunc = @(x) sin(pi*x(1,:)) .* cos(pi*x(2,:));
16 % coefficients = space.project(analyticFunc);
17 %
18 % % Evaluate projected function at specific points
19 % evalPoints = [0.25, 0.75; 0.5, 0.5];
20 % values = space.evaluate(evalPoints, coefficients);
21 %
22 % % Compute weighted average of a function over the domain
23 % quadraticFunc = @(x) x(1,:).^2 + x(2,:).^2;
24 % avgValue = space.average(quadraticFunc);
25 %
26 % See also:

76 function Y = evaluate(obj, varargin)
77 % EVALUATE Evaluate functions in the mesh space.
78 %
79 % Y = evaluate(obj, Z) evaluates using combined input matrix
80 % containing evaluation points and coefficients.
81 %
82 % Y = evaluate(obj, X, C) evaluates at physical points X
83 % using coefficient matrix C organized by elements.
84 %
85 % Y = evaluate(obj, f, xRef, ...) evaluates function handle f
86 % at reelementrence points on all mesh elements.
87 %
88 % Inputs:
89 % obj - The MeshSpace object
90 % varargin - Input arguments
91 %< Z - (nDims+nDofs, nPoints) combined matrix [X; C]
92 %< X - (nDims, nPoints) physical evaluation points
93 %< C - (nDofs, nElements*nFunctions) coefficient matrix
94 %< f - Function handle to evaluate
95 %< xRef - (nDims, nRefPoints) reelementrence points per element
96 %
97 % Outputs:
98 % Y - Function values at evaluation points
99
100 nArgs = length(varargin);
101
102 if nArgs == 1
103 Z = varargin{1};
104 Y = obj.element.evaluate(Z);
105 elseif nArgs == 2 && ~isa(varargin{1}, 'function_handle')
106 [X, C] = varargin{:};
107 N = obj.mesh.nTotalElements;
108 p = size(X, 2);
109 m = obj.element.nDofs;
110 C = reshape(C, m, []);
111 Y = obj.element.evaluate(X, C);
112 Y = reshape(Y.', N*p, []);
113 elseif nArgs >= 2 && isa(varargin{1}, 'function_handle')
114 [f, xRef] = varargin{1:2};
115 N = obj.mesh.nTotalElements;
116 p = size(xRef, 2);
117 I = obj.mesh.allElementMultiIndices;
118 X = obj.mesh.collocate(xRef, I);
119 Y = f(X, varargin{3:end});
120 Y = reshape(Y, N*p, []);
121 end
122 end

124 function A = average(obj, f)
125 % AVERAGE Compute weighted average of a function over the mesh.
126 %
127 % A = average(obj, f) computes the weighted average of the
128 % function f using the integration weights across all mesh
129 % elements. This provides a measure of the function's mean
130 % value over the computational domain.
131 %
132 % Inputs:
133 % obj - The MeshSpace object
134 % f - Function handle to compute average
135 %
136 % Outputs:
137 % A - (nElements, nComponents) weighted average values
138
139 n = obj.mesh.nTotalElements;
140 w = obj.element.volumeData.weights;
141 I = obj.mesh.allElementMultiIndices;
142 xRef = obj.element.volumeData.nodes;
143 X = obj.mesh.collocate(xRef, I);
144 A = f(X);
145 A = w * reshape(A, length(w), []);
146 A = reshape(A, n, []);
147 end

149 function U = project(obj, f, varargin)
150 % PROJECT Project a function onto the finite element space.
151 %
152 % U = project(obj, f) projects the function f onto the finite
153 % element space using the appropriate Galerkin projection
154 % method. The projection method (modal or nodal) is
155 % determined by the type of projector associated with the
156 % finite element.
157 %
158 % Inputs:
159 % obj - The MeshSpace object
160 % f - Function handle to project onto the space
161 % varargin - Input arguments for function handle
162 %
163 % Outputs:
164 % U - (nDofs, nComponents) projected coefficients
165
166 P = obj.element.projector;
167 B = P.basis;
168
169 cls = 'core.function.SeparableFunction';
170 core.except.assert( ...
171 ~isa(B, cls) || ~iscell(B.factors), ...
172 'InvalidBasis', ...
173 'All basis functions must have the same type.');
174
175 if isempty(varargin)
176 I = obj.mesh.allElementMultiIndices;
177 xRef = obj.element.volumeData.nodes;
178 X = obj.mesh.collocate(xRef, I);
179 U = f(X);
180 else
181 U = f(varargin{:});
182 end
183
184 if isa(P, 'approx.project.ModalProjector')
185 D = obj.element.volumeData;
186 w = D.weights;
187 V = D.values;
188 F = P.embed(U, V, w);
189 end
190
191 if isa(P, 'approx.project.NodalProjector')
192 F = P.embed(U);
193 end
194
195 U = P.project(F);
196 U = reshape(U, obj.nGlobalDofs, []);
197 end
198 end
