1classdef C0FiniteElementClipper < fem.element.FiniteElementClipper
2 % C0FINITEELEMENTCLIPPER Geometric clipper for continuous finite elements.
3 %
4 % C0FiniteElementClipper implements geometric clipping operations
5 % for continuous finite elements in semi-Lagrangian methods. It
6 % computes the intersection between the current Eulerian element
7 % and the upstream footprint traced along characteristic curves,
8 % creating volume pieces for accurate transport calculations.
9 %
10 % The clipper handles the complex geometry resulting from backward
11 % characteristic tracing by decomposing the intersection region
12 % into simpler geometric pieces. Each piece is equipped with
13 % appropriate quadrature rules and function data for both Eulerian
14 % and upstream (Lagrangian) representations.
15 %
16 % This approach enables stable and accurate semi-Lagrangian schemes
17 % for convection-dominated problems without the CFL restriction
18 % that limits explicit Eulerian methods.
19 %
20 % Examples:
21 % % Create clipper for C0 element with specified velocity
22 % velocity = [0.5; 0.3];
23 % element = fem.element.C0FiniteElement(geometry, projector, integrator);
24 % clipper = C0FiniteElementClipper(element, velocity);
25 %
26 % % Set up volume piece data for time step and mesh spacing
27 % timeStep = 0.1;
28 % meshSpacing = [0.05; 0.05];
29 % clipper.setVolumePieceData(timeStep, meshSpacing);
30 %
31 % % Access computed pieces and data
32 % nPieces = clipper.nVolumePieces;
33 % eulerianData = clipper.volumeEulerianData;
34 % upstreamData = clipper.volumeUpstreamData;
35 %
36 % See also:
37 % fem.element.FiniteElementClipper, fem.element.L2FiniteElementClipper,
38 % fem.element.C0FiniteElement

53 function setVolumePieceData(obj, ht, hx)
54 % SETVOLUMEPIECEDATA Set up volume piece data for semi-Lagrangian transport.
55 %
56 % setVolumePieceData(obj, ht, hx) computes the geometric
57 % intersection between the current element and the upstream
58 % footprint, creating volume pieces with associated function
59 % data for accurate semi-Lagrangian integration.
60 %
61 % Inputs:
62 % obj - The C0FiniteElementClipper object
63 % ht - Time step for characteristic tracking (positive scalar)
64 % hx - Mesh spacing vector (column vector)
65 %
66 % Outputs:
67 % NULL
68
69 nd = obj.fe.nDims;
70 v = obj.velocity;
71
72 %< Compute clipping geometry
73 bbox = obj.fe.geometry.bbox;
74 ratio = v(:) .* ht ./ hx(:);
75 poi = ratio - floor(ratio) - 1 / 2;
76
77 %< Initialize volume pieces structure
78 obj.volumePieces = struct();
79 obj.volumePieces.xc = zeros(nd, 2^nd);
80 obj.volumePieces.hx = zeros(nd, 2^nd);
81 np = 0;
82
83 %< Generate intersection pieces
84 indexer = core.linalg.MultiIndexer(ones(1, nd) * 2);
85 m = indexer.generate();
86 for i = 1:2^nd
87 x1 = poi;
88 x2 = bbox((m(i, :) + (0:nd - 1) * 2)');
89 xc = (x1 + x2) / 2;
90 hx = abs(x2 - x1);
91
92 %< Skip degenerate pieces
93 if prod(hx) < 1e-8
94 continue;
95 end
96
97 np = np + 1;
98 obj.volumePieces.xc(:, np) = xc;
99 obj.volumePieces.hx(:, np) = hx;
100 end
101
102 %< Handle case with no valid pieces
103 if np == 0
104 obj.volumePieces.xc = zeros(nd, 1);
105 obj.volumePieces.hx = ones(nd, 1);
106 np = 1;
107 else
108 obj.volumePieces.xc = obj.volumePieces.xc(:, 1:np);
109 obj.volumePieces.hx = obj.volumePieces.hx(:, 1:np);
110 end
111
112 %< Compute piece boundaries
113 obj.volumePieces.xmin = obj.volumePieces.xc - obj.volumePieces.hx / 2;
114 obj.volumePieces.xmax = obj.volumePieces.xc + obj.volumePieces.hx / 2;
115 obj.nVolumePieces = np;
116
117 %< Create function data objects for each piece
118 I = obj.fe.volumeData.integrator;
119 obj.volumeEulerianData = arrayfun(@(i) fem.data.FunctionData(I.copy()), 1:np);
120 obj.volumeUpstreamData = arrayfun(@(i) fem.data.FunctionData(I.copy()), 1:np);
121
122 %< Set up integration points for each piece
123 nq = obj.fe.projector.basis.nFactorCodims;
124 for i = 1:np
125 a = obj.volumePieces.xmin(:, i);
126 b = obj.volumePieces.xmax(:, i);
127 obj.volumeEulerianData(i).setPoints(nq, false, a, b);
128 obj.volumeUpstreamData(i).setPoints(nq, false, -b, -a);
129 end
130
131 %< Compute basis function values at integration points
132 f = obj.fe.projector.basis;
133 for i = 1:np
134 De = obj.volumeEulerianData(i);
135 Du = obj.volumeUpstreamData(i);
136 Ie = De.integrator;
137 Iu = Du.integrator;
138 Xe = Ie.nodes(1:nd, :);
139 Xu = Iu.nodes(1:nd, :);
140
141 %< Set function values and derivatives
142 De.setValues(f, Xe);
143 Du.setValues(f, Xu);
144 De.setDerivatives(f, Xe, 1);
145 end
146
147 %< Compute index shifts for upstream element mapping
148 obj.volumeShifts = zeros(nd, np);
149 mm = ceil(ratio) .* (ratio > 0) + (floor(ratio) + 1) .* (ratio < 0);
150 mm0 = -mm;
151 mm1 = -mm + 1;
152 for i = 1:np
153 bb = obj.volumePieces.xc(:, i) <= 0;
154 obj.volumeShifts(:, i) = mm0 .* bb + mm1 .* (1 - bb);
155 end
156 end
157 end
158end
