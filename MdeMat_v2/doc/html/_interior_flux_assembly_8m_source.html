1classdef InteriorFluxAssembly < approx.assembly.Assembly
2 % INTERIORFLUXASSEMBLY Assembly for interior flux terms in DG methods.
3 %
4 % InteriorFluxAssembly handles the assembly of flux terms between
5 % neighboring elements in the interior of the computational domain.
6 % These flux terms are essential for providing inter-element coupling
7 % in discontinuous Galerkin (DG) methods, where the solution is
8 % allowed to be discontinuous across element boundaries.
9 %
10 % The class implements both inner and outer flux contributions:
11 % - Inner fluxes: contributions from the current element's boundary
12 % - Outer fluxes: contributions from neighboring elements' boundaries
13 %
14 % Each flux type (left-biased, right-biased) is supported, enabling
15 % the construction of upwind-like or downwind-like numerical schemes.
16 % The flux terms are assembled using triplet format for efficient
17 % sparse matrix construction.
18 %
19 % See also:
20 % approx.assembly.Assembly, approx.assembly.FluxAssembly,
21 % approx.element.L2ElementOperator
22

24 function T = assembleInnerLeftFlux(obj, dim, coe)
25 % ASSEMBLEINNNERLEFTFLUX Assemble inner left-biased flux terms.
26 %
27 % T = assembleInnerLeftFlux(obj, dim, coe) assembles the
28 % inner flux contributions for left-biased numerical fluxes
29 % along the specified dimension. Inner fluxes represent the
30 % contribution from the current element's boundary to the
31 % flux integral.
32 %
33 % Inputs:
34 % obj - The InteriorFluxAssembly object
35 % dim - Dimension index (positive integer)
36 % coe - Coefficient vector (nElements x 1)
37 %
38 % Outputs:
39 % T - Triplet matrix [row, col, val]
40
41 nl = obj.space.nLocalDofs;
42 P = obj.space.element.projector;
43 G = obj.operator.gradient;
44
45 m = obj.space.mesh.allElementMultiIndices;
46 e = m(:, dim) > 1 & m(:, dim) < obj.space.mesh.resolution(dim);
47 m = m(e, :);
48 l = obj.space.mesh.indexer.multiToLinear(m, 0);
49 ne = numel(l);
50
51 T = zeros(nl^2*ne, 3);
52
53 F = G.fluxData(1, 2*dim).matrix;
54 [i, j, v] = find(P.project(F));
55 nv = numel(v);
56 k = 1:(nv * ne);
57 T(k, 1) = reshape(bsxfun(@plus, (l(:).' - 1)*nl, i(:)), [], 1);
58 T(k, 2) = reshape(bsxfun(@plus, (l(:).' - 1)*nl, j(:)), [], 1);
59 T(k, 3) = kron(coe(l), v(:));
60 end
