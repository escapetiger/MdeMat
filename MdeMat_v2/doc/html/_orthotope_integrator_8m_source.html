1classdef OrthotopeIntegrator < approx.integrate.Integrator
2 % ORTHOTOPEINTEGRATOR Numerical integration over orthotope domains.
3 %
4 % OrthotopeIntegrator provides specialized integration capabilities
5 % for functions defined on multidimensional orthotopes
6 % (hyperrectangles). It uses separable integration rules that
7 % construct multidimensional quadrature by tensor products of
8 % univariate rules.
9 %
10 % Examples:
11 % % Create integrator with Gauss-Legendre rule for 3D orthotope
12 % integrator = OrthotopeIntegrator(3, 'gauss_legendre');
13 %
14 % % Create integrator with Gauss-Lobatto rule for 2D rectangle
15 % integrator = OrthotopeIntegrator(2, 'gauss_lobatto');
16 %
17 % % Using existing rule object
18 % rule = approx.integrate.SeparableRule.gaussLegendre(4);
19 % integrator = OrthotopeIntegrator(rule);
20 %
21 % Notes:
22 % Orthotopes are multidimensional generalizations of rectangles.
23 % The separable rules are most efficient for functions that can
24 % be expressed as products of univariate functions.
25 %
26 % See also:
27 % approx.integrate.Integrator, approx.integrate.SeparableRule,
28 % approx.integrate.OrthotopeFaceIntegrator

46 % Outputs:

49 if nargin == 1

52 [nDims, ruleId] = varargin{:};

61 error('Invalid number of arguments.');
62 end
63 obj@approx.integrate.Integrator(rule);
64 end
