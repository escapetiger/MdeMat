1classdef BilinearFormData < handle
2 % BILINEARFORMDATA A collection of data for a bilinear form.
3 %
4 % BilinearFormData represents a bilinear form through its matrix
5 % representation, computed from function data and derivatives. Each
6 % bilinear form corresponds to a weighted inner product between
7 % basis functions and their derivatives of the form:
8 %
9 % \f[
10 % a(u, v) = \int_{\Omega} \frac{\partial^i u}{\partial x^i}
11 % w(x) \frac{\partial^j v}{\partial x^j} \, dx
12 % \f]
13 %
14 % where \f$u\f$ and \f$v\f$ are basis functions, \f$i\f$ and \f$j\f$
15 % are derivative orders, and \f$w(x)\f$ are integration weights.
16 % The discrete representation is computed using quadrature rules.
17 %
18 % This class is fundamental in finite element methods for assembling
19 % system matrices from weak formulations of differential equations.
20 %
21 % Examples:
22 % % Create empty bilinear form
23 % form = BilinearFormData();
24 %
25 % % Create mass matrix (0,0 derivatives)
26 % massForm = BilinearFormData(functionData1, functionData2, 0, 0);
27 % M = massForm.matrix;
28 %
29 % % Create stiffness matrix (1,1 derivatives)
30 % stiffForm = BilinearFormData(functionData1, functionData2, 1, 1);
31 % K = stiffForm.matrix;
32 %
33 % % Mixed derivative form
34 % mixedForm = BilinearFormData(functionData1, functionData2, 0, 1);
35 %
36 % See also:
37 % approx.fem.data.FunctionData, approx.integrate.Integrator
38

40 matrix % Matrix representation of the bilinear form
41 vector % Vector representation of the weighted test function values
42 end

73 function obj = setMatrix(obj, D1, D2, i, j)
74 % SETMATRIX Set the matrix representation of the bilinear form.
75 %
76 % obj = setMatrix(obj, D1, D2, i, j) computes the matrix
77 % representation as the weighted inner product between
78 % function values or derivatives from two function data
79 % objects. The computation follows the discrete bilinear
80 % form:
81 %
82 % \f[
83 % A_{mn} = \sum_{k} U_k^{(i)}(x_m) w_k V_k^{(j)}(x_n)
84 % \f]
85 %
86 % where \f$U^{(i)}\f$ and \f$V^{(j)}\f$ are the i-th and j-th
87 % derivatives, and \f$w_k\f$ are integration weights.
88 %
89 % Inputs:
90 % obj - The BilinearFormData object
91 % D1 - First function data object (FunctionData)
92 % D2 - Second function data object (FunctionData)
93 % i - Derivative order for first function (0 for values)
94 % j - Derivative order for second function (0 for values)
95 %
96 % Outputs:
97 % obj - The BilinearFormData object
98
99 %< Create diagonal weight matrix
100 W = diag(D1.weights);
101
102 %< Extract function values or derivatives for first function
103 if i == 0
104 U = D1.values;
105 else
106 U = D1.derivatives{i};
107 end
108
109 %< Extract function values or derivatives for second function
110 if j == 0
111 V = D2.values;
112 else
113 V = D2.derivatives{j};
114 end
115
116 %< Compute bilinear form matrix: A = U^T * W * V
117 A = U * W * V.';
118 obj.matrix = A;
119
120 %< Compute weighted test function vector: b = U^T * W
121 obj.vector = U * W;
122 end
123 end
