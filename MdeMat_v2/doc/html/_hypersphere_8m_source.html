2 % HYPERSPHERE Geometry for (d-1)-sphere domains.
3 %
4 % Hypersphere represents a generalized geometric surface that extends
5 % the concept of circles and spherical surfaces to arbitrary
6 % dimensions. It represents the set of points at a constant distance
7 % (radius) from a center point in d-dimensional Euclidean space.
8 % Dimensional representations:
9 % - 0-sphere: A pair of points on a line
10 % - 1-sphere: A circle in a plane
11 % - 2-sphere: A sphere surface in 3D space
12 % - n-sphere: A surface in d-dimensional space
13 %
14 % Examples:
15 % % Create a 1-sphere (circle) with radius 3 centered at [1,2]
16 % sphere = core.geometry.Hypersphere([1,2], 3);
17 % circumference = sphere.magnitude(); % Returns 2*pi*3
18 %
19 % % Create a unit 2-sphere
20 % sphere2D = core.geometry.Hypersphere(2);

32 %
33 % obj = Hypersphere(nSphereDims) creates a unit hypersphere
34 % centered at the origin with unit radius. The dimension of
35 % the sphere surface is specified by nSphereDims, and the
36 % ambient space dimension is nSphereDims+1.
37 %

45 %< radius - Radius of the n-sphere (positive scalar)

47 % Outputs:
48 % obj - Constructed Hypersphere object

52 nSphereDims = varargin{1};

72 function m = magnitude(obj)
73 % MAGNITUDE Returns the surface area of the hypersphere.
74 %
75 % m = magnitude(obj) computes the (n-1)-dimensional surface
76 % area of the hypersphere using the general formula for
77 % hyperspheres. Special cases are optimized for common
78 % dimensions (0-sphere, 1-sphere, 2-sphere) for computational
79 % efficiency and accuracy.
80 %
81 % Inputs:
82 % obj - The Hypersphere object
83 %
84 % Outputs:
85 % m - Surface area of the hypersphere
86
87 n = obj.nSphereDims;
88
89 r = obj.radius;
90 switch n
91 case 0
92 tol = sqrt(eps);
93 m = 2 * (abs(r) > tol);
94 case 1
95 m = 2 * pi * r;
96 case 2
97 m = 4 * pi * r^2;
98 otherwise
99 m = 2 * pi^((n+1)/2) / gamma((n+1)/2) * r^n;
100 end
101 end

126 function TF = isOnBoundary(obj, X)
127 % ISONBOUNDARY Tests if points are on the hypersphere surface.
128 %
129 % TF = isOnBoundary(obj, X) determines whether points lie
130 % exactly on the hypersphere surface by checking if their
131 % distance from the center equals the radius within numerical
132 % tolerance. For a hypersphere, this is equivalent to testing
133 % if points lie on the sphere.
134 %
135 % Inputs:
136 % obj - The Hypersphere object
137 % X - Coordinates (matrix)
138 %
139 % Outputs:
140 % TF - True if the point is on the boundary
141
142 d = obj.nDims;
143 n = size(X);
144 core.except.assert(n(1) == d, 'DimensionMismatch', ...
145 'Point dimension must match geometry dimension.');
146
147 r = obj.radius;
148 c = obj.centroid(:);
149 tol = sqrt(eps);
150 TF = abs(sum((X - c).^2, 1)-r^2) <= tol;
151 end

188 %
189 % C = sphericalToCartesian(obj, S) transforms points from
190 % hyperspherical to Cartesian coordinates, receiving only
191 % angular coordinates and using the sphere's radius

193 % hypersphere surface.
194 %
195 % Inputs:
196 % obj - The Hypersphere object

199 % Outputs:
