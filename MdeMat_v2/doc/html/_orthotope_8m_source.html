32 function obj = Orthotope(bbox)
33 % ORTHOTOPE Constructor for Orthotope.
34 %
35 % obj = Orthotope(bbox) creates an orthotope (hyperrectangle)
36 % from a bounding box specification. The bounding box is
37 % defined as alternating lower and upper bounds for each
38 % dimension [a1, b1, a2, b2, ...] where ai is the lower bound
39 % and bi is the upper bound for dimension i.
40 %
41 % Inputs:
42 % bbox - Bounding box vector [a1, b1, a2, b2, ...]
43 %
44 % Outputs:
45 % obj - Constructed Orthotope object
46
47 core.except.assert(nargin >= 1, 'InvalidInput', ...
48 'Bounding box must be specified.');
49
50 core.except.assert(isvector(bbox), 'InvalidInput', ...
51 'Bounding box must be a vector.');
52
53 core.except.assert(mod(length(bbox), 2) == 0, 'InvalidInput', ...
54 'Bounding box must have even number of elements.');
55
56 core.except.assert(length(bbox) >= 2, 'InvalidInput', ...
57 'Bounding box must have at least 2 elements.');
58
59 bbox = bbox(:)';
60 nDims = length(bbox) / 2;
61
62
63 lower = bbox(1:2:end);
64 upper = bbox(2:2:end);
65
66 core.except.assert(all(lower <= upper), 'InvalidInput', ...
67 'Lower bounds must not exceed upper bounds.');
68
69 nVertices = 2^nDims;
70 vertices = zeros(nDims, nVertices);
71 for i = 0:(nVertices - 1)
72 binary = dec2bin(i, nDims);
73 for j = 1:nDims
74 if binary(j) == '0'
75 vertices(j, i+1) = lower(j);
76 else
77 vertices(j, i+1) = upper(j);
78 end
79 end
80 end
81
82 nFaces = 2 * nDims;
83 faces = cell(1, nFaces);
84 binaries = false(nVertices, nDims);
85 for i = 0:(nVertices - 1)
86 binStr = dec2bin(i, nDims);
87 for j = 1:nDims
88 binaries(i+1, j) = binStr(j) == '1';
89 end
90 end
91 nVerticesPerFace = 2^(nDims - 1);
92 for j = 1:nDims
93 f1 = zeros(1, nVerticesPerFace);
94 f2 = zeros(1, nVerticesPerFace);
95 k1 = 1;
96 k2 = 1;
97 for i = 1:2^nDims
98 if binaries(i, j) == false
99 f1(k1) = i;
100 k1 = k1 + 1;
101 else
102 f2(k2) = i;
103 k2 = k2 + 1;
104 end
105 end
106 faces{2*j-1} = f1;
107 faces{2*j} = f2;
108 end
109
110 obj@core.geometry.Polytope(vertices, faces);
111 obj.lower = lower;
112 obj.upper = upper;
113 end

156 function m = magnitude(obj)
157 % MAGNITUDE Returns the volume of the orthotope.
158 %
159 % m = magnitude(obj) computes the n-dimensional volume of the
160 % orthotope as the product of the extent in each dimension.
161 % For 1D this is length, for 2D this is area, for 3D this is
162 % volume, and for higher dimensions this is hypervolume.
163 %
164 % Inputs:
165 % obj - The Orthotope object
166 %
167 % Outputs:
168 % m - Volume of the orthotope

173 function TF = isInside(obj, X)
174 % ISINSIDE Tests if points are inside the orthotope.
175 %
176 % TF = isInside(obj, X) determines whether points lie strictly
177 % within the interior of the orthotope, excluding boundary
178 % faces. Points are inside if they satisfy ai < xi < bi for
179 % all dimensions i.
180 %
181 % Inputs:
182 % obj - The Orthotope object
183 % X - A matrix of size d×m where each column represents a point
184 %
185 % Outputs:
186 % TF - A logical vector indicating which points are inside
187
188 d = obj.nDims;
189 n = size(X);
190 core.except.assert(n(1) == d, 'DimensionMismatch', ...
191 'Point dimension must match geometry dimension.');
192
193 a = obj.lower;
194 b = obj.upper;
195 X = reshape(X, n(1), []);
196 TF = true(1, size(X, 2));
197 for i = 1:d
198 TF = TF & (X(i, :) > a(i)) & (X(i, :) < b(i));
199 end
200
201 if length(n) > 2, TF = reshape(TF, n(2:end)); end
202 end

204 function TF = isOnBoundary(obj, X)
205 % ISONBOUNDARY Tests if points are on the orthotope boundary.
206 %
207 % TF = isOnBoundary(obj, X) determines whether points lie
208 % exactly on the boundary faces of the orthotope. Points are
209 % on the boundary if they are within the orthotope bounds
210 % but not strictly inside.
211 %
212 % Inputs:
213 % obj - The Orthotope object
214 % X - A matrix of size d×m where each column represents a point
215 %
216 % Outputs:
217 % TF - A logical vector indicating which points are on boundary
218
219 d = obj.nDims;
220 n = size(X);
221 core.except.assert(n(1) == d, 'DimensionMismatch', ...
222 'Point dimension must match geometry dimension.');
223
224 a = obj.lower;
225 b = obj.upper;
226 X = reshape(X, n(1), []);
227 TF = true(1, size(X, 2));
228 for i = 1:d
229 TF = TF & (X(i, :) >= a(i)) & (X(i, :) <= b(i));
230 end
231
232 if length(n) > 2, TF = reshape(TF, n(2:end)); end
233
234 TF = TF & ~obj.isInside(X);
235 end
