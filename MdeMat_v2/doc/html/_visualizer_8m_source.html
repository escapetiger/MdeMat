2 % VISUALIZER Base class for visualizers.
3 %
4 % The Visualizer class provides comprehensive plotting capabilities
5 % for advection equation solutions, including support for exact
6 % solution comparisons and temporal interpolation. It manages plot
7 % timelines, rendering strategies, and coordinate visualization
8 % workflows.
9 %
10 % Key features include:
11 % - Multi-dimensional visualization support (1D, 2D, 3D)
12 % - Exact solution overlay capabilities
13 % - Temporal interpolation for smooth animations
14 % - Flexible component-based plotting configuration
15 % - Automatic subplot arrangement and styling
16 %
17 % Examples:
18 % % Basic 1D visualization setup
19 % config = struct('nDims', 1, 'density', [50], 'final', 1.0);
20 % visualizer = Visualizer(config);
21 %
22 % % With exact solution comparison
23 % config.exacts = struct('u', @(x,t) sin(x-t));
24 % config.components = struct('u', [1]);
25 % visualizer = Visualizer(config);
26 % visualizer.plot(space, dofs, tDisc);
27 %
28 % Notes:
29 % The class uses a handle-based design to manage visualization state
30 % efficiently across multiple time steps and solution updates.
31 %
32 % See Also:
33 % physics.advection.State, physics.advection.Simulator,
34 % profilers.visual.GeometryPlotter

117 function obj = reset(obj)
118 % RESET Reset visualizer to initial state.
119 %
120 % obj = reset(obj) reinitializes the visualizer by clearing
121 % datasets, resetting the timeline, and reconstructing the
122 % plotting strategy based on current component configuration.
123 %
124 % Inputs:
125 % obj - The Visualizer object
126 %
127 % Outputs:
128 % obj - The Visualizer object
129
130 strategy = profilers.visual.Strategy(obj.nDims, 'map');
131 dofNames = fieldnames(obj.components);
132 offset = 0;
133 for iField = 1:length(obj.nComponents)
134 nFieldComponents = obj.nComponents(iField);
135 dofName = dofNames{iField};
136 for iComponent = 1:nFieldComponents
137 componentName = sprintf('%s%d', dofName, ...
138 obj.components.(dofName)(iComponent));
139 axisIdx = (offset + iComponent - 1) * obj.nAxesPerComponent + 1;
140 strategy.bind(componentName, axisIdx);
141 end
142 offset = offset + nFieldComponents;
143 end
144 obj.plotter = profilers.visual.GeometryPlotter( ...
145 core.geometry.Orthotope.unit(obj.nDims), ...
146 obj.nRows, obj.nCols);
147 obj.plotter.setNodes(obj.density);
148 obj.plotter.setStrategy(strategy);
149 for iAxis = 1:obj.nAxes
150 obj.plotter.addAxisSpec(1, 1);
151 end
152 obj.plotter.draft();
153 obj.timeline.reset();
154 obj.isPrepared = false;
155 obj.dataset = struct();
156 end

194 function obj = plot(obj, space, dofs, tDisc)
195 % PLOT Generate visualization for current solution state.
196 %
197 % obj = plot(obj, space, dofs, tDisc) creates plots for the
198 % current solution state, including exact solution
199 % comparisons if configured. Handles temporal interpolation
200 % and multi-component visualization automatically.
201 %
202 % Inputs:
203 % obj - The Visualizer object
204 % space - Spatial discretization object
205 % dofs - Degrees of freedom structure containing solution data
206 % tDisc - Temporal discretization object
207 %
208 % Outputs:
209 % obj - The Visualizer object
210 %
211 % Notes:
212 % The method automatically handles coordinate preparation on
213 % first call and manages timeline advancement for animations.
214
215 if ~obj.isPrepared
216 x = space.mesh.collocate(obj.plotter.coords);
217 dofNames = fieldnames(obj.dataset);
218 for i = 1:length(dofNames)
219 dofName = dofNames{i};
220 obj.dataset.(dofName).setCoordinates(x);
221 end
222 obj.isPrepared = true;
223 end
224
225 xRef = obj.plotter.nodes;
226 t0 = tDisc.timeline.now;
227 t1 = tDisc.timeline.next;
228 t = obj.timeline.now;
229
230 dofNames = fieldnames(dofs);
231
232 if t1 > t0
233 k = min(tDisc.nSteps, tDisc.timeline.count);
234 V0 = cell(1, k);
235 for i = 1:k
236 W0 = reshape(tDisc.U0{i}, space.nGlobalDofs, []);
237 offset = 0;
238 for j = 1:length(dofNames)
239 dofName = dofNames{j};
240 nComponentsPerDof = size(dofs.(dofName), 2);
241 dof = W0(:, offset + (1:nComponentsPerDof));
242 V0{i}.(dofName) = space.evaluate(xRef, dof);
243 offset = offset + nComponentsPerDof;
244 end
245 end
246 else
247 V0 = [];
248 end
249
250 while t1 >= t - 1e-14
251 for i = 1:length(dofNames)
252 dofName = dofNames{i};
253 component = obj.components.(dofName);
254 if ~isempty(component)
255 dsNames = fieldnames(obj.dataset);
256 for j = 1:length(dsNames)
257 dsName = dsNames{j};
258 if strcmp(dsName, 'REF')
259 f = obj.exacts.(dofName);
260 U = space.evaluate(f, xRef, t);
261 U = U(:, component);
262 else
263 U = space.evaluate(xRef, dofs.(dofName)(:, component));
264 if ~isempty(V0)
265 U0 = arrayfun(@(i) V0{i}.(dofName)(:, component), 1:k, 'Un', 0);
266 U0 = cellfun(@(x) reshape(x, [1, size(x)]), U0, 'Un', 0);
267 n = size(U);
268 U = reshape(U, [1, n]);
269 U = cat(1, U0{:}, U);
270 ti = [t1 - cumsum(tDisc.timeline.h(k:-1:1)), t1].';
271 U = interp1(ti, U, t);
272 U = reshape(squeeze(U), n);
273 end
274 end
275 obj.dataset.(dsName).data.(dofName) = U;
276 end
277 end
278 end
279
280 %< Create title and render
281 if isempty(obj.experimentId) && isempty(obj.schemeId)
282 title = sprintf('t = %0.2f', t);
283 elseif ~isempty(obj.experimentId) && isempty(obj.schemeId)
284 title = sprintf('%s at t = %0.2f', obj.experimentId, t);
285 elseif ~isempty(obj.schemeId) && isempty(obj.experimentId)
286 title = sprintf('%s at t = %0.2f', obj.schemeId, t);
287 else
288 title = sprintf('%s with %s at t = %0.2f', obj.experimentId, obj.schemeId, t);
289 end
290
291 %< Render
292 obj.plotter.render(title, obj.dataset, obj.style, []);
293
294 %< Advance output timeline
295 obj.timeline.advance();
296 t = obj.timeline.now;
297 end
298 end
