1classdef ScaledAffineSpace < approx.space.AffineSpace
2 % SCALEDAFFINESPACE Scaled affine transformation of function space.
3 %
4 % ScaledAffineSpace extends the affine space with individual scaling
5 % parameters for both the coefficients and the directional component.
6 % Functions in this space have the form:
7 %
8 % \f[
9 % f(x) = \varepsilon_u \cdot u(x) + \varepsilon_g \cdot g(x)
10 % \f]
11 %
12 % where \f$s_u\f$ are scaling factors applied to the finite element
13 % coefficients, \f$u(x)\f$ is the finite element function, \f$s_g\f$
14 % is the scaling factor for the directional component, and \f$g\f$
15 % is the directional shift.
16 %
17 % Examples:
18 % % Create scaled affine space with coefficient and direction scaling
19 % coeffScales = [1.5; 2.0; 0.8]; % scale each FE coefficient
20 % directionScale = 0.5; % scale directional component
21 % scales = [coeffScales; directionScale];
22 % space = ScaledAffineSpace(finiteElement, scales);
23 %
24 % % Evaluate with automatic scaling applied
25 % points = [0, 0.5, 1.0; 0, 0, 0];
26 % coeffs = [1; 1; 1]; % unit coefficients
27 % direction = [1, 1, 1]; % unit direction
28 % values = space.evaluate(points, coeffs, direction);
29 % % Result: values = 1.5*element1 + 2.0*element2 + 0.8*element3 + 0.5*[1,1,1]
30 %
31 % See also:
32 % approx.space.AffineSpace, approx.space.FunctionSpace
