4 %
5 % AffineDiscretization provides a framework for constructing affine
6 % function spaces on spherical and orthotopal domains. It manages
7 % element construction with various basis functions including
8 % Legendre, Fourier, and spherical harmonics, along with appropriate
9 % integration schemes.
10 %
11 % The class supports C0 sphere elements with different dimensional
12 % reductions and scaling parameters. It encapsulates the complex
13 % process of coordinating geometry, basis functions, projectors,
14 % and integrators for spherical approximation problems.
15 %
16 % Examples:
17 % % Create 1D affine discretization with Legendre basis
18 % disc = AffineDiscretization();

54 function obj = setC0SphereElement(obj, varargin)
55 % SETC0SPHEREELEMENT Set C0 sphere element with specified
56 % configuration.
57 %
58 % obj = setC0SphereElement(obj, varargin) creates a C0 sphere
59 % element with the specified parameters and assigns it to the
60 % discretization. The method configures geometry, basis
61 % functions, integrators, and projectors based on dimensional
62 % and reduction parameters.
63 %
64 % Inputs:
65 % obj - The AffineDiscretization object
66 % varargin - Name-value parameter pairs:
67 %< 'nDims' - Number of spatial dimensions (1, 2, or 3)
68 %< 'reduction' - Reduction parameter (1 or 2, default: 1)
69 %< 'order' - Polynomial order (positive integer, default: 1)
70 %< 'radius' - Sphere radius (positive scalar, default: 1)
71 %< 'nPoints' - Number of integration points (positive integer)
72 %
73 % Outputs:
74 % obj - The AffineDiscretization object
75 %
76 % Notes:
77 % Reduction parameter affects basis choice:
78 % - nDims=1, reduction=1: Legendre basis on hypersphere
79 % - nDims=1, reduction=2: Legendre basis on orthotope
80 % - nDims=2, reduction=1: Fourier basis on circle
81 % - nDims=3: Spherical harmonic basis on sphere
82
83 p = inputParser;
84 addParameter(p, 'nDims', [], @(x) x >= 1 && x <= 3);
85 addParameter(p, 'reduction', 1, @(x) x >= 1 && x <= 2);
86 addParameter(p, 'order', 1, @(x) x >= 1);
87 addParameter(p, 'radius', 1, @(x) x > 0);
88 addParameter(p, 'nPoints', [], @(x) isnumeric(x));
89 parse(p, varargin{:});
90
91 d = p.Results.nDims;
92 n = p.Results.nPoints;
93 k = p.Results.order;
94 t = p.Results.reduction;
95 r = p.Results.radius;
96
97 if d == 1 && t == 1
98 G = core.geometry.Hypersphere(0);
99 F = approx.basis.OrthogonalBasisFunction( ...
100 k, 'legendre', 'canonical');
101 F.autoLoad();
102 E = core.function.ConstantFunction(1, 1/2);
103 I = approx.integrate.HypersphereIntegrator( ...
104 1, 'gauss_trapezoidal');
105 I.setPoints(n);
106 I.addWeightFunction(@(v) 1./E.evaluate(v));
107 elseif d == 1 && t == 2
108 G = core.geometry.Orthotope([-1, 1]);
109 F = approx.basis.OrthogonalBasisFunction( ...
110 k, 'legendre', 'canonical');
111 F.autoLoad();
112 E = core.function.ConstantFunction(1, 1/(2 * r));
113 I = approx.integrate.OrthotopeIntegrator( ...
114 1, 'gauss_legendre');
115 I.setPoints(n, -r, r);
116 I.addWeightFunction(@(v) 1./E.evaluate(v));
117 elseif d == 2 && t == 1
118 G = core.geometry.Hypersphere(1);
119 F = approx.basis.OrthogonalBasisFunction( ...
120 k, 'fourier', 'canonical');
121 F.autoLoad();
122 E = core.function.ConstantFunction(1, 1/(2 * pi * r));
123 I = approx.integrate.HypersphereIntegrator( ...
124 2, 'gauss_trapezoidal');
125 I.setPoints(n, zeros(1, 2), r, 3);
126 I.addWeightFunction(@(v) 1./E.evaluate(v));
127 else
128 G = core.geometry.Hypersphere(2);
129 F = approx.basis.OrthogonalBasisFunction( ...
130 k, 'spherical_harmonic', 'canonical');
131 F.autoLoad();
132 E = core.function.ConstantFunction(2, 1/(4 * pi * r^2));
133 I = approx.integrate.HypersphereIntegrator( ...
134 3, 'gauss_trapezoidal');
135 I.setPoints(n, zeros(1, 3), r, 3);
136 I.addWeightFunction(@(v) 1./E.evaluate(v));
137 end
138 B = core.function.ProductFunction(F, E);
139 P = approx.project.ModalProjector(B);
140 P.setMass(I.nodes, I.weights);
141 obj.element = approx.element.C0Element(G, P, I);
142 end
