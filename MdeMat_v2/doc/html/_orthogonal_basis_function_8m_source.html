1classdef OrthogonalBasisFunction < core.function.CompiledFunction
2 % ORTHOGONALBASISFUNCTION Compiled orthogonal polynomial basis
3 % functions.
4 %
5 % OrthogonalBasisFunction represents a set of orthogonal polynomial
6 % basis functions (such as Legendre or Chebyshev polynomials) with
7 % automatic metadata loading from the core.symbolic compilation
8 % system.
9 %
10 % The class automatically discovers appropriate metadata files based
11 % on the specified basis type, interval type, and number of basis
12 % functions, providing efficient evaluation of both function values
13 % and derivatives through precompiled function handles.
14 %
15 % Supported orthogonal polynomial families:
16 % - Legendre polynomials: orthogonal on \f$[-1,1]\f$ with weight
17 % \f$w(x) = 1\f$
18 % - Chebyshev polynomials: orthogonal on \f$[-1,1]\f$ with weight
19 % \f$w(x) = \frac{1}{\sqrt{1-x^2}}\f$
20 % - Monic Legendre: Legendre polynomials with leading coefficient = 1
21 %
22 % Examples:
23 % % Standard Legendre basis on unit interval
24 % basis = OrthogonalBasisFunction(5, 'legendre', 'unit');
25 % basis = basis.autoLoad();
26 % y = basis.evaluate([0.25]);
27 % dy = basis.derivative([0.25], [1]);
28 %
29 % % Monic Legendre polynomials on canonical interval
30 % basis = OrthogonalBasisFunction(4, 'monic_legendre', 'canonical');
31 % basis = basis.autoLoad();
32 %
33 % % Chebyshev polynomials with manual loading
34 % basis = OrthogonalBasisFunction(6, 'chebyshev', 'unit');
35 % basis = basis.load('chebyshev_unit_6');
36 %
37 % See Also:
38 % core.function.CompiledFunction,
39 % core.function.InterpolationBasisFunction
