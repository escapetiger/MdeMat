1classdef InterpolationBasisFunction < core.function.CompiledFunction
2 % INTERPOLATIONBASISFUNCTION Compiled interpolation basis functions.
3 %
4 % InterpolationBasisFunction represents a set of interpolation basis
5 % functions (such as Lagrange or Hermite polynomials) with automatic
6 % metadata loading from the core.symbolic compilation system.
7 %
8 % The class automatically discovers appropriate metadata files based
9 % on the specified basis type, interval type, node type, and number
10 % of basis functions, providing efficient evaluation of both function
11 % values and derivatives through precompiled function handles.
12 %
13 % Supported configurations:
14 % - Basis types: Lagrange, Hermite polynomials
15 % - Intervals: unit [0,1], canonical [-1,1]
16 % - Node types: Gauss-Legendre, Gauss-Lobatto quadrature points
17 %
18 % Examples:
19 % % Lagrange basis on unit interval with Gauss-Legendre nodes
20 % basis = InterpolationBasisFunction(5, 'lagrange', 'unit', 'gauss_legendre');
21 % basis = basis.autoLoad();
22 % y = basis.evaluate([0.25]);
23 % dy = basis.derivative([0.25], [1]);
24 %
25 % % Hermite basis on canonical interval with Gauss-Lobatto nodes
26 % basis = InterpolationBasisFunction(3, 'hermite', 'canonical', 'gauss_lobatto');
27 % basis = basis.autoLoad();
28 %
29 % % Manual metadata loading
30 % basis = InterpolationBasisFunction(4, 'lagrange', 'unit', 'gauss_legendre');
31 % basis = basis.load('lagrange_unit_gauss_legendre_4');
32 %

35 % core.function.OrthogonalBasisFunction
