2 % SEMILAGRANGIANCONVECTIONOPERATOR Assembly for semi-Lagrangian
3 % convection operator.
4 %
5 % SemiLagrangianConvectionOperator assembles linear transport
6 % operators for semi-Lagrangian methods. It handles characteristic
7 % tracing and flux reconstruction across element boundaries for
8 % advection-dominated problems, supporting both periodic and
9 % Dirichlet boundary conditions.
10 %
11 % The operator uses backward characteristic tracing to compute the
12 % contribution from departure points to arrival points, enabling
13 % stable time stepping for convection problems without CFL
14 % restrictions. The assembly process coordinates volume integrals for
15 % interior contributions and boundary integrals for flux conditions.
16 %
17 % Examples:
18 % % Create semi-Lagrangian convection operator with periodic BCs
19 % space = approx.space.MeshSpace(element, mesh);
20 % slOperator = approx.element.L2SemiLagrangianElementOperator(element, clipper);
21 % convOp = SemiLagrangianConvectionOperator(space, slOperator, 'periodic');
22 %
23 % % Assemble for pure advection (periodic case)
24 % A = convOp.linear();
25 %
26 % % Assemble with Dirichlet boundary conditions
27 % convOpDir = SemiLagrangianConvectionOperator(space, slOperator, 'dirichlet');
28 % tStart = 0.0; tEnd = 0.1;
29 % bcFunc = @(x, t) sin(pi * x);
30 % [b, A] = convOpDir.linear(tStart, tEnd, bcFunc);
31 %
32 % % Use in time stepping
33 % U_new = A * U_old + b;
34 %
35 % Notes:
36 % Element operator must be a semi-Lagrangian type. Boundary condition
37 % type determines the assembly strategy: periodic uses volume
38 % integrals only, while Dirichlet includes boundary condition
39 % contributions.
40 %
41 % See also:
42 % approx.assembly.Assembly,
43 % approx.element.SemiLagrangianElementOperator,

84 function [b, A] = linear(obj, varargin)
85 % LINEAR Assemble linear transport operator components.
86 %
87 % b = linear(obj) assembles boundary condition vector for the
88 % linear transport operator with current boundary conditions.
89 % For periodic boundary conditions, returns empty vector.
90 %
91 % [b, A] = linear(obj) assembles both the boundary condition
92 % vector and the linear transport operator matrix. The matrix
93 % represents the semi-Lagrangian discretization of the advection
94 % operator using characteristic tracing.
95 %
96 % [b, A] = linear(obj, tBegin, tEnd, f) assembles linear
97 % transport components with specified time interval and
98 % boundary condition function. Time interval [tBegin, tEnd]
99 % defines the characteristic tracing period.
100 %
101 % Inputs:
102 % obj - The SemiLagrangianConvectionOperator object
103 % varargin - Variable input arguments for Dirichlet conditions:
104 %< tBegin - Beginning time for characteristic tracking (scalar)
105 %< tEnd - End time for characteristic tracking (scalar)
106 %< f - Boundary condition function handle f(x, t)
107 %
108 % Outputs:
109 % b - Boundary contribution vector (sparse column vector)
110 % A - Interior contribution matrix (sparse matrix)
111
112 n = obj.space.nGlobalDofs;
113
114 %< Interior contribution via characteristic tracing
115 if nargout >= 2
116 T = obj.dynamic.assembleVolume();
117 A = core.linalg.sparseFromTriplets(T, n, n);
118 end
119
120 %< Boundary contribution for Dirichlet conditions
121 if nargout >= 1
122 b = [];
123 if strcmpi(obj.dynamic.BC_TYPE, 'dirichlet')
124 [tBegin, tEnd, f] = varargin{:};
125 if abs(tBegin-tEnd) < 1e-8, return; end
126 T = obj.dynamic.assembleBoundaryCondition(tBegin, tEnd, f);
127 b = core.linalg.sparseFromTriplets(T, n, 1);
128 end
129 end
130 end
