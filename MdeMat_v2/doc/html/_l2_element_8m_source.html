2 % L2ELEMENT Element-level L2-conforming function.
3 %
4 % L2Element implements an element-level function that is generally
5 % discontinuous across element boundaries and supports flux
6 % integration.
7 %
8 % The class manages both volume and flux integration data, making it
9 % suitable for discontinuous finite element methods where
10 % inter-element communication occurs through numerical fluxes
11 % computed at element boundaries. Each element boundary has its own
12 % integration data for efficient flux computations.
13 %
14 % Examples:
15 % % Create L2 element with volume and flux integration
16 % geometry = core.geometry.Orthotope.unit(2);
17 % basis = approx.basis.LagrangeBasis(2);
18 % projector = approx.project.NodalProjector(basis);
19 % volumeIntegrator = approx.integrate.GaussLegendre(3);
20 % fluxIntegrators = arrayfun(@(i) approx.integrate.FaceIntegrator(i), 1:4);
21 % element = L2Element(geometry, projector, volumeIntegrator, fluxIntegrators);
22 %
23 % % Set up data for DG method assembly
24 % element.setVolumeData(1); % Volume terms with derivatives
25 % element.setFluxData(0); % Flux terms with function values
26 %
27 % % Access flux data for boundary face i
28 % faceValues = element.fluxData(i).values;
29 % faceWeights = element.fluxData(i).weights;
30 %
31 % % Use static factory method for standard orthotope elements
32 % element = L2Element.Orthotope(2, 3, 2, 1);

76 % SETFLUXDATA Set function data for flux integrals.
77 %
78 % obj = setFluxData(obj, k) computes and stores function
79 % values and derivatives up to order @a k at boundary
80 % integration points for all element faces. This method
81 % prepares the data needed for flux computations in
82 % discontinuous Galerkin methods.
83 %
84 % Inputs:
85 % obj - The L2Element object
86 % k - Maximum derivative order (non-negative integer)
87 %
88 % Outputs:
89 % obj - The L2Element object
90
91 D = obj.fluxData;
92
93 for i = 1:obj.nFluxes
94 X = D(i).integrator.nodes(1:obj.nDims, :);
95 D(i).setValues(obj.projector.basis, X);
96 if k > 0, D(i).setDerivatives(obj.projector.basis, X, k); end
97 end
