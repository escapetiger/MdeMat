1classdef Sdirk4Integrator < approx.odeint.DirkIntegrator
2 % SDIRK4INTEGRATOR Fourth-order SDIRK integrator.
3 %
4 % Sdirk4Integrator implements a fourth-order SDIRK method with five
5 % stages for solving ordinary differential equations of the form:
6 %
7 % \f[
8 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = Lu + S,
9 % \f]
10 %
11 % where \f$M\f$ is a mass operator, \f$L\f$ is a linear operator, and
12 % \f$S\f$ is a source term. This method provides high-order accuracy
13 % with good stability properties for stiff problems. The diagonal
14 % elements of the Butcher tableau are identical.
15 %
16 % The method uses \f$\gamma = 1/4\f$, which provides A-stability and
17 % good damping properties. The method requires five implicit solves
18 % per time step but achieves fourth-order accuracy, making it suitable
19 % for problems requiring high precision.
20 %
21 % Examples:
22 % % Basic usage
23 % integrator = approx.odeint.Sdirk4Integrator(1.0);
24 % U_new = integrator.step(L, S, M);
25 % integrator.update(U_new);
26 % integrator.advance();
27 %
28 % % Pure diffusion (only linear term)
29 % U_new = integrator.step(L, [], []);
30 %
31 % % Pure forcing (only source term)
32 % U_new = integrator.step([], S, []);
33 %
34 % % With mass matrix
35 % U_new = integrator.step(L, S, M);
36 %
37 % See also:
38 % approx.odeint.DirkIntegrator, approx.odeint.Sdirk2Integrator,
39 % approx.odeint.Sdirk3Integrator
40
