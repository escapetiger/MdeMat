31 % SIMPLEX Constructor for Simplex.
32 %
33 % obj = Simplex(V) creates a simplex from the given vertices,
34 % with strict validation of vertex configuration. The vertices
35 % must be affinely independent and form a valid simplex. A
36 % d-dimensional simplex requires exactly d+1 vertices.
37 %
38 % Inputs:
39 % V - A matrix of size d×(d+1) where each column is a vertex
40 %
41 % Outputs:
42 % obj - Constructed Simplex object
43
44 core.except.assert(nargin >= 1, 'InvalidInput', ...
45 'Vertices must be specified.');
46
47 core.except.assert(ismatrix(V), 'InvalidInput', ...
48 'Vertices must be a matrix.');
49
50 [d, n] = size(V);
51 core.except.assert(d+1 == n, 'InvalidInput', ...
52 'Vertices must be of shape (%d, %d).', d, d+1);
53
54 E = V(:, 2:end) - V(:, 1);
55 core.except.assert(rank(E) >= d, 'InvalidInput', ...
56 'Vertices must be affinely independent.');

58 F = arrayfun(@(i) setdiff(1:n, i), 1:n, 'Un', 0);

61 end

77 function N = get.outerNormals(obj)
78 % GET.OUTERNORMALS Returns the outer normals of boundaries.
79 %
80 % N = get.outerNormals(obj) computes the outward-pointing
81 % normal vectors for each boundary face of the simplex. Each
82 % normal is a unit vector perpendicular to the corresponding
83 % face and pointing away from the simplex interior.
84 %
85 % Outputs:
86 % N - A matrix where each column is an outer normal vector
87
88 V = obj.vertices;
89 d = obj.nDims;
90 N = zeros(d, d+1);
91 for i = 1:(d + 1)
92 idx = setdiff(1:(d + 1), i);
93 U = V(:, idx);
94 U0 = U(:, 1);
95 E = U(:, 2:end) - U0;
96 ni = null(E.');
97 if dot(U0-V(:, i), ni) < 0, ni = -ni; end
98 N(:, i) = ni / norm(ni);
99 end
100 end

102 function m = magnitude(obj)
103 % MAGNITUDE Returns the volume of the simplex.
104 %
105 % m = magnitude(obj) computes the n-dimensional volume of the
106 % simplex using the determinant formula. The volume is
107 % calculated as |det(E)| / d! where E is the matrix of edge
108 % vectors from one vertex to all others.
109 %
110 % Inputs:
111 % obj - The Simplex object
112 %
113 % Outputs:
114 % m - Volume of the simplex
115
116 d = obj.nDims;
117 V = obj.vertices;
118 E = V(:, 2:end) - V(:, 1);
119 m = abs(det(E)) / factorial(d);
120 end

172 function B = cartesianToBarycentric(obj, C)
173 % CARTESIANTOBARYCENTRIC Converts Cartesian to barycentric coordinates.
174 %
175 % B = cartesianToBarycentric(obj, C) transforms points from
176 % Cartesian coordinates to barycentric coordinates with
177 % respect to the simplex vertices. Barycentric coordinates
178 % sum to 1 and are non-negative for interior points. Each
179 % coordinate represents the weight of the corresponding vertex.
180 %
181 % Inputs:
182 % obj - The Simplex object
183 % C - A matrix of size d×m where each column represents a point
184 %
185 % Outputs:
186 % B - A matrix of size (d+1)×m containing barycentric coordinates
187
188 d = obj.nDims;
189 n = size(C);
190 core.except.assert(n(1) == d, 'DimensionMismatch', ...
191 'Point dimension must match the simplex dimension.');
192
193 C = reshape(C, d, []);
194 V = obj.vertices;
195 A = [V; ones(1, d+1)];
196 B = [C; ones(1, size(C, 2))];
197 B = A \ B;
198 n(1) = d + 1;
199 B = reshape(B, n);
200 end

202 function C = barycentricToCartesian(obj, B)
203 % BARYCENTRICTOCARTESIAN Converts barycentric to Cartesian coordinates.
204 %
205 % C = barycentricToCartesian(obj, B) transforms points from
206 % barycentric coordinates back to Cartesian coordinates. The
207 % barycentric coordinates must sum to 1 and be non-negative
208 % for valid points within or on the simplex.
209 %
210 % Inputs:
211 % obj - The Simplex object
212 % B - A matrix of size (d+1)×m representing barycentric coordinates
213 %
214 % Outputs:
215 % C - A matrix of size d×m where each column represents a point
216
217 d = obj.nDims;
218 n = size(B);
219 core.except.assert(n(1) == d+1, 'DimensionMismatch', ...
220 'Barycentric coordinates must be %d', d+1);
221
222 tol = sqrt(eps);
223 B = reshape(B, d+1, []);
224 core.except.assert(all(abs(sum(B, 1)-1) <= tol), ...
225 'InvalidInput', 'Barycentric coordinates must sum to 1.');
226 core.except.assert(all(B >= -tol), 'InvalidInput', ...
227 'Barycentric coordinates should be non-negative.');
228 V = obj.vertices;
229 C = V * B;
230 n(1) = d;
231 C = reshape(C, n);
232 end
