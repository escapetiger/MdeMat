1classdef C0ElementOperator < approx.element.ElementOperator
2 % C0ELEMENTOPERATOR Operator for continuous finite elements.
3 %
4 % C0ElementOperator provides differential operators such as gradient
5 % for C0 elements. It manages bilinear form data for volume integrals
6 % and constructs discrete operator matrices used in conforming finite
7 % element methods.
8 %
9 % The class computes discrete representations of differential
10 % operators by evaluating weighted inner products between basis
11 % function derivatives. These operators are fundamental for
12 % assembling system matrices in finite element discretizations of
13 % partial differential equations.
14 %
15 % Examples:
16 % % Create C0 element and its operator
17 % element = approx.element.C0Element(geometry, projector, integrator);
18 % element.setVolumeData(1); % Include first derivatives
19 % operator = C0ElementOperator(element);
20 %
21 % % Set up operator data and access gradient
22 % operator.setVolumeData();
23 % gradOp = operator.gradient;
24 %
25 % % Use gradient components for stiffness matrix assembly
26 % gradX = gradOp.volumeData(1).matrix; % x-derivative component
27 % gradY = gradOp.volumeData(2).matrix; % y-derivative component
28 %
29 % % Access complete gradient structure
30 % nDims = length(gradOp.volumeData);
31 % stiffnessContrib = sum(arrayfun(@(i) gradOp.volumeData(i).matrix, 1:nDims), 3);
32 %
