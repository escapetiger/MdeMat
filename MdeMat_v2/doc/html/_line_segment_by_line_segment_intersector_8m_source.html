2 % LINESEGMENTBYLINESEGMENTINTERSECTOR Computes intersections between line segments.
3 %
4 % LineSegmentByLineSegmentIntersector computes intersection points
5 % between pairs of line segments in n-dimensional space (n >= 2). The
6 % line segments are defined by their endpoints, and intersections are
7 % computed using a least-squares approach.
8 %
9 % The line segments are parameterized as:
10 % L1(s) = X1 + s*(X2 - X1)
11 % L2(t) = X3 + t*(X4 - X3)
12 %
13 % An intersection is considered valid only when:
14 % 1. Two non-collinear line segments intersect at exactly one point
15 % 2. Two collinear line segments intersect at exactly one point
16 % (e.g., at an endpoint)
17 %
18 % If line segments would have more than one intersection point
19 % (e.g., they overlap), an error is thrown.
20 %
21 % Examples:
22 % % Create a 2D line-line intersector
23 % intersector = LineSegmentByLineSegmentIntersector(2);
24 %
25 % % Define two line segments
26 % X1 = [0; 0]; % First endpoint of first line
27 % X2 = [1; 1]; % Second endpoint of first line
28 % X3 = [0; 1]; % First endpoint of second line
29 % X4 = [1; 0]; % Second endpoint of second line
30 %
31 % % Compute intersection
32 % [intersectionPoint, isValid] = intersector.intersect(X1, X2, X3, X4);
33 %
34 % Notes:
35 % For collinear overlapping segments, the method throws an error
36 % since there would be infinitely many intersection points.
37 %
38 % See also:
39 % core.geometry.Intersector,
40 % core.geometry.LineSegmentByPlaneIntersector

116 function [X, TF] = intersectND(obj, X1, X2, X3, X4, tol)
117 % INTERSECTND Internal method for n-dimensional intersection computation.
118 %
119 % [X, TF] = intersectND(obj, X1, X2, X3, X4, tol) performs
120 % the actual intersection computation for n-dimensional line
121 % segments using vectorized operations.
122 %
123 % Inputs:
124 % obj - The LineSegmentByLineSegmentIntersector object
125 % X1 - d×m matrix of first endpoints of first line segments
126 % X2 - d×m matrix of second endpoints of first line segments
127 % X3 - d×m matrix of first endpoints of second line segments
128 % X4 - d×m matrix of second endpoints of second line segments
129 % tol - Numerical tolerance
130 %
131 % Outputs:
132 % X - d×k matrix of intersection points
133 % TF - 1×m logical vector indicating valid intersections
134
135 [d, n] = size(X1);
136 X = nan(d, n);
137 TF = false(1, n);
138 P = nan(2, n);
139 for i = 1:n
140 p1 = X1(:, i);
141 p2 = X2(:, i);
142 p3 = X3(:, i);
143 p4 = X4(:, i);
144 v1 = p2 - p1;
145 v2 = p4 - p3;
146 len1 = norm(v1);
147 len2 = norm(v2);
148
149 %< First, check if line segments coincide exactly (same endpoints)
150 if (norm(p1-p3) < tol && norm(p2-p4) < tol) || (norm(p1-p4) < tol && norm(p2-p3) < tol)
151 %< Line segments have the same endpoints (in same or reversed order)
152 if len1 < tol && len2 < tol
153 %< Both are points - valid single intersection
154 TF(i) = true;
155 X(:, i) = p1;
156 P(:, i) = [0; 0];
157 else
158 %< At least one is not a point - infinite intersections
159 core.except.assert(0, 'CoincidentSegments', ...
160 ['Line segments coincide exactly, ' ...
161 'resulting in more than one intersection point.']);
162 end
163 continue;
164 end
165
166 %< Handle zero-length segments (points)
167 if len1 < tol && len2 < tol
168 %< Both segments are points - check if they're the same point
169 if norm(p1-p3) < tol
170 TF(i) = true;
171 X(:, i) = p1;
172 P(:, i) = [0; 0];
173 end
174 continue;
175 elseif len1 < tol
176 %< First segment is a point - check if it lies on second segment
177 [onLine, t] = obj.pointOnLineSegment(p1, p3, p4, tol);
178 if onLine
179 TF(i) = true;
180 X(:, i) = p1;
181 P(:, i) = [0; t];
182 end
183 continue;
184 elseif len2 < tol
185 %< Second segment is a point - check if it lies on first segment
186 [onLine, t] = obj.pointOnLineSegment(p3, p1, p2, tol);
187 if onLine
188 TF(i) = true;
189 X(:, i) = p3;
190 P(:, i) = [t; 0];
191 end
192 continue;
193 end
194
195 %< Handle regular line segments
196 %< Check if lines are parallel or collinear
197
198 %< Normalize direction vectors
199 u1 = v1 / len1;
200 u2 = v2 / len2;
201
202 %< Check if directions are parallel (dot product near ±1)
203 a = abs(dot(u1, u2));
204
205 if a > 1 - tol
206 %< Vectors are parallel - check if lines are collinear
207 %< Compute distance from p3 to the line through p1 and p2
208
209 %< For collinearity check, we need to see if p3 is on the line defined by p1 and p2
210 %< We do this by checking if the vector from p1 to p3 is parallel to v1
211
212 q = p3 - p1;
213
214 %< Project connect vector onto v1 and check orthogonal component
215 p = dot(q, u1) * u1;
216 c = q - p;
217
218 if norm(c) < tol
219 %< Lines are collinear - check for overlap
220
221 %< Project endpoints of second segment onto first segment
222 t3 = dot(p3-p1, u1) / len1;
223 t4 = dot(p4-p1, u1) / len1;
224
225 %< Check for single-point intersection cases
226
227 %< Case 1: p3 is at p1
228 if abs(t3) < tol
229 TF(i) = true;
230 X(:, i) = p1;
231 P(:, i) = [0; 0];
232 continue;
233 end
234
235 %< Case 2: p3 is at p2
236 if abs(t3-1) < tol
237 TF(i) = true;
238 X(:, i) = p2;
239 P(:, i) = [1; 0];
240 continue;
241 end
242
243 %< Case 3: p4 is at p1
244 if abs(t4) < tol
245 TF(i) = true;
246 X(:, i) = p1;
247 P(:, i) = [0; 1];
248 continue;
249 end
250
251 %< Case 4: p4 is at p2
252 if abs(t4-1) < tol
253 TF(i) = true;
254 X(:, i) = p2;
255 P(:, i) = [1; 1];
256 continue;
257 end
258
259 %< Check for overlap scenarios
260
261 %< Both endpoints of second segment inside first segment
262 bothInside = (t3 >= 0 && t3 <= 1 && t4 >= 0 && t4 <= 1);
263
264 %< Second segment contains first segment
265 contains = (t3 < 0 && t4 > 1) || (t3 > 1 && t4 < 0);
266
267 %< Segment spans across first segment
268 spans = (t3 < 0 && t4 > 0 && t4 <= 1) || (t3 >= 0 && t3 <= 1 && t4 > 1);
269
270 %< Error case: Multiple intersection points
271 core.except.assert( ...
272 ~bothInside && ~contains && ~spans, ...
273 'CollinearOverlap', ...
274 ['Line segments are collinear and overlapping, ' ...
275 'resulting in more than one intersection point.'])
276 end
277
278 %< Parallel but not collinear, or collinear with no intersection
279 continue;
280 end
281
282 %< Non-parallel lines - find closest points
283 %< Set up the coefficient matrix for the least squares problem
284 A = [v1, -v2];
285 b = p3 - p1;
286
287 %< Solve for parameters [s; t]
288 %< We use the Moore-Penrose pseudoinverse for the general case
289 theta = pinv(A) * b;
290 s = theta(1);
291 t = theta(2);
292
293 %< Store parameters
294 P(:, i) = [s; t];
295
296 %< Compute residual (distance between closest points)
297 Y1 = p1 + s * v1;
298 Y2 = p3 + t * v2;
299
300 %< Check if closest points are within tolerance and on segments
301 if norm(Y1-Y2) <= tol && s >= 0 && s <= 1 && t >= 0 && t <= 1
302 TF(i) = true;
303
304 %< Use midpoint between closest points as intersection
305 X(:, i) = (Y1 + Y2) / 2;
306 end
307 end
308 X = X(:, TF);
309 end

311 function [onLine, t] = pointOnLineSegment(obj, point, lineStart, lineEnd, tol)
312 % POINTONLINESEGMENT Determines if a point lies on a line segment.
313 %
314 % [onLine, t] = pointOnLineSegment(obj, point, lineStart, lineEnd, tol)
315 % checks if a point lies on the line segment from lineStart to lineEnd.
316 %
317 % Inputs:
318 % obj - The LineSegmentByLineSegmentIntersector object
319 % point - Point to check for inclusion on line segment
320 % lineStart - Start point of line segment
321 % lineEnd - End point of line segment
322 % tol - Tolerance for numerical comparisons
323 %
324 % Outputs:
325 % onLine - True if point lies on the line segment
326 % t - Parameter value where point = lineStart + t*(lineEnd-lineStart)
327
328 %< If tolerance not provided, use default
329 if nargin < 5 || isempty(tol)
330 tol = sqrt(eps);
331 end
332
333 %< Get line direction
334 lineDir = lineEnd - lineStart;
335 lineLength = norm(lineDir);
336
337 %< If line has zero length, check if point is at start
338 if lineLength < tol
339 onLine = norm(point-lineStart) < tol;
340 t = 0;
341 return;
342 end
343
344 %< Compute projection parameter
345 t = dot(point-lineStart, lineDir) / (lineLength^2);
346
347 %< Check if parameter is in [0,1]
348 if t < 0 || t > 1
349 onLine = false;
350 return;
351 end
352
353 %< Compute closest point on line
354 closestPoint = lineStart + t * lineDir;
355
356 %< Check if point is close enough to line
357 onLine = norm(point-closestPoint) < tol;
358 end
359 end
