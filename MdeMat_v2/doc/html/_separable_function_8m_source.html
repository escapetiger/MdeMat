352 function Y = evaluateDense(obj, X)
353 % EVALUATEDENSE Dense evaluation using full tensor product.
354 %
355 % Y = evaluateDense(obj, X) evaluates the function using the
356 % full tensor product structure. Each factor function is
357 % evaluated and results are combined using Kronecker or
358 % Khatri-Rao products depending on input format.
359 %
360 % Inputs:
361 % obj - The SeparableFunction object
362 % X - Coordinates (nDims x nPoints or cell array)
363 %
364 % Outputs:
365 % Y - Function values (nCodims x nPoints)
366
367 F = obj.factors;
368 k = length(F);
369 Z = cell(1, k);
370
371 if iscell(X)
372 for i = 1:k
373 if iscell(F)
374 Fi = F{i};
375 else
376 Fi = F(i);
377 end
378 mi = Fi.nCodims;
379 Z{i} = Fi.evaluate(X{i});
380 Z{i} = reshape(Z{i}, mi, []);
381 end
382 Y = core.linalg.kronecker(Z);
383 else
384 j = 1;
385 for i = 1:k
386 if iscell(F)
387 Fi = F{i};
388 else
389 Fi = F(i);
390 end
391 ni = Fi.nDims;
392 mi = Fi.nCodims;
393 J = j:(j + ni - 1);
394 Z{i} = Fi.evaluate(X(J, :));

396 j = j + ni;
397 end
