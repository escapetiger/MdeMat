2 % GRIDASSEMBLER Grid-based assembler for structured meshes.
3 %
4 % GridAssembler provides assembly methods for finite element
5 % operators on structured grid meshes. This class extends the base
6 % Assembler to handle grid-specific operations and boundary condition
7 % treatments.
8 %
9 % The assembler supports different boundary condition types and
10 % provides specialized methods for handling Dirichlet boundary
11 % conditions on structured grids.
12 %
13 % Examples:
14 % % Create grid assembler with periodic boundaries
15 % assembler = GridAssembler(fe, gridMesh, op, 0);
16 %
17 % % Create grid assembler with Dirichlet boundaries
18 % assembler = GridAssembler(fe, gridMesh, op, 1);

54 function T = assembleLeftDirichletBoundaryConditionTerms(obj, d, c, f, varargin)
55 % ASSEMBLELEFTDIRICHLETBOUNDARYCONDITIONTERMS Assemble left
56 % boundary Dirichlet condition terms.
57 %
58 % T = assembleLeftDirichletBoundaryConditionTerms(obj, d, c,
59 % f) assembles the contribution from Dirichlet boundary
60 % conditions on the left boundary of the specified dimension.
61 %
62 % Inputs:
63 % obj - The GridAssembler object
64 % d - Spatial dimension index (1, 2, or 3)
65 % c - Coefficient vector
66 % f - Boundary condition function handle
67 % varargin - Additional arguments for boundary function
68 %
69 % Outputs:
70 % T - Triplet matrix for boundary condition terms
71
72 np = obj.nLocalDofs;
73 P = obj.fe.projector;
74 G = obj.op.gradient;
75
76 m = obj.mesh.allElementMultiIndices;
77 e = m(:, d) == 1;
78 ne = numel(find(e));
79 m = m(e, :);
80 l = obj.mesh.indexer.multiToLinear(m, 1);
81 X = obj.mesh.collocate(obj.fe.fluxData(2*d-1).nodes, m);
82
83 T = zeros(np^2*ne, 3);
84
85 F = G.fluxData(1, 2*d-1).vector;
86 U = f(X, varargin{:});
87 U = reshape(U, size(F, 2), []);
88 Q = P.project(-F*U);
89 nq = numel(Q);
90 k = 1:nq;
91 T(k, 1) = reshape(bsxfun(@plus, (l(:).' - 1)*np, (1:np).'), [], 1);
92 T(k, 2) = ones(size(T(k, 1)));
93 T(k, 3) = kron(c(l), ones(np, 1)) .* Q(:);
94 end
