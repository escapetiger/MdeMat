1classdef Analyzer < handle
2 % ANALYZER Comprehensive error analysis and metrics computation.
3 %
4 % Analyzer provides sophisticated error analysis capabilities for
5 % numerical methods, including absolute error computation against
6 % exact solutions and relative error analysis using Richardson
7 % extrapolation. This class is essential for validating numerical
8 % schemes and quantifying their accuracy.
9 %
10 % The analyzer supports multiple error reduction types (L1, L2,
11 % Lx), handles multi-component solutions, and can work with both
12 % exact solutions and grid-based Richardson extrapolation for
13 % problems where analytical solutions are unavailable.
14 %
15 % Examples:
16 % % Create analyzer for convergence study
17 % analyzer = Analyzer('nLevels', 4, ...
18 % 'density', [10, 10], ...
19 % 'reductions', {'L1', 'L2'}, ...
20 % 'components', struct('u', 1, 'v', 2), ...
21 % 'exacts', struct('u', @(x) sin(x(1,:))));
22 %
23 % % Compute absolute error against exact solution
24 % analyzer.setLevel(mesh);
25 % analyzer.absolute(space, dofs);
26 %
27 % % Compute relative error using Richardson extrapolation
28 % analyzer.relative(fineSpace, fineDofs, coarseSpace, coarseDofs);
29 %
30 % % Generate analysis results
31 % results = analyzer.analyze();
32 %
33 % See also:
34 % profilers.analysis.ConvergenceProfiler, profilers.metrics.MeshSpaceAbsoluteMetric

53 function obj = Analyzer(varargin)
54 % ANALYZER Constructor for Analyzer.
55 %
56 % obj = Analyzer('Parameter', Value, ...) creates an analyzer
57 % with specified configuration for error analysis. The
58 % analyzer supports various error metrics and can handle
59 % multi-component solutions with optional exact solutions.
60 %
61 % Inputs:
62 % varargin - Input arguments
63 %< nLevels - Number of refinement levels (positive integer)
64 %< density - (1×nDims) error sampling density per element (optional)
65 %< reductions - Cell array of error reduction types (optional)
66 %< components - Structure mapping field names to indices (optional)
67 %< exacts - Structure of exact solution functions (optional)
68 %
69 % Outputs:
70 % obj - Constructed Analyzer object
71
72 p = inputParser;
73 addParameter(p, 'nLevels', [], @(x) isnumeric(x) && isscalar(x));
74 addParameter(p, 'reductions', [], @(x) iscell(x));
75 addParameter(p, 'density', [], @(x) isnumeric(x));
76 addParameter(p, 'components', [], @(x) isstruct(x));
77 addParameter(p, 'exacts', [], ...
78 @(x) isempty(x) || isa(x, 'function_handle') || isstruct(x));
79 parse(p, varargin{:});
80
81 obj.profiler = profilers.analysis.ConvergenceProfiler(p.Results.nLevels);
82 obj.density = p.Results.density;
83 obj.reductions = p.Results.reductions;
84 obj.components = p.Results.components;
85 obj.exacts = p.Results.exacts;
86
87 %< Initialize error storage structure
88 fieldNames = fieldnames(p.Results.components);
89 nFields = length(fieldNames);
90 args = cell(2, nFields);
91 args(1, :) = fieldNames;
92 args(2, :) = {{cell(p.Results.nLevels, 1)}};
93 obj.errors = struct(args{:});
94 obj.count = 1;
95 end

153 function obj = absolute(obj, space, dofs, varargin)
154 % ABSOLUTE Compute absolute error against exact solutions.
155 %
156 % obj = absolute(obj, space, dofs) computes absolute errors
157 % by comparing numerical solutions with exact analytical
158 % solutions. The method evaluates both numerical and exact
159 % solutions at the same sample points and computes specified
160 % error metrics.
161 %
162 % Inputs:
163 % obj - The Analyzer object
164 % space - Function space for solution evaluation
165 % dofs - Structure containing solution coefficient dofs
166 % varargin - Additional arguments for exact solution evaluation
167 %
168 % Outputs:
169 % obj - The Analyzer object
170
171 metric = profilers.metrics.MeshSpaceAbsoluteMetric(obj.reductions, space);
172 metric.setPoints(obj.density);
173
174 X = metric.nodes;
175 fieldNames = fieldnames(dofs);
176 for iField = 1:length(fieldNames)
177 fieldName = fieldNames{iField};
178 component = obj.components.(fieldName);
179 C = dofs.(fieldName);
180 f = obj.exacts.(fieldName);
181 U = space.evaluate(X, C);
182 U = U(:, component);
183 U0 = space.evaluate(f, X, varargin{:});
184 U0 = U0(:, component);
185 obj.errors.(fieldName){obj.count} = metric.evaluate(U, U0);
186 end
187 obj.count = obj.count + 1;
188 end

190 function obj = relative(obj, fSpace, fDofs, cSpace, cDofs)
191 % RELATIVE Compute relative error using Richardson
192 % extrapolation.
193 %
194 % obj = relative(obj, fSpace, fDofs, cSpace, cDofs) computes
195 % relative errors using Richardson extrapolation between fine
196 % and coarse grid solutions. This method is useful when exact
197 % solutions are not available.
198 %
199 % Inputs:
200 % obj - The Analyzer object
201 % fSpace - Fine grid function space
202 % fDofs - Structure containing fine grid solution dofs
203 % cSpace - Coarse grid function space
204 % cDofs - Structure containing coarse grid solution dofs
205 %
206 % Outputs:
207 % obj - The Analyzer object
208
209 metric = profilers.metrics.GridSpaceRichardsonMetric( ...
210 obj.reductions, fSpace, cSpace);
211 metric.setPoints(obj.density);
212
213 X = metric.nodes{1};
214 X0 = metric.nodes{2};
215
216 fieldNames = fieldnames(fDofs);
217 for iField = 1:length(fieldNames)
218 fieldName = fieldNames{iField};
219 component = obj.components.(fieldName);
220 C = fDofs.(fieldName)(metric.iNestDofs, :);
221 C0 = cDofs.(fieldName);
222 U = fSpace.evaluate(X, C);
223 U = U(:, component);
224 U0 = cSpace.evaluate(X0, C0);
225 U0 = U0(:, component);
226 obj.errors.(fieldName){obj.count} = metric.evaluate(U, U0);
227 end
228 obj.count = obj.count + 1;
229 end

231 function results = analyze(obj)
232 % ANALYZE Generate comprehensive error analysis results.
233 %
234 % results = analyze(obj) processes all collected error data
235 % and generates structured results including convergence
236 % tables for each field component and error reduction type.
237 % The results can be used for convergence verification and
238 % method validation.
239 %
240 % Inputs:
241 % obj - The Analyzer object
242 %
243 % Outputs:
244 % results - Convergence tables
245
246 results = struct();
247 nLevels = obj.profiler.nLevels;
248 nReductions = length(obj.reductions);
249 fieldNames = fieldnames(obj.errors);
250 for iField = 1:length(fieldNames)
251 fieldName = fieldNames{iField};
252 component = obj.components.(fieldName);
253 E = cell2mat(obj.errors.(fieldName));
254 for iComponent = 1:size(E, 2)
255 Ej = reshape(E(:, iComponent), nReductions, nLevels).';
256 name = sprintf('%s%d', fieldName, component(iComponent));
257 results.(name) = obj.profiler.buildTable(Ej, obj.reductions);
258 end
259 end
260 end
261 end
