1classdef ModalProjector < approx.project.GalerkinProjector
2 % MODALPROJECTOR Galerkin projector based on modal embedding.
3 %
4 % ModalProjector implements a Galerkin projection method that uses
5 % inner products between basis functions to construct the mass
6 % operator. This approach is particularly suitable for orthogonal
7 % and well-conditioned basis functions.
8 %
9 % The Galerkin system is defined as:
10 % \f[
11 % \langle u(x), p_j(x)\rangle = \sum_{i=1}^{n} c_i \langle p_i(x), p_j(x)\rangle, \forall j = 1, ..., m
12 % \f]
13 % where \f$\langle\cdot,\cdot\rangle\f$ denotes the inner product
14 % defined by integration weights, \f$p_i\f$ are basis functions, and
15 % \f$c_i\f$ are the coefficients to be determined.
16 %
17 % The mass matrix \f$M_{ij} = \langle p_i, p_j\rangle\f$ is
18 % constructed using numerical integration with specified quadrature
19 % points and weights.
20 %
21 % Examples:
22 % % Create modal projector with Legendre basis
23 % basis = OrthogonalBasisFunction(5, 'legendre', 'unit');
24 % basis = basis.autoLoad();
25 % projector = ModalProjector(basis);
26 %
27 % % Set up mass operator with Gauss-Legendre quadrature
28 % [X, w] = gauss_legendre_quadrature(10); % 10 quadrature points
29 % projector.setMass(X, w);
30 %
31 % % Project function data
32 % U = sin(pi * X); % Function values at quadrature points
33 % B = basis.evaluate(X); % Basis values at quadrature points
34 % F = projector.embed(U, B, w);
35 % coeffs = projector.project(F);
36 %
37 % See Also:
38 % approx.project.GalerkinProjector, approx.project.NodalProjector

41 function obj = setMass(obj, X, w)
42 % SETMASS Set up the mass operator using numerical integration.
43 %
44 % obj = setMass(obj, X, w) constructs the mass matrix using
45 % integration points and weights. The (i,j) entry of the mass
46 % matrix is \f$\langle p_i, p_j\rangle\f$ where the inner
47 % product is defined by the weighted integration: \f$\langle
48 % f,g\rangle = \sum_k w_k f(x_k) g(x_k)\f$.
49 %
50 % Inputs:
51 % obj - The ModalProjector object
52 % X - Integration points (n x q)
53 % w - Integration weights (1 x q)
54 %
55 % Outputs:
56 % obj - The ModalProjector object
57
58 n = obj.basis.nDims;
59 X = reshape(X, size(X, 1), []);
60
61 core.except.assert(size(X, 1) >= n, 'InvalidInput', ...
62 'Integration points must have at least %d rows.', n);
63
64 X = X(1:n, :);
65 w = w(:);
66
67 q = length(w);
68 core.except.assert(size(X, 2) == q, 'InvalidInput', ...
69 'Number of integration points (%d) must match number of weights (%d).', ...
70 size(X, 2), q);
71
72 core.except.assert(all(w >= 0), 'InvalidInput', ...
73 'Integration weights must be non-negative.');
74
75 B = obj.basis.evaluate(X);
76 W = diag(w);
77 M = B * W * B.';
78 M = core.linalg.zeroing(M);
79
80 obj.mass = M;
81 end

83 function F = embed(obj, U, B, w)
84 % EMBED Embed data into the latent space using modal approach.
85 %
86 % F = embed(obj, U, B, w) embeds function values into the
87 % latent space using the modal approach. Computes the weighted
88 % inner product between basis functions and function values
89 % at integration points:
90 % \f$F_i = \langle U, p_i\rangle = \sum_k w_k U_k p_i(x_k)\f$.
91 %
92 % Inputs:
93 % obj - The ModalProjector object
94 % U - Function values at integration points (q x p)
95 % B - Basis function values at integration points (m x q)
96 % w - Integration weights (1 x q)
97 %
98 % Outputs:
99 % F - Latent space representation (m x p)
100
101 M = obj.mass;
102 core.except.assert(~isempty(M), 'EmptyMass', ...
103 'The mass operator has not been set. Call setMass() first.');
104
105 m = size(M, 1);
106 core.except.assert(size(B, 1) == m, 'InvalidInput', ...
107 'Basis function rows (%d) must match mass operator size (%d).', ...
108 size(B, 1), m);
109
110 q = length(w);
111 core.except.assert(size(B, 2) == q, 'InvalidInput', ...
112 'Basis function columns (%d) must match number of weights (%d).', ...
113 size(B, 2), q);
114
115 U = reshape(U, q, []);
116 B = reshape(B, [], q);
117 W = diag(w(:));
118 F = B * W * U;
119 end
120 end
