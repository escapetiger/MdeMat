1classdef OdeIntegrator < handle
2 % ODEINTEGRATOR Abstract base class for ODE integrators.
3 %
4 % OdeIntegrator provides a common interface and shared functionality
5 % for all ODE integration methods. This base class manages solution
6 % history, timeline operations, and linear system solving
7 % infrastructure. Concrete integrators must implement the abstract
8 % methods for specific integration schemes.
9 %
10 % The class focuses on solving ordinary differential equations of
11 % the form:
12 %
13 % \f[
14 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = Lu + F(u) + S(t),
15 % \f]
16 %
17 % where \f$M\f$ is a mass operator, \f$L\f$ is a linear operator,
18 % \f$F(u)\f$ is a nonlinear operator, and \f$S(t)\f$ is a source term.
19 % Any of these terms can be empty. In particular, M = [] means the
20 % mass operator reduces to an identity operator; {L, F, S} = [] means
21 % they vanish from the equation.
22 %
23 % The contribution of mass operator M is determined by the physical
24 % simulation framework:
25 % * In the Eulerian framework, M is space-only so that it can be
26 % represented by a unique static matrix.
27 % * In the Semi-Lagrangian framework, M is time-dependent. The LHS
28 % mass term is forced to be a unique static matrix as in Eulerian
29 % framework. However, the RHS mass term should be updated as time
30 % evolves. In particular, M is a matrix under periodic boundary
31 % conditions, while M is a pair formed by a matrix and a vector
32 % under Dirichlet boundary conditions.
33 %
34 % See also:
35 % approx.odeint.BeIntegrator, approx.odeint.BdfIntegrator,
36 % approx.odeint.DirkIntegrator, approx.odeint.ExrkIntegrator

52 % ODEINTEGRATOR Constructor for OdeIntegrator.
53 %
54 % obj = OdeIntegrator(nSteps, nStages, final) creates an ODE
55 % integrator with the specified parameters. This constructor
56 % is called by concrete subclasses to initialize the base
57 % class functionality.
58 %
59 % Inputs:
60 % nSteps - Number of time steps required for the method (positive integer)
61 % nStages - Number of stages required for the method (positive integer)
62 % final - Final time for integration (positive scalar)

64 % Outputs:
65 % obj - Constructed OdeIntegrator object
66
67 obj.nSteps = nSteps;
68 obj.nStages = nStages;
69 obj.timeline = approx.mesh.DynamicTimeline(final, nSteps);
70 obj.reset();
71 end

73 function obj = reset(obj)
74 % RESET Reset the integrator to initial state.
75 %
76 % obj = reset(obj) clears solution history, resets the
77 % timeline, and initializes storage arrays for left-hand side
78 % matrices and right-hand side vectors. Also creates a fresh
79 % linear solver instance. This method is called automatically
80 % during construction and can be called to restart
81 % integration.
82 %
83 % Inputs:
84 % obj - The OdeIntegrator object
85 %
86 % Outputs:
87 % obj - The OdeIntegrator object
88
89 obj.U0 = cell(obj.nStages, obj.nSteps);
90 obj.timeline.reset();
91
92 %< Initialize storage based on method characteristics
93 if obj.nStages > 1 && obj.nSteps == 1
94 %< Multi-stage, single-step methods (RK-type)
95 obj.lhs = cell(obj.nStages + 1, obj.nSteps);
96 obj.rhs = cell(obj.nStages + 1, obj.nSteps);
97 elseif obj.nStages == 1 && obj.nSteps > 1
98 %< Single-stage, multi-step methods (BDF-type)
99 obj.lhs = cell(1, obj.nSteps);
100 obj.rhs = cell(1, obj.nSteps);
101 else
102 %< Single-stage, single-step methods
103 obj.lhs = [];
104 obj.rhs = [];
105 end
106
107 %< Initialize mass term storage
108 if obj.nStages > 1 && obj.nSteps == 1
109 obj.MLhs = [];
110 obj.MRhs = cell(obj.nStages + 1, 1);
111 elseif obj.nStages == 1 && obj.nSteps > 1
112 obj.MLhs = [];
113 obj.MRhs = cell(1, obj.nSteps);
114 else
115 obj.MLhs = [];
116 obj.MRhs = [];
117 end
118
119 %< Create linear solver instance
120 obj.solver = core.linalg.LinearSolver();
121 end

123 function obj = update(obj, U)
124 % UPDATE Update the solution history with a new solution.
125 %
126 % obj = update(obj, U) shifts the existing solution history
127 % and adds the new solution at the most recent position. For
128 % multi-step methods, this maintains the required number of
129 % previous solutions. For single-step methods, simply stores
130 % the current solution.
131 %
132 % Inputs:
133 % obj - The OdeIntegrator object
134 % U - The new solution to add to history
135 %
136 % Outputs:
137 % obj - The OdeIntegrator object
138
139 %< Shift history for multi-step methods
140 if obj.nSteps > 1
141 obj.U0 = circshift(obj.U0, [0, 1]);
142 end
143
144 %< Store new solution at all stage positions
145 for i = 1:obj.nStages
146 obj.U0{i, 1} = U;
147 end
148 end
