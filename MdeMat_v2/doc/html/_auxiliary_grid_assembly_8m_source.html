2 % AUXILIARYGRIDASSEMBLY Grid-based assembly for auxiliary variables.
3 %
4 % AuxiliaryGridAssembly implements Dirichlet boundary treatment for
5 % auxiliary variables in mixed finite element systems. This class
6 % handles coupling terms between primal and auxiliary variables with
7 % both interior and extraction contributions.
8 %
9 % The assembly specializes in handling jump terms and boundary
10 % conditions specific to auxiliary variable formulations in mixed
11 % methods and discontinuous Galerkin schemes.
12 %
13 % Examples:
14 % % Create auxiliary assembly with left flux and Dirichlet BC
15 % assembly = AuxiliaryGridAssembly(fe, mesh, op, 1, 1);
16 %
17 % % Assemble outer jump operator

27 function A = outerJump(obj, d, c)
28 % OUTERJUMP Assemble outer jump operator for auxiliary
29 % variables.
30 %
31 % A = outerJump(obj, d, c) creates the outer jump operator
32 % matrix for auxiliary variables along the specified
33 % dimension. This handles jump discontinuities at element
34 % boundaries for mixed formulations.
35 %
36 % Inputs:
37 % obj - The AuxiliaryGridAssembly object
38 % d - Spatial dimension index (1, 2, or 3)
39 % c - Coefficient vector or scalar
40 %
41 % Outputs:
42 % A - Assembled outer jump sparse matrix
43
44 ne = obj.nTotalElements;
45
46 %< ------------------------------------------------------------
47 %< COEFFICIENTS
48 %< ------------------------------------------------------------
49 if isa(obj.mesh, 'approx.mesh.UniformGrid')
50 h = obj.mesh.spacings{d};
51 c = ones(1, ne) .* c ./ h;
52 end
53
54 if isa(obj.mesh, 'approx.mesh.NonuniformGrid')
55 h = obj.mesh.spacings;
56 [h{:}] = ndgrid(h{:});
57 c = c ./ h{d}(:);
58 end
59
60 c = c(:);
61
62 %< ------------------------------------------------------------
63 %< FLUX CONTRIBUTION
64 %< ------------------------------------------------------------
65 if isa(obj.fe, 'fem.element.L2FiniteElement')
66 switch obj.fluxType
67 case 1
68 T = obj.assembleOuterLeftFluxDirichletBoundaryJumpTerms(d, c);
69 case 2
70 T = obj.assembleOuterRightFluxDirichletBoundaryJumpTerms(d, c);
71 case 3
72 T1 = obj.assembleOuterLeftFluxDirichletBoundaryJumpTerms(d, c/2);
73 T2 = obj.assembleOuterRightFluxDirichletBoundaryJumpTerms(d, c/2);
74 T = [T1; T2];
75 end
76 end
77
78 %< ------------------------------------------------------------
79 %< TRIPLETS TO SPARSE MATRIX
80 %< ------------------------------------------------------------
81 n = obj.nGlobalDofs;
82 A = obj.sparseFromTriplets(T, n, n);
83 end
84 end

87 function T = assembleInnerLeftFluxDirichletBoundaryTerms(obj, d, c)
88 % ASSEMBLEINNERLEFTFLUXDIRICHLETBOUNDARYTERMS Assemble inner
89 % left flux Dirichlet boundary terms.
90 %
91 % T = assembleInnerLeftFluxDirichletBoundaryTerms(obj, d, c)
92 % computes the inner contribution to a left flux along the
93 % specified dimension on boundaries with Dirichlet boundary
94 % conditions for auxiliary variables.
95 %
96 % Inputs:
97 % obj - The AuxiliaryGridAssembly object
98 % d - Spatial dimension index (1, 2, or 3)
99 % c - Coefficient vector, one per element
100 %
101 % Outputs:
102 % T - Triplet matrix [row, col, value] for boundary contribution
103
104 np = obj.nLocalDofs;
105 P = obj.fe.projector;
106 G = obj.op.gradient;
107
108 m = obj.mesh.allElementMultiIndices;
109 e = m(:, d) == 1 | m(:, d) == obj.mesh.resolution(d);
110 ne = numel(find(e));
111 m = m(e, :);
112 l = obj.mesh.indexer.multiToLinear(m, 1);
113
114 T = zeros(np^2*ne, 3);
115
116 F = G.fluxData(1, 2*d).matrix;
117 [i, j, v] = find(P.project(F));
118 nv = numel(v);
119 k = 1:(nv * ne);
120 T(k, 1) = reshape(bsxfun(@plus, (l(:).' - 1)*np, i(:)), [], 1);
121 T(k, 2) = reshape(bsxfun(@plus, (l(:).' - 1)*np, j(:)), [], 1);
122 T(k, 3) = kron(c(l), v(:));
123 end

125 function T = assembleOuterLeftFluxDirichletBoundaryTerms(obj, d, c)
126 % ASSEMBLEOUTERLEFTFLUXDIRICHLETBOUNDARYTERMS Assemble outer
127 % left flux Dirichlet boundary terms.
128 %
129 % T = assembleOuterLeftFluxDirichletBoundaryTerms(obj, d, c)
130 % computes the outer contribution to a left flux along the
131 % specified dimension on boundaries with Dirichlet boundary
132 % conditions. Combines interior and extraction contributions.
133 %
134 % Inputs:
135 % obj - The AuxiliaryGridAssembly object
136 % d - Spatial dimension index (1, 2, or 3)
137 % c - Coefficient vector, one per element
138 %
139 % Outputs:
140 % T - Combined triplet matrix for boundary contribution
141
142 np = obj.nLocalDofs;
143 P = obj.fe.projector;
144 G = obj.op.gradient;
145
146 %< part 1: interior
147 m = obj.mesh.allElementMultiIndices;
148 e = m(:, d) == obj.mesh.resolution(d);
149 ne = numel(find(e));
150 m = m(e, :);
151 l1 = obj.mesh.indexer.multiToLinear(m, 1);
152 m(:, d) = m(:, d) - 1;
153 l2 = obj.mesh.indexer.multiToLinear(m, 1);
154
155 T1 = zeros(np^2*ne, 3);
156
157 F = G.fluxData(2, 2*d-1).matrix;
158 [i, j, v] = find(-P.project(F));
159 nv = numel(v);
160 k = 1:(nv * ne);
161 T1(k, 1) = reshape(bsxfun(@plus, (l1(:).' - 1)*np, i(:)), [], 1);
162 T1(k, 2) = reshape(bsxfun(@plus, (l2(:).' - 1)*np, j(:)), [], 1);
163 T1(k, 3) = kron(c(l1), v(:));
164
165 %< part 2: extraction
166 m = obj.mesh.allElementMultiIndices;
167 e = m(:, d) == 1;
168 ne = numel(find(e));
169 m = m(e, :);
170 l = obj.mesh.indexer.multiToLinear(m, 1);
171 T2 = zeros(np^2*ne, 3);
172
173 F = G.fluxData(1, 2*d-1).matrix;
174 [i, j, v] = find(-P.project(F));
175 nv = numel(v);
176 k = 1:(nv * ne);
177 T2(k, 1) = reshape(bsxfun(@plus, (l(:).' - 1)*np, i(:)), [], 1);
178 T2(k, 2) = reshape(bsxfun(@plus, (l(:).' - 1)*np, j(:)), [], 1);
179 T2(k, 3) = kron(c(l), v(:));
180
181 T = [T1; T2];
182 end
