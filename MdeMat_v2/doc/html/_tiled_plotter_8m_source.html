4 % TiledPlotter provides infrastructure for creating and managing

6 % This class simplifies the creation of complex visualization layouts

107 function obj = draft(obj, varargin)
108 % DRAFT Create the tiledlayout and axes.
109 %
110 % obj = draft(obj) creates the MATLAB tiledlayout object and
111 % generates all axes according to the added specifications.
112 % This method must be called before rendering can occur.
113 %
114 % obj = draft(obj, 'ParameterName', ParameterValue) passes
115 % additional arguments to the tiledlayout constructor.
116 %
117 % Inputs:
118 % obj - The TiledPlotter object
119 % varargin - Additional arguments for tiledlayout
120 %
121 % Outputs:
122 % obj - The TiledPlotter object
123
124 if ~isempty(obj.axes), return; end
125
126 nRows = obj.size(1);
127 nCols = obj.size(2);
128 obj.tiledObj = tiledlayout(nRows, nCols, varargin{:});
129 obj.computeTileIndices();
130
131 nAxes = length(obj.specs);
132 obj.axes = gobjects(1, nAxes);
133 for iAxis = 1:nAxes
134 spec = obj.specs(iAxis);
135 tileIdx = obj.tileIndices(iAxis);
136 ax = nexttile(obj.tiledObj, tileIdx, [spec.rows, spec.cols]);
137 obj.axes(iAxis) = ax;
138 end
139 end

195 function obj = computeTileIndices(obj)
196 % COMPUTETILEINDICES Compute tile placement for axis
197 % specifications.
198 %
199 % computeTileIndices(obj) automatically determines where to
200 % place each axis specification in the tiled layout, ensuring
201 % no overlaps occur. Uses a greedy placement algorithm.
202 %
203 % Inputs:
204 % obj - The TiledPlotter object
205 %
206 % Outputs:
207 % obj - The TiledPlotter object
208
209 rows = obj.size(1);
210 cols = obj.size(2);
211 occupied = false(rows, cols);
212 nSpecs = length(obj.specs);
213 obj.tileIndices = zeros(1, nSpecs);
214
215 for iSpec = 1:nSpecs
216 spec = obj.specs(iSpec);
217 placed = false;
218
219 %< Try to place the current specification
220 for idx = 1:(rows * cols)
221 row = floor((idx - 1) / cols) + 1;
222 col = mod(idx - 1, cols) + 1;
223
224 %< Check if specification fits at this position
225 if row + spec.rows - 1 <= rows && col + spec.cols - 1 <= cols
226 block = occupied(row:row+spec.rows-1, col:col+spec.cols-1);
227 if all(block(:) == false)
228 occupied(row:row+spec.rows-1, col:col+spec.cols-1) = true;
229 obj.tileIndices(iSpec) = idx;
230 placed = true;
231 break;
232 end
233 end
234 end
235
236 core.except.assert(placed, 'BadAxisSpec', ...
237 'Cannot place axis spec %d in the grid.', iSpec);
238 end
