1classdef FunctionData < handle
2 % FUNCTIONDATA A collection of discrete data for a function.
3 %
4 % FunctionData stores function values and derivatives evaluated at
5 % integration nodes, along with the associated quadrature weights.
6 % This class provides the fundamental data structure for numerical
7 % integration and finite element computations.
8 %
9 % The class manages function evaluations at quadrature points and
10 % supports efficient computation of integrals involving functions
11 % and their derivatives. It serves as a bridge between continuous
12 % function representations and discrete numerical computations.
13 %
14 % Examples:
15 % % Create function data with Gauss quadrature
16 % integrator = approx.integrate.GaussLegendre(5);
17 % funcData = FunctionData(integrator);
18 %
19 % % Set integration points and evaluate function
20 % funcData.setPoints(-1, 1);
21 % f = @(x) sin(pi*x);
22 % funcData.setValues(f, funcData.nodes);
23 %
24 % % Compute derivatives
25 % df = DifferentiableFunction(f);
26 % funcData.setDerivatives(df, funcData.nodes, 2);
27 %
28 % % Access computed data
29 % values = funcData.values;
30 % firstDeriv = funcData.derivatives{1};
31 %
32 % See also:
33 % approx.integrate.Integrator, approx.fem.data.BilinearFormData

128 function obj = setDerivatives(obj, f, X, k)
129 % SETDERIVATIVES Set the function derivatives at specified
130 % points.
131 %
132 % obj = setDerivatives(obj, f, X, k) computes derivatives of
133 % the function @a f up to order @a k at the specified points
134 % @a X using multi-index notation. The derivatives are stored
135 % in the derivatives cell array.
136 %
137 % Inputs:
138 % obj - The FunctionData object
139 % f - DifferentiableFunction object with derivative method
140 % X - Evaluation nodes (column vector)
141 % k - Maximum derivative order (non-negative integer)
142 %
143 % Outputs:
144 % obj - The FunctionData object
145
146 n = size(X, 1);
147
148 %< Create multi-index generator for derivative orders
149 L = core.linalg.L1MultiIndexer();
150 L.setCache(n, n + k);
151
152 %< Extract multi-indices (exclude zero-th order)
153 M = L.cache(2:end, :) - 1;
154 m = size(M, 1);
155
156 %< Compute derivatives for each multi-index
157 dY = arrayfun(@(i) f.derivative(X, M(i, :)), 1:m, 'Un', 0);
158 obj.derivatives = dY;
159 end
160 end
