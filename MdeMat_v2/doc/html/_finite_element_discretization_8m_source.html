1classdef FiniteElementDiscretization < handle
2 % FINITEELEMENTDISCRETIZATION Finite element spatial discretization.
3 %
4 % FiniteElementDiscretization contains all components needed for
5 % finite element methods: element, mesh, space, and operator.
6 % It provides a complete framework for discontinuous Galerkin (DG)
7 % and continuous finite element methods on orthotopal meshes.
8 %
9 % The class manages the complete finite element discretization
10 % including geometric mesh, function spaces, and differential
11 % operators. It supports various basis function types (Legendre,
12 % Lagrange) and integration schemes for high-order accurate
13 % computations.
14 %
15 % Examples:
16 % % Create FE discretization with L2 orthotope element
17 % disc = FiniteElementDiscretization();
18 % disc.setL2OrthotopeElement('nDims', 2, 'order', 3, 'basisType', 1);
19 % disc.setElementOperator();
20 % disc.setDerivativeOrder(1);
21 %
22 % % Set up uniform mesh and function space
23 % disc.setUniformGrid([10, 10], [0, 1; 0, 1]);
24 % disc.setMeshSpace();
25 %
26 % % Access gradient operator for stiffness matrix assembly
27 % gradOp = disc.operator.gradient;
28 % volGradData = gradOp.volumeData; % Volume gradient terms
29 % fluxGradData = gradOp.fluxData; % Boundary flux terms
30 %
31 % % Mesh refinement
32 % fineMesh = disc.refineMesh(2); % Refine by factor of 2
33 %
34 % Notes:
35 % Components must be set in order: element, then operator, then
36 % derivative order. Mesh and space can be set independently but
37 % both are needed for mesh-based computations.
38 %
39 % See also:
40 % approx.element.L2Element, approx.element.L2ElementOperator,
41 % approx.mesh.UniformGrid, approx.mesh.NonuniformGrid,

69 function obj = setL2OrthotopeElement(obj, varargin)
70 % SETL2ORTHOTOPEELEMENT Set L2 orthotope element with specified
71 % configuration.
72 %
73 % obj = setL2OrthotopeElement(obj, varargin) creates a
74 % discontinuous finite element on the unit orthotope with
75 % specified polynomial order and basis configuration. This
76 % method encapsulates the complex element construction
77 % process including geometry, basis functions, integrators,
78 % and projectors for DG methods.
79 %
80 % Inputs:
81 % obj - The FiniteElementDiscretization object
82 % varargin - Name-value parameter pairs:
83 %< 'nDims' - Number of spatial dimensions (positive integer)
84 %< 'order' - Polynomial order (positive integer, default: 1)
85 %< 'basisType' - Basis function type (1, 2, or 3, default: 1)
86 %< 'basisPattern' - Basis ordering pattern (1 or 2, default: 1)
87 %
88 % Outputs:
89 % obj - The FiniteElementDiscretization object
90 %
91 % Notes:
92 % Basis types:
93 % 1: Legendre orthogonal basis (modal)
94 % 2: Lagrange-Gauss-Legendre basis (nodal)
95 % 3: Lagrange-Gauss-Lobatto basis (nodal)
96 %
97 % Basis patterns:
98 % 1: L_infinity norm ordering (tensor product)
99 % 2: L_1 norm ordering (total degree)
100
101 p = inputParser;
102 addParameter(p, 'nDims', [], @(x) x >= 1);
103 addParameter(p, 'order', 1, @(x) x >= 1);
104 addParameter(p, 'basisType', 1, @(x) ismember(x, 1:3));
105 addParameter(p, 'basisPattern', 1, @(x) ismember(x, 1:2));
106 parse(p, varargin{:});
107
108 d = p.Results.nDims;
109 k = p.Results.order;
110 t = p.Results.basisType;
111 s = p.Results.basisPattern;
112
113 %< Create unit orthotope geometry
114 G = core.geometry.Orthotope.unit(d);
115
116 %< Create basis functions based on type
117 switch t
118 case 1
119 U = approx.basis.OrthogonalBasisFunction( ...
120 k, 'monic_legendre', 'unit');
121 case 2
122 U = approx.basis.InterpolationBasisFunction( ...
123 k, 'lagrange', 'unit', 'gauss_legendre');
124 case 3
125 U = approx.basis.InterpolationBasisFunction( ...
126 k, 'lagrange', 'unit', 'gauss_lobatto');
127 end
128 U = repmat(U, 1, d);
129
130 %< Create separable basis with specified pattern
131 switch s
132 case 1
133 B = approx.basis.SeparableBasisFunction(U, 'lx');
134 case 2
135 B = approx.basis.SeparableBasisFunction(U, 'l1');
136 end
137 B.autoLoad();
138
139 %< Create volume integrator
140 switch t
141 case {1, 2}
142 Q = 'gauss_legendre';
143 case 3
144 Q = 'gauss_lobatto';
145 end
146 VI = approx.integrate.OrthotopeIntegrator(G.nDims, Q);
147 VI.setPoints(B.nFactorCodims, G.lower, G.upper);
148
149 %< Create face integrators for all boundaries
150 FI = arrayfun(@(i) approx.integrate.OrthotopeFaceIntegrator( ...
151 G.nDims, Q, i), 1:(2 * G.nDims));
152 for i = 1:(2 * G.nDims)
153 FI(i).setPoints(B.nFactorCodims, G.lower, G.upper);
154 end
155
156 %< Create projector based on basis type
157 switch t
158 case 1
159 P = approx.project.ModalProjector(B);
160 P.setMass(VI.nodes, VI.weights);
161 case {2, 3}
162 P = approx.project.NodalProjector(B);
163 P.setMass(VI.nodes);
164 end
165
166 %< Construct L2 orthotope finite element
167 obj.element = approx.element.L2Element(G, P, VI, FI);
168 end

198 % SETDERIVATIVEORDER Set derivative order for all components.
199 %

201 % for volume and flux integrals with the specified maximum

221 obj.element.setFluxData(max(0, order-1));

225 obj.operator.setVolumeData();

227 %< Set flux data for L2 operator

288 function newObj = refineMesh(obj, varargin)
289 % REFINEMESH Create refined mesh discretization.
290 %
291 % newObj = refineMesh(obj, varargin) creates a new finite
292 % element discretization with refined mesh while preserving
293 % the element and operator configurations. This enables
294 % h-refinement studies and adaptive mesh refinement.
295 %
296 % Inputs:
297 % obj - The FiniteElementDiscretization object
298 % varargin - Refinement parameters:
299 %< For uniform grid: k - Refinement factor per dimension
300 %< For nonuniform grid: k - Number of subdivisions per element
301 %

304
305 newObj = approx.discretization.FiniteElementDiscretization();

307 newObj.mesh = obj.mesh.refine(varargin{:});
