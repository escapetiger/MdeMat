1classdef PointSet < handle
2 % POINTSET Point distributions within geometric domains.
3 %
4 % PointSet generates and manages point distributions within various
5 % geometric domains using appropriate integration rules. It supports
6 % orthotope (rectangular) and hypersphere geometries with multiple
7 % point generation strategies including deterministic quadrature
8 % and Monte Carlo methods.
9 %
10 % Examples:
11 % % Create points in 2D rectangle using Gauss-Legendre rule
12 % orthotope = msclab.geometry.Orthotope([-1, 0], [1, 2]);
13 % pointSet = PointSet(orthotope, [5, 5], 'RuleType', 'gauss_legendre');
14 %
15 % % Create points on 3D sphere using Lebedev rule
16 % sphere = msclab.geometry.Hypersphere([0, 0, 0], 1);
17 % pointSet = PointSet(sphere, 110, 'RuleType', 'lebedev');
18 %
19 % % Monte Carlo points with custom seed
20 % pointSet = PointSet(sphere, 1000, 'RuleType', 'monte_carlo', 'Seed', 42);
21 %
22 % Notes:
23 % The class automatically selects appropriate integrators based on
24 % geometry type and provides a unified interface for accessing
25 % nodes, weights, and point set properties.
26 %
27 % See also:
28 % approx.integrate.OrthotopeIntegrator, approx.integrate.HypersphereIntegrator,
29 % msclab.geometry.Orthotope, msclab.geometry.Hypersphere

171 end

173 I.setPoints(n, normalization, G.centroid, G.radius, coordSys);
