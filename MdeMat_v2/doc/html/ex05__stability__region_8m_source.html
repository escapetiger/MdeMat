1%% ex05_stability_regions.m
2% Analyze and visualize stability regions for all integrator types
3%
4% Tests A-stability using the Dahlquist test equation: du/dt = λu
5% A method is A-stable if |R(z)| ≤ 1 for all z with Re(z) ≤ 0
6% where R(z) is the stability function and z = λ*dt
7% Includes explicit, implicit, and IMEX integrators.
8
9clear; close all; clc;
10
11%% Setup for stability analysis
12fprintf('=== STABILITY REGION ANALYSIS ===\n\n');
13
14% Complex plane grid
15lambda_real = linspace(-8, 2, 200);
16lambda_imag = linspace(-5, 5, 200);
17[LR, LI] = meshgrid(lambda_real, lambda_imag);
18Z = LR + 1i * LI;
19
20%% Define integrators to analyze
21final_time = 5.0;
22% Implicit integrators
23implicit_integrators = {
24 approx.odeint.BeIntegrator(final_time)
25 approx.odeint.Bdf2Integrator(final_time)
26 approx.odeint.Sdirk2Integrator(final_time)
27 approx.odeint.Sdirk3Integrator(final_time)
28};
29
30% Explicit integrators
31explicit_integrators = {
32 approx.odeint.FeIntegrator(final_time)
33 approx.odeint.HeunIntegrator(final_time)
34 approx.odeint.Exrk4Integrator(final_time)
35 approx.odeint.Ssprk3Integrator(final_time)
36};
37
38% IMEX integrators (for A-stability, we consider their implicit part)
39imex_integrators = {
40 approx.odeint.Ars111Integrator(final_time)
41 approx.odeint.Ars222Integrator(final_time)
42 approx.odeint.Ars443Integrator(final_time)
43};

45% Combine all integrators
46all_integrators = [implicit_integrators; explicit_integrators; imex_integrators];
47n_methods = length(all_integrators);

49% Extract integrator names and types

53for i = 1:n_methods
54 class_name = class(all_integrators{i});
55 name_parts = strsplit(class_name, '.');
56 base_name = name_parts{end};
57 integrator_names{i} = strrep(base_name, 'Integrator', '');

59 % Determine type
60 if i <= length(implicit_integrators)

62 elseif i <= length(implicit_integrators) + length(explicit_integrators)
63 integrator_types{i} = 'Explicit';
64 else
65 integrator_types{i} = 'IMEX';
66 end
67end

69%% Compute stability functions
70R = cell(n_methods, 1);
71
72% Implicit methods
73% Backward Euler: R(z) = 1/(1-z)
74R{1} = 1 ./ (1 - Z);
75
76% BDF2: R(z) = (3-2z)/(1 - 2z/3)^2 (corrected BDF2 stability function)
77R{2} = (3 - 2*Z) ./ (1 - (2/3)*Z).^2;
78
79% SDIRK2: R(z) = (1 + (1-2γ)z) / (1 - γz)^2
80gamma2 = 1 - sqrt(2)/2;
81R{3} = (1 + (1-2*gamma2)*Z) ./ (1 - gamma2*Z).^2;
82
83% SDIRK3: Approximate stability function
84gamma3 = 0.435866521508459;
85R{4} = 1 ./ (1 - gamma3*Z).^3; % Simplified approximation
86
87% Explicit methods
88% Forward Euler: R(z) = 1 + z
89R{5} = 1 + Z;
90
91% Heun (RK2): R(z) = 1 + z + z^2/2
92R{6} = 1 + Z + Z.^2/2;
93
94% RK4: R(z) = 1 + z + z^2/2 + z^3/6 + z^4/24
95R{7} = 1 + Z + Z.^2/2 + Z.^3/6 + Z.^4/24;
96
97% SSPRK3: Approximate stability function (3rd order)
98R{8} = 1 + Z + Z.^2/2 + Z.^3/6;
99
100% IMEX methods (analyze implicit part for A-stability)
101% ARS111: Similar to Backward Euler for implicit part
102R{9} = 1 ./ (1 - Z);
103
104% ARS222: Similar to SDIRK2 for implicit part
105gamma_ars2 = 1 - sqrt(2)/2;
106R{10} = (1 + (1-2*gamma_ars2)*Z) ./ (1 - gamma_ars2*Z).^2;
107
108% ARS443: More complex, approximate as 3rd order DIRK
109R{11} = 1 ./ (1 - 0.5*Z).^3; % Simplified
