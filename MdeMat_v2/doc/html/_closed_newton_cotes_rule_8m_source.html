2 % CLOSEDNEWTONCOTESRULE Closed Newton-Cotes numerical integration rule.
3 %
4 % ClosedNewtonCotesRule implements numerical integration using the
5 % closed Newton-Cotes method, which includes the interval endpoints.
6 % This method creates equally spaced nodes and calculates weights
7 % using Lagrange interpolation polynomials.
8 %
9 % Examples:
10 % % Create a closed Newton-Cotes rule
11 % rule = ClosedNewtonCotesRule();
12 %
13 % % Generate 5 nodes and weights for interval [0, 1]
14 % [x, w] = rule.generate(5);
15 %
16 % % Generate nodes and weights for custom interval
17 % [x, w] = rule.generate(5, -1, 1);
18 %
19 % Notes:
20 % The closed Newton-Cotes formula includes both endpoints of the
21 % integration interval. For n nodes, the method has polynomial
22 % exactness of degree n-1 for odd n and n for even n.
23 %
24 % See also:
25 % approx.integrate.UnivariateRule,
26 % approx.integrate.GaussLegendreRule,

31 % GENERATE Generate 1D closed Newton-Cotes nodes and weights.
32 %
33 % [x, w] = generate(obj, n) generates n equally spaced nodes
34 % and corresponding weights for the interval [0, 1].
35 %
36 % [x, w] = generate(obj, n, a, b) generates nodes and weights
37 % for the interval [a, b].
38 %
39 % Inputs:
40 % n - Number of quadrature nodes (n >= 1)
41 % a - Lower integration bound (optional, default: 0)
42 % b - Upper integration bound (optional, default: 1)
43 %
44 % Outputs:
45 % x - Row vector of quadrature nodes within [a, b]
46 % w - Row vector of quadrature weights
47
48 if nargin < 3, [a, b] = deal(0, 1); end

50 core.except.assert(isscalar(a) && isscalar(b), ...
51 'InvalidInput', 'a and b must be scalars.');
52 core.except.assert(a <= b, 'InvalidInput', ...
53 'a <= b is required.');
54
55 if a == b
56 [x, w] = deal(a, 1);
57 return;
58 end
59
60 x = linspace(0, 1, n);
61 w = zeros(1, n);
62 for i = 1:n
63 L = 1;
64 for j = 1:n
65 if i ~= j
66 L = conv(L, [1, -x(j)]) / (x(i) - x(j));
67 end
68 end
69 I = polyint(L);
70 w(i) = polyval(I, 1) - polyval(I, 0);
71 end
72
73 x = a + (b - a) * x;
74 w = w * (b - a);
75 end
76 end
