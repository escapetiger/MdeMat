2 % SEMILAGRANGIANDIRICHLETASSEMBLY Assembly for semi-Lagrangian dynamic
3 % with Dirichlet boundary conditions.
4 %
5 % SemiLagrangianDirichletAssembly implements the assembly of
6 % transport operators for semi-Lagrangian methods with Dirichlet
7 % boundary conditions. It handles characteristic tracing, boundary
8 % intersections, and the incorporation of boundary data into the
9 % discrete system.
10 %
11 % The class manages two key operations: volume assembly for interior
12 % characteristic tracing and boundary assembly for enforcing
13 % Dirichlet conditions. Characteristics that exit the computational
14 % domain are traced to boundary intersections where boundary data is
15 % evaluated and incorporated into the system.
16 %
17 % Notes:
18 % Boundary condition assembly involves geometric intersection
19 % computations and requires careful handling of characteristic
20 % trajectories that exit the domain.
21 %
22 % See also:
23 % approx.assembly.Assembly,
24 % approx.assembly.SemiLagrangianPeriodicAssembly,
25 % core.geometry.LineSegmentByPlaneIntersector

32 function T = assembleVolume(obj)
33 % ASSEMBLEVOLUME Assemble volume terms for semi-Lagrangian
34 % transport.
35 %
36 % T = assembleVolume(obj) assembles the triplet
37 % representation of the transport operator matrix for
38 % interior elements. The method handles characteristic
39 % tracing and interpolation for semi-Lagrangian methods,
40 % excluding elements that intersect domain boundaries.
41 %
42 % Inputs:
43 % obj - The SemiLagrangianDirichletAssembly object
44 %
45 % Outputs:
46 % T - Triplet matrix representation [row, col, val]
47
48 nl = obj.space.nLocalDofs;
49 np = obj.operator.clipper.nVolumePieces;
50 ne = obj.space.mesh.nTotalElements;
51
52 %< Project mass matrices for each volume piece
53 fM = @(i) obj.space.element.projector.project(obj.operator.mass(i).matrix);
54 M = arrayfun(fM, 1:np, 'Un', 0);
55 s = obj.operator.clipper.volumeShifts;
56 d = obj.space.nDims;
57
58 T = cell(np, 1);
59 for p = 1:np
60 T{p} = zeros(nl^2*ne, 3);
61 [i, j, v] = find(M{p});
62 nv = numel(v);
63 m = obj.space.mesh.allElementMultiIndices;
64 b = m + s(:, p).';
65
66 %< Identify boundary interactions
67 b = (b < 1) .* (2 * (1:d) - 1) + ...
68 (b > obj.space.mesh.resolution) .* (2 * (1:d));
69
70 %< Keep only interior elements
71 m = m(sum(b ~= 0, 2) == 0, :);
72 nb = size(m, 1);
73 k = 1:(nv * nb);
74
75 %< Compute global indices for assembly
76 l1 = obj.space.mesh.indexer.multiToLinear(m, 1);
77 l2 = obj.space.mesh.indexer.multiToLinear(m + s(:, p).', 1);
78 T{p}(k, 1) = reshape(bsxfun(@plus, (l1(:).' - 1)*nl, i(:)), [], 1);
79 T{p}(k, 2) = reshape(bsxfun(@plus, (l2(:).' - 1)*nl, j(:)), [], 1);
80 T{p}(k, 3) = kron(ones(nb, 1), v(:));
81 end
82 T = vertcat(T{:});
83 end

85 function T = assembleBoundaryCondition(obj, tBegin, tEnd, f)
86 % ASSEMBLEBOUNDARYCONDITION Assemble Dirichlet boundary
87 % condition terms.
88 %
89 % T = assembleBoundaryCondition(obj, tBegin, tEnd, f)
90 % assembles the boundary condition contribution for
91 % semi-Lagrangian transport. The method traces
92 % characteristics that exit the domain to their boundary
93 % intersections and evaluates the boundary condition function
94 % at those points.
95 %
96 % Inputs:
97 % obj - The SemiLagrangianDirichletAssembly object
98 % tBegin - Beginning time for characteristic integration (scalar)
99 % tEnd - End time for characteristic integration (scalar)
100 % f - Boundary condition function handle f(x, t)
101 %
102 % Outputs:
103 % T - Triplet matrix representation [row, col, val]
104 %
105 % Notes:
106 % Uses geometric intersection algorithms to find where
107 % characteristics meet domain boundaries. Handles
108 % multi-dimensional boundary intersections correctly.
109
110 nl = obj.space.nLocalDofs;
111 np = obj.operator.clipper.nVolumePieces;
112 nd = obj.space.nDims;
113 ht = tBegin - tEnd;
114 v = obj.operator.clipper.velocity;
115
116 T = cell(1, np);
117 for p = 1:np
118 %< Find elements whose pth volume piece exits domain
119 m = obj.space.mesh.allElementMultiIndices;
120 s = obj.operator.clipper.volumeShifts(:, p).';
121 b = (m + s < 1) .* (2 * (1:nd) - 1) + ...
122 (m + s > obj.space.mesh.resolution) .* (2 * (1:nd));
123 b = m(sum(b ~= 0, 2) > 0, :);
124 nb = size(b, 1);
125
126 %< Initialize triplet structure
127 k = obj.space.mesh.indexer.multiToLinear(b, 1);
128 T{p} = zeros(nb*nl, 3);
129 T{p}(:, 1) = reshape(bsxfun(@plus, (k(:)-1).'*nl, (1:nl).'), [], 1);
130 T{p}(:, 2) = 1;
131
132 if nb == 0, continue; end
133
134 %< Set up characteristic end points
135 nq = obj.operator.clipper.volumeEulerianData(1, p).nPoints;
136 X1 = zeros(nd + 1, nb*nq);
137 X2 = zeros(nd + 1, nb*nq);
138 Xq = obj.operator.clipper.volumeEulerianData(1, p).nodes;
139 X2(1:nd, :) = obj.space.mesh.collocate(Xq, b);
140 X2(nd+1, :) = tBegin;
141 X1(1:nd, :) = X2(1:nd, :) - ht * v(:);
142 X1(nd+1, :) = tEnd;
143
144 %< Find boundary intersections of characteristics
145 I = core.geometry.LineSegmentByPlaneIntersector(nd+1);
146 bbox = reshape(obj.space.mesh.bbox, 2, []);
147 bbox = -bbox(2*(1:nd) - (v > 0));
148 Xb = zeros(nd + 1, nb*nq);
149
150 for d = 1:nd
151 a = zeros(nd+1, 1);
152 a(d) = 1;
153 [X, TF] = I.intersect(X1, X2, a, bbox(d));
154 Xb(:, TF) = X;
155 end
156
157 %< Evaluate boundary function and project to element space
158 U = f(Xb(1:nd, :), Xb(nd+1, :));
159 U = reshape(U, nq, nb);
160 P = obj.operator.volumePieceData(1, p).vector;
161 T{p}(:, 3) = reshape(obj.space.element.projector.project(P * U), [], 1);
162 end
163 T = vertcat(T{:});
164 end
165 end
166end
