1classdef SeparableGraph < core.linalg.Separable
2 % SEPARABLEGRAPH Memory-efficient structured graph representation.
3 %
4 % SeparableGraph provides a memory-efficient representation of
5 % structured graphs using tensor decomposition concepts. Instead of
6 % storing all vertices and edges explicitly, it stores projections
7 % along each dimension as separate Graph objects, enabling efficient
8 % operations on large structured grids.
9 %
10 % Examples:
11 % % Create 2D grid graph
12 % V = {[0, 1, 2], [0, 1]}; % 3×2 grid vertices
13 % E = {[1,2; 2,3], [1,2]}; % Edge connectivity per dimension
14 % sepGraph = SeparableGraph(V, E);
15 %
16 % % Query properties
17 % nVerts = sepGraph.nTotalVertices;
18 % nEdges = sepGraph.nTotalEdges;
19 %
20 % % Convert to full graph representation
21 % fullGraph = sepGraph.full();
22 %
23 % Notes:
24 % This representation is most efficient for structured grids where
25 % connectivity follows a tensor product pattern. Memory usage scales
26 % as O(sum of factor sizes) rather than O(product of factor sizes).
27 %
28 % See also:
29 % core.linalg.SeparableObject, approx.mesh.Graph,
30 % approx.mesh.Grid

173 function E = getFullEdges(obj)
174 % GETFULLEDGES Generate full edge connectivity matrix.
175 %
176 % E = getFullEdges(obj) creates the complete edge
177 % connectivity matrix by expanding factor edges to the full
178 % tensor product graph structure.
179 %
180 % Inputs:
181 % obj - The SeparableGraph object
182 %
183 % Outputs:
184 % E - Matrix of all edge connections (nEdges × 2)
185
186 D = obj.nDims;
187
188 if D == 1
189 E = obj.factors(1).edges;
190 return;
191 end
192
193 N = obj.nVertices;
194 indexer = core.linalg.MultiIndexer(N);
195
196 E = zeros(obj.nTotalEdges, 2);
197 iEdge = 0;
198 for d = 1:D
199 factorEdges = obj.factors(d).edges;
200 nFactorEdges = size(factorEdges, 1);
201
202 if nFactorEdges == 0
203 continue;
204 end
205
206 I = 1:D;
207 I(d) = [];
208 S = N(I);
209 P = prod(S);
210 batchSize = min(10000, P);
211
212 for batchStart = 1:batchSize:P
213 batchEnd = min(batchStart + batchSize - 1, P);
214 batchSize = batchEnd - batchStart + 1;
215
216 indexer2 = core.linalg.MultiIndexer(S);
217 L = (batchStart:batchEnd)';
218 M = indexer2.linearToMulti(L);
219
220 for e = 1:nFactorEdges
221 v1 = factorEdges(e, 1);
222 v2 = factorEdges(e, 2);
223
224 newEdges = zeros(batchSize, 2);
225
226 for j = 1:batchSize
227 idx1 = ones(1, D);
228 idx2 = ones(1, D);
229 idx1(I) = M(j, :);
230 idx2(I) = M(j, :);
231 idx1(d) = v1;
232 idx2(d) = v2;
233 L1 = indexer.multiToLinear(idx1);
234 L2 = indexer.multiToLinear(idx2);
235 newEdges(j, :) = [L1, L2];
236 end
237
238 E(iEdge+1:iEdge+batchSize, :) = newEdges;
239 iEdge = iEdge + batchSize;
