1classdef DirkIntegrator < approx.odeint.OdeIntegrator
2 % DIRKINTEGRATOR Diagonally Implicit Runge-Kutta integrator.
3 %
4 % DirkIntegrator implements Diagonally Implicit Runge-Kutta (DIRK)
5 % methods for solving ordinary differential equations of the form:
6 %
7 % \f[
8 % \frac{\mathrm{d}(Mu)}{\mathrm{d}t} = Lu + S,
9 % \f]
10 %
11 % where \f$M\f$ is a mass operator, \f$L\f$ is a linear operator, and
12 % \f$S\f$ is a source term. Any of these terms can be empty. For mass
13 % operator, it means it reduces to an identity. For linear and source
14 % term, it means they vanish from the equation.
15 %
16 % DIRK methods are characterized by a lower triangular Butcher
17 % tableau with identical diagonal elements, allowing each stage to be
18 % solved independently. This makes them more efficient than fully
19 % implicit Runge-Kutta methods while maintaining good stability
20 % properties.
21 %
22 % Examples:
23 % % Basic usage
24 % integrator = approx.odeint.Sdirk2Integrator(1.0);
25 % U_new = integrator.step(L, S, M);
26 % integrator.update(U_new);
27 % integrator.advance();
28 %
29 % % Pure diffusion (only linear term)
30 % U_new = integrator.step(L, [], []);
31 %
32 % % Pure forcing (only source term)
33 % U_new = integrator.step([], S, []);
34 %
35 % % Identity evolution (no dynamics)
36 % U_new = integrator.step([], [], []);
37 %
38 % % With mass matrix
39 % U_new = integrator.step(L, S, M);
40 %
41 % See also:
42 % approx.odeint.OdeIntegrator, approx.odeint.BeIntegrator,
43 % approx.odeint.BdfIntegrator
44
45 properties
46 A % Butcher tableau coefficient matrix (lower triangular)
47 b % Stage weights vector for final update
48 c % Stage time coefficients vector
49 K % Cell array of derivative approximations for each stage
50 end

85 function obj = setLhs(obj, i, L, M)
86 % SETLHS Set the left-hand side matrix for stage i.
87 %
88 % obj = setLhs(obj, i, L, M) constructs the left-hand side
89 % matrix as M - dt*A(i,i)*L for stage i of the DIRK system.
90 %
91 % Inputs:
92 % obj - The DirkIntegrator object
93 % i - Stage index (positive integer)
94 % L - Linear operator (matrix or empty)
95 % M - Mass operator (matrix or empty)
96 %
97 % Outputs:
98 % obj - The DirkIntegrator object
99
100 dt = obj.timeline.dt;
101
102 %< Add mass term
103 obj.lhs{i} = M;
104
105 %< Add linear term contribution with diagonal coefficient
106 if ~isempty(L)
107 obj.lhs{i} = obj.lhs{i} - dt * obj.A(i, i) * L;
108 end
109 end

111 function obj = setRhs(obj, i, S, M)
112 % SETRHS Set the right-hand side vector for stage i.
113 %
114 % obj = setRhs(obj, i, S, M) constructs the right-hand side
115 % vector using mass operator, accumulated previous stages,
116 % and source term.
117 %
118 % Inputs:
119 % obj - The DirkIntegrator object
120 % i - Stage index (positive integer)
121 % S - Source term (vector, function, or empty)
122 % M - Mass operator (matrix, function, or empty)
123 %
124 % Outputs:
125 % obj - The DirkIntegrator object
126
127 dt = obj.timeline.dt;
128 U0 = obj.U0{1};
129
130 %< Add mass term contribution from previous solution
131 if isempty(M)
132 obj.rhs{i} = U0;
133 elseif iscell(M)
134 obj.rhs{i} = M{2} * U0;
135 if ~isempty(M{1})
136 obj.rhs{i} = obj.rhs{i} + M{1};
137 end
138 else
139 obj.rhs{i} = M * U0;
140 end
141
142 %< Add accumulated term from previous stages
143 for j = 1:(i - 1)
144 obj.rhs{i} = obj.rhs{i} + dt * obj.A(i, j) * obj.K{j};
145 end
146
147 %< Add source term contribution with diagonal coefficient
148 if ~isempty(S)
149 obj.rhs{i} = obj.rhs{i} + dt * obj.A(i, i) * S;
150 end
151 end

153 function U = stage(obj, i, L, S, M)
154 % STAGE Compute stage i of the DIRK method.
155 %
156 % U = stage(obj, i, L, S, M) computes the solution at stage i
157 % by solving the implicit system and computing the derivative
158 % approximation for use in subsequent stages.
159 %
160 % Inputs:
161 % obj - The DirkIntegrator object
162 % i - Stage index (positive integer)
163 % L - Linear operator (matrix, function, or empty)
164 % S - Source term (vector, function, or empty)
165 % M - Mass operator (matrix, function, or empty)
166 %
167 % Outputs:
168 % U - Solution approximation at stage i
169
170 t0 = obj.timeline.now;
171 dt = obj.timeline.dt;
172 U0 = obj.U0{1};
173
174 %< Evaluate linear term at stage time
175 if isempty(L)
176 Le = [];
177 elseif isa(L, 'function_handle')
178 Le = L(t0 + obj.c(i) * dt);
179 else
180 Le = L;
181 end
182
183 %< Evaluate source term at stage time
184 if isempty(S)
185 Se = [];
186 elseif isa(S, 'function_handle')
187 Se = S(t0 + obj.c(i) * dt);
188 else
189 Se = S;
190 end
191
192 %< Update LHS matrix if step size changed
193 if obj.timeline.hasStepSizeChanged
194 %< Evaluate LHS mass term
195 if isempty(M)
196 Me = speye(length(U0));
197 elseif isa(M, 'function_handle')
198 if isempty(obj.MLhs)
199 if nargin(M) == 1
200 obj.MLhs = M(t0 + obj.c(i) * dt);
201 elseif nargin(M) == 2
202 [~, obj.MLhs] = M(t0 + obj.c(i) * dt, t0 + obj.c(i) * dt);
203 end
204 end
205 Me = obj.MLhs;
206 else
207 Me = M;
208 end
209
210 %< Construct LHS matrix
211 obj.setLhs(i, Le, Me);
212 end
213
214 %< Evaluate RHS mass term at previous time level
215 if isempty(M)
216 Me = [];
217 elseif isa(M, 'function_handle')
218 if nargin(M) == 1
219 if isempty(obj.MRhs{i})
220 obj.MRhs{i} = M(t0);
221 end
222 Me = obj.MRhs{i};
223 elseif nargin(M) == 2
224 Me = cell(1, 2);
225 if isempty(obj.MRhs{i}) || obj.timeline.hasStepSizeChanged
226 [Me{1:2}] = M(t0 + obj.c(i) * dt, t0);
227 obj.MRhs{i} = Me{2};
228 else
229 Me{2} = obj.MRhs{i};
230 Me{1} = M(t0 + obj.c(i) * dt, t0);
231 end
232 end
233 else
234 Me = M;
235 end
236
237 %< Construct RHS vector
238 obj.setRhs(i, Se, Me);
239
240 %< Solve linear system for stage approximation
241 U = obj.solver.solve(obj.lhs{i}, obj.rhs{i});
242
243 %< Compute derivative approximation for this stage
244 obj.K{i} = 0;
245 if ~isempty(Le)
246 obj.K{i} = obj.K{i} + Le * U;
247 end
248 if ~isempty(Se)
249 obj.K{i} = obj.K{i} + Se;
250 end
251 end

253 function U = step(obj, L, S, M)
254 % STEP Advance one time step using the DIRK method.
255 %
256 % U = step(obj, L, S, M) performs a complete time step by
257 % computing all stages and combining them according to the
258 % Butcher tableau weights.
259 %
260 % Inputs:
261 % obj - The DirkIntegrator object
262 % L - Linear operator (matrix, function, or empty)
263 % S - Source term (vector, function, or empty)
264 % M - Mass operator (matrix, function, or empty)
265 %
266 % Outputs:
267 % U - Solution at the next time step
268
269 t0 = obj.timeline.now;
270 dt = obj.timeline.dt;
271 U0 = obj.U0{1};
272 s = obj.nStages;
273
274 %< Compute all stage approximations
275 for i = 1:s
276 [~] = obj.stage(i, L, S, M);
277 end
278
279 %< Update LHS matrix for final solve if step size changed
280 if obj.timeline.hasStepSizeChanged
281 %< Evaluate LHS mass term
282 if isempty(M)
283 Me = speye(length(U0));
284 elseif isa(M, 'function_handle')
285 if isempty(obj.MLhs)
286 if nargin(M) == 1
287 obj.MLhs = M(t0 + dt);
288 elseif nargin(M) == 2
289 [~, obj.MLhs] = M(t0 + dt, t0 + dt);
290 end
291 end
292 Me = obj.MLhs;
293 else
294 Me = M;
295 end
296 obj.lhs{s + 1} = Me;
297 end
298
299 %< Evaluate mass term for final solve
300 if isempty(M)
301 Me = [];
302 elseif isa(M, 'function_handle')
303 if nargin(M) == 1
304 if isempty(obj.MRhs{s + 1})
305 obj.MRhs{s + 1} = M(t0);
306 end
307 Me = obj.MRhs{s + 1};
308 elseif nargin(M) == 2
309 Me = cell(1, 2);
310 if isempty(obj.MRhs{s + 1}) || obj.timeline.hasStepSizeChanged
311 [Me{1:2}] = M(t0 + dt, t0);
312 obj.MRhs{s + 1} = Me{2};
313 else
314 Me{2} = obj.MRhs{s + 1};
315 Me{1} = M(t0 + dt, t0);
316 end
317 end
318 else
319 Me = M;
320 end
321
322 %< Add mass term contribution from previous solution
323 if isempty(Me)
324 obj.rhs{s + 1} = U0;
325 elseif iscell(Me)
326 obj.rhs{s + 1} = Me{2} * U0;
327 if ~isempty(Me{1})
328 obj.rhs{s + 1} = obj.rhs{s + 1} + Me{1};
329 end
330 else
331 obj.rhs{s + 1} = Me * U0;
332 end
333
334 %< Add weighted stage derivatives
335 for i = 1:s
336 obj.rhs{s + 1} = obj.rhs{s + 1} + dt * obj.b(i) * obj.K{i};
337 end
338
339 %< Solve for final step approximation
340 U = obj.solver.solve(obj.lhs{s + 1}, obj.rhs{s + 1});
341 end
342 end
