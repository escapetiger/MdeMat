1classdef FluxAssembly < approx.assembly.Assembly
2 % FLUXASSEMBLY Assembly for flux terms in DG methods.
3 %
4 % FluxAssembly handles the assembly of flux terms that arise in
5 % discontinuous Galerkin (DG) formulations. These flux terms
6 % provide inter-element coupling in DG methods, where the solution
7 % is allowed to be discontinuous across element boundaries.
8 %
9 % The class supports different flux types (left-biased, right-biased,
10 % central) and boundary condition types (periodic, Dirichlet).
11 % It manages both interior flux assembly (between neighboring
12 % elements) and boundary flux assembly (at domain boundaries).
13 %
14 % Flux terms are essential for stability and accuracy in DG methods,
15 % as they control the flow of information across element interfaces
16 % and enforce appropriate boundary conditions. The choice of flux
17 % type affects the stability properties and numerical diffusion
18 % characteristics of the resulting scheme.
19 %
20 % See also:
21 % approx.assembly.Assembly, approx.assembly.VolumeAssembly,
22 % approx.assembly.InteriorFluxAssembly
23

35 % obj = FluxAssembly(space, operator, bcType, fluxType)

132 function T = assembleImplicitBoundaryJump(obj, dim, coe)
133 % ASSEMBLEIMPLICITBOUNDARYJUMP Assemble implicit boundary jump terms.
134 %
135 % T = assembleImplicitBoundaryJump(obj, dim, coe) creates
136 % triplets for the jump contribution of boundary flux along
137 % the dimension dim with the coefficient coe. Jump terms
138 % arise in DG formulations to handle discontinuities at
139 % domain boundaries.
140 %
141 % Inputs:
142 % obj - The FluxAssembly object
143 % dim - Dimension index (positive integer)
144 % coe - Coefficient vector (nElements x 1)
145 %
146 % Outputs:
147 % T - Triplet matrix [row, col, val] for jump terms
148
149 switch lower(obj.fluxType)
150 case 'left'
151 T = obj.boundary.assembleImplicitLeftJump(dim, coe);
152 case 'right'
153 T = obj.boundary.assembleImplicitRightJump(dim, coe);
154 case 'central'
155 T1 = obj.boundary.assembleImplicitLeftJump(dim, coe/2);
156 T2 = obj.boundary.assembleImplicitRightJump(dim, coe/2);
157 T = [T1; T2];
158 end
159 end

184 function T = assembleLeftFlux(obj, dim, coe)
185 % ASSEMBLELEFTFLUX Assemble left-biased flux terms.
186 %
187 % T = assembleLeftFlux(obj, dim, coe) combines interior and
188 % boundary contributions for left-biased flux terms along the
189 % specified dimension. Left-biased fluxes provide upwind-like
190 % characteristics that can enhance stability for convection-
191 % dominated problems.
192 %
193 % Inputs:
194 % obj - The FluxAssembly object
195 % dim - Dimension index (positive integer)
196 % coe - Coefficient vector (nElements x 1)
197 %
198 % Outputs:
199 % T - Combined triplet matrix for left flux terms
200
201 T1 = obj.interior.assembleInnerLeftFlux(dim, coe);
202 T2 = obj.interior.assembleOuterLeftFlux(dim, coe);
203 T3 = obj.boundary.assembleInnerLeftFlux(dim, coe);
204 T4 = obj.boundary.assembleOuterLeftFlux(dim, coe);
205 T = [T1; T2; T3; T4];
206 end
