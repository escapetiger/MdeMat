1classdef HermiteBasisFunction < core.symbolic.InterpolationPolynomialBasisFunction
2 % HERMITEBASISFUNCTION Hermite interpolation basis functions.
3 %
4 % HermiteBasisFunction generates Hermite interpolation basis functions
5 % that interpolate both function values and derivatives at specified
6 % nodes. Each basis function satisfies specific conditions at the
7 % interpolation nodes for both the function and its derivatives.
8 %
9 % For \f$n\f$ nodes, there are \f$2n\f$ Hermite basis functions:
10 % Index \f$1\f$ to \f$n\f$: function value interpolation
11 % Index \f$n+1\f$ to \f$2n\f$: derivative interpolation
12 %
13 % Examples:
14 % % Function value basis at first node
15 % nodes = [0, 1];
16 % basis = core.symbolic.HermiteBasisFunction(nodes, 1);
17 %
18 % % Derivative basis in monic form
19 % basis = core.symbolic.HermiteBasisFunction(nodes, 3, sym('t'), true);
20 %
21 % See also:
22 % core.symbolic.InterpolationPolynomialBasisFunction

25 function expr = generate(obj)
26 % GENERATE Generates the Hermite basis function.
27 %
28 % expr = generate(obj) creates the symbolic expression for
29 % the Hermite interpolation basis function. Handles both
30 % function value and derivative interpolation conditions
31 % based on the index.
32 %
33 % Inputs:
34 % obj - The HermiteBasisFunction object
35 %
36 % Outputs:
37 % expr - Symbolic expression for the Hermite basis function
38
39 x = obj.variables{1};
40 n = length(obj.nodes);
41 i = obj.index;
42 nodes = obj.nodes;
43
44
45 if i <= n
46 H = obj.generateFunctionBasis(i, nodes, x);
47 else
48 nodeIdx = i - n;
49 H = obj.generateDerivativeBasis(nodeIdx, nodes, x);
50 end
51
52 if obj.monic
53 coeffs = sym2poly(H);
54 leadingCoeff = coeffs(1);
55 expr = horner(simplify(H / leadingCoeff), x);
56 else
57 expr = horner(simplify(H), x);
58 end
59 end
60 end

63 function H = generateFunctionBasis(~, nodeIdx, nodes, x)
64 % GENERATEFUNCTIONBASIS Generates Hermite function value basis.
65
66 n = length(nodes);
67 xi = nodes(nodeIdx);
68
69 L = sym(1);
70 for j = 1:n
71 if j ~= nodeIdx
72 L = L * (x - nodes(j)) / (xi - nodes(j));
73 end
74 end
75
76 Lp_xi = subs(diff(L, x), x, xi);
77
78 H = L * (1 - 2 * Lp_xi * (x - xi));
79 end

82 % GENERATEDERIVATIVEBASIS Generates Hermite derivative basis.
83
84 n = length(nodes);

91 end
92 end
