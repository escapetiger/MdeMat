1classdef FiniteElement < handle
2 % FINITEELEMENT Abstract base class for finite element representations.
3 %
4 % FiniteElement provides the foundation for finite element methods
5 % with domain geometry, function space, and integration capabilities.
6 % This abstract base class defines the common interface for all
7 % finite element implementations in the finite element method
8 % framework.
9 %
10 % The class encapsulates the essential components of a finite
11 % element: the geometric domain and the function space projector. It
12 % provides access to geometric properties and function evaluation
13 % capabilities that are inherited by all concrete finite element
14 % types.
15 %
16 % Finite elements discretize continuous function spaces by defining
17 % basis functions over geometric domains and providing numerical
18 % integration capabilities for computing integrals in weak
19 % formulations.
20 %
21 % See also:
22 % fem.element.C0FiniteElement, fem.element.L2FiniteElement,
23 % core.geometry.Geometry, approx.project.Projector

105 function Y = evaluate(obj, varargin)
106 % EVALUATE Evaluate functions in the finite element space.
107 %
108 % Y = evaluate(obj, Z) evaluates the basis functions at
109 % points specified in the combined matrix Z = [X; C], where X
110 % contains evaluation points and C contains coefficients.
111 %
112 % Y = evaluate(obj, X, C) evaluates the finite element
113 % function defined by coefficients C at the evaluation points
114 % X.
115 %
116 % Inputs:
117 % obj - The FiniteElement object
118 % varargin - Input arguments
119 %< Z - Combined matrix [X; C] where X are points and C are coefficients
120 %< X - Evaluation points (nDims x nPoints matrix)
121 %< C - Coefficient vectors (nDofs x nFunctions matrix)
122 %
123 % Outputs:
124 % Y - Function values at evaluation points
125 %< * If nargin = 2: (1 x nPoints) vector of basis evaluations
126 %< * If nargin = 3: (nFunctions x nPoints) matrix of function values
127
128 Y = obj.projector.evaluate(varargin{:});
129 end
