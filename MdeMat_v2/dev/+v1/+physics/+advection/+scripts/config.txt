OPTION=outDir=assets

# Time discretization

OPTION=tDiscId='be'

# CFL number
OPTION=cfl=0.5

# Space discretization
# -----------------------------------------------------------------------

# xDiscId='dg': discontinuous Galerkin
OPTION=xDiscId='dg'

# spatial order
OPTION=xBasisOrder=1

# xBasisId=1: Monic Legendre basis,

# xBasisId=3: Lagrange basis on Gauss-Lobatto nodes
OPTION=xBasisId=1

# xBasisPattern=1: Qk polynomial space
# xBasisPattern=2: Pk polynomial space
OPTION=xBasisPattern=2

# xAdvectionType=1: Upwind advection
# xAdvectionType=2: Semi-Lagrangian advection
OPTION=xAdvectionType=1
