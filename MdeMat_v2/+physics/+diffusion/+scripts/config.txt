# xPrmFluxType='right': alternating-right flux
# xPrmFluxType='central': central flux

# xAuxBoundaryJumpType='none': no jump at boundary
# xAuxBoundaryJumpType='implicit': implicit jump at boundary
# xAuxBoundaryJumpType='explicit': explicit jump at boundary (parabolic CFL)
OPTION=xAuxBoundaryJumpType='implicit'
